    I = sp.Point(I)

    area = abs(sp.Polygon(A, B, C).area)
    r = area / s
    # Contact point D on BC: projection of I onto line BC
    # param t for B + t*(C-B)
    # D = B + t*(C - B) where (I - D) dot (C - B) = 0
    BC = C - B
    t = ((I - B).dot(BC)) / (BC.dot(BC))
    D = B + t * BC
    D = sp.Point(D)

    AB = B - A

    AC = C - A
    t3 = ((I - A).dot(AC)) / (AC.dot(AC))
    E = A + t3 * AC
    E = sp.Point(E)
    return {'I': I, 'r': r, 'D': D, 'E': E, 'F': F}

def circumcenter(p1,p2,p3):
    # returns (ox, oy)
    (x1,y1), (x2,y2), (x3,y3) = p1,p2,p3
    d = 2*(x1*(y2-y3) + x2*(y3-y1) + x3*(y1-y2))
    if abs(d) < 1e-12:
        raise ValueError("Collinear points")
    ux = ((x1*x1+y1*y1)*(y2-y3) + (x2*x2+y2*y2)*(y3-y1) + (x3*x3+y3*y3)*(y1-y2)) / d
    uy = ((x1*x1+y1*y1)*(x3-x2) + (x2*x2+y2*y2)*(x1-x3) + (x3*x3+y3*y3)*(x2-x1)) / d
    return (ux, uy)

def dist(p,q):
    return math.hypot(p[0]-q[0], p[1]-q[1])

def incircle_center(A,B,C):
    # side lengths opposite vertices
    a = dist(B,C)
    b = dist(C,A)
    c = dist(A,B)
    # weighted sum: (a*A + b*B + c*C)/(a+b+c)
    x = (a*A[0] + b*B[0] + c*C[0])/(a+b+c)
    y = (a*A[1] + b*B[1] + c*C[1])/(a+b+c)
    return (x,y)

def foot_point(P, L1, L2):
    # projection of point P onto line through L1 and L2
    # return foot point
    x1,y1 = L1
    x2,y2 = L2
    xp,yp = P
    # vector v = L2-L1
    vx = x2 - x1
    vy = y2 - y1
    # if L1==L2: degenerate
    if abs(vx) < 1e-12 and abs(vy) < 1e-12:
        return L1
    # t = ((P - L1)·v)/|v|^2
    t = ((xp - x1)*vx + (yp - y1)*vy) / (vx*vx + vy*vy)
    foot = (x1 + t*vx, y1 + t*vy)
    return foot

def circle_intersections(o1,r1,o2,r2):
    # returns list of intersection points (0, 1, or 2)
    x1,y1 = o1
    x2,y2 = o2
    dx = x2 - x1
    dy = y2 - y1
    d = math.hypot(dx, dy)
    # no intersection
    if d > r1 + r2 + 1e-12 or d < abs(r1 - r2) - 1e-12 or d < 1e-12:
        return []
    # a = (r1^2 - r2^2 + d^2)/(2d)
    a = (r1*r1 - r2*r2 + d*d) / (2*d)

def line_intersection(line1, line2):
    # line1: (P1, dir1), line2: (P2, dir2)
    (x1,y1), (dx1,dy1) = line1
    (x2,y2), (dx2,dy2) = line2

    # Solve for t1,t2

    A = [[dx1, -dx2], [dy1, -dy2]]

    x2 = O_T[0] - sqrt_disc
    # choose the one not equal to B.x (0)
    eps = 1e-8
    T = None
    if abs(x1) > eps:
        T = (x1, 0.0)
    elif abs(x2) > eps:
        T = (x2, 0.0)
    else:
        # both are zero => degenerate
        return None
    # compute distances
    CT = dist(C, T)
    BT = dist(B, T)
    NB = dist(N, B)
    NE = dist(N, E)
    if BT == 0 or NE == 0:
        return None
    R_val = (CT * NB) / (BT * NE)
    return {
        'A': A, 'B': B, 'C': C, 'I': I, 'r': r,
        'D': D, 'E': E, 'F': F,
        'K': K, 'Kp': Kp, 'N': N,
        'cyclic_error': cyclic_error,
        'R': R_val,
        'T': T,
        'CT': CT, 'BT': BT, 'NB': NB, 'NE': NE,
        'R_cyc_err': cyclic_error
    }

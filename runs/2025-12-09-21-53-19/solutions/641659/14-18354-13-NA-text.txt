def compute_geometry(n, s):
    # compute geometry values, return dict
    # Fibonacci numbers
    # compute F_n for n
    def fib(k):
        # simple
        a,b=0,1
        for i in range(k):
            a,b=b,a+b
        return a
    f = fib(n)
    g = fib(n+1)
    a = fib(n+2)
    # side lengths
    b = s - f
    c = s - g
    # check positivity
    if b <= 0 or c <= 0:
        raise ValueError("Invalid side lengths")
    # coordinates
    xA = (a**2 + c**2 - b**2) / (2*a)
    # compute yA via sqrt
    y_sq = c**2 - xA**2
    if y_sq <= 0:
        raise ValueError("Negative y^2")
    yA = mp.sqrt(y_sq)
    # points
    B = mp.matrix([0,0])
    C = mp.matrix([a,0])
    A = mp.matrix([xA, yA])
    # incircle radius
    s_semiperim = s # note s is semiperimeter already defined as s variable
    # Actually s is semiperimeter by definition
    # area using Heron's formula
    Δ = mp.sqrt(s*(s-a)*(s-b)*(s-c))
    r = Δ / s
    I = mp.matrix([f, r])
    # foot from I to AB
    vAB = A - B
    t = (I - B).dot(vAB) / vAB.dot(vAB)
    F = B + t * vAB
    # foot from I to AC
    vAC = C - A
    t2 = (I - A).dot(vAC) / vAC.dot(vAC)
    E = A + t2 * vAC
    # foot from D onto EF, D = (f,0)
    D = mp.matrix([f,0])
    vEF = F - E
    t3 = (D - E).dot(vEF) / vEF.dot(vEF)
    N = E + t3 * vEF
    # compute circumcenter of ABC
    # use function
    def circumcenter(P,Q,R):
        # compute circumcenter of triangle PQR using perpendicular bisectors
        # P,Q,R are vectors
        d = 2*(P[0]*(Q[1]-R[1]) + Q[0]*(R[1]-P[1]) + R[0]*(P[1]-Q[1]))
        if abs(d) < 1e-12:
            raise ValueError("Collinear")
        ux = ((P[0]**2+P[1]**2)*(Q[1]-R[1]) + (Q[0]**2+Q[1]**2)*(R[1]-P[1]) + (R[0]**2+R[1]**2)*(P[1]-Q[1]))/d
        uy = ((P[0]**2+P[1]**2)*(R[0]-Q[0]) + (Q[0]**2+Q[1]**2)*(P[0]-R[0]) + (R[0]**2+R[1]**2)*(Q[0]-P[0]))/d
        return mp.matrix([ux, uy])
    O_ABC = circumcenter(A,B,C)
    R_ABC = mp.norm(A - O_ABC)
    # circumcenter of AEF
    O_AEF = circumcenter(A,E,F)
    R_AEF = mp.norm(A - O_AEF)
    # intersection of circles O_ABC,R_ABC and O_AEF,R_AEF
    # compute intersection points
    d_centers = mp.norm(O_AEF - O_ABC)
    # if no intersection, return None
    if d_centers > R_ABC + R_AEF + 1e-9:
        # no intersection
        return None
    # distance from O_ABC to line of intersection points along connecting line
    a_len = (R_ABC**2 - R_AEF**2 + d_centers**2) / (2*d_centers)
    # height
    h_sq = R_ABC**2 - a_len**2
    if h_sq < 0:
        h = 0
    else:
        h = mp.sqrt(h_sq)
    # base point
    ex = (O_AEF - O_ABC) / d_centers
    # perpendicular vector
    perp = mp.matrix([-ex[1], ex[0]])
    P = O_ABC + a_len * ex
    # two intersection points
    inter1 = P + h * perp
    inter2 = P - h * perp
    # Determine which is A
    # distance to A
    if mp.norm(inter1 - A) < mp.norm(inter2 - A):
        K = inter2  # other is K
    else:
        K = inter1
    # reflect K across line EF
    # reflect point Q across line through P1,P2
    def reflect_point_across_line(Q, P1, P2):
        # foot M of Q onto line P1P2
        v = P2 - P1
        # parameter t for foot
        t = (Q - P1).dot(v) / v.dot(v)
        M = P1 + t * v
        # reflected point
        Qr = 2*M - Q
        return Qr
    K_prime = reflect_point_across_line(K, E, F)
    # compute cyclic determinant for B,N,K,Kprime
    # B = (0,0)

        M = mp.matrix([[P1[0], P1[1], P1[0]**2 + P1[1]**2, 1],
                       [P2[0], P2[1], P2[0]**2 + P2[1]**2, 1],
                       [P3[0], P3[1], P3[0]**2 + P3[1]**2, 1],
                       [P4[0], P4[1], P4[0]**2 + P4[1]**2, 1]])
        # compute determinant
        return mp.det(M)
    det_val = cyclic_det(B, N, K, K_prime)

    # Solve for x: x = ox +/- sqrt(R_T^2 - oy^2)
    # One solution is x=0 (B). So choose the other.
    ox, oy = O_T[0], O_T[1]
    rad_sq = R_T**2

    discr = rad_sq - oy**2

    vAB = [A[0]-B[0], A[1]-B[1]]
    t = dot([I[0]-B[0], I[1]-B[1]], vAB) / dot(vAB, vAB)
    F = [B[0] + t * vAB[0], B[1] + t * vAB[1]]
    # foot to AC
    vAC = [C[0]-A[0], C[1]-A[1]]
    t2 = dot([I[0]-A[0], I[1]-A[1]], vAC) / dot(vAC, vAC)
    E = [A[0] + t2 * vAC[0], A[1] + t2 * vAC[1]]
    # foot from D to EF
    D = [f,0.0]
    vEF = [F[0]-E[0], F[1]-E[1]]
    t3 = dot([D[0]-E[0], D[1]-E[1]], vEF) / dot(vEF, vEF)
    N = [E[0] + t3 * vEF[0], E[1] + t3 * vEF[1]]
    # circumcenter functions
    def circumcenter(P,Q,R):
        # using formula for coordinates
        d = 2*(P[0]*(Q[1]-R[1]) + Q[0]*(R[1]-P[1]) + R[0]*(P[1]-Q[1]))
        if abs(d) < 1e-12:
            raise ValueError("Collinear points")
        ux = ((P[0]**2+P[1]**2)*(Q[1]-R[1]) + (Q[0]**2+Q[1]**2)*(R[1]-P[1]) + (R[0]**2+R[1]**2)*(P[1]-Q[1]))/d
        uy = ((P[0]**2+P[1]**2)*(R[0]-Q[0]) + (Q[0]**2+Q[1]**2)*(P[0]-R[0]) + (R[0]**2+R[1]**2)*(Q[0]-P[0]))/d
        return [ux, uy]
    O_ABC = circumcenter(A,B,C)
    R_ABC = norm([A[0]-O_ABC[0], A[1]-O_ABC[1]])
    O_AEF = circumcenter(A,E,F)
    R_AEF = norm([A[0]-O_AEF[0], A[1]-O_AEF[1]])
    # circle intersections
    # vector between centers
    d_centers = norm([O_AEF[0]-O_ABC[0], O_AEF[1]-O_ABC[1]])
    # check for no intersection
    if d_centers > R_ABC + R_AEF + 1e-9:
        # no intersection
        return None
    # distance from O_ABC to line of intersection
    a_len = (R_ABC**2 - R_AEF**2 + d_centers**2) / (2*d_centers)
    # height
    h_sq = R_ABC**2 - a_len**2
    if h_sq < 0:
        h = 0.0
    else:
        h = mp.sqrt(h_sq)
    # unit vector along line connecting centers
    ex = [(O_AEF[0]-O_ABC[0])/d_centers, (O_AEF[1]-O_ABC[1])/d_centers]
    # perpendicular unit vector
    perp = [-ex[1], ex[0]]
    # base point
    P = [O_ABC[0] + a_len * ex[0], O_ABC[1] + a_len * ex[1]]
    inter1 = [P[0] + h * perp[0], P[1] + h * perp[1]]
    inter2 = [P[0] - h * perp[0], P[1] - h * perp[1]]
    # pick K as the one not equal to A
    if norm([inter1[0]-A[0], inter1[1]-A[1]]) < norm([inter2[0]-A[0], inter2[1]-A[1]]):
        K = inter2
    else:
        K = inter1
    # reflect K across line EF
    def reflect(Q, P1, P2):
        v = [P2[0]-P1[0], P2[1]-P1[1]]
        # foot M
        t = dot([Q[0]-P1[0], Q[1]-P1[1]], v) / dot(v, v)
        M = [P1[0] + t * v[0], P1[1] + t * v[1]]
        # reflection
        Qr = [2*M[0] - Q[0], 2*M[1] - Q[1]]
        return Qr
    K_prime = reflect(K, E, F)
    # cyclic determinant
    def cyclic_det(P1,P2,P3,P4):
        # matrix rows: x, y, x^2+y^2, 1
        M = [
            [P1[0], P1[1], P1[0]**2 + P1[1]**2, 1.0],
            [P2[0], P2[1], P2[0]**2 + P2[1]**2, 1.0],
            [P3[0], P3[1], P3[0]**2 + P3[1]**2, 1.0],
            [P4[0], P4[1], P4[0]**2 + P4[1]**2, 1.0],
        ]
        # compute determinant
        return mp.det(M)
    det_val = cyclic_det(B, N, K, K_prime)
    # Circle tangent to BN at B through K
    BN = [N[0]-B[0], N[1]-B[1]]
    # perpendicular to BN: rotate 90 deg CCW
    perp_BN = [-BN[1], BN[0]]
    # Compute lambda for center O_T = B + λ * perp_BN
    # Solve |λ perp|^2 = |λ perp - K|^2 => λ^2*|perp|^2 = λ^2*|perp|^2 - 2λ perp·K + |K|^2 => 0 = -2λ perp·K + |K|^2 => λ = |K|^2/(2*perp·K)
    dot_perp_K = dot(perp_BN, K)
    if abs(dot_perp_K) < 1e-12:
        O_T = None
        BT = None
        CT = None
        NB = None
        NE = None
        ratio = None
    else:
        lam = (K[0]**2 + K[1]**2) / (2*dot_perp_K)
        O_T = [lam * perp_BN[0], lam * perp_BN[1]]
        R_T = norm(O_T)  # radius
        # Intersection with BC (y=0)
        ox, oy = O_T[0], O_T[1]
        disc = R_T**2 - oy**2
        if disc < 0:
            T = None
        else:
            sqrt_disc = mp.sqrt(disc)
            x1 = ox + sqrt_disc
            x2 = ox - sqrt_disc
            # identify B at x=0
            if abs(x1) < 1e-8:
                t_coord = x2
            elif abs(x2) < 1e-8:
                t_coord = x1
            else:
                # Choose the one not 0
                # if both not near zero, pick the positive one maybe?
                # but B is at 0, so there must be one root near 0, else error
                t_coord = x1 if abs(x1) > 1e-6 else x2
            T = [t_coord, 0.0]
        # compute distances
        NB = norm([N[0]-B[0], N[1]-B[1]])
        NE = norm([N[0]-E[0], N[1]-E[1]])
        BT = norm([B[0]-T[0], B[1]-T[1]]) # should be |t_coord|
        CT = norm([C[0]-T[0], C[1]-T[1]])
        ratio = (CT * NB) / (BT * NE) if BT != 0 and NE != 0 else None
    # compile
    return {
        'A':A,'B':B,'C':C,'E':E,'F':F,'D':D,'I':I,'N':N,'K':K,'K_prime':K_prime,
        'det':det_val,'ratio':ratio,'s':s,'r':r,'f':f,'g':g,'a':a,
        'T':T,'BT':BT,'CT':CT,'NB':NB,'NE':NE
    }

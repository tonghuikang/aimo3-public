def triangle_from_params(x,y,s):
    # x = BD = BF, y = CD = CE, a = BC = x+y
    a = x+y
    # b = CA = s - x
    b = s - x
    c = s - y
    # Check positivity
    if not (b>0 and c>0):
        return None
    # incircle radius r = sqrt[(s-a)*(s-b)*(s-c)/s]
    r = sp.sqrt((s-a)*(s-b)*(s-c)/s)
    # define angles φ, ψ
    # using sympy's atan
    φ = 2*sp.atan(r/x)
    ψ = 2*sp.atan(y/r)
    # direction vectors
    dirAB = sp.Matrix([sp.cos(φ), sp.sin(φ)])
    dirCA = sp.Matrix([sp.cos(ψ), sp.sin(ψ)])  # from C to A
    # coordinates of B and C
    B = sp.Matrix([0,0])
    C = sp.Matrix([a,0])
    # Solve B + t1*dirAB = C + t2*dirCA
    t1, t2 = sp.symbols('t1 t2')
    sol = sp.solve(B + t1*dirAB - (C + t2*dirCA), (t1, t2), dict=True)
    if not sol:
        return None
    sol = sol[0]
    t1_val = sol[t1]
    # compute A
    A = B + t1_val*dirAB
    # compute F and E: F = B + x*unit_AB (unit vector same as dirAB)
    F = B + x*dirAB
    # E = C + y*dirCA
    E = C + y*dirCA
    # incircle center I = (x, r)
    I = sp.Matrix([x, r])
    # Now compute circumcircles of ABC and AFE
    # Determine circle eq: center and radius for each.
    def circumcenter(P1,P2,P3):
        # use sympy geometry? We'll compute via perpendicular bisectors.
        # Solve for center (h,k) such that distances to three points equal.
        h,k = sp.symbols('h k')
        eqs = [
            (P1[0]-h)**2 + (P1[1]-k)**2 - ((P2[0]-h)**2 + (P2[1]-k)**2),
            (P1[0]-h)**2 + (P1[1]-k)**2 - ((P3[0]-h)**2 + (P3[1]-k)**2)
        ]
        solc = sp.solve(eqs, (h,k), dict=True)
        if not solc:
            return None

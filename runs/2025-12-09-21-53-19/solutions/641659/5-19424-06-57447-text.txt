    a,b=0,1
    for _ in range(n):
        a,b=b,a+b
    return a

def find_s_for_n(n, s_guess_low, s_guess_high):
    # Find s such that det ~ 0 using root finding (bisection)
    # We'll use simple bisection on sign of det
    # Ensure det changes sign over interval
    f_low = geometry_values(n, s_guess_low)
    f_high = geometry_values(n, s_guess_high)
    if f_low is None or f_high is None:
        return None
    det_low = f_low['det']
    det_high = f_high['det']
    # Try to adjust range until sign change
    # We'll sample log spaced values
    # but for now just try bisection assuming sign changes.
    if det_low == 0:
        return s_guess_low, f_low
    if det_high == 0:
        return s_guess_high, f_high
    # if same sign, try expand range
    if det_low*det_high > 0:
        # try extend high
        for mult in [2,4,8,16,32,64,128,256,512,1024]:
            new_high = s_guess_high*mult
            f_high = geometry_values(n, new_high)
            if f_high is None: continue
            det_high = f_high['det']
            if det_low*det_high < 0:
                s_guess_high = new_high
                break
        else:
            # try lower
            for div in [2,4,8,16,32,64,128,256,512,1024]:
                new_low = s_guess_low/div
                f_low = geometry_values(n, new_low)
                if f_low is None: continue
                det_low = f_low['det']
                if det_low*det_high < 0:
                    s_guess_low = new_low
                    break
            else:
                # no sign change found
                return None
    # now bisection for root
    for _ in range(60):
        mid = (s_guess_low + s_guess_high)/2
        f_mid = geometry_values(n, mid)
        if f_mid is None:
            break
        det_mid = f_mid['det']
        if det_mid == 0:
            return mid, f_mid
        if det_low*det_mid < 0:
            s_guess_high = mid
            det_high = det_mid
        else:
            s_guess_low = mid
            det_low = det_mid
    # return mid approximate
    return (s_guess_low + s_guess_high)/2, geometry_values(n, (s_guess_low + s_guess_high)/2)

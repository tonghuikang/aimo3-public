def v2(n):
    cnt = 0
    while n % 2 == 0:
        cnt += 1
        n //= 2
    return cnt

def odd_part(n):
    while n % 2 == 0:
        n //= 2
    return n

def compute_D_from_indices(indices):
    if not indices:
        return None
    s = v2(indices[0])
    for n in indices:
        if v2(n) != s:
            raise ValueError("Indices have different v2")
    # compute lcm of odd parts
    odds = [odd_part(n) for n in indices]
    L = odds[0]
    for o in odds[1:]:
        L = math.lcm(L, o)
    D = (2**(s-1)) * L
    return D

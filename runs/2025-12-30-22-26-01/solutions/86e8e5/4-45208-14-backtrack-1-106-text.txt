def get_ratio(p,q):
    pq = p*q
    D = pq + p + q
    g = math.gcd(pq, D)
    Drem = D // g
    # check if Drem power of 3
    if Drem <= 0:
        return None
    while Drem % 3 == 0:
        Drem //= 3
    if Drem != 1:
        return None
    # ratio as rational (pq/D) simplified
    # compute simplified numerator and denominator
    numerator = pq // g
    denominator = D // g
    return numerator, denominator

def search(limit):
    best = None
    best_pair = None
    for p in range(2, limit+1):
        for q in range(p+1, limit+1):
            r = get_ratio(p,q)
            if r is not None:
                num, den = r
                # ratio = num/den
                ratio = num/den
                if best is None or ratio < best:
                    best = ratio
                    best_pair = (p,q,num,den)
    return best, best_pair

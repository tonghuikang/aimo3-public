def divisors_of_binomial(d):
    # Return set of Poly divisors of x^d + 1 with integer coefficients
    poly = sp.Poly(x**d + 1, x, domain='ZZ')
    # Factor
    factors = sp.factor_list(poly)[1]
    # each factor appears exponent 1 (squarefree)
    irr = [f[0] for f in factors]
    # generate subsets
    divs = set()
    n = len(irr)
    for mask in range(1 << n):
        prod = sp.Poly(1, x, domain='ZZ')

                prod = prod * sp.Poly(irr[i], x, domain='ZZ')
        prod = sp.Poly(prod.as_expr().expand(), x, domain='ZZ')
        divs.add(prod)
    return divs

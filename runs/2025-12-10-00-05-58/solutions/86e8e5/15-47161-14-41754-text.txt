def get_M_mod_S(S):
    # compute M mod S where M = 3^E, E huge (2025!)
    # If S==1 return 0
    if S == 1:
        return 0
    # factor S
    fac = sp.factorint(S)
    a = 1
    for p, exp in fac.items():
        if p == 3:
            a = 3**exp
    t = S // a
    # a is power of 3; t coprime to 3
    # M ≡ 0 mod a (since E >= exp)
    # M ≡ 1 mod t (since exponent is multiple of phi(t))
    # Solve using CRT
    # Find x such that x ≡ 0 (mod a), x ≡ 1 (mod t)
    # Use sympy.crt
    if t == 1:
        # only power of 3
        return 0
    else:
        x = sp.crt([a, t], [0, 1])[0]  # returns solution mod lcm(a,t)=S
        return int(x)

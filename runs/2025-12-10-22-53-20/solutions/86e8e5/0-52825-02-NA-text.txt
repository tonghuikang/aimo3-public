    temp = T
    e = 0
    while temp % 3 == 0:
        temp //= 3
        e += 1
    T3 = 3**e
    Tprime = T // T3
    if Tprime == 1:
        return 0
    # compute inverse of T3 modulo Tprime
    inv_T3 = sp.mod_inverse(T3, Tprime)
    return (T3 * inv_T3) % T

def n_mod_T(T, case):

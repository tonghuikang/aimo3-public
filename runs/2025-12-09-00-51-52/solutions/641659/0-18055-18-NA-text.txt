<|end|><|start|>assistant<|channel|>analysis to=python code<|message|>def circumcenter_three(p1, p2, p3):
    # compute circumcenter of three non-collinear points
    # using perpendicular bisector intersection
    # p1,p2,p3 are mp.matrix([x,y])
    # define vectors
    v12 = p2 - p1
    v13 = p3 - p1
    # midpoints
    m12 = (p1 + p2) / 2
    m13 = (p1 + p3) / 2
    # perpendicular directions (rotate by 90 deg)
    # rotate vector (dx,dy) -> (-dy, dx)
    perp12 = mp.matrix([-v12[1], v12[0]])
    perp13 = mp.matrix([-v13[1], v13[0]])
    # Solve for t,s: m12 + t*perp12 = m13 + s*perp13
    # Solve 2x2 linear system for t and s.
    # Write equation: t*perp12 - s*perp13 = m13 - m12
    # unknowns [t, s]
    A_mat = mp.matrix([[perp12[0], -perp13[0]], [perp12[1], -perp13[1]]])
    b_vec = m13 - m12
    # Solve linear system
    # Use mp.linalg? Not built-in; solve manually
    # Compute determinant
    det = A_mat[0,0]*A_mat[1,1] - A_mat[0,1]*A_mat[1,0]
    if mp.almosteq(det, 0):
        raise ValueError("Points are collinear")
    t = (b_vec[0]*A_mat[1,1] - b_vec[1]*A_mat[0,1]) / det
    # s = (A_mat[0,0]*b_vec[1] - A_mat[1,0]*b_vec[0]) / det   # not needed
    O = m12 + t * perp12
    return O

<|end|><|start|>assistant<|channel|>analysis to=python code<|message|>def circle_intersections(O1, R1, O2, R2):
    # returns list of intersection points (as mp.matrix)
    # Using standard formula
    # distance between centers
    d = mp.sqrt((O2[0]-O1[0])**2 + (O2[1]-O1[1])**2)
    # check for no intersection?
    # Compute a = (R1^2 - R2^2 + d^2) / (2d)
    a = (R1**2 - R2**2 + d**2) / (2*d)
    # height from chord to intersection points
    h_sq = R1**2 - a**2
    if h_sq < 0:
        h_sq = 0  # handle numerical issues
    h = mp.sqrt(h_sq)
    # point P2 = O1 + a*(O2-O1)/d
    P2 = O1 + a * (O2 - O1) / d
    # perpendicular vector (dx, dy) from O1 to O2 is (dx, dy)
    dx = (O2[0] - O1[0]) / d
    dy = (O2[1] - O1[1]) / d
    # perpendicular vectors: (-dy, dx) and (dy, -dx)
    # Intersection points:
    intersect1 = mp.matrix([P2[0] - h*dy, P2[1] + h*dx])
    intersect2 = mp.matrix([P2[0] + h*dy, P2[1] - h*dx])
    return [intersect1, intersect2]

<|end|><|start|>assistant<|channel|>analysis to=python code<|message|>def reflect_point_across_line(P, Q1, Q2):
    # Reflect point P across line defined by Q1, Q2.
    # Compute unit direction of line
    v = Q2 - Q1
    v_norm = mp.sqrt(v[0]**2 + v[1]**2)
    u = v / v_norm
    # projection of (P - Q1) onto direction
    proj_len = (P[0] - Q1[0])*u[0] + (P[1] - Q1[1])*u[1]
    proj = Q1 + proj_len * u
    # perpendicular component
    perp = P - proj
    # reflection
    P_ref = P - 2*perp
    return P_ref

def compute_K_prime(K, E, F):

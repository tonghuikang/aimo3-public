def circumcenter(p1, p2, p3):
    # Returns circumcenter (cx,cy) of triangle p1,p2,p3
    # Using perpendicular bisector intersection.
    (x1,y1) = p1
    (x2,y2) = p2
    (x3,y3) = p3
    # Compute determinants:
    d = 2*(x1*(y2 - y3) + x2*(y3 - y1) + x3*(y1 - y2))
    if abs(d) < 1e-12:
        # Points collinear or nearly
        return None
    ux = ((x1**2 + y1**2)*(y2 - y3) + (x2**2 + y2**2)*(y3 - y1) + (x3**2 + y3**2)*(y1 - y2)) / d
    uy = ((x1**2 + y1**2)*(x3 - x2) + (x2**2 + y2**2)*(x1 - x3) + (x3**2 + y3**2)*(x2 - x1)) / d
    return (ux, uy)

def intersect_circles(c0, r0, c1, r1):
    (x0,y0) = c0
    (x1,y1) = c1
    dx = x1 - x0
    dy = y1 - y0
    d = math.hypot(dx, dy)
    if d < 1e-12:
        return []  # coincident centers
    # check for no intersection
    if d > r0 + r1 + 1e-9 or d < abs(r0 - r1) - 1e-9:
        return []  # no intersection
    # compute a
    a = (r0**2 - r1**2 + d**2) / (2*d)
    # point P2
    xm = x0 + a * dx / d
    ym = y0 + a * dy / d
    # distance from P2 to intersection points
    h_sq = max(r0**2 - a**2, 0.0)
    h = math.sqrt(h_sq)
    # offset (rx, ry) perpendicular to line
    rx = -dy * (h / d)
    ry = dx * (h / d)
    p_int1 = (xm + rx, ym + ry)
    p_int2 = (xm - rx, ym - ry)
    return [p_int1, p_int2]

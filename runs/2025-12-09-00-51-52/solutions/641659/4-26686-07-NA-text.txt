def intersection_two_circles(c1, r1, c2, r2):
    # returns two intersection points (x,y) as tuples; may be complex if no intersection.

    # p1, p2, p3 are np arrays shape (2,)
    x1, y1 = p1
    x2, y2 = p2
    x3, y3 = p3
    M = np.array([[2*(x2 - x1), 2*(y2 - y1)],
                  [2*(x3 - x1), 2*(y3 - y1)]], dtype=float)
    RHS = np.array([x2**2 + y2**2 - x1**2 - y1**2,
                    x3**2 + y3**2 - x1**2 - y1**2], dtype=float)

    except np.linalg.LinAlgError:
        return None, None
    center = np.array([cx, cy], dtype=float)
    radius = np.linalg.norm(center - p1)
    return center, radius

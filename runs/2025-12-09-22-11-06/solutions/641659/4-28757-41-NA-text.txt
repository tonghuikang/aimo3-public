    # Using circle-circle intersection formulas.
    # Let O1=O_abc, r1=R_abc; O2=O_afe, r2=R_afe.
    O1x,O1y = O_abc
    O2x,O2y = O_afe
    d_centers = math.hypot(O2x-O1x, O2y-O1y)
    # If circles intersect at two points, compute intersection.
    # Compute a = (r1^2 - r2^2 + d^2) / (2d)
    # h = sqrt(r1^2 - a^2)
    a_len = (R_abc**2 - R_afe**2 + d_centers**2) / (2*d_centers)
    h_sq = max(R_abc**2 - a_len**2, 0.0)
    h = math.sqrt(h_sq)
    # point P2 = O1 + a * (O2-O1)/d
    Px = O1x + a_len * (O2x - O1x) / d_centers
    Py = O1y + a_len * (O2y - O1y) / d_centers
    # Intersection points:
    # offset = ( - (Oy2-Oy1) * (h/d), (Ox2-Ox1) * (h/d) )
    offset_x = - (O2y - O1y) * (h/d_centers)
    offset_y = (O2x - O1x) * (h/d_centers)
    int1 = (Px + offset_x, Py + offset_y)
    int2 = (Px - offset_x, Py - offset_y)
    # Determine which is A (xA,yA)
    def dist(p,q):
        return math.hypot(p[0]-q[0], p[1]-q[1])
    if dist(int1,(xA,yA)) < 1e-8:
        Kx,Ky = int2
    else:
        Kx,Ky = int1
    # reflection of K across line EF
    # line EF defined by points F (Fx,Fy) and E (Ex,Ey). Compute reflection formula.
    # Compute projection of K onto EF, then reflect.
    # projection of K onto line EF:
    # vector FE = (Ex-Fx, Ey-Fy)
    FE_x, FE_y = Ex-Fx, Ey-Fy
    # parameter t = ((K-F)·FE)/|FE|^2
    t_proj = ((Kx-Fx)*FE_x + (Ky-Fy)*FE_y) / (FE_x*FE_x + FE_y*FE_y)
    proj_x = Fx + t_proj*FE_x
    proj_y = Fy + t_proj*FE_y
    K_prime_x = 2*proj_x - Kx
    K_prime_y = 2*proj_y - Ky
    # compute diff for cyclic condition: distance(K',center of circle through B,K,N) vs radius
    # Find circumcenter of triangle B(0,0), K, N.
    O_bkn = circumcenter((0.0,0.0),(Kx,Ky),(Nx,Ny))
    if O_bkn is None:
        return None
    R_bkn = math.hypot(Kx-O_bkn[0], Ky-O_bkn[1])  # should be same for N
    diff = math.hypot(K_prime_x-O_bkn[0], K_prime_y-O_bkn[1]) - R_bkn
    # Compute NB, NE, etc
    NB = math.hypot(Nx, Ny)
    NE = math.hypot(Nx-Ex, Ny-Ey)
    # compute Oprime for tangent circle:
    # NB vector = (Nx, Ny)
    # compute denominator D = cross(NB, K) = Nx*Ky - Ny*Kx
    D = Nx*Ky - Ny*Kx
    # K squared
    K_sq = Kx*Kx + Ky*Ky
    # lambda = (K_sq/2)/D
    lam = (K_sq / 2.0) / D
    # Oprime coordinates = lam * (-Ny, Nx)
    Oprime_x = -lam * Ny
    Oprime_y = lam * Nx
    # T point is (2*Oprime_x, 0)
    T_x = 2*Oprime_x
    # compute CT, BT
    CT = abs(a - T_x)
    BT = abs(T_x) # B is at 0
    # compute ratio
    ratio_val = (CT * NB) / (BT * NE) if BT != 0 else None
    # also return diff
    return {
        'c':c,
        'diff':diff,
        'ratio':ratio_val,
        'K':(Kx,Ky),
        'Kprime':(K_prime_x,K_prime_y),
        'Oprime':(Oprime_x,Oprime_y),
        'T':(T_x,0.0),
        'N':(Nx,Ny),
        'NB':NB,
        'NE':NE,
        'CT':CT,
        'BT':BT
    }

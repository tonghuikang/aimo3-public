def compute_triangle_numeric(n, s):
    Fn = fib(n)
    Fn1 = fib(n+1)
    Fn2 = fib(n+2)
    a = Fn2
    b = s - Fn
    c = s - Fn1
    # Coordinates:
    B = (0.0, 0.0)
    C = (a*1.0, 0.0)
    # x-coordinate of A:
    xA = (c**2 - b**2 + a**2) / (2*a)
    # y-coordinate:
    y_sq = c**2 - xA**2
    if y_sq < 0:
        y_sq = 0.0
    yA = math.sqrt(y_sq)
    A = (xA, yA)
    # incircle radius:
    # area via Heron:
    s_val = s
    area = math.sqrt(s_val * (s_val - a) * (s_val - b) * (s_val - c))
    r = area / s_val
    # incircle center:
    D = (Fn*1.0, 0.0)
    I = (Fn*1.0, r)
    # points F and E:
    BF = Fn  # distance from B to F along AB
    # BF = s-b = Fn
    # So F = B + (BF/c)*(A-B)
    F = (BF/c * (A[0] - B[0]), BF/c * (A[1] - B[1]))
    # CE = Fn1
    CE = Fn1
    # E = C + (CE/b)*(A - C)
    E = (C[0] + CE/b * (A[0] - C[0]), C[1] + CE/b * (A[1] - C[1]))
    return {
        'A': A, 'B': B, 'C': C, 'D': D, 'I': I,
        'F': F, 'E': E, 'r': r, 'a': a, 'b': b, 'c': c,
        's': s_val
    }

def circle_center_radius(p1, p2, p3):
    # compute circumcenter and radius using geometry
    # Using perpendicular bisectors
    (x1,y1), (x2,y2), (x3,y3) = p1, p2, p3
    # compute using determinant formula
    # Solve for center (Ux, Uy):
    # Use linear equations: (x1 - Ux)^2 + (y1 - Uy)^2 = (x2 - Ux)^2 + (y2 - Uy)^2
    # and similarly for points 2 and 3.
    # Derive linear system:
    # 2*(x2 - x1)*Ux + 2*(y2 - y1)*Uy = x2^2 + y2^2 - x1^2 - y1^2
    # 2*(x3 - x2)*Ux + 2*(y3 - y2)*Uy = x3^2 + y3^2 - x2^2 - y2^2
    A_mat = [[2*(x2 - x1), 2*(y2 - y1)], [2*(x3 - x2), 2*(y3 - y2)]]
    B_vec = [x2**2 + y2**2 - x1**2 - y1**2, x3**2 + y3**2 - x2**2 - y2**2]
    # Solve linear system:
    det = A_mat[0][0]*A_mat[1][1] - A_mat[0][1]*A_mat[1][0]
    if abs(det) < 1e-12:
        # points nearly collinear
        return None, None
    Ux = (B_vec[0]*A_mat[1][1] - B_vec[1]*A_mat[0][1]) / det
    Uy = (A_mat[0][0]*B_vec[1] - A_mat[1][0]*B_vec[0]) / det
    R = math.hypot(x1 - Ux, y1 - Uy)
    return (Ux, Uy), R

def intersect_two_circles(O1, R1, O2, R2):
    # returns intersection points (list of 0-2 points)
    (x0, y0), (x1, y1) = O1, O2
    dx, dy = x1 - x0, y1 - y0
    d = math.hypot(dx, dy)
    # No intersection
    if d > R1 + R2 + 1e-12 or d < abs(R1 - R2) - 1e-12:
        return []
    # midpoint distance from O0
    a = (R1**2 - R2**2 + d**2) / (2*d) if d != 0 else 0
    h_sq = max(R1**2 - a**2, 0.0)
    h = math.sqrt(h_sq)
    # point P2
    xm = x0 + a * dx / d if d != 0 else x0
    ym = y0 + a * dy / d if d != 0 else y0
    # Intersection points:
    if h < 1e-12:
        return [(xm, ym)]
    # offset vector perpendicular to line O0O1
    rx = -dy * (h / d)
    ry = dx * (h / d)
    p1 = (xm + rx, ym + ry)
    p2 = (xm - rx, ym - ry)
    return [p1, p2]

def compute_K_numeric(data):
    A = data['A']; B = data['B']; C = data['C']; E = data['E']; F = data['F']
    # circumcircle ABC
    O1, R1 = circle_center_radius(A,B,C)
    # circle AEF
    O2, R2 = circle_center_radius(A,E,F)
    # intersect
    inter = intersect_two_circles(O1,R1,O2,R2)
    # pick point not equal to A
    # compute distance from A
    A_pt = A
    for p in inter:
        if math.hypot(p[0]-A_pt[0], p[1]-A_pt[1]) > 1e-6:
            return p
    # if not found, maybe due to numeric rounding
    # return the other point
    if len(inter) == 2:
        return inter[0] if math.hypot(inter[0][0]-A_pt[0], inter[0][1]-A_pt[1]) > 1e-6 else inter[1]
    raise ValueError("No K found")

def foot_of_perpendicular(P, line_pt1, line_pt2):
    # Return foot of perpendicular from point P to line defined by two points.
    (x0,y0) = P
    (x1,y1) = line_pt1
    (x2,y2) = line_pt2
    # line direction
    dx, dy = x2 - x1, y2 - y1
    # param t of projection of P onto line
    if dx == dy == 0:
        return (x1, y1)
    t = ((x0 - x1)*dx + (y0 - y1)*dy) / (dx*dx + dy*dy)
    # foot point:
    xf = x1 + t*dx
    yf = y1 + t*dy
    return (xf, yf)

def reflect_point_across_line(P, line_pt1, line_pt2):
    # Reflect point P across line through line_pt1, line_pt2
    # Use projection method: reflect across line = point + 2*(foot - point)
    foot = foot_of_perpendicular(P, line_pt1, line_pt2)
    return (2*foot[0] - P[0], 2*foot[1] - P[1])

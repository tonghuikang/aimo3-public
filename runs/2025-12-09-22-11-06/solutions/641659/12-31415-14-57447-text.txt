def compute_ratio_for_n(n, numeric=True):
    # Compute F_n, F_{n+1}, F_{n+2}
    # Use Fibonacci sequence (standard F_0=0, F_1=1)
    # We'll compute using python ints.
    def fib(k):
        a, b = 0, 1
        for _ in range(k):
            a, b = b, a+b
        return a
    Fn = fib(n)
    Fnp1 = fib(n+1)
    Fnp2 = fib(n+2)
    # define symbols
    r = sp.symbols('r', positive=True)
    
    # slopes
    m1 = 2*Fn*r/(Fn**2 - r**2)
    m2 = -2*Fnp1*r/(Fnp1**2 - r**2)
    # coordinates for A
    xA = (-m2*Fnp1 - m1*Fn) / (m1 - m2)
    yA = m1*xA + (m1*Fn - r)  # using AB line eq

    # line AB: A1 x + B1 y + C1 = 0
    A1 = m1
    B1 = -1
    C1 = m1*Fn - r
    # foot of perp from origin onto line AB: point F
    denom1 = A1**2 + B1**2
    xF = -A1*C1/denom1
    yF = -B1*C1/denom1
    
    # line AC: A2 x + B2 y + C2 = 0, from earlier: m2 x - y - (m2*Fnp1 + r) =0
    A2 = m2
    B2 = -1
    C2 = -(m2*Fnp1 + r)
    denom2 = A2**2 + B2**2
    xE = -A2*C2/denom2
    yE = -B2*C2/denom2
    
    # circumcenter O of ABC: using coordinates of A, B, C. B = (-Fn, -r), C = (Fnp1, -r)
    Bpt = sp.Point(-Fn, -r)
    Cpt = sp.Point(Fnp1, -r)
    Apt = sp.Point(xA, yA)
    # compute circumcenter via sympy geometry
    O = sp.Point(sp.geometry.Triangle(Apt, Bpt, Cpt).circumcircle.center)
    R = sp.geometry.Triangle(Apt, Bpt, Cpt).circumcircle.radius
    
    # circumcenter O1 of AEF
    Ept = sp.Point(xE, yE)
    Fpt = sp.Point(xF, yF)
    O1 = sp.Point(sp.geometry.Triangle(Apt, Ept, Fpt).circumcircle.center)
    R1 = sp.geometry.Triangle(Apt, Ept, Fpt).circumcircle.radius
    
    # compute K: second intersection of circles O and O1 (points A and K)
    # Use vector projection method: M = projection of A onto line O O1
    # line direction vector d = O1 - O
    d = sp.Matrix([O1.x - O.x, O1.y - O.y])
    # compute projection of vector AO onto d: t = ((A-O)·d)/(d·d)
    AO = sp.Matrix([xA - O.x, yA - O.y])
    t = (AO.dot(d)) / (d.dot(d))

def circumcenter(p1, p2, p3):
    # return (cx,cy) circumcenter of three points
    (x1,y1), (x2,y2), (x3,y3) = p1, p2, p3
    d = 2*(x1*(y2-y3) + x2*(y3-y1) + x3*(y1-y2))
    if abs(d) < 1e-12:
        return None
    ux = ((x1**2+y1**2)*(y2-y3) + (x2**2+y2**2)*(y3-y1) + (x3**2+y3**2)*(y1-y2)) / d
    uy = ((x1**2+y1**2)*(x3-x2) + (x2**2+y2**2)*(x1-x3) + (x3**2+y3**2)*(x2-x1)) / d
    return (ux, uy)

def foot_point_to_line(P, A, B):
    # foot of perpendicular from point P to line AB
    # line defined by points A, B
    # compute projection
    (x0,y0) = P
    (x1,y1) = A
    (x2,y2) = B
    # vector AB
    dx = x2 - x1
    dy = y2 - y1
    # if AB is degenerate
    if abs(dx) < 1e-12 and abs(dy) < 1e-12:
        return A
    # parameter t of projection of P onto line AB:
    t = ((x0 - x1)*dx + (y0 - y1)*dy) / (dx*dx + dy*dy)
    # foot coordinates:
    xf = x1 + t*dx
    yf = y1 + t*dy
    return (xf, yf)

def reflect_point_across_line(P, A, B):
    # reflect point P across line AB
    # Compute foot H of P onto AB, then reflected point = 2H - P
    H = foot_point_to_line(P, A, B)
    return (2*H[0] - P[0], 2*H[1] - P[1])

    # Compute Ux, Uy
    x1sq = x1*x1 + y1*y1

def circle_intersections(c1, r1, c2, r2):
    x1, y1 = c1

def line_projection(P, A, B):
    # project point P onto line through A,B
    # returns foot point
    # vector from A to B
    vx = B[0] - A[0]
    vy = B[1] - A[1]
    # vector from A to P
    wx = P[0] - A[0]
    wy = P[1] - A[1]
    # projection scalar t = (w·v)/|v|^2
    denom = vx*vx + vy*vy
    if denom == 0:
        return A
    t = (wx*vx + wy*vy) / denom
    proj = (A[0] + t*vx, A[1] + t*vy)
    return proj

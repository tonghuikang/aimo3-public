    # compute fibonacci numbers up to n+2
    F = [0,1]
    for i in range(2, n+3):
        F.append(F[-1] + F[-2])
    BD = sp.Rational(F[n])
    CD = sp.Rational(F[n+1])
    a = BD + CD
    Fn_minus1 = sp.Rational(F[n-1])
    c = sp.Rational(b_val) - Fn_minus1
    # ensure c positive
    # coordinates of A:
    xA = (a**2 + c**2 - sp.Rational(b_val)**2) / (2*a)
    yA_sq = c**2 - xA**2
    if yA_sq <= 0:
        return None
    yA = sp.sqrt(yA_sq)
    # points
    B = sp.Point(0,0)
    C = sp.Point(a,0)
    A = sp.Point(xA, yA)
    # incircle radius r = sqrt((b - CD) * BD * CD / (b + BD))
    # but we only need D, N, etc.
    # contact points:
    # F = B + (BD / c) * (A - B)
    F_pt = sp.Point(BD * xA / c, BD * yA / c)
    # E = C + (CD / b) * (A - C)
    E_pt = sp.Point(a + CD * (xA - a) / sp.Rational(b_val), CD * yA / sp.Rational(b_val))
    # D = (BD, 0)
    D_pt = sp.Point(BD, 0)
    # N foot from D onto EF
    line_EF = sp.Line(E_pt, F_pt)
    N_pt = line_EF.projection(D_pt)
    # circles
    circ_ABC = sp.Circle(A, B, C)
    circ_AFE = sp.Circle(A, F_pt, E_pt)
    # intersection
    inters = circ_ABC.intersection(circ_AFE)
    # find K not equal to A
    K_pt = None
    for P in inters:
        if not P.equals(A):
            K_pt = P
    if K_pt is None:
        return None
    # K' reflection of K across line EF
    K_ref = K_pt.reflect(sp.Line(E_pt, F_pt))
    # circle through B, K, N
    try:
        circ_BKN = sp.Circle(B, K_pt, N_pt)

def circumcenter(p1, p2, p3):
    # p1,p2,p3 are tuples (x,y)
    (x1,y1) = p1
    (x2,y2) = p2
    (x3,y3) = p3
    # Compute circumcenter using perpendicular bisector intersection
    d = 2*(x1*(y2 - y3) + x2*(y3 - y1) + x3*(y1 - y2))
    if abs(d) < 1e-12:
        return None
    ux = ((x1**2 + y1**2)*(y2 - y3) + (x2**2 + y2**2)*(y3 - y1) + (x3**2 + y3**2)*(y1 - y2))/d
    uy = ((x1**2 + y1**2)*(x3 - x2) + (x2**2 + y2**2)*(x1 - x3) + (x3**2 + y3**2)*(x2 - x1))/d
    return (ux, uy)

def circumradius(center, p):
    return math.hypot(center[0]-p[0], center[1]-p[1])

def circle_intersections(c1, r1, c2, r2):
    # returns list of intersection points (each tuple)
    (x0,y0) = c1
    (x1,y1) = c2
    dx = x1 - x0
    dy = y1 - y0
    d = math.hypot(dx, dy)
    if d > r1 + r2 + 1e-12 or d < abs(r1 - r2) - 1e-12:
        return []  # no intersection
    # a = (r1^2 - r2^2 + d^2) / (2d)
    a = (r1**2 - r2**2 + d**2) / (2*d)
    # h = sqrt(r1^2 - a^2)
    # ensure non-negative due to floating errors
    h_sq = max(r1**2 - a**2, 0.0)
    h = math.sqrt(h_sq)
    # point P2 = c1 + a*(c2-c1)/d
    xm = x0 + a*dx/d
    ym = y0 + a*dy/d
    # intersection points:
    # (xm +/- h* (-dy/d), ym +/- h*(dx/d))
    rx = -dy * (h/d)
    ry = dx * (h/d)
    p_int1 = (xm + rx, ym + ry)
    p_int2 = (xm - rx, ym - ry)
    # If h ~ 0, both points same
    if h < 1e-12:
        return [p_int1]
    else:
        return [p_int1, p_int2]

def reflect_point_across_line(P, L1, L2):
    # reflect point P across line defined by points L1 and L2
    # line in vector form: L1 + t * v, where v = L2-L1
    (x0,y0) = P
    (x1,y1) = L1
    (x2,y2) = L2
    vx = x2 - x1
    vy = y2 - y1
    # line normal vector n = (vy, -vx) (perpendicular)
    # projection of P onto line:
    # compute t = ((P-L1)·v) / (v·v)
    t = ((x0 - x1)*vx + (y0 - y1)*vy) / (vx*vx + vy*vy)
    proj_x = x1 + t*vx
    proj_y = y1 + t*vy
    # reflection: Q = 2*proj - P
    qx = 2*proj_x - x0
    qy = 2*proj_y - y0
    return (qx, qy)

def foot_of_perpendicular(P, L1, L2):

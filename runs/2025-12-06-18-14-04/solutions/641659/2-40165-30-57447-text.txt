def circumcenter(P1, P2, P3):
    (x1, y1), (x2, y2), (x3, y3) = P1, P2, P3
    d = 2*(x1*(y2 - y3) + x2*(y3 - y1) + x3*(y1 - y2))
    if abs(d) < 1e-12:
        raise ValueError("Degenerate")
    ux = ((x1**2 + y1**2)*(y2 - y3) + (x2**2 + y2**2)*(y3 - y1) + (x3**2 + y3**2)*(y1 - y2)) / d
    uy = ((x1**2 + y1**2)*(x3 - x2) + (x2**2 + y2**2)*(x1 - x3) + (x3**2 + y3**2)*(x2 - x1)) / d
    R = math.hypot(ux - x1, uy - y1)
    return (ux, uy), R

def intersect_circles(O1, R1, O2, R2):
    (x0, y0) = O1
    (x1, y1) = O2
    dx = x1 - x0
    dy = y1 - y0
    d = math.hypot(dx, dy)
    if d > R1 + R2 + 1e-12 or d < abs(R1 - R2) - 1e-12:
        raise ValueError("No intersect")
    # ex unit vector
    ex = (dx / d, dy / d)
    # distance from O1 to point along line
    x = (d**2 + R1**2 - R2**2) / (2*d)
    # height from line
    h_sq = R1**2 - x**2
    h = math.sqrt(max(h_sq, 0.0))
    # point P
    px = x0 + x * ex[0]
    py = y0 + x * ex[1]
    # perpendicular vector ey
    ey = (-ex[1], ex[0])
    p1 = (px + h*ey[0], py + h*ey[1])
    p2 = (px - h*ey[0], py - h*ey[1])
    return p1, p2

def project_point(P, P0, d):
    # project P onto line through P0 with direction d
    v0 = (P[0] - P0[0], P[1] - P0[1])
    dot = v0[0]*d[0] + v0[1]*d[1]
    d_norm_sq = d[0]**2 + d[1]**2
    t = dot / d_norm_sq
    proj = (P0[0] + t*d[0], P0[1] + t*d[1])
    return proj

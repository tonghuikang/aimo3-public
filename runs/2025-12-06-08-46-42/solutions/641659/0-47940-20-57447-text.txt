def f_det(s, n):
    # compute determinant for given s (float) and n
    a = mp.mpf(sp.fibonacci(n+2))
    d = mp.mpf(sp.fibonacci(n))
    h = a - d
    # s must be > a
    if s <= a:
        return mp.mpf('inf')
    # geometry
    r = mp.sqrt(d*h*(s - a) / s)
    mB = 2*d*r/(d**2 - r**2)
    mC = -2*h*r/(h**2 - r**2)
    denom = d*(h**2 - r**2) + h*(d**2 - r**2)
    xA = h*a*(d**2 - r**2) / denom
    yA = 2*d*h*a*r / denom
    B = (mp.mpf(0), mp.mpf(0))
    C = (a, mp.mpf(0))
    A = (xA, yA)
    c_len = mp.sqrt(xA**2 + yA**2)
    b_len = mp.sqrt((xA - a)**2 + yA**2)
    # F and E
    F = (d/c_len * xA, d/c_len * yA)
    E = (a + (xA - a)*(h/b_len), yA * (h/b_len))
    D = (d, mp.mpf(0))
    # N foot from D onto line EF
    # projection onto EF
    vx = F[0] - E[0]
    vy = F[1] - E[1]
    wx = D[0] - E[0]
    wy = D[1] - E[1]
    t_proj = (wx*vx + wy*vy) / (vx*vx + vy*vy)
    N = (E[0] + t_proj*vx, E[1] + t_proj*vy)
    # circumcircles intersection (K)
    # circumcenter functions
    def circumcenter(P1,P2,P3):
        x1,y1=P1; x2,y2=P2; x3,y3=P3
        dval = 2*(x1*(y2-y3)+x2*(y3-y1)+x3*(y1-y2))
        ux = ((x1**2+y1**2)*(y2-y3)+(x2**2+y2**2)*(y3-y1)+(x3**2+y3**2)*(y1-y2))/dval
        uy = ((x1**2+y1**2)*(x3-x2)+(x2**2+y2**2)*(x1-x3)+(x3**2+y3**2)*(x2-x1))/dval
        return (ux, uy)
    O1 = circumcenter(A,B,C)
    R1 = mp.sqrt((A[0]-O1[0])**2 + (A[1]-O1[1])**2)
    O2 = circumcenter(A,F,E)
    R2 = mp.sqrt((A[0]-O2[0])**2 + (A[1]-O2[1])**2)
    # circle intersection

                   [row(Kprime)[i] for i in range(4)]])

    return det

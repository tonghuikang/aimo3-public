def convolve(a,b):
    # convolution of two coefficient lists a,b (list of ints)
    n = len(a)
    m = len(b)
    res = [0]*(n+m-1)
    for i, ai in enumerate(a):
        if ai == 0: continue
        for j, bj in enumerate(b):
            if bj == 0: continue
            res[i+j] += ai*bj
    return res

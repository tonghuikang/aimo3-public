def product_terms(alpha, beta):
    # alpha: dict exponent->coeff (int)
    # beta: dict exponent->coeff (int)
    res = {}
    for e1,c1 in alpha.items():
        for e2,c2 in beta.items():
            e = e1 + e2
            res[e] = res.get(e, 0) + c1*c2
    # remove zeros
    res = {e:c for e,c in res.items() if c != 0}
    return res

def is_two_monomials_one_each(prod):
    # check if dict has exactly two entries, each coeff = 1
    if len(prod) != 2:
        return False
    return all(c == 1 for c in prod.values())

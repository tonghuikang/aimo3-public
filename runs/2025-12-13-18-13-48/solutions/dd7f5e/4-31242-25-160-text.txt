# define cyclotomic polynomial coefficients as dict: exponent->coeff
def poly_to_dict(poly):
    # sympy poly expand
    expr = sp.expand(poly)
    # get terms
    d = {}
    for term in expr.as_ordered_terms():
        coeff = term.as_coeff_mul(sp.Symbol('x'))[0]
        # find exponent
        # term could be coeff*x**exp
        # separate coefficient and monomial
        monom = term/coeff
        if monom == 1:
            exp = 0
        else:
            exp = sp.degree(monom, sp.Symbol('x'))
        d[exp] = int(coeff)
    return d

# Helper to multiply polynomials (dict)
def poly_mul(p, q):
    r = {}
    for i,a in p.items():
        for j,b in q.items():
            r[i+j] = r.get(i+j,0) + a*b
    # remove zero coefficients
    return {e:c for e,c in r.items() if c!=0}

def poly_add(p,q):
    r = p.copy()
    for e,c in q.items():
        r[e] = r.get(e,0)+c
        if r[e]==0:
            del r[e]
    return r

def poly_pow(p,exp):
    r = {0:1}
    for _ in range(exp):
        r = poly_mul(r,p)
    return r

def subsets_with_degree_limit(items, limit):
    # items is list of (n, deg)
    # return list of subsets (as list of ns) meeting sum degree <= limit
    valid = []
    n = len(items)
    for mask in range(1<<n):
        total_deg = 0
        subset = []
        for i in range(n):
            if mask & (1<<i):
                total_deg += items[i][1]
                subset.append(items[i][0])
        if total_deg <= limit:
            valid.append((subset,total_deg))
    return valid

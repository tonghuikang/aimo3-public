    a_val = BD + CD
    s_min = a_val + 1e-6
    s_max = a_val * s_max_factor + 10  # ensure far enough
    # sample
    xs = [s_min + (s_max - s_min) * i / steps for i in range(steps+1)]
    prev_val = cyclic_error(BD, CD, xs[0])
    # if prev_val is None, maybe degenerate at start - skip
    for i in range(1, len(xs)):
        val = cyclic_error(BD, CD, xs[i])
        if val is None or prev_val is None:
            # skip this interval
            prev_val = val
            continue
        if prev_val == 0:
            return xs[i-1]
        if val == 0:
            return xs[i]
        # detect sign change
        if prev_val * val < 0:
            # root between xs[i-1] and xs[i]; find by bisection
            a = xs[i-1]
            b = xs[i]
            # refine

                if fmid is None:

                    a = mid
                    continue

    return None

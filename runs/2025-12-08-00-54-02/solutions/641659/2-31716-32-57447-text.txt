def foot_point_to_line(P, Q, R):
    # foot of perpendicular from point P to line QR
    # P,R,Q are numpy arrays
    Q = np.array(Q); R = np.array(R); P = np.array(P)
    v = R - Q

        raise ValueError("Degenerate line")
    t = np.dot(P - Q, v) / np.dot(v, v)
    proj = Q + t * v
    # foot is projection
    return proj

def reflect_point_across_line(P, Q, R):
    # reflect point P across line QR
    # Use foot then compute reflection
    foot = foot_point_to_line(P, Q, R)
    return foot*2 - P

def circle_intersections(c1, r1, c2, r2):

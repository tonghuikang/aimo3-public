    # => (t^2 * (v·v) - 2 t (v·K) + K·K) = t^2
    # Since v is unit vector, v·v = 1
    # => t^2 - 2 t (v·K) + K·K - t^2 = 0 => - 2 t (v·K) + K·K = 0 => t = K·K / (2 v·K)
    # Wait, sign? Let's compute carefully.
    # O_T = t v (since B is origin)
    # Condition: distance from O_T to K equals |t|.
    # |t v - K|^2 = t^2
    # Expand: t^2 v·v - 2 t v·K + K·K = t^2
    # Since v·v=1: t^2 - 2 t (v·K) + K·K = t^2 => cancel t^2: -2 t (v·K) + K·K = 0 => t = K·K / (2 v·K)
    # So t = (K·K) / (2 (v·K)). This works if v·K > 0; else t negative (but radius absolute). Use absolute later.
    # Compute v·K.
    v_x = -y_N
    v_y = x_N
    # Normalize v
    norm_v = mp.sqrt(v_x**2 + v_y**2)
    v_xn = v_x / norm_v
    v_yn = v_y / norm_v
    # dot product v·K
    dot_vK = v_xn * x_K + v_yn * y_K
    if abs(dot_vK) < 1e-12:
        # degenerate case maybe
        return None
    t = (x_K**2 + y_K**2) / (2 * dot_vK)
    # Center coordinates
    O_Tx = t * v_xn
    O_Ty = t * v_yn
    # radius
    R_T = mp.sqrt(O_Tx**2 + O_Ty**2)  # should be |t|
    # Intersection of this circle with line BC (y=0) besides B (0,0)
    # Solve (x - O_Tx)^2 + (0 - O_Ty)^2 = R_T^2
    # => (x - O_Tx)^2 = R_T^2 - O_Ty**2 = O_Tx**2 + O_Ty**2 - O_Ty**2 = O_Tx**2
    # => (x - O_Tx)^2 = O_Tx**2 => x - O_Tx = +/- O_Tx => x = O_Tx +/- O_Tx => x = 0 (B) or x = 2*O_Tx
    # So T_x = 2 * O_Tx (if O_Tx != 0). Indeed the other intersection is at x = 2*O_Tx.
    T_x = 2 * O_Tx
    # Compute distances
    CT = a - T_x
    BT = T_x
    # compute X
    X = (CT * NB) / (BT * NE)
    return X

def max_X_for_n(n, s_vals):
    p = fibonacci(n)
    q = fibonacci(n+1)
    # we need s > a = p+q
    a = p+q
    max_val = -1
    max_s = None
    for s in s_vals:
        if s <= a:
            continue
        X = compute_X(p,q,s)
        if X is None:
            continue
        if X > max_val:
            max_val = X
            max_s = s
    return max_val, max_s

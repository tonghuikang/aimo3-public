    A = sp.Point(xA, yA)
    return A, B, C, a, b, c

def incircle_contact_points(A, B, C, a, b, c):
    # Compute semiperimeter s (we have it but recompute)
    s_val = (a + b + c) / 2
    # incircle center I
    I = sp.Point((a*A.x + b*B.x + c*C.x) / (a+b+c), (a*A.y + b*B.y + c*C.y) / (a+b+c))
    r = a * A.y / (2 * s_val)  # incircle radius from earlier derived (or compute distance from I to BC)
    # D foot to BC
    D = sp.Point(I.x, 0)
    # contact point on AB: foot of perpendicular from I to AB
    # AB line from B to A
    AB_vec = sp.Matrix([A.x, A.y])
    # projection of I onto AB
    t = (I.dot(AB_vec)) / (A.x**2 + A.y**2)
    F = sp.Point(t*A.x, t*A.y)
    # contact point on AC: foot from I onto AC
    AC_vec = sp.Matrix([A.x - C.x, A.y - C.y])
    IC = sp.Matrix([I.x - C.x, I.y - C.y])
    t2 = (IC.dot(AC_vec)) / (AC_vec.dot(AC_vec))
    E = sp.Point(C.x + t2*AC_vec[0], C.y + t2*AC_vec[1])
    return I, D, E, F, r

def intersect_circles(c1, c2):
    # Return intersection points of circles c1, c2 (list of Points)
    return list(c1.intersection(c2))

def foot_of_perpendicular(P, line):
    # Return foot of perpendicular from point P to line (sp.Line)
    return line.projection(P)

def reflect_point_across_line(P, line):
    # return reflection of point P across line
    # formula: reflect = 2*foot - P
    foot = foot_of_perpendicular(P, line)
    return sp.Point(2*foot.x - P.x, 2*foot.y - P.y)

def compute_K(A, B, C, E, F):
    # Circle through A, E, F
    circ1 = sp.Circle(A, E, F)
    # Circumcircle of ABC
    circ2 = sp.Circle(A, B, C)
    # intersection points
    pts = list(circ1.intersection(circ2))
    # Should be two points: A and K
    # Identify K (not equal to A)
    K = None
    for pt in pts:
        if not pt.equals(A):
            K = pt
    if K is None:
        raise ValueError("K not found")
    return K

def compute_T(B, K, N, line_BC):
    # Circle through B and K, tangent to BN at B.
    # The condition is that line BN is tangent to the circle at B.
    # For a circle passing through B and K, the center O_T lies on the perpendicular bisector of BK.
    # Also radius is distance from center to B.
    # Tangent at B is perpendicular to radius at B, so line from center to B is perpendicular to BN.
    # So center O_T is intersection of perpendicular bisector of BK and line through B perpendicular to BN.
    BK_mid = sp.Point((B.x + K.x)/2, (B.y + K.y)/2)
    # direction of BK
    # perpendicular bisector: line through BK_mid with direction perpendicular to BK (i.e., slope negative reciprocal)
    # Let's compute line objects:
    line_perp_bis = sp.Line(BK_mid, BK_mid + sp.Point(K.y - B.y, B.x - K.x))  # vector orthogonal to BK
    # line through B perpendicular to BN: BN direction is N - B
    BN_dir = sp.Point(N.x - B.x, N.y - B.y)
    # Perpendicular direction to BN is rotate 90 deg
    perp_to_BN = sp.Point(BN_dir.y, -BN_dir.x)  # (dy, -dx)
    line_perp_BN = sp.Line(B, B + perp_to_BN)
    # Intersection gives center O_T
    O_T = line_perp_bis.intersection(line_perp_BN)[0]
    # radius
    R_T = O_T.distance(B)
    # Circle O_T radius R_T intersects BC at B and T. Find second intersection with line BC.
    # Solve intersection of line BC with circle O_T
    line_BC = sp.Line(sp.Point(0,0), sp.Point(line_BC.p2.x, line_BC.p2.y))  # line from B to C
    # Actually BC is line through B and C.
    # Intersection:
    inter_pts = sp.Circle(O_T, R_T).intersection(line_BC)
    # Should be B and another point T.
    T = None
    for pt in inter_pts:
        if not pt.equals(B):
            T = pt
    if T is None:
        raise ValueError("T not found")
    return T

def compute_ratio(C, T, N, B, NE):
    # compute CT * NB / (BT * NE)
    # CT distance between C and T
    CT = C.distance(T)
    NB = N.distance(B)
    BT = B.distance(T)
    # NE distance between N and E (but we need NE length)
    # NE distance computed from N to point E
    # NE = distance N to E
    # Already given NE as a number? We'll compute.
    return CT * NB / (BT * NE)

def circle_center(p1, p2, p3):
    # p1,p2,p3 are arrays length 2
    # Solve linear system for center.
    # (x - x1)^2 + (y - y1)^2 = (x - x2)^2 + (y - y2)^2
    # => -2x1*x -2y1*y + x1^2 + y1^2 = -2x2*x -2y2*y + x2^2 + y2^2
    # => 2(x2 - x1)*x + 2(y2 - y1)*y = x2^2 + y2^2 - x1^2 - y1^2
    # Similarly for p1,p3.
    x1,y1 = p1
    x2,y2 = p2
    x3,y3 = p3
    A = np.array([[2*(x2 - x1), 2*(y2 - y1)],
                  [2*(x3 - x1), 2*(y3 - y1)]], dtype=float)
    b = np.array([x2**2 + y2**2 - x1**2 - y1**2,
                  x3**2 + y3**2 - x1**2 - y1**2], dtype=float)
    # solve for center
    center = np.linalg.solve(A, b)
    return center

def circle_radius(center, p):
    return np.linalg.norm(center - p)

def intersect_circles(c1_center, r1, c2_center, r2):
    # Return intersection points (0,1,2)
    # Compute distance between centers
    d = np.linalg.norm(c2_center - c1_center)
    # no intersection if d > r1+r2 or d < |r1-r2|
    if d > r1 + r2 + 1e-12 or d < abs(r1 - r2) - 1e-12:
        return []
    # find midpoint line
    # ex = (c2 - c1)/d
    ex = (c2_center - c1_center) / d
    # a = (r1^2 - r2^2 + d^2) / (2d)
    a = (r1**2 - r2**2 + d**2) / (2*d)
    # height from line to intersection points
    h_sq = r1**2 - a**2
    if h_sq < -1e-12:
        return []
    h = np.sqrt(max(h_sq,0))
    # point P2 = c1 + a*ex
    p2 = c1_center + a * ex
    # intersection points
    # offset = h * ( -ex[1], ex[0] )
    offset = np.array([-ex[1], ex[0]]) * h
    inter1 = p2 + offset
    inter2 = p2 - offset
    if np.allclose(inter1, inter2):
        return [inter1]
    else:
        return [inter1, inter2]

def foot_of_perp(P, line_pt1, line_pt2):
    # foot of perpendicular from point P to line through line_pt1 and line_pt2
    # vector from line_pt1 to line_pt2
    v = line_pt2 - line_pt1
    w = P - line_pt1
    t = np.dot(w, v) / np.dot(v, v)
    proj = line_pt1 + t * v
    # foot of perp is proj (since we project onto line)
    # Actually foot of perpendicular is projection onto line. Yes.
    return proj

def reflect_point(P, line_pt1, line_pt2):
    # reflect P across line through line_pt1, line_pt2
    foot = foot_of_perp(P, line_pt1, line_pt2)
    return 2*foot - P

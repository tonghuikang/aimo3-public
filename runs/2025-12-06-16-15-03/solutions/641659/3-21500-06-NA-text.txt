    if n == 0: return 0

    # reflect point P across line through A and B
    # compute unit normal vector
    d = sp.Matrix([B.x - A.x, B.y - A.y])
    # normal vector (perpendicular)
    n = sp.Matrix([d[1], -d[0]])  # one perpendicular direction

    # reflection
    Pprime = sp.Matrix([P.x, P.y]) - 2 * proj * n_unit
    return sp.Point(sp.simplify(Pprime[0]), sp.simplify(Pprime[1]))

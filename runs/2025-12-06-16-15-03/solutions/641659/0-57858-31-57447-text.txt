    # Use high precision mp (maybe 50 digits)

    # compute Fibonacci numbers (as mp.mpf)
    # Use python integers for exact but convert

    Fn1 = mp.mpf(fib[n+1])
    Fn2 = mp.mpf(fib[n+2])

    c_len = u + Fn
    b_len = u + Fn1
    # incircle radius
    r = mp.sqrt(u * Fn * Fn1 / (a + u))
    tan_phi2 = r / Fn
    phi = 2 * mp.atan(tan_phi2)
    # points
    B = (mp.mpf('0'), mp.mpf('0'))
    C = (a, mp.mpf('0'))
    D = (Fn, mp.mpf('0'))
    F_pt = (Fn * mp.cos(phi), Fn * mp.sin(phi))
    A_pt = (c_len * mp.cos(phi), c_len * mp.sin(phi))

    AC_y = A_pt[1] - C[1]

            C[1] + (w / b_len) * AC_y)

    EF_y = E_pt[1] - F_pt[1]

    OK = mp.sqrt((O_pt[0]-K_pt[0])**2 + (O_pt[1]-K_pt[1])**2)
    return OB - OK

### 1.  Geometry and the cyclicity condition  

For a triangle \(ABC\) with incircle contact lengths  

\[
BD=F_n,\qquad CD=F_{n+1},
\]

let  

\[
a=BC=F_{n+2},\qquad 
b=CA=F_{n+1}+u,\qquad 
c=AB=F_{n}+u,\qquad 
u=s-F_{n+2}>0 .
\]

All other points (\(F,E,N,K,K',T\)) are functions of the single parameter  

\[
t=\frac{u}{F_{n+2}}\;(>0).
\]

The condition that the quadrilateral \(B,K,N,K'\) is cyclic is equivalent to the single scalar equation  

\[
|OB|^{2}=|OK|^{2},
\]

where  

* \(O\) is the intersection of the line \(EF\) (the perpendicular bisector of \(KK'\))  
  with the perpendicular bisector of the segment \(BN\);
* \(K\) is the second intersection of the circles \((A,F,E)\) and \((A,B,C)\).

Thus for each \(n\) there is a unique value \(t_n\) solving the above equation.

    # line EF direction
    dEF = (EF[0],EF[1])
    # unit direction
    normEF = mp.sqrt(dEF[0]**2 + dEF[1]**2)
    uEF = (dEF[0]/normEF, dEF[1]/normEF)

    # projection onto EF

    Kprime = (K[0]-2*perp[0], K[1]-2*perp[1])
    # O = intersection of EF and perp bisector of BN
    MBN = ((B[0]+N_pt[0])/2, (B[1]+N_pt[1])/2)
    BN = (N_pt[0]-B[0], N_pt[1]-B[1])
    perpBN = (-BN[1], BN[0])
    # solve F + s*EF = MBN + t*perpBN
    A_mat = [[EF[0], -perpBN[0]],
             [EF[1], -perpBN[1]]]
    b_vec = [MBN[0]-F_pt[0], MBN[1]-F_pt[1]]
    det = A_mat[0][0]*A_mat[1][1] - A_mat[0][1]*A_mat[1][0]
    s_val = (b_vec[0]*A_mat[1][1] - b_vec[1]*A_mat[0][1]) / det
    O_pt = (F_pt[0] + s_val*EF[0], F_pt[1] + s_val*EF[1])
    # distances
    OB = mp.sqrt(O_pt[0]**2+O_pt[1]**2)
    OK = mp.sqrt((O_pt[0]-K[0])**2+(O_pt[1]-K[1])**2)
    ON = mp.sqrt((O_pt[0]-N_pt[0])**2+(O_pt[1]-N_pt[1])**2)
    OKp = mp.sqrt((O_pt[0]-Kprime[0])**2+(O_pt[1]-Kprime[1])**2)
    return float(OB), float(OK), float(ON), float(OKp)

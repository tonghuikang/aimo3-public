def circumcenter(P1, P2, P3):
    x1,y1 = P1
    x2,y2 = P2
    x3,y3 = P3
    # Solve linear system
    A = [[2*(x1 - x2), 2*(y1 - y2)],
         [2*(x1 - x3), 2*(y1 - y3)]]
    B = [x1**2 - x2**2 + y1**2 - y2**2,
         x1**2 - x3**2 + y1**2 - y3**2]
    # Solve using determinant
    det = A[0][0]*A[1][1] - A[0][1]*A[1][0]
    if abs(det) < 1e-12:
        raise ValueError("Points are collinear")
    ox = (B[0]*A[1][1] - B[1]*A[0][1]) / det
    oy = (A[0][0]*B[1] - A[1][0]*B[0]) / det
    R = math.hypot(ox - x1, oy - y1)
    return (ox, oy, R)

def intersection_line_circles(O1, R1, O2, R2, known_point):
    # Returns the two intersection points of circles (O1,R1) and (O2,R2).
    # Compute using geometry
    x1,y1 = O1
    x2,y2 = O2
    d = math.hypot(x2 - x1, y2 - y1)
    if d == 0:
        raise ValueError("Same center")
    # Check solvable
    # a = (R1^2 - R2^2 + d^2) / (2d)
    a = (R1**2 - R2**2 + d**2) / (2*d)
    # height from the line connecting centers
    h_sq = R1**2 - a**2
    if h_sq < -1e-12:
        # no intersection
        return []
    h = math.sqrt(max(h_sq,0))

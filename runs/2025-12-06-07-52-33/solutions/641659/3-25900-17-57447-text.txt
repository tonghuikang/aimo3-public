def fibonacci(n):
    # using fast doubling for exact integer
    def fib_pair(k):
        if k == 0:
            return (0, 1)
        else:
            a, b = fib_pair(k >> 1)
            c = a * ( (b << 1) - a )
            d = a*a + b*b
            if k & 1:
                return (d, c + d)
            else:
                return (c, d)
    return mp.mpf(fib_pair(n)[0])

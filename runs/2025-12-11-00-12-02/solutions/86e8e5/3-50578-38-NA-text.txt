    # return minimal lcm of distinct positive ints a,b,c with a+b+c=n
    # if no triple (should be always possible for n>=6)
    import math
    best = None
    for a in range(1,n-2):
        for b in range(a+1,n-a):
            c = n-a-b
            if c<=b: continue # ensure distinct and increasing a<b<c
            # distinct positive
            # compute lcm
            l = abs(a*b)//math.gcd(a,b)
            l = abs(l*c)//math.gcd(l,c)
            if best is None or l < best:
                best = l
    return best

def compute_f_values(limit):
    return [f_bruteforce(n) for n in range(limit+1)]

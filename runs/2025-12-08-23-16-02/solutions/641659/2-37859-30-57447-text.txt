def circle_intersections(O1, R1, O2, R2):
    # returns up to two points
    O1 = np.array(O1); O2 = np.array(O2)
    d = np.linalg.norm(O2 - O1)
    if d == 0:
        return []
    # check for intersection
    # using law of cos
    # ensure |R1 - R2| <= d <= R1 + R2
    if d > R1 + R2 + 1e-12 or d < abs(R1 - R2) - 1e-12:
        return []
    # ex vector
    ex = (O2 - O1) / d
    # distance from O1 to line of intersection points along ex
    x = (R1**2 - R2**2 + d**2) / (2 * d)
    # height from line
    h2 = R1**2 - x**2
    if h2 < 0:
        h2 = 0
    h = math.sqrt(h2)
    p = O1 + x * ex
    ey = np.array([-ex[1], ex[0]])  # perpendicular unit
    pt1 = p + h * ey
    pt2 = p - h * ey
    return [pt1, pt2]

def circumcenter(P, Q, R):
    P = np.array(P); Q = np.array(Q); R = np.array(R)
    # Solve linear equations: (Q-P)·O = (|Q|^2 - |P|^2)/2 etc.
    A = np.vstack([Q-P, R-P])
    b = np.array([(np.dot(Q,Q) - np.dot(P,P)) / 2.0, (np.dot(R,R) - np.dot(P,P)) / 2.0])
    # Solve
    try:
        O = np.linalg.solve(A, b)
    except np.linalg.LinAlgError:
        # Singular (points collinear) - shouldn't happen
        O = np.linalg.lstsq(A, b, rcond=None)[0]
    return O

def incircle_geometry(Fn, Fn1):
    # Returns symbolic expressions for coordinates etc given n parameters
    # We'll keep variables xA, yA, r.
    a = Fn + Fn1  # BC length = F_{n+2} = Fn+Fn1
    x0 = Fn  # BD = Fn, incircle touch point D at (Fn,0)
    # variables:
    xA, yA, r = sp.symbols('xA yA r', real=True)
    # define points:
    B = sp.Point(0,0)
    C = sp.Point(a,0)
    I = sp.Point(x0, r)
    # distance from I to line AB equals r:
    # line AB: passes through B and A
    line_AB = sp.Line(sp.Point(0,0), sp.Point(xA,yA))
    # distance:
    dist_AB = line_AB.distance(I)
    # distance from I to line AC equals r:
    line_AC = sp.Line(sp.Point(a,0), sp.Point(xA,yA))
    dist_AC = line_AC.distance(I)
    # equations:
    eq1 = sp.Eq(dist_AB, r)
    eq2 = sp.Eq(dist_AC, r)
    return eq1, eq2, B, C, I, xA, yA, r

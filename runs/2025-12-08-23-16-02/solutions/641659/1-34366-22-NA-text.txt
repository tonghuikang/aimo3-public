def foot_of_perpendicular(P, Q, R):
    # foot from point P onto line QR
    # P: point tuple (x0,y0)
    # line through Q (x1,y1) and R (x2,y2)
    x0,y0 = P
    x1,y1 = Q
    x2,y2 = R
    # vector v = R - Q
    vx = x2 - x1
    vy = y2 - y1
    # projection of (P-Q) onto v
    t = ((x0 - x1)*vx + (y0 - y1)*vy) / (vx*vx + vy*vy)
    # foot point = Q + t*v
    xf = x1 + t*vx
    yf = y1 + t*vy
    return (xf, yf)

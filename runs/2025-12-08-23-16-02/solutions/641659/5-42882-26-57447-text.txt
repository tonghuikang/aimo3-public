def solve_for_n(n):
    # compute Fibonacci numbers up to n+2
    fib = [0,1]
    for i in range(2, n+3):
        fib.append(fib[-1] + fib[-2])
    Fn = fib[n]
    Fnp1 = fib[n+1]
    Fnp2 = fib[n+2]
    a = Fnp2
    # define s as variable
    s = sp.symbols('s', positive=True)
    # ensure s > a
    # compute b,c,P
    b = s - Fn
    c = s - Fnp1
    P = 2*s
    # x_A, y_A
    xA = (a**2 - b**2 + c**2) / (2*a)
    # yA^2 = c^2 - xA^2
    yA_sq = c**2 - xA**2
    # define yA positive sqrt
    yA = sp.sqrt(yA_sq)
    # incircle radius r = a*yA / P
    r = a * yA / P
    # points
    D = sp.Point(Fn, 0)
    B = sp.Point(0,0)
    C = sp.Point(a,0)
    A = sp.Point(xA, yA)
    # E and F
    # compute unit vectors
    # F = B + (BF/c)*(A-B) where BF = Fn
    Fx = Fn * xA / c
    Fy = Fn * yA / c
    F_pt = sp.Point(Fx, Fy)
    # E = C + (Fnp1/b)*(A-C)
    Ex = a + (Fnp1)*(xA - a) / b
    Ey = (Fnp1)*y_A / b
    E_pt = sp.Point(Ex, Ey)
    # foot N from D onto line EF
    line_EF = sp.Line(E_pt, F_pt)

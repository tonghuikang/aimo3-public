def incircle_and_contacts(A,B,C):
    # sides
    a = B.distance(C)
    b = C.distance(A)
    c = A.distance(B)
    # semiperimeter

    # incenter coordinates formula: weighted by side lengths
    I = sp.Point((a*A.x + b*B.x + c*C.x)/(a+b+c), (a*A.y + b*B.y + c*C.y)/(a+b+c))
    # incircle radius = area / s
    # but we can compute foot from I to sides as contact points
    # foot on BC (line through B,C)
    line_BC = sp.Line(B,C)
    D = line_BC.projection(I)
    # foot on CA
    line_CA = sp.Line(C,A)
    E = line_CA.projection(I)
    # foot on AB
    line_AB = sp.Line(A,B)
    F = line_AB.projection(I)
    return I, D, E, F, a,b,c,s

    # sympy Circle; might be None if collinear

def second_intersection(circle1,circle2,pointA):
    # find intersection points of two circles; return point that is not pointA
    inter = circle1.intersection(circle2)
    # inter is list of Points (0,1,2)
    # filter out pointA
    for pt in inter:
        if not pt.equals(pointA):
            return pt
    return None

def reflect_point_across_line(P, line):
    # reflect point P across line
    # using sympy: line.projection(P) is foot; then reflected = foot*2 - P
    foot = line.projection(P)
    # foot is sympy Point
    # compute vector
    reflected = sp.Point(2*foot.x - P.x, 2*foot.y - P.y)
    return reflected

def circumcenter(P1,P2,P3):
    # P1, P2, P3 are tuples (x,y)
    (x1,y1),(x2,y2),(x3,y3) = P1,P2,P3
    # Using determinant method
    d = 2*(x1*(y2 - y3) + x2*(y3 - y1) + x3*(y1 - y2))
    if abs(d) < 1e-12:

    ux = ((x1**2 + y1**2)*(y2 - y3) + (x2**2 + y2**2)*(y3 - y1) + (x3**2 + y3**2)*(y1 - y2))/d

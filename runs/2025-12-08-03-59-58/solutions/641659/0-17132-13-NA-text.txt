def cyclic_condition(points):
    B = points['B']
    K = points['K']
    Kp = points['Kp']
    N = points['N']
    # compute oriented angle between lines B-K and B-N
    # angle(BK, BN) and angle(BK', BN)
    # use sympy: angle between vectors
    # v1 = K - B; v2 = N - B
    v1 = sp.Matrix([K.x - B.x, K.y - B.y])
    v2 = sp.Matrix([N.x - B.x, N.y - B.y])
    # compute angle cos? Use cross and dot

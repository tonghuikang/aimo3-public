def foot_of_perpendicular(point, line_p1, line_p2):
    # foot from point to line through line_p1,line_p2
    (x0,y0) = point
    (x1,y1) = line_p1
    (x2,y2) = line_p2
    # compute projection formula
    dx = x2 - x1
    dy = y2 - y1
    if dx==0 and dy==0:
        raise ValueError
    t = ((x0 - x1)*dx + (y0 - y1)*dy) / (dx*dx + dy*dy)
    # foot
    xf = x1 + t*dx
    yf = y1 + t*dy
    return (xf,yf)

def line_through(p1,p2):
    # returns line as (a,b,c) for ax+by+c=0
    (x1,y1) = p1
    (x2,y2) = p2
    a = y1 - y2
    b = x2 - x1
    c = x1*y2 - x2*y1
    return (a,b,c)

def reflect_point_across_line(P, line):
    # reflect point P across line ax+by+c=0
    x0,y0 = P
    a,b,c = line
    # formula: d = (a*x0 + b*y0 + c)/(a**2+b**2)
    d = (a*x0 + b*y0 + c)/(a*a + b*b)
    xr = x0 - 2*a*d
    yr = y0 - 2*b*d
    return (xr, yr)

def intersect_circles(c1, r1, c2, r2):
    # returns intersection points as list of tuples
    (x0,y0) = c1
    (x1,y1) = c2
    d = math.hypot(x1-x0, y1-y0)
    # assume circles intersect
    # distance from c1 to line between intersection points:
    a = (r1**2 - r2**2 + d**2) / (2*d)
    h = math.sqrt(max(r1**2 - a**2, 0))
    # point P2 = c1 + a*(c2-c1)/d
    x2 = x0 + a*(x1-x0)/d
    y2 = y0 + a*(y1-y0)/d
    # intersection points:
    rx = -(y1-y0) * (h/d)
    ry = (x1-x0) * (h/d)
    i1 = (x2 + rx, y2 + ry)
    i2 = (x2 - rx, y2 - ry)
    return i1,i2

def is_concyclic(points):
    # points: list of (x,y) 4 points; check if they are cyclic (determinant of extended matrix zero)
    M = []
    for (x,y) in points:
        M.append([x, y, x*x+y*y, 1])
    det = sp.Matrix(M).det()
    return sp.simplify(det)

def ratio_for_n_s_numeric(n, x):
    # compute ratio and cyclic condition for given n and x
    # Fibonacci numbers
    F = [0,1]
    for i in range(2, n+3):
        F.append(F[-1]+F[-2])
    d = F[n]
    a = F[n+2]
    s = x * a
    if s <= a:
        return None
    r_sq = (s - a) * F[n] * F[n+1] / s
    if r_sq <= 0:
        return None
    r = math.sqrt(r_sq)
    B = (0.0, 0.0)
    C = (a, 0.0)
    I = (d, r)
    # Compute A coordinates:
    Δ = d*d - a*d + r*r
    X_A = (d - a)*(d*d - r*r) / Δ
    Y_A = 2*d*r*(d - a) / Δ
    A = (X_A, Y_A)
    # compute K,F,E
    K, Fpt, E = compute_K_numeric(A, B, C, I)
    # N foot from D to EF
    D = (d, 0.0)
    N = foot_of_perpendicular(D, E, Fpt)
    # T using numeric method
    T = compute_T_numeric(B, C, N, K)
    if T is None:
        return None
    # compute ratio
    CT = math.hypot(C[0]-T[0], C[1]-T[1])
    BT = math.hypot(B[0]-T[0], B[1]-T[1])
    NB = math.hypot(N[0]-B[0], N[1]-B[1])
    NE = math.hypot(N[0]-E[0], N[1]-E[1])
    ratio = (CT * NB) / (BT * NE)
    # cyclic condition determinant
    line_EF = line_through(E, Fpt)
    Kprime = reflect_point_across_line(K, line_EF)
    det = float(is_concyclic([B, N, K, Kprime]))
    return ratio, det

def circumcenter(p1, p2, p3):
    # p1,p2,p3 tuples
    (x1,y1),(x2,y2),(x3,y3)=p1,p2,p3
    d = 2*(x1*(y2 - y3) + x2*(y3 - y1) + x3*(y1 - y2))
    if abs(d) < 1e-12:

    ux = ((x1**2 + y1**2)*(y2 - y3) + (x2**2 + y2**2)*(y3 - y1) + (x3**2 + y3**2)*(y1 - y2))/d
    uy = ((x1**2 + y1**2)*(x3 - x2) + (x2**2 + y2**2)*(x1 - x3) + (x3**2 + y3**2)*(x2 - x1))/d

def circle_intersections(c1, r1, c2, r2):
    # Returns intersection points (list of up to 2)
    (x0,y0) = c1
    (x1,y1) = c2
    dx = x1 - x0
    dy = y1 - y0
    d = math.hypot(dx, dy)
    if d > r1 + r2 + 1e-12 or d < abs(r1 - r2) - 1e-12:
        return []  # no intersection
    # a = (r1^2 - r2^2 + d^2) / (2d)
    a = (r1**2 - r2**2 + d**2) / (2*d)
    # h^2 = r1^2 - a^2
    h2 = r1**2 - a**2
    if h2 < -1e-12:

def foot_of_perpendicular(point, line_pt1, line_pt2):
    # foot of perpendicular from point to line through line_pt1, line_pt2
    x0,y0 = point
    x1,y1 = line_pt1
    x2,y2 = line_pt2
    # direction vector of line
    dx = x2 - x1
    dy = y2 - y1
    # if line is degenerate, return something
    if dx==0 and dy==0:
        return line_pt1
    # projection factor t = ((point - line_pt1)·(dx,dy))/((dx,dy)·(dx,dy))
    t = ((x0 - x1)*dx + (y0 - y1)*dy) / (dx*dx + dy*dy)
    # foot
    xf = x1 + t*dx
    yf = y1 + t*dy
    return (xf, yf)

def reflect_point_across_line(P, line_pt1, line_pt2):

    import math
    a = math.hypot(B[0]-C[0], B[1]-C[1]) # BC opposite A
    b = math.hypot(A[0]-C[0], A[1]-C[1]) # AC opposite B
    c = math.hypot(A[0]-B[0], A[1]-B[1]) # AB opposite C
    # angle A opposite side a: cos A = (b^2 + c^2 - a^2)/(2bc)
    cosA = (b*b + c*c - a*a)/(2*b*c)
    cosB = (a*a + c*c - b*b)/(2*a*c)
    cosC = (a*a + b*b - c*c)/(2*a*b)
    # clamp
    cosA = max(-1, min(1, cosA))
    cosB = max(-1, min(1, cosB))
    cosC = max(-1, min(1, cosC))
    Aang = math.degrees(math.acos(cosA))
    Bang = math.degrees(math.acos(cosB))
    Cang = math.degrees(math.acos(cosC))
    return Aang, Bang, Cang

def is_shifty(alpha_coeffs):
    # alpha_coeffs length 9, coefficients from 0..8
    # Build sympy polynomial
    A = sum(coeff * x**i for i, coeff in enumerate(alpha_coeffs))
    if A == 0:
        return False
    # Determine monomial factor: find smallest i with coeff !=0?
    # Actually we need to factor out x^r where r = min exponent with nonzero coefficient.
    # Since coefficients may be zero at low indices, we can factor out x^r.
    # Check if there is an integer r such that A = x**r * D(x) where D has constant term !=0.
    # Compute r = smallest i with coeff != 0.

def poly_mul(p,q):
    # p,q are dict exponent->coeff
    r={}
    for e1,c1 in p.items():
        for e2,c2 in q.items():
            e=e1+e2
            r[e]=r.get(e,0)+c1*c2
    # remove zero coeffs
    r={e:c for e,c in r.items() if c!=0}
    return r

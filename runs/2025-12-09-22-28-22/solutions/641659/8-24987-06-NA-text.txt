    N = foot_of_perpendicular_from_point_to_line(D, E, F)
    # compute ratio CT * NB / (BT * NE)
    # T: intersection of circle through B,K tangent to BN at B, with BC again.
    # Compute circle through B and K tangent to BN at B: center O_T is intersection of perpendicular bisector of BK and line perpendicular to BN through B.
    # Compute line BN direction
    BN = N - B
    # line through B perpendicular to BN: direction perpendicular to BN
    # direction vector for line L: rotate BN 90 deg
    # Actually we need the line (set of points X such that (X-B) dot BN = 0)
    # Center O_T satisfies (O_T - B) dot BN = 0.
    # Also O_T lies on perpendicular bisector of BK: (O_T - (B+K)/2) dot (K-B) = 0.
    # Solve for O_T (two linear equations)
    # Let unknown (ox,oy)
    # Equation1: (ox,oy) dot BN = 0   (since B at origin)
    # Equation2: (ox - (B+K)/2) dot (K - B) = 0  => (ox - (K/2)) dot K = 0.
    # Solve linear equations.
    # Write BN = (BNx, BNy)
    BNx, BNy = BN
    Kx, Ky = K
    # Solve linear system:
    # ox*BNx + oy*BNy = 0
    # (ox - Kx/2)*Kx + (oy - Ky/2)*Ky = 0
    # => ox*Kx + oy*Ky = (Kx^2 + Ky^2)/2
    # Solve for ox,oy.
    A_mat = np.array([[BNx, BNy], [Kx, Ky]], dtype=float)
    b_vec = np.array([0.0, (Kx**2 + Ky**2)/2.0], dtype=float)
    # Solve
    try:
        O_T = np.linalg.solve(A_mat, b_vec)
    except np.linalg.LinAlgError:
        return None
    # radius:
    R_T = np.linalg.norm(O_T - B)
    # Intersect this circle with BC (x-axis y=0). Solve (x - O_Tx)^2 + (0 - O_Ty)^2 = R_T^2.
    # => (x - O_Tx)^2 = R_T^2 - O_Ty^2 = (distance^2 - O_Ty^2). Since O_T is on perpendicular bisector of BK, O_Ty probably >0.
    # So x = O_Tx ± sqrt(R_T^2 - O_Ty^2)
    # Choose solution x != 0 (since B = (0,0) is one intersection). So T_x = O_Tx + sign*sqrt(...). We need to decide sign.
    disc = R_T**2 - O_T[1]**2
    if disc < -1e-9:
        return None
    if disc < 0:
        disc = 0.0
    delta = sqrt(disc)
    # two intersection x positions:
    T1 = O_T[0] + delta
    T2 = O_T[0] - delta
    # B at (0,0). Determine which is B: one should be near 0.
    # If O_T[0] - delta is near 0, then that is B. Check:
    if abs(T1) < 1e-6:
        T_x = T2
    elif abs(T2) < 1e-6:
        T_x = T1
    else:
        # none close to zero, maybe error?
        # Use whichever is not zero
        if abs(T1) > 1e-6:
            T_x = T1
        else:
            T_x = T2
    T = np.array([T_x, 0.0])
    # Compute distances:
    CT = np.linalg.norm(C - T)  # C=(h,0) so CT = |h - T_x|
    BT = np.linalg.norm(T - B)  # = |T_x|
    NB = np.linalg.norm(N - B)  # = distance from B to N (since B at origin)
    # NE = distance from N to E
    NE = np.linalg.norm(N - E)
    ratio = (CT * NB) / (BT * NE) if BT !=0 and NE!=0 else None
    # Check cyclic condition for K,N,K',B: compute circumcircle of K,N,B and see if distance to K' matches radius.
    # Compute circumcenter of K,N,B:
    O_c, R_c = circle_center_radius(K, N, B)
    cyclic = False
    if O_c is not None:
        # distance from O_c to K' should be approx equal R_c
        if abs(np.linalg.norm(Kp - O_c) - R_c) < 1e-6:
            cyclic = True
    # Return all
    return {
        'params': params,
        'K':K,
        'Kp':Kp,
        'N':N,
        'T':T,
        'ratio':ratio,
        'cyclic':cyclic,
        'O_T':O_T,
        'O_c':O_c,
        'R_c':R_c
    }

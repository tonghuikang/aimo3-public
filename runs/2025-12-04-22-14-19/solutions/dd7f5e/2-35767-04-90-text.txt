def is_shifty_alpha(alpha):
    # alpha dict mapping i->value (nonzero)
    # search for beta and k,l

        for l in range(-10,11):
            if k==l: continue
            # choose beta support size 1 to 3
            for support_size in range(1,4):
                # choose j indices
                for J in combinations(range(-10,11), support_size):
                    # choose beta values
                    for bs in product(range(-5,6), repeat=support_size):
                        if all(b==0 for b in bs): continue
                        beta = {j:b for j,b in zip(J, bs)}

                        out = {}

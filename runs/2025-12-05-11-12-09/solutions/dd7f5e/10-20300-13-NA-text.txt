def cross_corr(a, b, n_range=None):
    # a,b are dict mapping index -> val
    # compute f(n)= sum_{t} a(t+n) * b(t)
    # assume indices maybe negative.
    # Determine n_range: min_n = min(index_a - max_t, index_b? but easier: compute all n that can have contribution:
    # n can be from min_a - max_t to max_a - min_t
    if n_range is None:
        min_a = min(a) if a else 0
        max_a = max(a) if a else 0
        min_b = min(b) if b else 0
        max_b = max(b) if b else 0
        min_n = min_a - max_b
        max_n = max_a - min_b
    else:
        min_n, max_n = n_range
    f={}
    for n in range(min_n, max_n+1):
        s=0
        for t,bt in b.items():
            s += a.get(t+n,0)*bt
        if s!=0:
            f[n]=s
    return f

def cross_corr_coeffs(alpha, beta):
    # alpha: dict of index->val (support [0,8])
    # beta: dict of index->val (support any)
    # compute f(n)= sum_{t} alpha(t+n) beta(t)
    min_a = min(alpha) if alpha else 0
    max_a = max(alpha) if alpha else 0
    min_b = min(beta) if beta else 0
    max_b = max(beta) if beta else 0
    min_n = min_a - max_b
    max_n = max_a - min_b
    f={}
    for n in range(min_n, max_n+1):
        s=0
        for t,bt in beta.items():
            s += alpha.get(t+n,0)*bt
        if s!=0:
            f[n]=s
    return f

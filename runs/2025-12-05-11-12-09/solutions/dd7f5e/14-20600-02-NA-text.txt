def find_factorizations(d, coeff_range=range(-2,3)):
    x=sp.symbols('x')
    polys=[]
    for g_coeffs in itertools.product(coeff_range, repeat=d+1):
        # leading coefficient g_d maybe zero allowed
        # but we want at least one non-zero? We'll handle after
        g=sp.expand(sum(g_coeffs[i]*x**i for i in range(d+1)))
        if not any(g_coeffs): continue
        for h_coeffs in itertools.product(coeff_range, repeat=d+1):
            h=sp.expand(sum(h_coeffs[i]*x**i for i in range(d+1)))
            if sp.expand(g*h) == 1 + x**d:
                polys.append((g_coeffs,h_coeffs))
    return polys

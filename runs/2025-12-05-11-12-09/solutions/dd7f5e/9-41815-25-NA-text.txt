    # compute P
    # alpha is tuple length 9: alpha0..alpha8
    coeffs = list(alpha)  # length 9
    # P(z) = alpha_8 + alpha_7 z + ... + alpha_0 z^8
    poly_P = sp.sympify(0)
    for i, coeff in enumerate(alpha):
        exp = 8 - i
        poly_P += coeff * x**exp
    # find r such that P divides z^8 + z^{8-r}
    # r can be any integer (positive/negative). Let's search for r from -20..20 excluding 0.
    for r in range(-20, 21):

            continue
        num = x**8 + x**(8 - r)
        # check if divisible
        q, rem = sp.div(num, poly_P, domain=sp.ZZ)

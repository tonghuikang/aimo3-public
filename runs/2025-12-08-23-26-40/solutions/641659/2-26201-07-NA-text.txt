def solve_triangle(n):
    F_n = fib(n)
    F_np1 = fib(n+1)
    a = fib(n+2)
    delta = fib(n-1) if n-1>=0 else 0
    r, c = sp.symbols('r c', positive=True, real=True)
    # sinθ, cosθ
    sinθ = 2*F_n*r/(F_n**2 + r**2)
    cosθ = (F_n**2 - r**2)/(F_n**2 + r**2)
    # sinψ, cosψ
    sinψ = 2*r*F_np1/(r**2 + F_np1**2)
    cosψ = (r**2 - F_np1**2)/(r**2 + F_np1**2)
    b = c + delta
    eq1 = sp.Eq(c*cosθ, a + b*cosψ)
    eq2 = sp.Eq(c*sinθ, b*sinψ)
    # Solve for c,r numeric approximate
    # Use nsolve
    try:
        sol = sp.nsolve((eq1, eq2), (c, r), (a*0.8, 0.5))
        return sol
    except Exception as e:
        print("Error:", e)
        return None

for n in range(2,8):
    sol = solve_triangle(n)
    print(n, sol)

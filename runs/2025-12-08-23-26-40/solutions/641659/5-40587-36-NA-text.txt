def fib(n):
    # use fast doubling for large n
    def fib_pair(k):
        if k == 0:
            return (0,1)
        else:
            a,b = fib_pair(k//2)
            c = a*(2*b - a)
            d = a*a + b*b
            if k%2 == 0:
                return (c,d)
            else:
                return (d,c+d)
    return fib_pair(n)[0]

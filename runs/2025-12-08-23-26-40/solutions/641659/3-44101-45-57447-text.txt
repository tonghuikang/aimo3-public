def ratio_and_residual(n, x_val):
    # compute geometry numerically using sympy with given x_val (float)
    # n integer >=2
    fib = [0,1]
    while len(fib) <= n+2:
        fib.append(fib[-1]+fib[-2])
    Fn = fib[n]
    Fn1 = fib[n+1]
    Fn2 = fib[n+2]
    a = Fn2  # BC length
    # coordinates
    B = sp.Point(0,0)
    C = sp.Point(a,0)
    # compute u, v:
    x = sp.nsimplify(x_val) # maybe keep as float
    # Use float for everything
    x = float(x_val)
    # compute u:
    u = (a**2 - ((Fn1 + x)**2 - (Fn + x)**2)) / (2*a)
    # compute v:
    v_sq = (Fn + x)**2 - u**2
    if v_sq <= 0:
        return None,None
    v = (v_sq)**0.5
    A = sp.Point(u, v)
    # incircle contact points:
    # F = (Fn/(Fn + x))*A
    F = sp.Point(u * Fn/(Fn + x), v * Fn/(Fn + x))

    inters = circ_ABC.intersection(circ_AFE)
    # find point not equal to A (within tolerance)
    K = None
    for pt in inters:

    if K is None:

def circumcenter(p1, p2, p3):
    # p1,p2,p3 tuples
    (x1,y1) = p1
    (x2,y2) = p2
    (x3,y3) = p3
    # Compute using perpendicular bisectors linear solve
    d = 2*(x1*(y2 - y3) + x2*(y3 - y1) + x3*(y1 - y2))
    if abs(d) < 1e-12:
        return None
    ux = ((x1**2 + y1**2)*(y2 - y3) + (x2**2 + y2**2)*(y3 - y1) + (x3**2 + y3**2)*(y1 - y2)) / d
    uy = ((x1**2 + y1**2)*(x3 - x2) + (x2**2 + y2**2)*(x1 - x3) + (x3**2 + y3**2)*(x2 - x1)) / d
    r = math.hypot(ux - x1, uy - y1)
    return (ux, uy, r)

def circle_intersections(c1, r1, c2, r2):
    # return list of intersection points (tuples), may be empty,1,2
    (x0,y0) = c1
    (x1,y1) = c2
    dx = x1 - x0
    dy = y1 - y0
    d = math.hypot(dx, dy)
    if d > r1 + r2 + 1e-12 or d < abs(r1 - r2) - 1e-12 or d == 0:
        # no intersection or coincident
        return []
    # a = (r1^2 - r2^2 + d^2) / (2d)
    a = (r1**2 - r2**2 + d**2) / (2*d)
    # h^2 = r1^2 - a^2
    h_sq = r1**2 - a**2
    if h_sq < -1e-12:
        return []
    h = math.sqrt(max(0, h_sq))
    # midpoint between intersection points
    xm = x0 + a*dx/d
    ym = y0 + a*dy/d
    # intersection points
    rx = -dy * (h/d)
    ry = dx * (h/d)
    p1 = (xm + rx, ym + ry)
    p2 = (xm - rx, ym - ry)
    if h < 1e-12:
        return [p1]  # tangent
    else:
        return [p1, p2]

    (x1,y1) = p1
    (dx1,dy1) = d1
    (x2,y2) = p2
    (dx2,dy2) = d2
    # Solve: x1 + t dx1 = x2 + s dx2; y1 + t dy1 = y2 + s dy2
    # => t dx1 - s dx2 = x2 - x1; t dy1 - s dy2 = y2 - y1
    # Solve 2x2 linear system for t,s.
    A = [[dx1, -dx2],
         [dy1, -dy2]]
    B = [x2 - x1, y2 - y1]
    # Solve using determinant
    det = A[0][0]*A[1][1] - A[0][1]*A[1][0]
    if abs(det) < 1e-12:
        return None
    t = (B[0]*A[1][1] - B[1]*A[0][1]) / det
    # s = (dx1* (y2-y1) - dy1*(x2-x1)) / det maybe not needed
    # compute intersection point
    ix = x1 + t*dx1
    iy = y1 + t*dy1
    return (ix, iy)

def compute_ratio_and_residual_numeric(n, x):
    # compute geometry for given n and x
    # compute Fibonacci
    fib = [0,1]
    while len(fib) <= n+2:
        fib.append(fib[-1]+fib[-2])
    Fn = fib[n]
    Fn1 = fib[n+1]
    Fn2 = fib[n+2]
    a = Fn2 # BC length
    # points B,C
    B = (0.0,0.0)
    C = (float(a), 0.0)
    # compute coordinates of A
    # u = (a^2 - ((Fn1 + x)^2 - (Fn + x)^2)) / (2a)
    u = (a*a - ((Fn1 + x)**2 - (Fn + x)**2)) / (2.0*a)
    v_sq = (Fn + x)**2 - u*u
    if v_sq <= 0:
        return None, None
    v = math.sqrt(v_sq)
    A = (u, v)
    # D point
    D = (Fn, 0.0)
    # incircle contact points
    # F = (Fn/(Fn + x))*A
    F = (u*Fn/(Fn + x), v*Fn/(Fn + x))
    # E = ((u*Fn1 + a*x)/(Fn1 + x), v*Fn1/(Fn1 + x))
    E = ((u*Fn1 + a*x)/(Fn1 + x), v*Fn1/(Fn1 + x))
    # Compute circumcenter of ABC
    circ_ABC = circumcenter_numeric(A, B, C)
    if circ_ABC is None:
        return None, None
    (ox, oy, r_ABC) = circ_ABC
    # circumcenter of AFE
    circ_AFE = circumcenter_numeric(A, F, E)
    if circ_AFE is None:
        return None, None
    (ox2, oy2, r_AFE) = circ_AFE
    # intersect circles to get K
    inters = circle_intersections_numeric((ox, oy), r_ABC, (ox2, oy2), r_AFE)
    if not inters:
        return None, None
    # choose K not equal to A
    K = None
    for pt in inters:
        if math.hypot(pt[0] - A[0], pt[1] - A[1]) > 1e-8:
            K = pt
            break
    if K is None:
        return None, None
    # K' reflect across line EF
    Kp = reflect_across_line(K, E, F)
    # N foot from D onto EF
    N = projection_point(D, E, F)
    # circle through B,K,N
    circ_BKN = circumcenter_numeric(B, K, N)
    if circ_BKN is None:
        return None, None
    (ox_BKN, oy_BKN, r_BKN) = circ_BKN
    residual = math.hypot(Kp[0] - ox_BKN, Kp[1] - oy_BKN) - r_BKN
    # Now compute T: circle through B,K tangent to line BN at B
    # line BN direction vector
    BN_dir = (N[0] - B[0], N[1] - B[1])
    # perpendicular direction at B (for center)
    perp_B_dir = (-BN_dir[1], BN_dir[0])  # rotated 90 deg
    # midpoint of BK
    mid_BK = ((B[0] + K[0])/2.0, (B[1] + K[1])/2.0)
    # direction vector of BK
    BK_dir = (K[0] - B[0], K[1] - B[1])
    # perpendicular direction for bisector
    perp_BK_dir = (-BK_dir[1], BK_dir[0])
    # Intersection of line B + t*perp_B_dir and mid_BK + s*perp_BK_dir
    O_T = intersect_lines(B, perp_B_dir, mid_BK, perp_BK_dir)
    if O_T is None:
        return None, None
    # T_x = 2*O_T.x
    T = (2.0 * O_T[0], 0.0)
    # distances
    C_pt = C
    CT = math.hypot(C_pt[0] - T[0], C_pt[1] - T[1])
    BT = math.hypot(B[0] - T[0], B[1] - T[1])
    NB = math.hypot(N[0] - B[0], N[1] - B[1])
    NE = math.hypot(N[0] - E[0], N[1] - E[1])
    ratio = (CT * NB) / (BT * NE)
    return ratio, residual

def compute_residual_mp(n, x):
    # high-precision using mpmath
    fib = [mp.mpf(0), mp.mpf(1)]
    while len(fib) <= n+2:
        fib.append(fib[-1] + fib[-2])
    Fn = fib[n]
    Fn1 = fib[n+1]
    Fn2 = fib[n+2]
    a = Fn2
    # coordinates
    B = (mp.mpf(0), mp.mpf(0))
    C = (a, mp.mpf(0))
    u = (a*a - ((Fn1 + x)**2 - (Fn + x)**2)) / (2*a)
    v_sq = (Fn + x)**2 - u*u
    if v_sq <= 0:
        return None
    v = mp.sqrt(v_sq)
    A = (u, v)
    F = (u*Fn/(Fn + x), v*Fn/(Fn + x))
    E = ((u*Fn1 + a*x)/(Fn1 + x), v*Fn1/(Fn1 + x))
    # circumcenter function using mpmath
    def circumcenter_mp(p1,p2,p3):
        (x1,y1),(x2,y2),(x3,y3) = p1,p2,p3
        d = 2*(x1*(y2 - y3) + x2*(y3 - y1) + x3*(y1 - y2))
        if mp.almosteq(d,0):
            return None
        ux = ((x1*x1 + y1*y1)*(y2 - y3) + (x2*x2 + y2*y2)*(y3 - y1) + (x3*x3 + y3*y3)*(y1 - y2)) / d
        uy = ((x1*x1 + y1*y1)*(x3 - x2) + (x2*x2 + y2*y2)*(x1 - x3) + (x3*x3 + y3*y3)*(x2 - x1)) / d
        r = mp.sqrt((ux - x1)**2 + (uy - y1)**2)
        return (ux, uy, r)
    circ_ABC = circumcenter_mp(A,B,C)
    circ_AFE = circumcenter_mp(A,F,E)
    # intersection of circles
    def circle_intersections_mp(c1,r1,c2,r2):
        (x0,y0),(x1,y1) = c1,c2
        dx = x1 - x0
        dy = y1 - y0
        d = mp.sqrt(dx*dx + dy*dy)
        if d > r1 + r2 + mp.mpf('1e-30') or d < abs(r1 - r2) - mp.mpf('1e-30') or d == 0:
            return []
        a = (r1*r1 - r2*r2 + d*d) / (2*d)
        h_sq = r1*r1 - a*a
        if h_sq < 0:
            h_sq = mp.mpf(0)
        h = mp.sqrt(h_sq)
        xm = x0 + a*dx/d
        ym = y0 + a*dy/d
        rx = -dy * (h/d)
        ry = dx * (h/d)
        p1 = (xm + rx, ym + ry)
        p2 = (xm - rx, ym - ry)
        if h == 0:
            return [p1]
        else:
            return [p1, p2]
    inters = circle_intersections_mp((circ_ABC[0],circ_ABC[1]), circ_ABC[2],
                                     (circ_AFE[0],circ_AFE[1]), circ_AFE[2])
    # choose K not equal to A
    K = None
    for pt in inters:
        if mp.sqrt((pt[0]-A[0])**2 + (pt[1]-A[1])**2) > mp.mpf('1e-20'):
            K = pt
            break
    if K is None:
        return None
    # reflect K across line EF
    # line EF param eqn, direction vector
    ux = F[0] - E[0]
    uy = F[1] - E[1]
    # projection of K-E onto line direction
    dot = (K[0]-E[0])*ux + (K[1]-E[1])*uy
    denom = ux*ux + uy*uy
    projx = E[0] + dot*ux/denom
    projy = E[1] + dot*uy/denom
    Kp = (2*projx - K[0], 2*projy - K[1])
    # N foot from D onto EF
    D = (Fn, mp.mpf(0))
    # projection of D onto line EF:
    dotD = (D[0]-E[0])*ux + (D[1]-E[1])*uy
    projxD = E[0] + dotD*ux/denom
    projyD = E[1] + dotD*uy/denom
    N = (projxD, projyD)
    # circle through B,K,N
    circ_BKN = circumcenter_mp(B,K,N)
    if circ_BKN is None:
        return None
    # residual: distance from Kp to center minus radius
    res = mp.sqrt((Kp[0]-circ_BKN[0])**2 + (Kp[1]-circ_BKN[1])**2) - circ_BKN[2]
    return res

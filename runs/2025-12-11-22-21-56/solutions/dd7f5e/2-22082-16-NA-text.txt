            for combo in combinations(lst, r):
                deg = sum(d for _, d in combo)
                if deg > 8: continue
                # product polynomial of cyclotomics in combo
                poly = sp.Poly(1, x, domain='ZZ')
                for n,_ in combo:
                    poly = sp.polys.polytools.mul(poly, cyclotomic_polys[n])
                # coefficients as list
                coeffs = [0]*poly.degree()
                # Actually get coefficients properly
                coeffs = [0]*(poly.degree()+1)
                for mon, coeff in zip(poly.monoms(), poly.coeffs()):
                    coeffs[mon[0]] = int(coeff)
                # shift positions
                for shift in range(0, 9 - deg):
                    # shift left by shift (multiply by x^shift)
                    alpha = [0]*shift + coeffs
                    # pad to length 9
                    alpha += [0]*(9 - len(alpha))
                    # sign variants
                    for sign in (1,-1):
                        alphas.add(tuple(sign*v for v in alpha))
    return alphas

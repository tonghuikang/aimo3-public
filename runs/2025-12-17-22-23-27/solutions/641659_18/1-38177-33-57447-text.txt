    # CD = Fn1 = cot(C/2) => C = 2*arctan(1/Fn1)

    # Points on incircle: set D at angle 0

    # Then F->E arc = pi - A
    # Then E->D arc = pi - C

def circumcircle_center(P1,P2,P3):
    # Return center and radius of circumcircle of triangle defined by complex points.
    # Use complex geometry: perpendicular bisectors.
    # Convert to real coordinates
    x1,y1 = P1.real,P1.imag
    x2,y2 = P2.real,P2.imag
    x3,y3 = P3.real,P3.imag
    # Solve for center (h,k) such that distances equal.
    A = 2*(x2 - x1)
    B = 2*(y2 - y1)
    C = x2**2 + y2**2 - x1**2 - y1**2
    D = 2*(x3 - x2)
    E = 2*(y3 - y2)
    F = x3**2 + y3**2 - x2**2 - y2**2
    # Solve linear system A h + B k = C, D h + E k = F
    det = A*E - B*D
    if abs(det) < 1e-12:
        raise ValueError("Collinear points")
    h = (C*E - B*F) / det
    k = (A*F - C*D) / det
    center = complex(h,k)
    radius = abs(P1 - center)
    return center, radius

def circle_intersections(c1,r1,c2,r2):
    # returns intersection points of two circles (if any)
    # Return list of complex points (0,1,2 points)
    # Based on geometry
    # compute distance between centers
    d = abs(c2 - c1)
    if d > r1 + r2 + 1e-12 or d < abs(r1 - r2) - 1e-12:
        return []  # no intersection
    # find point p2 = c1 + a*(c2-c1)/d where a = (r1^2 - r2^2 + d^2) / (2d)
    a = (r1**2 - r2**2 + d**2) / (2*d)
    # point p2
    p2 = c1 + a*(c2 - c1)/d
    # distance h from p2 to intersection points
    h_sq = r1**2 - a**2
    if abs(h_sq) < 1e-12:
        return [p2]
    h = math.sqrt(max(h_sq, 0.0))
    # offset vector perpendicular
    # (c2 - c1) rotated by 90 degrees
    dx = (c2 - c1)/d
    # perpendicular
    perp = complex(-dx.imag, dx.real)
    i1 = p2 + h*perp
    i2 = p2 - h*perp
    return [i1,i2]

def foot_of_perpendicular(P, line_pt1, line_pt2):
    # foot from P onto line defined by points line_pt1, line_pt2
    # projection formula
    # line direction vector
    v = line_pt2 - line_pt1
    u = P - line_pt1
    t = (u.real*v.real + u.imag*v.imag) / (v.real**2 + v.imag**2)
    foot = line_pt1 + t*v
    return foot

def reflect_point_across_line(P, line_pt1, line_pt2):
    # reflect point P across line defined by two points
    foot = foot_of_perpendicular(P, line_pt1, line_pt2)
    reflected = foot*2 - P
    return reflected

    # Compute B and C based on incircle radius r
    B = 2*math.atan(r / Fn)
    C = 2*math.atan(r / Fn1)
    # A = pi - B - C (must be >0)
    A = math.pi - B - C
    if A <= 0:
        # invalid triangle
        return None
    # incircle radius = r, center at origin
    # Determine angles for D,E,F as before: D at 0; E at pi - C; F at pi + B
    d_angle = 0.0
    e_angle = (math.pi - C) % (2*math.pi)
    f_angle = (math.pi + B) % (2*math.pi)
    D = complex(r*math.cos(d_angle), r*math.sin(d_angle))
    E = complex(r*math.cos(e_angle), r*math.sin(e_angle))
    F = complex(r*math.cos(f_angle), r*math.sin(f_angle))
    # tangent lines: x cosθ + y sinθ = r, where cosθ = P.x / r, sinθ = P.y / r
    def tangent_coeff(P):
        # assume P on incircle radius r, so cosθ = P.real/r, sinθ = P.imag/r
        a = P.real / r
        b = P.imag / r
        # line: a x + b y = r
        return (a,b)
    aD,bD = tangent_coeff(D)
    aE,bE = tangent_coeff(E)
    aF,bF = tangent_coeff(F)
    # intersect D-F -> B
    det = aD*bF - aF*bD
    xB = (r*bF - r*bD) / det
    yB = (aD*r - aF*r) / det
    B_pt = complex(xB, yB)
    # intersect D-E -> C
    det2 = aD*bE - aE*bD

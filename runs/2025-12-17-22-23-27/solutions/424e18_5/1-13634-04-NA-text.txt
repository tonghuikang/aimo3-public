def generate_pairings(group):
    # group is list of players (IDs)
    # generate all possible pairings (perfect matchings)
    # returns list of list of pairs (tuple sorted)
    # naive recursion
    if not group:
        return [[]]
    # pick first element
    first = group[0]
    res = []
    for i in range(1, len(group)):
        second = group[i]
        rest = group[1:i] + group[i+1:]
        for sub in generate_pairings(rest):
            res.append([(first, second)] + sub)
    return res

import math

def circumcenter(p1, p2, p3):
    x1,y1 = p1
    x2,y2 = p2
    x3,y3 = p3
    d = 2*(x1*(y2 - y3) + x2*(y3 - y1) + x3*(y1 - y2))
    if abs(d) < 1e-12:
        return None, None
    ux = ((x1**2 + y1**2)*(y2 - y3) + (x2**2 + y2**2)*(y3 - y1) + (x3**2 + y3**2)*(y1 - y2)) / d
    uy = ((x1**2 + y1**2)*(x3 - x2) + (x2**2 + y2**2)*(x1 - x3) + (x3**2 + y3**2)*(x2 - x1)) / d
    return (ux, uy), math.hypot(ux - x1, uy - y1)

def circle_intersections(O1,R1,O2,R2):
    # return list of intersection points (0,1,2)
    x1,y1 = O1
    x2,y2 = O2
    dx, dy = x2 - x1, y2 - y1
    d = math.hypot(dx, dy)
    if d > R1 + R2 + 1e-12 or d < abs(R1 - R2) - 1e-12 or d < 1e-12:
        return []  # no intersection or coincident
    # a = (R1^2 - R2^2 + d^2) / (2d)
    a = (R1**2 - R2**2 + d**2) / (2*d)
    h_sq = R1**2 - a**2
    if h_sq < -1e-12:
        return []
    h = math.sqrt(max(h_sq,0))
    # point P2
    x3 = x1 + a * dx / d
    y3 = y1 + a * dy / d
    # offset
    rx = -dy * (h / d)
    ry = dx * (h / d)
    p_int1 = (x3 + rx, y3 + ry)
    p_int2 = (x3 - rx, y3 - ry)
    if h < 1e-12:
        return [p_int1]  # tangent
    else:
        return [p_int1, p_int2]

def foot_of_perpendicular(P, Q, R):
    # foot from point P onto line QR
    # returns point foot H
    # line QR direction vector v
    x0,y0 = P
    x1,y1 = Q
    x2,y2 = R
    vx, vy = x2 - x1, y2 - y1
    # if degenerate line
    norm2 = vx**2 + vy**2
    if norm2 == 0:
        return (x1,y1)
    # projection of vector QP onto v
    t = ((x0 - x1)*vx + (y0 - y1)*vy) / norm2
    # foot H = Q + t*v
    hx = x1 + t*vx
    hy = y1 + t*vy
    return (hx, hy)

def reflect_point_across_line(P, Q, R):
    # reflect P across line QR
    H = foot_of_perpendicular(P, Q, R)
    # reflection = 2H - P
    rx = 2*H[0] - P[0]
    ry = 2*H[1] - P[1]
    return (rx, ry)

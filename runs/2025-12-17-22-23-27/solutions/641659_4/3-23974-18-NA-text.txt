def circumcenter(p1, p2, p3):
    # each p is tuple (x,y)
    (x1,y1), (x2,y2), (x3,y3) = p1, p2, p3
    # Compute using perpendicular bisector intersection formula
    d = 2 * (x1*(y2 - y3) + x2*(y3 - y1) + x3*(y1 - y2))

    xm = x0 + a * dx / d
    ym = y0 + a * dy / d
    # intersection points
    if h < 1e-12:
        # one intersection (tangent)
        return [(xm, ym)]
    else:

        ry = dx * (h / d)
        p1 = (xm + rx, ym + ry)
        p2 = (xm - rx, ym - ry)

def reflect_point_across_line(P, A, B):
    # reflect point P across line AB (through A and B)
    # compute unit direction vector u of AB
    (xA, yA) = A
    (xB, yB) = B
    dx = xB - xA
    dy = yB - yA
    # normalize
    norm = math.hypot(dx, dy)
    if norm == 0:
        return None
    ux = dx / norm
    uy = dy / norm
    # vector from A to P
    APx = P[0] - xA
    APy = P[1] - yA
    # projection length
    proj = APx * ux + APy * uy
    # projection point Q
    Qx = xA + proj * ux
    Qy = yA + proj * uy
    # reflected point
    Rx = 2*Qx - P[0]
    Ry = 2*Qy - P[1]
    return (Rx, Ry)

def foot_point(P, A, B):
    # foot of perpendicular from point P onto line AB
    return reflect_point_across_line(P, A, B)  # Actually not; reflection across line is twice the projection; foot is projection; but we have formula: Q = projection
    # but we need foot point; we could compute using projection as above

def projection_point(P, A, B):
    # projection of P onto line AB (foot)
    (xA, yA) = A
    (xB, yB) = B
    dx = xB - xA
    dy = yB - yA
    norm2 = dx*dx + dy*dy
    if norm2 == 0:
        return A
    t = ((P[0] - xA) * dx + (P[1] - yA) * dy) / norm2
    return (xA + t*dx, yA + t*dy)

def compute_triangle_and_ratio(n, c):
    # returns dictionary of geometry for given n and side length c (AB)
    a = fib(n+2)  # BC length
    d = fib(n)    # BD length
    # delta = b - c = fib(n-1)
    delta = fib(n-1)
    b = c + delta
    # coordinates
    B = (0.0, 0.0)
    C = (float(a), 0.0)
    # compute A coordinates using law of cosines formula
    xA = (c*c - b*b + a*a) / (2.0 * a)
    # ensure c > xA for y positive
    val = c*c - xA*xA
    if val <= 0:
        return None
    yA = math.sqrt(val)
    A = (xA, yA)
    # incircle radius
    # area
    area = a * yA / 2.0
    s = (a + b + c) / 2.0
    r = area / s
    I = (float(d), r)
    # foot of I onto AB (since B is origin)
    # AB direction v = (xA, yA)
    dot = I[0] * xA + I[1] * yA
    vnorm2 = xA*xA + yA*yA
    tF = dot / vnorm2
    F = (tF * xA, tF * yA)
    # foot onto AC: line A->C direction v = (a - xA, -yA)
    vACx = a - xA
    vACy = -yA
    vACnorm2 = vACx*vACx + vACy*vACy
    # projection of I - A onto vAC
    IA_x = I[0] - xA
    IA_y = I[1] - yA
    tE = (IA_x * vACx + IA_y * vACy) / vACnorm2
    E = (xA + tE * vACx, yA + tE * vACy)
    # circle (AEF) circumcenter
    O_AEF = circumcenter(A, E, F)
    if O_AEF is None:
        return None
    R_AEF = math.hypot(O_AEF[0] - A[0], O_AEF[1] - A[1])
    # circumcircle of ABC:
    O_ABC = (a/2.0, (xA*xA + yA*yA - a*xA) / (2.0*yA))
    R_ABC = math.hypot(O_ABC[0] - B[0], O_ABC[1] - B[1])
    # compute intersection of circles (AEF) and (ABC)
    inters = circle_intersections(O_ABC, R_ABC, O_AEF, R_AEF)
    # we need the intersection not equal to A (within tolerance)
    K = None
    for pt in inters:
        if math.hypot(pt[0] - A[0], pt[1] - A[1]) > 1e-8:
            K = pt
            break
    if K is None:
        # maybe A is one of the intersection but not found? Could be due to numeric?
        # fallback: compute both, pick far from A
        if len(inters) == 2:
            K = inters[1] if math.hypot(inters[0][0] - A[0], inters[0][1] - A[1]) < 1e-8 else inters[0]
        else:
            return None
    # reflect K across line EF to get Kprime
    Kprime = reflect_point_across_line(K, E, F)
    # N: foot from D onto line EF
    Dpt = (float(d), 0.0)
    N = projection_point(Dpt, E, F)
    # Check cyclic condition: K,N,Kprime,B concyclic
    # Compute circle through K,N,B
    O_KNB = circumcenter(K, N, B)
    if O_KNB is None:
        cyclic = False
    else:
        R_KNB = math.hypot(O_KNB[0] - K[0], O_KNB[1] - K[1])
        # distance from O_KNB to Kprime
        dist_Kprime = math.hypot(O_KNB[0] - Kprime[0], O_KNB[1] - Kprime[1])
        cyclic = abs(dist_Kprime - R_KNB) < 1e-6
    # Compute T: circle through B,K tangent to line BN at B
    # Find center O_T: intersection of line through B perp to BN and perpendicular bisector of BK
    # line BN:
    # direction of BN:
    BN_dx = N[0] - B[0]
    BN_dy = N[1] - B[1]
    # line L1: through B with direction perpendicular to BN: direction vector = (-BN_dy, BN_dx)
    # param eq: O_T = B + s * (-BN_dy, BN_dx)
    # line L2: perpendicular bisector of BK: midpoint M = ((B[0]+K[0])/2, (B[1]+K[1])/2)
    # direction vector of BK = (K[0]-B[0], K[1]-B[1]) = (Kx, Ky)
    # perpendicular direction = (-Ky, Kx)
    # So O_T = M + t * (-Ky, Kx)
    # Solve for s,t where O_T from both param eqs equal.
    # Solve linear equations: B + s*(-BN_dy, BN_dx) = M + t*(-Ky, Kx)
    # Unknowns s,t.
    # Write as: s*(-BN_dy) - t*(-Ky) = Mx - Bx
    #            s*BN_dx - t*Kx = My - By
    # Let's solve.
    Mx = (B[0] + K[0]) / 2.0
    My = (B[1] + K[1]) / 2.0
    # Solve linear system:
    # [-BN_dy, Ky] * [s, t]^T = Mx - Bx
    # [BN_dx, -Kx] * [s, t]^T = My - By
    # Use matrix
    a11 = -BN_dy
    a12 = Ky
    a21 = BN_dx
    a22 = -Kx
    rhs1 = Mx - B[0]
    rhs2 = My - B[1]
    det = a11 * a22 - a12 * a21
    if abs(det) < 1e-12:
        # degenerate? fallback
        O_T = None
    else:
        s = (rhs1 * a22 - a12 * rhs2) / det
        # t = (a11 * rhs2 - rhs1 * a21) / det
        O_T = (B[0] + s * (-BN_dy), B[1] + s * (BN_dx))
    if O_T is None:
        return None
    # radius
    R_T = math.hypot(O_T[0] - B[0], O_T[1] - B[1])
    # Intersection of circle with BC (y=0)
    # Equation: (x - O_T.x)^2 + (0 - O_T.y)^2 = R_T^2
    # => (x - O_T.x)^2 = R_T^2 - O_T.y^2
    rhs = R_T**2 - O_T[1]**2
    if rhs < -1e-9:
        # no real intersection
        return None
    # sqrt
    dx = math.sqrt(max(0.0, rhs))
    # solutions: x = O_T.x +/- dx
    x1 = O_T[0] + dx
    x2 = O_T[0] - dx
    # B is at x=0. So the intersection at B corresponds to one of these x being 0 (within tolerance). Choose the other as T.
    # Determine which is B:
    if abs(x1) < 1e-6:
        xT = x2
    elif abs(x2) < 1e-6:
        xT = x1
    else:
        # Choose the one not near zero.
        if abs(x1) > 1e-6:
            # pick the one not zero? Actually B is one intersection at x=0. Our computation may not give exactly zero due numeric.
            # Let's compute distances to B to decide.
            if abs(x1) < abs(x2):
                # whichever closer to zero is B
                xT = x2
            else:
                xT = x1
        else:
            xT = x2
    T = (xT, 0.0)
    # compute distances for ratio
    CT = math.hypot(C[0] - T[0], C[1] - T[1])
    BT = math.hypot(T[0] - B[0], T[1] - B[1])
    NB = math.hypot(N[0] - B[0], N[1] - B[1])
    NE = math.hypot(N[0] - E[0], N[1] - E[1])
    ratio = (CT * NB) / (BT * NE) if BT != 0 and NE != 0 else None
    return {
        'c': c,
        'b': b,
        'A': A,
        'I': I,
        'F': F,
        'E': E,
        'K': K,
        'Kprime': Kprime,
        'N': N,
        'cyclic': cyclic,
        'ratio': ratio,
        'O_KNB': O_KNB,
        'R_KNB': R_KNB
    }

def cyclic_residual_c(c, n):
    geom = compute_triangle_and_ratio(n, c)
    if geom is None:
        return None
    # compute residual for cyclic condition as distance difference:
    if not geom['cyclic']:
        # compute signed diff: distance from O_KNB to Kprime minus radius
        O = geom['O_KNB']
        if O is None:
            return None
        R = geom['R_KNB']
        # distance from O to Kprime
        Kprime = geom['Kprime']
        diff = math.hypot(O[0]-Kprime[0], O[1]-Kprime[1]) - R
        return diff
    else:
        return 0.0

def find_c_for_n(n, debug=False):
    # search for c that makes cyclic condition
    # lower bound for c > (a - delta)/2
    a = fib(n+2)
    delta = fib(n-1)
    lower_c = (a - delta) / 2.0
    # we need some upper bound; we can try up to some multiple
    # For large c, geometry changes; let's search up to say lower_c + 1000 * maybe.
    max_c = lower_c + 5000.0
    # Use sign change scanning.
    import numpy as np
    N_steps = 20000
    cs = np.linspace(lower_c + 1e-6, max_c, N_steps)
    prev_c = None
    prev_val = None
    interval = None
    for ci in cs:
        val = cyclic_residual_c(ci, n)
        if val is None:
            continue
        if prev_val is not None and val*prev_val < 0:
            interval = (prev_c, ci)
            break
        prev_c = ci
        prev_val = val
    if interval is None:
        if debug:
            print("No sign change found for n=", n)
        return None
    a_c, b_c = interval
    # bisection
    for _ in range(80):
        mid = (a_c + b_c) / 2.0
        fmid = cyclic_residual_c(mid, n)
        if fmid is None:
            a_c = mid
            continue
        if abs(fmid) < 1e-12:
            return mid
        f_a = cyclic_residual_c(a_c, n)
        if f_a is None:
            a_c = mid
            continue
        if f_a * fmid <= 0:
            b_c = mid
        else:
            a_c = mid
    return (a_c + b_c) / 2.0

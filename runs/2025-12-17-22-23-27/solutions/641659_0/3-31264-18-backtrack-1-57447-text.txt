def incircle_coords(x,y,s):
    # x = BD = F_n, y = CD = F_{n+1}, a = x+y
    a = x+y
    # Compute coordinates X,Y of vertex A
    X = y + s*(x - y)/a
    # side lengths
    c = s - y  # AB
    # compute Y from AB length squared = X^2 + Y^2 = c^2
    # ensure c^2 >= X^2
    if c*c < X*X:
        # numerical issues: triangle impossible?
        raise ValueError("c^2 < X^2")
    Y = math.sqrt(c*c - X*X)
    # incircle radius r = (area)/s
    # area = (a * Y)/2 (since base a on x-axis)
    area = a * Y / 2.0
    r = area / s

    I = (x, r)
    return (X,Y), I, r, a

def foot_of_perpendicular(P, Q, R):
    # foot from point P onto line QR
    # returns point foot
    # using projection
    # vector Q->R
    qx,qy = Q

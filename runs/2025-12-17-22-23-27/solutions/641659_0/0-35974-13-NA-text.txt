def circle_from_three(p1, p2, p3):
    # Returns center (cx,cy) and radius r.
    (x1,y1) = p1
    (x2,y2) = p2
    (x3,y3) = p3
    # Using perpendicular bisector method: solve linear equations for center.
    # (cx - x1)^2 + (cy - y1)^2 = (cx - x2)^2 + (cy - y2)^2
    # (cx - x1)^2 + (cy - y1)^2 = (cx - x3)^2 + (cy - y3)^2
    # Expand and simplify to linear equations:
    # 2*(x2 - x1)*cx + 2*(y2 - y1)*cy = x2^2 + y2^2 - x1^2 - y1^2
    # 2*(x3 - x1)*cx + 2*(y3 - y1)*cy = x3^2 + y3^2 - x1^2 - y1^2
    A = np.array([[2*(x2-x1), 2*(y2-y1)], [2*(x3-x1), 2*(y3-y1)]], dtype=float)
    b = np.array([x2*x2 + y2*y2 - x1*x1 - y1*y1,
                  x3*x3 + y3*y3 - x1*x1 - y1*y1], dtype=float)
    # Solve
    try:
        cx, cy = np.linalg.solve(A, b)
    except np.linalg.LinAlgError:
        # Points are collinear or nearly.
        return None
    r = math.hypot(cx - x1, cy - y1)
    return (cx, cy, r)

def intersect_two_circles(c1, c2):
    # c1, c2 = (cx,cy,r)
    (x0, y0, r0) = c1
    (x1, y1, r1) = c2
    # Compute distance between centers
    dx = x1 - x0
    dy = y1 - y0
    d = math.hypot(dx, dy)
    # no intersection cases not considered
    if d > r0 + r1 + 1e-9 or d < abs(r0 - r1) - 1e-9:
        return []  # no intersection
    # intersection points formula
    # a = (r0^2 - r1^2 + d^2) / (2d)
    a = (r0*r0 - r1*r1 + d*d) / (2*d) if d != 0 else 0
    # height h
    h_sq = r0*r0 - a*a
    if h_sq < -1e-12:
        return []
    h = math.sqrt(max(0.0, h_sq))
    # midpoint of line between intersection points
    xm = x0 + a * dx / d if d != 0 else x0
    ym = y0 + a * dy / d if d != 0 else y0
    # offset vector
    rx = -dy * (h/d) if d != 0 else 0
    ry = dx * (h/d) if d != 0 else 0
    p1 = (xm + rx, ym + ry)
    p2 = (xm - rx, ym - ry)
    # return both (could be same if h=0)
    return [p1, p2]

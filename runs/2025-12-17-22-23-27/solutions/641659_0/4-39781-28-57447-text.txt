    # P,Q,R are arrays or tuples of floats
    # compute using linear algebra
    # Solve (Q-P)·O = (|Q|^2 - |P|^2)/2, (R-P)·O = (|R|^2 - |P|^2)/2
    # Actually we can solve linear equations: 2*(Q-P)·O = |Q|^2 - |P|^2, etc.
    P = np.array(P, dtype=float)
    Q = np.array(Q, dtype=float)
    R = np.array(R, dtype=float)
    A_mat = np.array([2*(Q-P), 2*(R-P)]) # 2x2 matrix
    b_vec = np.array([np.dot(Q, Q) - np.dot(P, P), np.dot(R, R) - np.dot(P, P)])
    # solve linear system A O = b
    O = np.linalg.solve(A_mat, b_vec)
    return O

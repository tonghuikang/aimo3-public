def tournament(num_players):
    # num_players must be power of 2
    rounds = num_players.bit_length() - 1
    # assign distinct speeds: a permutation of [1..num_players], higher number = faster
    speeds = list(range(num_players))
    random.shuffle(speeds)  # speed ranking; higher = faster
    # participants labelled by index
    participants = list(range(num_players))
    # each participant's current score = 0
    scores = [0]*num_players
    # maintain groups: dict score -> list of participants
    groups = {0: participants[:]}
    for i in range(1, rounds+1):
        weight = 2**(rounds-i)  # round i weight
        new_groups = {}
        # for each score group
        for scr, members in groups.items():
            # members length should be even
            # random pairing
            random.shuffle(members)
            # pair them
            for a,b in zip(members[::2], members[1::2]):
                # determine winner by speed
                if speeds[a] > speeds[b]:
                    winner, loser = a,b
                else:
                    winner, loser = b,a
                # winner gets weight added
                scores[winner] += weight
                # assign to new groups based on new scores
                new_groups.setdefault(scores[winner], []).append(winner)
                new_groups.setdefault(scores[loser], []).append(loser)
        groups = new_groups
    # final scores
    return speeds, scores

However, as we argued, final scores are all distinct, so there are no ties. So ranking yields a total order. So the number of possible "orderings" is just the number of possible permutations of participants, i.e., (2^{20})!.

Thus answer is 62140.

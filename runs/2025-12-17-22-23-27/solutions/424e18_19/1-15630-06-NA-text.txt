def tournament_ordering(participants, speeds, pairing):
    # participants: list of labels
    # speeds: dict label->speed (higher better)
    # pairing: list of pairs for round1; each pair is (label1,label2)
    # round1: compute winners and losers
    # round2: winners group and losers group each have 2 participants; they fight each other.
    # Return final ordering list (from highest score to lowest)
    # Determine scores per participant
    scores = {p:0 for p in participants}
    # round 1
    for a,b in pairing:
        if speeds[a] > speeds[b]:
            win, lose = a, b
        else:
            win, lose = b, a
        scores[win] += 2**(2-1) # n=2, round1 points = 2^{2-1} = 2
    # After round1, partition winners and losers
    winners = [win for (a,b) in pairing for win in [a if speeds[a]>speeds[b] else b]]
    losers = [lose for (a,b) in pairing for lose in [b if speeds[a]>speeds[b] else a]]
    # round2: pair winners and losers groups arbitrarily (only possible pairing is (w1,w2) and (l1,l2))
    w1,w2 = winners
    if speeds[w1] > speeds[w2]:
        win, lose = w1,w2
    else:
        win, lose = w2,w1
    scores[win] += 2**(2-2) # +1
    # losers group
    l1,l2 = losers
    if speeds[l1] > speeds[l2]:
        win_l, lose_l = l1,l2
    else:
        win_l, lose_l = l2,l1
    scores[win_l] += 1
    # Now order by scores descending
    ordered = sorted(participants, key=lambda p: scores[p], reverse=True)
    return ordered

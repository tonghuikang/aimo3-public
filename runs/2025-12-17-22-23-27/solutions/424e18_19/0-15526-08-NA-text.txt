    speed = {i: participants[i] for i in range(n)}  # speed values arbitrary
    # initialize scores dict
    score = {i:0 for i in range(n)}
    # rounds k = log2 n
    k = int(round(math.log2(n)))
    for i in range(1, k+1):
        # group participants by current score
        groups = {}

            s = score[p]
            groups.setdefault(s, []).append(p)
        # each group must have even size
        # randomly pair each group

            random.shuffle(members)

                # determine winner based on speed (higher speed value wins)
                if speed[a] > speed[b]:

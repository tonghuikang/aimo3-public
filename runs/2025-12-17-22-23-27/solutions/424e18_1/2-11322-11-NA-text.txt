def all_matchings(players):
    # returns list of list of pairs (as tuple) covering all players
    # players is a tuple/list of distinct elements
    # algorithm: recursively pick first element and pair with any of the others, then recurs.
    if not players:
        return [[]]
    players = list(players)
    first = players[0]
    rest = players[1:]
    matchings = []
    for i, other in enumerate(rest):
        # pair first with other
        new_rest = rest[:i] + rest[i+1:]
        for sub in all_matchings(new_rest):
            matchings.append([(first, other)] + sub)
    return matchings

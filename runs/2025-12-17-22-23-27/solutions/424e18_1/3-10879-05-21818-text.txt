def vp_catalan(m, p):

    def vp_int(x):
        cnt = 0
        while x % p == 0:
            x //= p
            cnt += 1
        return cnt
    return vp_factorial(2*m, p) - 2*vp_factorial(m, p) - vp_int(m+1)

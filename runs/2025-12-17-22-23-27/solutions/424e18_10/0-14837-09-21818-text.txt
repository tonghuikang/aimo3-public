def pairings_of_four():
    # returns list of list of 2 pairs each, each pair as tuple of two elements (order irrelevant)
    # participants as 0,1,2,3
    participants = [0,1,2,3]
    pairings = []
    # choose first pair as any unordered pair; then second pair is remaining two.
    for a,b in itertools.combinations(participants,2):
        remaining = [x for x in participants if x not in (a,b)]
        c,d = remaining
        # sort pairs (smallest first to ensure canonical)
        p1 = tuple(sorted((a,b)))
        p2 = tuple(sorted((c,d)))
        # ensure ordering of pairs consistent (sorted lexicographically)
        if p1 <= p2:
            pairings.append([p1,p2])
        else:
            pairings.append([p2,p1])
    # Remove duplicates (some may be same after sorting)
    # Use set of frozensets
    uniq = []
    seen = set()
    for p in pairings:
        s = frozenset(p)
        if s not in seen:
            seen.add(s)
            uniq.append(list(p))
    return uniq

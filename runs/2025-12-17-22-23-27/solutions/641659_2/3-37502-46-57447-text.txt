def compute_ratio(BD, CD, r):
    # define points as sympy rational?
    # But we will do numeric (float)
    BD = float(BD); CD = float(CD); r = float(r)
    B = sp.Point(-BD, -r)
    C = sp.Point(CD, -r)
    D = sp.Point(0, -r)
    # compute F, E
    F = sp.Point(-2 * r**2 * BD / (BD**2 + r**2), r * (BD**2 - r**2) / (BD**2 + r**2))
    E = sp.Point(2 * r**2 * CD / (CD**2 + r**2), r * (CD**2 - r**2) / (CD**2 + r**2))
    # compute A intersection of lines AB and AC: solve linear equations
    cosF = -2 * r * BD / (BD**2 + r**2)
    sinF = (BD**2 - r**2) / (BD**2 + r**2)
    cosE = 2 * r * CD / (CD**2 + r**2)
    sinE = (CD**2 - r**2) / (CD**2 + r**2)
    # Solve linear system
    # cosF * x + sinF * y = r
    # cosE * x + sinE * y = r
    # Solve for x,y
    # Use sympy linear solve
    x, y = sp.symbols('x y')
    sol = sp.solve([cosF*x + sinF*y - r, cosE*x + sinE*y - r], (x,y))
    A = sp.Point(sol[x], sol[y])
    # circumcircle of ABC:
    O_c = sp.geometry.Triangle(A,B,C).circumcircle.center
    # radius:
    R_c = sp.geometry.Triangle(A,B,C).circumcircle.radius
    # circumcircle of AEF
    circ_AEF = sp.geometry.Triangle(A,E,F).circumcircle
    # Intersection of circles (Omega and circ_AEF)
    # Solve for intersection points
    # Use sympy geometry intersect
    inter = sp.geometry.Circle(O_c, R_c).intersection(circ_AEF)
    # inter should be list of points
    # Find K not equal to A
    K = None
    for pt in inter:
        # Compare pt to A
        if pt.distance(A) > 1e-8:
            K = pt
            break
    # If not found, return None
    if K is None:
        return None
    # reflect K across line EF
    line_EF = sp.Line(E, F)
    K_ref = K.mirror(line_EF) # sympy has mirror? maybe use reflect?
    # Actually use reflect method: reflect about line
    K_prime = K.reflect(line_EF) # Mirror about line
    # foot N from D to EF
    N = sp.Line(E, F).projection(D)
    # Now compute circle tangent to BN at B and passing through K.
    # The circle passes through B and K, and tangent to BN at B => its center lies on line perpendicular to BN at B.
    line_BN = sp.Line(B, N)
    # direction perpendicular to BN:
    # line perpendicular through B:
    perp_line = line_BN.perpendicular_line(B)
    # perpendicular bisector of BK:
    mid_BK = sp.Point((B.x+K.x)/2, (B.y+K.y)/2)
    perp_bk = sp.Line(B, K).perpendicular_line(mid_BK)
    # Intersection gives center O_t
    O_t = perp_line.intersection(perp_bk)[0]
    # radius
    R_t = O_t.distance(B)
    # Intersection of this circle with line BC (y = -r):
    line_BC = sp.Line(B, C)  # should be horizontal line y = -r
    # Solve for intersection points
    circ_t = sp.Circle(O_t, R_t)
    inter_T = circ_t.intersection(line_BC)
    # Choose T not equal B
    T = None
    for pt in inter_T:
        if pt.distance(B) > 1e-8:
            T = pt
            break
    if T is None:

    # Compute distances

def circumcenter(p1, p2, p3):
    # p1,p2,p3 are (x,y)
    (x1,y1),(x2,y2),(x3,y3) = p1,p2,p3
    # Compute using perpendicular bisectors intersection
    # Solve linear system: (x - x1)^2 + (y - y1)^2 = (x - x2)^2 + (y - y2)^2
    # => 2*(x2 - x1)*x + 2*(y2 - y1)*y = x2^2 + y2^2 - x1^2 - y1^2
    # Similarly for p2,p3
    A = [[2*(x2 - x1), 2*(y2 - y1)],
         [2*(x3 - x2), 2*(y3 - y2)]]
    B = [x2*x2 + y2*y2 - x1*x1 - y1*y1,
         x3*x3 + y3*y3 - x2*x2 - y2*y2]
    # Solve linear equations
    det = A[0][0]*A[1][1] - A[0][1]*A[1][0]
    if abs(det) < 1e-12:
        return None # points collinear or nearly so
    invA = [[A[1][1]/det, -A[0][1]/det],
            [-A[1][0]/det, A[0][0]/det]]
    cx = invA[0][0]*B[0] + invA[0][1]*B[1]
    cy = invA[1][0]*B[0] + invA[1][1]*B[1]
    return (cx,cy)

def circle_intersections(c1,r1,c2,r2):
    # returns list of intersection points (x,y)
    x1,y1 = c1; x2,y2 = c2
    dx = x2 - x1; dy = y2 - y1
    d = math.hypot(dx,dy)
    if d > r1 + r2 + 1e-12 or d < abs(r1 - r2) - 1e-12:
        return []  # no intersection
    if d == 0 and abs(r1-r2) < 1e-12:
        return []  # infinite intersections
    # a = (r1^2 - r2^2 + d^2) / (2d)
    a = (r1*r1 - r2*r2 + d*d) / (2*d)
    # point P2 = c1 + a*(c2-c1)/d
    x3 = x1 + a * dx / d
    y3 = y1 + a * dy / d
    # height h = sqrt(r1^2 - a^2)
    h_sq = r1*r1 - a*a
    if h_sq < -1e-12:
        return []
    h = math.sqrt(max(h_sq, 0.0))
    # offset (rx,ry) = (-dy*(h/d), dx*(h/d))
    rx = -dy * (h/d)
    ry = dx * (h/d)
    p_int1 = (x3 + rx, y3 + ry)
    p_int2 = (x3 - rx, y3 - ry)
    # If h close to 0, both points same
    if h < 1e-12:
        return [p_int1]
    else:
        return [p_int1, p_int2]

def reflect_point_across_line(P, A, B):

    dot = wx*vx + wy*vy
    len2 = vx*vx + vy*vy

    t = dot/len2
    projx = ax + t*vx
    projy = ay + t*vy
    rx = 2*projx - px
    ry = 2*projy - py
    return (rx, ry)

    }

    # compute Fibonacci numbers F0=0, F1=1
    if n < 0:
        raise ValueError
    a, b = 0, 1
    for i in range(n):
        a, b = b, a + b
    return a

    # distance from P to line through L1-L2
    (x0,y0) = P
    (x1,y1) = L1
    (x2,y2) = L2

    # returns intersection points of two circles (x,y) as list
    (x0,y0) = center1
    (x1,y1) = center2
    R0 = radius1
    R1 = radius2
    # distance between centers
    d = math.hypot(x1-x0, y1-y0)
    if d > R0+R1 or d < abs(R0-R1) or d == 0:
        return []  # no or infinite intersections
    # a = (R0^2 - R1^2 + d^2)/(2d)
    a = (R0**2 - R1**2 + d**2) / (2*d)
    # h^2 = R0^2 - a^2
    h_sq = R0**2 - a**2
    h = math.sqrt(max(0, h_sq))
    # point P2 = center0 + a*(center1-center0)/d
    x2 = x0 + a * (x1 - x0) / d
    y2 = y0 + a * (y1 - y0) / d
    # intersection points
    # offset perpendicular vector
    rx = -(y1 - y0) * (h / d)
    ry = (x1 - x0) * (h / d)
    p1 = (x2 + rx, y2 + ry)
    p2 = (x2 - rx, y2 - ry)
    return [p1, p2]

def f_brutal_one(n):
    best = None
    best_trip = None
    for a in range(1, n//3+1):
        for b in range(a+1, (n-a)//2+1):
            c = n - a - b
            if c <= b: continue
            l = math.lcm(a,b,c)
            if best is None or l < best:
                best = l
                best_trip = (a,b,c)
    return best, best_trip

    if math.gcd(3, S) == 1:
        phi = sum(1 for i in range(1, S) if math.gcd(i, S) == 1)

def is_power_of_3(x):
    # check if x is power of 3
    if x <=0:
        return False
    while x % 3 == 0:
        x //=3
    return x ==1

def find_min_ratio(limit_g=50, limit_p1=50):
    best = None
    best_params = None
    for g in range(1, limit_g+1):
        for p1 in range(1, limit_p1+1):
            for q1 in range(1, limit_p1+1):
                if p1 == q1:

                r = (g * p1 * q1) / D1
                if best is None or r < best:
                    best = r
                    best_params = (g, p1, q1, D1, r)
    return best, best_params

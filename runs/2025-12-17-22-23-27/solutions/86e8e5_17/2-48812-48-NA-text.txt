def brute_f_min(n, limit=1000000):
    best = None
    bestN = None
    for N in range(1, limit+1):
        # list divisors
        divs = []
        for d in range(1, int(math.sqrt(N))+1):
            if N % d == 0:
                divs.append(d)

                for k in range(j+1, L):

                if found: break
            if found: break
        if found:

            break

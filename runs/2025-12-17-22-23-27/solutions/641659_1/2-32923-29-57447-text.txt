def dist(P,Q):
    return math.hypot(P[0]-Q[0], P[1]-Q[1])

def incircle_center(A,B,C):
    # side lengths opposite vertices
    a = dist(B,C)
    b = dist(C,A)
    c = dist(A,B)
    # incenter as weighted sum
    p = a + b + c
    I = ((a*A[0] + b*B[0] + c*C[0]) / p,
         (a*A[1] + b*B[1] + c*C[1]) / p)
    return I

def foot_of_perpendicular(P, U, V):
    # projection of P onto line UV
    # compute vector UV
    ux, uy = U
    vx, vy = V
    # direction vector
    dx = vx - ux
    dy = vy - uy
    # dot product of (P-U) with direction
    px, py = P
    t = ((px-ux)*dx + (py-uy)*dy) / (dx*dx + dy*dy)
    proj = (ux + t*dx, uy + t*dy)
    return proj

def circumcenter(P, Q, R):
    # returns center (x,y) and radius
    # Using formula from perpendicular bisector intersection
    # Solve using linear algebra: solve for center O satisfying |O-P|^2 = |O-Q|^2 and |O-P|^2 = |O-R|^2
    # Equivalent to linear equations
    (x1,y1),(x2,y2),(x3,y3)=P,Q,R
    d = 2*(x1*(y2-y3)+x2*(y3-y1)+x3*(y1-y2))
    if abs(d) < 1e-12:
        # collinear
        return None
    ux = ((x1*x1+y1*y1)*(y2-y3)+(x2*x2+y2*y2)*(y3-y1)+(x3*x3+y3*y3)*(y1-y2))/d
    uy = ((x1*x1+y1*y1)*(x3-x2)+(x2*x2+y2*y2)*(x1-x3)+(x3*x3+y3*y3)*(x2-x1))/d
    R = math.hypot(ux-x1, uy-y1)
    return (ux, uy, R)

def circle_intersections(c1, c2):
    # c1, c2 are (x,y,R)
    x1,y1,R1 = c1
    x2,y2,R2 = c2
    # distance between centers
    d = math.hypot(x2-x1, y2-y1)
    # if no intersections or infinite?
    if d > R1+R2+1e-12 or d < abs(R1-R2)-1e-12:
        return []  # no intersections
    # find a = (R1^2 - R2^2 + d^2) / (2d)
    a = (R1*R1 - R2*R2 + d*d) / (2*d)
    # height from chord to intersection points
    h_sq = R1*R1 - a*a
    if h_sq < -1e-12:
        return []
    h = math.sqrt(max(h_sq,0.0))
    # point P2 = P0 + a*(P1-P0)/d
    x3 = x1 + a*(x2-x1)/d
    y3 = y1 + a*(y2-y1)/d
    # intersection points
    rx = -(y2-y1) * (h/d)
    ry = (x2-x1) * (h/d)
    pt1 = (x3 + rx, y3 + ry)
    pt2 = (x3 - rx, y3 - ry)
    return [pt1, pt2]

def reflect_point_across_line(P, A, B):
    # reflect point P across line AB
    # foot of perp
    foot = foot_of_perpendicular(P, A, B)
    # reflected point = foot*2 - P
    return (2*foot[0] - P[0], 2*foot[1] - P[1])

        vy = R[1] - Q[1]

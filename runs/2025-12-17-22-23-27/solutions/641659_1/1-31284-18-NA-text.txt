def compute_cyclic_residual(t_val, lam_val):
    # using mpmath for high precision
    # BD=1, CD=lam
    r = 1.0 / t_val
    # B, C, D
    B = mp.matrix([r, 1.0])
    C = mp.matrix([r, -lam_val])
    D = mp.matrix([r, 0.0])
    # F coordinates
    xF = r * (r**2 - 1.0) / (r**2 + 1.0)
    yF = (2 * r) / (r**2 + 1.0)
    F = mp.matrix([xF, yF])
    # E coordinates
    xE = r * (r**2 - lam_val**2) / (r**2 + lam_val**2)
    yE = - (2 * r * lam_val) / (r**2 + lam_val**2)
    E = mp.matrix([xE, yE])
    # Intersection of lines BF and CE to get A
    # Solve B + u*(F-B) = C + v*(E-C)
    # 2 eqs for u,v
    # Write as matrix
    # (F-B) * u - (E-C) * v = C - B
    # unknowns u,v
    FB = F - B
    EC = E - C
    # Solve linear system
    # [[FB.x, -EC.x], [FB.y, -EC.y]] * [u, v]^T = C - B
    M = mp.matrix([[FB[0], -EC[0]], [FB[1], -EC[1]]])
    rhs = C - B
    try:
        sol = mp.lu_solve(M, rhs)
    except Exception as e:
        # degenerate
        return None
    u = sol[0]
    # v = sol[1]  # not needed
    A = B + FB * u
    # Function to compute circle center and radius from 3 points
    def circumcenter(P1, P2, P3):
        # using formula from perpendicular bisectors
        # Compute using linear algebra
        # Solve for center O = (ox, oy)
        # |O - Pi|^2 = |O - Pj|^2 -> (Pi-Pj)·O = (|Pi|^2 - |Pj|^2)/2
        # set up two equations
        x1, y1 = P1[0], P1[1]
        x2, y2 = P2[0], P2[1]
        x3, y3 = P3[0], P3[1]
        # Equation (P2-P1).O = (|P2|^2 - |P1|^2)/2
        # Equation (P3-P1).O = (|P3|^2 - |P1|^2)/2
        A_mat = mp.matrix([[x2 - x1, y2 - y1], [x3 - x1, y3 - y1]])
        B_vec = mp.matrix([ (x2**2 + y2**2 - x1**2 - y1**2)/2,
                            (x3**2 + y3**2 - x1**2 - y1**2)/2 ])
        O = mp.lu_solve(A_mat, B_vec)
        return O
    # circles
    O_ABC = circumcenter(A, B, C)
    R_ABC = mp.norm(A - O_ABC)
    O_AFE = circumcenter(A, F, E)
    R_AFE = mp.norm(A - O_AFE)
    # Intersection of circles: solve for points
    # Use circle-circle intersection formula
    # distance between centers d
    d = mp.norm(O_ABC - O_AFE)
    if d == 0:
        return None
    # ex = (O_AFE - O_ABC) / d
    ex = (O_AFE - O_ABC) / d
    # a = (R_ABC**2 - R_AFE**2 + d**2) / (2*d)
    a = (R_ABC**2 - R_AFE**2 + d**2) / (2*d)
    # point P2 = O_ABC + a*ex
    P2 = O_ABC + ex * a
    # h = sqrt(R_ABC**2 - a**2)
    # If negative due to rounding, set to 0
    h_sq = R_ABC**2 - a**2
    if h_sq < 0:
        h = 0.0
    else:
        h = mp.sqrt(h_sq)
    # orthogonal vector ey = rotate ex by 90 degrees
    ey = mp.matrix([-ex[1], ex[0]])
    # intersection points = P2 ± h*ey
    inter1 = P2 + ey * h
    inter2 = P2 - ey * h
    # Choose K as the one not close to A
    # compute distances to A
    d1 = mp.norm(inter1 - A)
    d2 = mp.norm(inter2 - A)
    if d1 > 1e-8:
        K = inter1 if d1 > d2 else inter2
    else:
        K = inter2
    # line EF
    # N: foot of perpendicular from D to line EF
    # projection formula:
    # Let v = E->F vector
    v = F - E
    # projection of D-E onto v:
    t_proj = mp.dot(D - E, v) / mp.dot(v, v)
    proj = E + v * t_proj
    N = proj
    # K' reflection of K across line EF:
    # projection of K onto line EF
    t_proj_K = mp.dot(K - E, v) / mp.dot(v, v)
    projK = E + v * t_proj_K
    Kprime = projK * 2 - K
    # Circle through B,N,K
    # compute circumcenter
    O_BNK = circumcenter(B, N, K)
    R_BNK = mp.norm(B - O_BNK)
    # residual = distance(Kprime,O_BNK) - R_BNK
    residual = mp.norm(Kprime - O_BNK) - R_BNK
    return residual

def pairings(lst):
    """Generate all perfect matchings of list of items; each matching is list of pairs (unordered)"""
    # recursion: pick first, pair with each other.
    if not lst:
        yield []
        return
    a = lst[0]
    for i in range(1, len(lst)):
        b = lst[i]
        rest = lst[1:i] + lst[i+1:]
        for rest_match in pairings(rest):
            yield [(a,b)] + rest_match

def tournament_rankings(n):
    participants = list(range(2**n))
    # speeds: assign speed = label (higher is faster)
    speeds = participants[:] # same as label
    # tournament rounds: round i from 1..n, points = 2^(n-i)
    # state: dict participant -> score
    from collections import defaultdict
    rankings_set = set()
    # use recursion over rounds
    def rec(round_idx, scores):
        # scores: dict participant -> score
        if round_idx > n:
            # compute ranking: sort participants by descending score, tie by maybe speed? but ties shouldn't happen.
            order = tuple(sorted(participants, key=lambda p: (-scores[p], p))) # p as tie-breaker but no ties expected
            rankings_set.add(order)
            return
        # group participants by current score
        groups = defaultdict(list)
        for p in participants:
            groups[scores[p]].append(p)
        # for each group, must produce a pairing (matching) within group
        # all groups are independent; total matchings are product of matchings per group
        # we need to iterate over all combos.
        # For each group, generate all matchings
        group_matchings_list = [] # list of list of matchings for each group (each matching is list of pairs)
        group_scores = [] # corresponding score key
        for sc, group in groups.items():
            # group size must be even (since number of participants is power of 2, each round splits evenly)
            # generate all pairings of group
            matchings = list(pairings(group))
            group_matchings_list.append(matchings)
            group_scores.append(sc)
        # iterate over cartesian product of matchings across groups
        for combo in itertools.product(*group_matchings_list):
            # combo[i] is matching for group i (group_scores[i] group)
            # compute new scores after this round
            new_scores = scores.copy()
            points = 2**(n-round_idx)
            # process each group matching
            for sc, matching in zip(group_scores, combo):
                for a,b in matching:
                    # determine winner (higher speed)
                    if speeds[a] > speeds[b]:
                        winner = a
                        loser = b
                    else:
                        winner = b
                        loser = a
                    new_scores[winner] += points
                    # loser gets nothing
            rec(round_idx+1, new_scores)
    rec(1, {p:0 for p in participants})
    return rankings_set

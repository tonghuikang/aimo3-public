def all_matchings(lst):
    # lst is tuple of items
    # yields list of pairs (tuple)
    if not lst:
        yield []
        return
    # pick first element and pair with any other
    first = lst[0]
    for i in range(1, len(lst)):
        second = lst[i]
        rest = lst[1:i] + lst[i+1:]
        for sub in all_matchings(rest):
            yield [(first, second)] + sub

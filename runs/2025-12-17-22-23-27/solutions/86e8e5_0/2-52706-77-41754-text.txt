                continue
            # distinct automatically a<b<c
            if best is None or c < best:

    allowed = {}
    for g in range(1, maxD+1):
        for u in range(1, maxD+1):

def inv_mod(a, m):

def M_mod_D(D):
    # compute M = 3^{2025!} mod D
    e, t = factor_three_pow(D)
    # M ≡ 0 mod 3^e
    mod3e = pow(3, e, D) if e>0 else 1  # not needed

    # M ≡ 1 mod t (since t coprime to 3)
    if t == 1:

        # M ≡ 0 mod D

    # Compute solution x ≡ 0 mod 3^e and x ≡ 1 mod t
    # Use CRT: x = 3^e * inv_mod(3^e, t) * 1 (mod D)
    # Since 3^e and t are coprime.
    inv = inv_mod(pow(3, e, t), t)  # compute inverse of 3^e mod t
    x = (pow(3, e) * inv) % D

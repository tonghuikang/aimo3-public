def compute_points(A, B, C):
    # side lengths opposite vertices: a = BC, b = CA, c = AB
    a = sp.sqrt((B[0]-C[0])**2 + (B[1]-C[1])**2)
    b = sp.sqrt((C[0]-A[0])**2 + (C[1]-A[1])**2)
    c = sp.sqrt((A[0]-B[0])**2 + (A[1]-B[1])**2)
    # incenter
    P = a + b + c
    I = (a*A[0] + b*B[0] + c*C[0]) / P, (a*A[1] + b*B[1] + c*C[1]) / P
    # foot D from I onto BC
    # param eq for BC: B + t*(C-B)
    Bv = sp.Matrix(B)
    Cv = sp.Matrix(C)

    d = (mp.mpf(3) - mp.sqrt(5))/2
    r = mp.mpf(r_val)
    # slopes
    mB = (2*d*r) / (d**2 - r**2)
    mC = (2*(d-1)*r) / ((d-1)**2 - r**2)
    # A
    xA = -mC/(mB - mC)
    yA = mB*xA
    # F
    tF = (d + r*mB) / (1 + mB**2)
    xF = tF
    yF = tF*mB
    # E
    tE = ((d-1) + r*mC) / (1 + mC**2)
    xE = 1 + tE
    yE = tE*mC
    # N (foot from D onto EF)
    ux = xE - xF
    uy = yE - yF
    dx = d - xF
    dy = - yF
    tN = (dx*ux + dy*uy) / (ux**2 + uy**2)
    xN = xF + tN*ux
    yN = yF + tN*uy
    # O1, O2
    ox1 = mp.mpf('0.5')
    if abs(yA) < mp.mpf('1e-30'):
        oy1 = mp.mpf('0')
    else:
        oy1 = (xA**2 + yA**2 - 2*ox1*xA) / (2*yA)
    # O2
    detM = (2*xA)*(2*yE) - (2*yA)*(2*xE)
    ox2 = ((xA**2 + yA**2)*(2*yE) - (2*yA)*(xE**2 + yE**2)) / detM
    oy2 = ((2*xA)*(xE**2 + yE**2) - (xA**2 + yA**2)*(2*xE)) / detM
    # Intersection K
    dO = mp.sqrt((ox2-ox1)**2 + (oy2-oy1)**2)
    R1_sq = ox1**2 + oy1**2
    R2_sq = ox2**2 + oy2**2
    a = (R1_sq - R2_sq + dO**2) / (2*dO)
    h_sq = R1_sq - a**2
    h = mp.sqrt(max(h_sq, mp.mpf('0')))
    ex = (ox2-ox1)/dO
    ey = (oy2-oy1)/dO
    px = ox1 + a*ex
    py = oy1 + a*ey
    # two possible K points
    K1x = px + h*(-ey)
    K1y = py + h*ex
    K2x = px - h*(-ey)
    K2y = py - h*ex
    # choose K not equal to A
    def dist_sq(x,y):

        solutions.append((ox, oy, radius, dist_line))

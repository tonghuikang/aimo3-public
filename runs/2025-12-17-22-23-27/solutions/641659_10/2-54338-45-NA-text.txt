def bisect_root(func, a, b, tol=1e-12, max_iter=100):
    fa = func(a); fb = func(b)
    if fa is None or fb is None:
        return None
    if fa * fb > 0:
        # no sign change
        return None
    for _ in range(max_iter):
        m = (a+b)/2
        fm = func(m)
        if fm is None:
            # can't evaluate
            break
        if abs(fm) < tol:
            return m
        if fa * fm <= 0:
            b = m
            fb = fm
        else:
            a = m
            fa = fm
    return (a+b)/2

def circle_center_radius(P1, P2, P3):
    # Compute circumcenter of triangle given three points (x,y) as numpy arrays
    # Solve using perpendicular bisectors
    # Using formula based on determinant
    x1, y1 = P1
    x2, y2 = P2
    x3, y3 = P3
    # Compute circumcenter using linear algebra
    d = 2*(x1*(y2 - y3) + x2*(y3 - y1) + x3*(y1 - y2))
    if abs(d) < 1e-12:
        raise ValueError("Points collinear")
    ux = ((x1**2 + y1**2)*(y2 - y3) + (x2**2 + y2**2)*(y3 - y1) + (x3**2 + y3**2)*(y1 - y2)) / d
    uy = ((x1**2 + y1**2)*(x3 - x2) + (x2**2 + y2**2)*(x1 - x3) + (x3**2 + y3**2)*(x2 - x1)) / d
    center = np.array([ux, uy])
    radius = np.linalg.norm(P1 - center)
    return center, radius

def intersect_circles(c1, r1, c2, r2):
    # Return intersection points of two circles (if any)
    # c1,c2 as numpy arrays
    d = np.linalg.norm(c2 - c1)
    if d > r1 + r2 + 1e-12 or d < abs(r1 - r2) - 1e-12:
        return []  # no intersection
    # Compute a = (r1^2 - r2^2 + d^2) / (2d)
    a = (r1**2 - r2**2 + d**2) / (2*d)
    # Height h = sqrt(r1^2 - a^2)
    h_sq = max(r1**2 - a**2, 0.0)
    h = np.sqrt(h_sq)
    # point P2 = c1 + a*(c2 - c1)/d
    P2 = c1 + a*(c2 - c1)/d
    # offset vector = h * perpendicular
    # perp vector (dx, dy) = (-(c2 - c1)[1], (c2 - c1)[0]) / d
    perp = np.array([-(c2 - c1)[1], (c2 - c1)[0]]) / d
    inter1 = P2 + h*perp
    inter2 = P2 - h*perp
    return [inter1, inter2]

    r=1.0

def reflect_point_across_line(P, L1, L2):
    # reflect point P across line through L1,L2
    # formula using projection
    # line vector
    v = L2 - L1
    v = v / np.linalg.norm(v)
    # projection of P-L1 onto v
    proj = np.dot(P - L1, v) * v
    # perpendicular component
    perp = P - L1 - proj
    # reflection = P - 2*perp
    return P - 2*perp

    # mat: list of lists of mp.mpf, size 4x4
    # Use Gaussian elimination
    n = 4
    M = [list(row) for row in mat]  # copy
    det = mp.mpf(1)

    for i in range(n):

        pivot = i

    # find smallest positive integer m with three distinct positive divisors summing to n
    # m must be >= something. We'll try up to limit.
    for m in range(1, limit+1):
        # compute divisors of m
        divs = []
        i = 1
        while i*i <= m:
            if m % i == 0:
                divs.append(i)

        # get distinct divisors

        for a,b,c in combinations(divs,3):
            if a+b+c == n:

    return None

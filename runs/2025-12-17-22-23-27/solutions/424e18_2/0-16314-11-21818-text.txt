    n = 2**k
    # participants speed: 1..n (fastest = n)
    participants = list(range(1,n+1))
    # simulate all possible pairing choices at each round.
    # The number of possible pairings grows huge quickly.
    # For small k (like k=2 or 3) we can brute force using recursion.
    def rec(round_num, scores, groups):
        # round_num: 1-indexed round number (1..k)
        # scores: dict participant->score (int)
        # groups: list of lists of participants that have same score at start of round
        if round_num > k:
            # final ordering: participants sorted by descending scores (tie-breaking not needed because scores distinct)
            # return tuple of participants sorted by descending scores.
            ordered = sorted(participants, key=lambda p: scores[p], reverse=True)
            return {tuple(ordered)}
        res = set()
        # for each group, we need to generate all pairings (perfect matchings) among members.
        # Since groups are independent, we can generate combinations across groups via Cartesian product.
        # For each group, generate list of possible pairings (list of list of (winner, loser) pairs)
        all_group_pairings = []
        for group in groups:
            # generate all perfect matchings for this group
            # group size is power of 2, may be large. For small k we can brute force.
            # Use recursion to generate matchings.
            def matchings(lst):
                if not lst:
                    yield []
                else:
                    a = lst[0]
                    for i in range(1,len(lst)):
                        b = lst[i]
                        # pair a,b
                        rest = lst[1:i] + lst[i+1:]
                        for rest_match in matchings(rest):
                            yield [(a,b)] + rest_match
            # each matching yields list of pairs (a,b). Winner is faster (higher speed label)
            # We need to consider ordering? For each pair, winner = max(a,b), loser = min(a,b)
            group_matchings = []
            for pairing in matchings(group):
                group_matchings.append(pairing)
            all_group_pairings.append(group_matchings)
        # Now iterate over Cartesian product of group matchings
        for combo in itertools.product(*all_group_pairings):
            # combo is a tuple of matchings for each group
            new_scores = scores.copy()
            # apply winners gain points for this round
            points = 2**(k - round_num)  # round i gives 2^{k-i}
            for grp_match in combo:
                for a,b in grp_match:
                    winner = a if a > b else b
                    loser = b if a > b else a
                    new_scores[winner] += points
            # compute next groups: participants with same score

            score_to_part = {}

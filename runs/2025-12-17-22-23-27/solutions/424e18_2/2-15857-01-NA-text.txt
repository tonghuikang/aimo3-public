        # choose left subset of size half (ordered left vs right)
        # we consider each unordered left subset; right is complement
        for left in combinations(group, half):
            left_set = set(left)
            right = tuple(x for x in group if x not in left_set)
            # recursively generate orderings for left and right
            left_orders = rec(left)
            right_orders = rec(right)
            # combine each pair: order is left order then right order
            for lo in left_orders:
                for ro in right_orders:
                    orderings.append(lo + ro)  # concatenation
        return orderings
    return rec(tuple(elements))

def distinct_orderings(n):
    els = list(range(1, n+1))  # 1 fastest, n slowest
    orders = generate_orderings(els)
    # convert each ordering to tuple
    uniq = set(tuple(o) for o in orders)
    return uniq

def compute_primitive_ratios(limit):
    ratio = {}
    # For each S, compute minimal LCM for sum S (using brute up to S maybe 200).
    # Since S up to 500, brute triple O(S^3) maybe large but okay for 500? Let's try.
    # We'll compute min LCM using brute for S up to 500.
    for S in range(6, limit+1):
        best = None
        for a in range(1, S):
            for b in range(a+1, S-a):
                c = S - a - b
                if c <= b: continue
                # compute LCM
                l = a*b//math.gcd(a,b)
                l = l*c//math.gcd(l,c)
                if best is None or l < best:
                    best = l
        ratio[S] = best / S
    return ratio

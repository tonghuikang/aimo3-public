<|end|><|start|>assistant<|channel|>analysis to=python code<|message|>def m_bruteforce(t, limit=200):
    # compute minimal lcm of three distinct positive ints summing to t
    best = None
    best_trip = None
    for a in range(1, t):
        for b in range(a+1, t):
            c = t - a - b
            if c <= b: # ensure c > b for distinct and > b
                continue
            # distinct: a<b<c
            # compute lcm
            from math import gcd
            l = a*b // gcd(a,b)
            l = l*c // gcd(l,c)
            if best is None or l < best:
                best = l
                best_trip = (a,b,c)
    return best, best_trip

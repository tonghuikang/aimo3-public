def simulate_pairing(pairing):
    # pairing: list of pairs (tuple of 2 IDs)
    # IDs 1..4 with speed ranking: lower number faster (1 fastest)
    # Round1 points = 2
    # Winners get 2 points, losers 0.
    scores = {i:0 for i in range(1,5)}
    winners = []
    losers = []
    for a,b in pairing:
        if a < b: # a faster
            winner, loser = a, b
        else:
            winner, loser = b, a
        scores[winner] += 2
        winners.append(winner)
        losers.append(loser)
    # Round2: winners pair together; losers pair together
    # points = 1
    # winners group:
    a,b = winners
    if a < b:
        winner, loser = a, b
    else:
        winner, loser = b, a
    scores[winner] += 1
    # losers group:
    a,b = losers
    if a < b:
        winner2, loser2 = a, b
    else:
        winner2, loser2 = b, a
    scores[winner2] += 1
    # produce final ordering (by decreasing score, tie break not possible)
    # sort by score descending then maybe by speed? but scores unique
    ordering = sorted(range(1,5), key=lambda i: -scores[i])
    return tuple(ordering), scores

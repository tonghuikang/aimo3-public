def all_matchings(lst):
    # return all perfect matchings of list (list of elements)
    if not lst:
        yield []
        return
    a = lst[0]
    for i in range(1, len(lst)):
        b = lst[i]
        remaining = lst[1:i] + lst[i+1:]
        for rest in all_matchings(remaining):
            yield [(a,b)] + rest

def tournament_outcomes(n):
    # n = exponent, number of participants 2^n
    participants = list(range(2**n))  # speeds order ascending (0 slowest)
    # initial scores all zero
    outcomes = []  # list of final scores dict
    # recursive simulation
    def recurse(scores, round_i):
        # scores: dict participant -> current score
        # round_i: 1-indexed round number
        if round_i > n:
            # final
            outcomes.append(scores.copy())
            return
        # group participants by current score
        groups = {}
        for p, s in scores.items():
            groups.setdefault(s, []).append(p)
        # For each group, we need to consider all possible matchings
        # The total matchings across groups are product of matchings per group
        # We'll generate all possibilities recursively
        # For each group, get list of possible matchings (list of list of pairs)
        group_matchings = {}
        for s, group in groups.items():
            # group size always power of two, we generate all matchings
            group_matchings[s] = list(all_matchings(group))
        # Now iterate over cartesian product of matchings across groups
        # Use recursion
        def assign_group_score(current_scores, groups_items):
            if not groups_items:
                # all groups processed, go to next round
                recurse(current_scores, round_i+1)
                return
            s, matchings = groups_items[0]
            for matching in matchings:
                # copy scores dict
                new_scores = current_scores.copy()
                for a, b in matching:
                    # winner is max speed (larger index)
                    winner = max(a,b)  # since speed ascending index
                    # loser is min
                    # winner gets points: 2^(n - round_i)
                    points = 2**(n - round_i)
                    new_scores[winner] = new_scores[winner] + points
                    # loser stays same
                assign_group_score(new_scores, groups_items[1:])
        assign_group_score(scores, list(group_matchings.items()))

    recurse(initial_scores, 1)

    d = mp.mpf(1)
    f = mp.mpf(k)
    a = d + f
    L = t
    s = a + L
    r = mp.sqrt(L * d * f / s)
    # slopes
    m_B = 2 * d * r / (d**2 - r**2)
    M = 2 * r * f / (f**2 - r**2)
    # A coordinates
    xA = M * a / (m_B + M)
    yA = m_B * xA
    # points
    Bpt = (mp.mpf(0), mp.mpf(0))
    Cpt = (a, mp.mpf(0))
    Apt = (xA, yA)
    # F
    Fpt = (xA/(1+L), yA/(1+L))
    # E
    factor = f/(L+f)
    Ept = (a + factor*(xA - a), factor*yA)
    # D
    Dpt = (d, mp.mpf(0))
    # line EF
    v = sub(Fpt, Ept)
    # N projection of D onto EF
    w = sub(Dpt, Ept)
    t_proj = dot(w, v) / dot(v, v)
    Npt = add(Ept, mul_scalar(t_proj, v))
    # circumcenters
    O1 = circumcenter(Bpt, Apt, Cpt)
    O2 = circumcenter(Apt, Fpt, Ept)
    # K = reflect A across line O1O2
    Kpt = reflect_point_across_line(Apt, O1, sub(O2, O1))
    # K' = reflect K across line EF
    Kprime = reflect_point_across_line(Kpt, Ept, v)
    # Circle through B,K,N: circumcenter O3
    O3 = circumcenter(Bpt, Kpt, Npt)
    # BT = 2 * O3_x (since B is origin)
    BT = 2 * O3[0]
    # CT = a - BT
    CT = a - BT
    # NB distance
    NB = mp.sqrt(Npt[0]**2 + Npt[1]**2)
    # NE distance
    NE = mp.sqrt((Npt[0] - Ept[0])**2 + (Npt[1] - Ept[1])**2)
    # ratio
    ratio = (CT * NB) / (BT * NE)
    return ratio

    ===========

    include in the formula may be given.

    t = phi / 3

    f = phi

    L = t
    s = a + L

    # slopes
    m_B = 2 * d * r / (d**2 - r**2)
    M = 2 * r * f / (f**2 - r**2)
    # A
    xA = M * a / (m_B + M)
    yA = m_B * xA
    A = (xA, yA)
    B = (mp.mpf(0), mp.mpf(0))
    C = (a, mp.mpf(0))
    # F
    F = (xA/(1+L), yA/(1+L))
    # E = C/4 + (3/4)A
    E = (C[0]/4 + 3*xA/4, 3*yA/4)
    # D
    D = (d, mp.mpf(0))
    # N projection of D onto EF
    v = (F[0] - E[0], F[1] - E[1])
    w = (D[0] - E[0], D[1] - E[1])
    t_proj = (w[0]*v[0] + w[1]*v[1]) / (v[0]**2 + v[1]**2)
    N = (E[0] + t_proj * v[0], E[1] + t_proj * v[1])
    # circumcenters O1, O2 numeric using circumcenter_numeric
    O1 = circumcenter_numeric(A, B, C)
    O2 = circumcenter_numeric(A, F, E)
    # line O1O2 direction vector w1 = O2 - O1
    w1 = (O2[0] - O1[0], O2[1] - O1[1])
    # reflect A across line through O1 with direction w1 to get K
    # compute unit direction u = w1 / |w1|
    w1_norm = mp.sqrt(w1[0]**2 + w1[1]**2)
    u = (w1[0]/w1_norm, w1[1]/w1_norm)
    # projection of (A - O1) onto u
    AO1 = (A[0] - O1[0], A[1] - O1[1])
    proj_len = AO1[0]*u[0] + AO1[1]*u[1]
    proj = (O1[0] + proj_len*u[0], O1[1] + proj_len*u[1])
    K = (2*proj[0] - A[0], 2*proj[1] - A[1])
    # Now compute lambda using formula earlier
    Nx, Ny = N
    Kx, Ky = K
    # compute denominator = -Ny*Kx + Nx*Ky
    denom = -Ny*Kx + Nx*Ky
    lambda_val = (Kx**2 + Ky**2) / (2 * denom)
    # O3 = lambda * (-Ny, Nx)
    O3 = (-lambda_val*Ny, lambda_val*Nx)
    # BT = 2 * |O3.x|

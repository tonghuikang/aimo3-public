    # d = BD, e = CD, a = d+e
    a = d+e
    # coordinates
    B = sp.Point(0,0)
    C = sp.Point(a,0)
    D = sp.Point(d,0)
    # incircle center I
    I = sp.Point(d, r)
    # F coordinates
    F = sp.Point(d*(d**2 - r**2)/(d**2 + r**2), 2*d**2*r/(d**2 + r**2))
    # E coordinates
    E = sp.Point(a - e*(e**2 - r**2)/(e**2 + r**2), 2*e**2*r/(e**2 + r**2))
    # Compute A as intersection of lines BF and CE
    lineBF = sp.Line(B, F)
    lineCE = sp.Line(C, E)
    A = lineBF.intersection(lineCE)[0]
    # circumcircle of ABC
    circ_ABC = sp.Circle(A, B, C)
    # circumcircle of AFE
    circ_AFE = sp.Circle(A, F, E)

    inter = circ_ABC.intersection(circ_AFE)

    # N foot from D onto line EF

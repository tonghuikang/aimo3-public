def fibonacci(n):

    if n==0: return 0
    a,b=0,1

def compute_residual(r):
    d = d_value()
    e = 1 - d
    # Points
    # incircle radius r
    # compute coordinates
    Fx = d*(d*d - r*r)/(d*d + r*r)
    Fy = 2*r*d*d/(d*d + r*r)
    Ex = d + (2*e*r*r)/(e*e + r*r)
    Ey = 2*e*e*r/(e*e + r*r)
    # compute A
    # line BF: passes through B (0,0) and F
    # line CE: passes through C (1,0) and E

    t = 1/coeff
    A = (t*Fx, t*Fy)
    # compute K: intersection of circles (A,F,E) and (A,B,C) besides A

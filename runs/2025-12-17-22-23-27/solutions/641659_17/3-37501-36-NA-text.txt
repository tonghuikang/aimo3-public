def compute_points(A,B,C):
    # points as sympy 2D vectors
    # return D,E,F incircle contact points, etc.
    # Compute incircle center I and radius r:
    # Sides: a = BC length, b = CA, c = AB
    # Use formula for incenter coordinates: weighted sum of vertices by side lengths: I = (a*A + b*B + c*C) / (a+b+c)
    a = B.distance(C)
    b = C.distance(A)
    c = A.distance(B)
    I = (a*A + b*B + c*C) / (a+b+c)
    # incircle radius r = distance from I to BC line:
    r = sp.Line(B,C).distance(I)
    # Contact points:
    # D is foot from I onto BC
    D = sp.Line(B,C).projection(I)
    # E foot onto CA
    E = sp.Line(C,A).projection(I)
    # F foot onto AB
    F = sp.Line(A,B).projection(I)
    return I, D, E, F, a,b,c

def circle_from_three_points(P1, P2, P3):
    # returns center and radius
    c = sp.Circle(P1, P2, P3)
    return c.center, c.radius

def intersect_circles(c1_center, c1_rad, c2_center, c2_rad):
    # returns intersection points of circles
    # Use sympy geometry circle intersection:
    circ1 = sp.Circle(c1_center, c1_rad)
    circ2 = sp.Circle(c2_center, c2_rad)
    inter = circ1.intersection(circ2)
    return inter

def second_intersection_AFE_with_circumcircle(A,B,C):

    # The intersection includes A itself (common point)

def reflect_point_across_line(P, L):
    # reflect point across line L
    # L is sympy Line
    # Use projection onto line and then reflect
    proj = L.projection(P)
    # reflection: P' = 2*proj - P
    return sp.Point(2*proj.x - P.x, 2*proj.y - P.y)

    fa = func(a)
    fb = func(b)
    if fa is None or fb is None:
        raise ValueError("None in endpoints")
    if fa*fb > 0:
        raise ValueError("No sign change")
    for i in range(max_iter):

        fc = func(c)

            # shrink interval a bit
            c = (a+b)/2
            continue
        if abs(fc) < tol:
            return c
        if fa*fc < 0:
            b = c
            fb = fc
        else:
            a = c
            fa = fc
    return (a+b)/2

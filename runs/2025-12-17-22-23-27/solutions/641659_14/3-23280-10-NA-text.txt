def compute_geometry_mp(b,c,r):
    B = mp.matrix([-b, -r])
    C = mp.matrix([c, -r])
    D = mp.matrix([0, -r])
    u = c
    v = b
    # E
    xE = 2*u*r**2/(r**2 + u**2)
    yE = r*(u**2 - r**2)/(r**2 + u**2)
    # F
    xF = -2*v*r**2/(r**2 + v**2)
    yF = r*(v**2 - r**2)/(r**2 + v**2)
    E = mp.matrix([xE,yE])
    F = mp.matrix([xF,yF])
    # A
    Delta = xF*yE - xE*yF
    xA = r**2 * (yE - yF) / Delta
    yA = r**2 * (xF - xE) / Delta
    A = mp.matrix([xA, yA])
    # circumcenter
    def circumcenter(p1,p2,p3):
        dx21 = p2[0] - p1[0]
        dy21 = p2[1] - p1[1]
        dx31 = p3[0] - p1[0]
        dy31 = p3[1] - p1[1]
        RHS1 = (p2[0]**2 + p2[1]**2 - p1[0]**2 - p1[1]**2)/2
        RHS2 = (p3[0]**2 + p3[1]**2 - p1[0]**2 - p1[1]**2)/2
        det = dx21*dy31 - dy21*dx31
        Ox = (RHS1*dy31 - dy21*RHS2) / det
        Oy = (dx21*RHS2 - RHS1*dx31) / det
        return mp.matrix([Ox, Oy])
    O_ABC = circumcenter(A,B,C)
    R_ABC_sq = (A[0]-O_ABC[0])**2 + (A[1]-O_ABC[1])**2
    O_AEF = circumcenter(A,E,F)
    R_AEF_sq = (A[0]-O_AEF[0])**2 + (A[1]-O_AEF[1])**2
    # Intersection of circles
    Ox1, Oy1 = O_ABC[0], O_ABC[1]
    Ox2, Oy2 = O_AEF[0], O_AEF[1]
    Acoef = Ox2 - Ox1
    Bcoef = Oy2 - Oy1
    C0 = (R_ABC_sq - R_AEF_sq - Ox1**2 + Ox2**2 - Oy1**2 + Oy2**2)/2
    intersections = []
    if abs(Bcoef) > 1e-12:
        # Solve using quadratic in x.
        Bc2 = Bcoef**2
        # Quadratic coefficients
        a = Bc2 + Acoef**2
        b = -2*Bc2*Ox1 - 2*Acoef*(C0 - Bcoef*Oy1)
        c = Bc2*Ox1**2 + (C0 - Bcoef*Oy1)**2 - Bc2*R_ABC_sq
        disc = b*b - 4*a*c
        if disc < 0:
            # no intersection
            return None
        sqrt_disc = mp.sqrt(disc)
        x1 = (-b + sqrt_disc) / (2*a)
        x2 = (-b - sqrt_disc) / (2*a)
        y1 = (C0 - Acoef*x1) / Bcoef
        y2 = (C0 - Acoef*x2) / Bcoef
        intersections.append(mp.matrix([x1,y1]))
        intersections.append(mp.matrix([x2,y2]))
    else:
        if abs(Acoef) < 1e-12:
            return None
        x_val = C0 / Acoef
        val = R_ABC_sq - (x_val - Ox1)**2
        if val < 0:
            return None
        sqrt_val = mp.sqrt(val)
        y1 = Oy1 + sqrt_val
        y2 = Oy1 - sqrt_val
        intersections.append(mp.matrix([x_val,y1]))
        intersections.append(mp.matrix([x_val,y2]))
    # Choose K not equal A
    K = None
    for P in intersections:
        if mp.sqrt((P[0]-A[0])**2 + (P[1]-A[1])**2) > 1e-8:
            K = P
            break
    if K is None:
        return None
    # Reflection across line EF
    vEF = mp.matrix([F[0]-E[0], F[1]-E[1]])
    t = (dot(K - E, vEF)) / dot(vEF, vEF)
    proj = mp.matrix([E[0] + vEF[0]*t, E[1] + vEF[1]*t])
    Kp = mp.matrix([2*proj[0] - K[0], 2*proj[1] - K[1]])
    # Foot N from D onto line EF
    tN = (dot(D - E, vEF)) / dot(vEF, vEF)
    N = mp.matrix([E[0] + vEF[0]*tN, E[1] + vEF[1]*tN])
    # O1 circumcenter of B,K,Kp
    O1 = circumcenter(B, K, Kp)
    R1_sq = (B[0]-O1[0])**2 + (B[1]-O1[1])**2
    diff = (N[0]-O1[0])**2 + (N[1]-O1[1])**2 - R1_sq
    return diff, {'B':B,'C':C,'A':A,'D':D,'E':E,'F':F,'K':K,'Kp':Kp,'N':N,'O1':O1}

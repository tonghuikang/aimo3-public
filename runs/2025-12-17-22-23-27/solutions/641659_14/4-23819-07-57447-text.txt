def circumcircle(p1,p2,p3):
    # returns center and radius
    (x1,y1),(x2,y2),(x3,y3) = p1,p2,p3
    D = 2*(x1*(y2-y3) + x2*(y3-y1) + x3*(y1-y2))
    if abs(D) < 1e-12:
        raise ValueError("Collinear")
    ux = ((x1**2+y1**2)*(y2-y3) + (x2**2+y2**2)*(y3-y1) + (x3**2+y3**2)*(y1-y2)) / D
    uy = ((x1**2+y1**2)*(x3-x2) + (x2**2+y2**2)*(x1-x3) + (x3**2+y3**2)*(x2-x1)) / D
    R = math.hypot(ux-x1, uy-y1)
    return (ux,uy),R

def intersect_circles(c1,r1,c2,r2):
    # returns intersection points of two circles
    (x0,y0) = c1
    (x1,y1) = c2
    dx, dy = x1-x0, y1-y0
    d = math.hypot(dx, dy)
    if d > r1+r2 or d < abs(r1-r2):
        return [] # no intersection or one inside the other
    # find a = (r1^2 - r2^2 + d^2) / (2d)
    a = (r1*r1 - r2*r2 + d*d) / (2*d)
    # h = sqrt(r1^2 - a^2)
    h_sq = r1*r1 - a*a
    if h_sq < 0: h = 0
    else: h = math.sqrt(h_sq)
    # point p2 = p0 + a * (dx/d, dy/d)
    xm = x0 + a*dx/d
    ym = y0 + a*dy/d
    # intersection points
    rx = -dy * (h/d)
    ry = dx * (h/d)
    p_int1 = (xm+rx, ym+ry)
    p_int2 = (xm-rx, ym-ry)
    return [p_int1, p_int2]

def foot_of_perpendicular(P, L1, L2):
    # foot from point P to line through L1 L2
    (x0,y0) = P
    (x1,y1) = L1
    (x2,y2) = L2
    # line direction
    dx, dy = x2-x1, y2-y1
    # projection
    t = ((x0-x1)*dx + (y0-y1)*dy) / (dx*dx+dy*dy)
    foot = (x1 + t*dx, y1 + t*dy)
    return foot

def reflect_point_across_line(P, L1, L2):
    # reflect point P across line L1-L2
    foot = foot_of_perpendicular(P, L1, L2)
    # vector from P to foot
    vx, vy = foot[0]-P[0], foot[1]-P[1]
    # reflected point = P + 2*(foot-P)
    return (P[0] + 2*vx, P[1] + 2*vy)

def is_cyclic_quad(p1,p2,p3,p4, tol=1e-6):
    # check determinant condition
    def row(p):

    M = sp.Matrix([row(p1), row(p2), row(p3), row(p4)])
    det = float(M.det())
    return abs(det) < tol

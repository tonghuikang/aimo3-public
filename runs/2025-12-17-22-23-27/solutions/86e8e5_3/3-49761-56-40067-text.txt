    base = {}
    # find primitive pairs (s,r) with gcd(s,r)=1 and u>r, and D = s+r+u
    for s in range(2,limit_s+1):
        for r in range(s+1,limit_s+1):
            if math.gcd(s,r) != 1:
                continue
            u = s*r//math.gcd(s,r)
            if u <= r:
                continue
            D = s+r+u
            # skip if D is multiple of a smaller base (i.e., D % base_D == 0)
            # But we will collect all D and later filter.
            base.setdefault(D, (s,r,u))
    # Now filter base D's such that D not divisible by any other D' smaller than D that also in base
    base_Ds = sorted(base.keys())
    primitive_Ds = []
    for D in base_Ds:
        is_multiple = False
        for D0 in primitive_Ds:
            if D % D0 == 0:
                is_multiple = True
                break
        if not is_multiple:
            primitive_Ds.append(D)
    # Build list of (D, u, ratio)
    prim = []
    for D in primitive_Ds:
        s,r,u = base[D]
        prim.append((D, u, u/D))
    prim.sort(key=lambda x: x[2])  # sort by ratio
    return prim

    # fact is dict prime->exponent for odd primes (excluding 3)
    primes = list(fact.keys())
    exps = [list(range(e+1)) for e in fact.values()]

    for combo in product(*exps):

        for p,exp in zip(primes, combo):
            d *= p**exp
        divisors.append(d)
    return sorted(divisors)

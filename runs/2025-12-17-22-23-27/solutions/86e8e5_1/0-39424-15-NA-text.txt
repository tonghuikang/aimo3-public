    # brute force up to some limit, increase as needed

        # compute divisors of X

        i=1
        while i*i<=X:
            if X%i==0:

        if len(divs)>=3:
            # check all triples of distinct
            from itertools import combinations
            for a,b,c in combinations(divs,3):
                if a+b+c==N:
                    return X
        X+=1
        if X>limit:

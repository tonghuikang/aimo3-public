def triangle_points(a,b,c):
    # B = (0,0), C = (a,0)
    # compute A = (h, v) where h = (c^2 - b^2 + a^2)/(2a)
    # ensure triangle inequality b + c > a, etc.
    h = (c**2 - b**2 + a**2) / (2*a)
    # v = sqrt(c^2 - h^2)
    temp = c**2 - h**2
    if temp < 0:
        return None
    v = math.sqrt(temp)
    A = (h, v)
    B = (0.0, 0.0)
    C = (a, 0.0)
    return A, B, C

def incircle_data(A,B,C):
    # compute side lengths a,b,c as per standard notation: a = BC, b = CA, c = AB
    a = math.dist(B,C)
    b = math.dist(C,A)
    c = math.dist(A,B)
    s = (a+b+c)/2
    # area via Heron
    area = math.sqrt(max(s*(s-a)*(s-b)*(s-c),0.0))
    r = area/s if s != 0 else 0
    # incircle center coordinates: (x,y) where D on BC at BD = s - b
    BD = s - b
    # D coordinate
    D = (BD, 0.0)
    I = (BD, r)
    return {"a":a,"b":b,"c":c,"s":s,"area":area,"r":r,"I":I,"D":D}

def foot_of_perp(P, Q, R):
    # foot of perpendicular from point P to line QR
    # Q,R are two points on line
    # returns foot point H
    # param: compute projection of vector QP onto QR then subtract
    # Using vector math: foot = Q + ((P-Q)·(R-Q) / |R-Q|^2) * (R-Q)
    # But that's projection onto line, not foot of perpendicular.
    # Actually foot of perpendicular is projection onto line: H = Q + ((P-Q)·(R-Q) / |R-Q|^2)*(R-Q)
    # That is foot (closest point).
    Qx,Qy = Q
    Rx,Ry = R
    Px,Py = P
    vx = Rx - Qx
    vy = Ry - Qy
    wx = Px - Qx
    wy = Py - Qy
    dot = wx*vx + wy*vy
    vv = vx*vx + vy*vy
    if vv == 0:
        return Q
    t = dot/vv
    Hx = Qx + t*vx
    Hy = Qy + t*vy
    return (Hx,Hy)

def reflect_point_across_line(P, Q, R):
    # reflect point P across line QR
    # Compute foot H of perpendicular, then reflect: P' = (2*H - P)
    H = foot_of_perp(P,Q,R)
    Px,Py = P
    Hx,Hy = H
    return (2*Hx - Px, 2*Hy - Py)

def circumcenter(P, Q, R):
    # Compute circumcenter of triangle PQR
    # Use perpendicular bisectors intersection
    (x1,y1) = P
    (x2,y2) = Q
    (x3,y3) = R
    # Compute using formula for circumcenter coordinates:
    d = 2*(x1*(y2-y3) + x2*(y3-y1) + x3*(y1-y2))
    if abs(d) < 1e-12:
        return None
    ux = ((x1**2 + y1**2)*(y2-y3) + (x2**2+y2**2)*(y3-y1) + (x3**2+y3**2)*(y1-y2)) / d
    uy = ((x1**2 + y1**2)*(x3-x2) + (x2**2+y2**2)*(x1-x3) + (x3**2+y3**2)*(x2-x1)) / d
    return (ux, uy)

def circumradius(P,Q,R):
    # radius = a*b*c/(4*area)
    a = math.dist(Q,R)
    b = math.dist(R,P)
    c = math.dist(P,Q)
    s = (a+b+c)/2
    area = math.sqrt(max(s*(s-a)*(s-b)*(s-c),0.0))
    if area == 0:
        return None
    return a*b*c/(4*area)

def circle_intersections(c1, r1, c2, r2):
    # return intersection points of two circles (as list of two points), may have 0,1,2 solutions
    x0,y0 = c1
    x1,y1 = c2
    dx = x1 - x0
    dy = y1 - y0
    d = math.hypot(dx, dy)
    # no intersection or one
    if d > r1+r2+1e-9 or d < abs(r1-r2)-1e-9:
        return []
    # one or two
    # distance from c1 to line joining intersection points
    # a = (r1^2 - r2^2 + d^2) / (2d)
    a = (r1**2 - r2**2 + d**2) / (2*d) if d != 0 else 0
    # height from point a to intersection points
    h_sq = r1**2 - a**2
    if h_sq < -1e-9:
        return []
    h = math.sqrt(max(h_sq,0.0))
    # point p2 = c1 + a * (c2 - c1)/d
    xm = x0 + a * dx / d if d != 0 else x0
    ym = y0 + a * dy / d if d != 0 else y0
    # intersection points
    rx = -dy * (h / d) if d != 0 else 0
    ry = dx * (h / d) if d != 0 else 0
    P1 = (xm + rx, ym + ry)
    P2 = (xm - rx, ym - ry)
    # if h~0, they are same point
    if h < 1e-12:
        return [P1]
    else:
        return [P1, P2]

def is_concyclic(Ps):
    # check if four points are concyclic: compute pairwise distances to circumcenter from three points, see if fourth satisfies radius equality
    # Use determinant test: For points (xi, yi), they lie on a circle if the determinant of 
    # |xi^2+yi^2, xi, yi, 1| is zero for any four points. For four points, compute determinant of 4x4 matrix. 
    # For numeric test, compute circle through three points, check distance of fourth.
    if len(Ps) != 4:
        return False
    # compute circumcenter for first three points
    O = circumcenter(Ps[0], Ps[1], Ps[2])
    if O is None:
        return False
    R = math.dist(O, Ps[0])
    return abs(math.dist(O, Ps[3]) - R) < 1e-6

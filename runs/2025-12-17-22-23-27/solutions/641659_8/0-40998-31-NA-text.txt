        return []
    if d > r1 + r2 + 1e-12 or d < abs(r1 - r2) - 1e-12:
        return []
    # a
    a = (r1**2 - r2**2 + d**2) / (2*d)
    # h^2
    h2 = r1**2 - a**2
    if h2 < -1e-12:
        return []
    h = math.sqrt(max(0.0, h2))
    # point P
    xm = x1 + a * dx / d
    ym = y1 + a * dy / d
    # intersection points
    rx = -dy * (h / d)
    ry = dx * (h / d)
    p1 = (xm + rx, ym + ry)
    p2 = (xm - rx, ym - ry)
    return [p1, p2]

def reflection_point(P, line_pt1, line_pt2):
    # reflect point P across line defined by two points
    # Using formula
    x0,y0 = P
    x1,y1 = line_pt1
    x2,y2 = line_pt2
    # line equation ax+by+c=0
    a = y2 - y1
    b = x1 - x2
    c = x2*y1 - x1*y2
    d = (a*x0 + b*y0 + c) / (a*a + b*b)
    xr = x0 - 2*a*d
    yr = y0 - 2*b*d
    return (xr, yr)

def cyclic_det(P, Q, R, S):
    # determinant condition for four points concyclic
    # using matrix as before
    M = np.array([
        [P[0], P[1], P[0]**2 + P[1]**2, 1.0],
        [Q[0], Q[1], Q[0]**2 + Q[1]**2, 1.0],
        [R[0], R[1], R[0]**2 + R[1]**2, 1.0],
        [S[0], S[1], S[0]**2 + S[1]**2, 1.0],
    ], dtype=float)
    return np.linalg.det(M)

def compute_ratio_and_condition_numeric(n, c):
    d1 = fibonacci(n)    # BD
    d2 = fibonacci(n+1)  # CD
    a = d1 + d2          # BC
    b = c + (d2 - d1)    # CA
    # Compute coordinates
    xA = (c*c - b*b + a*a) / (2*a)
    # ensure valid
    if c*c - xA*xA <= 0:
        return None
    yA = math.sqrt(c*c - xA*xA)
    # incircle radius
    # r = sqrt((c - d1) * d1 * d2 / (c + d2))
    # ensure positive under sqrt
    if c - d1 <= 0 or c + d2 <= 0:
        return None
    r = math.sqrt((c - d1) * d1 * d2 / (c + d2))
    # Points:
    B = (0.0,0.0)
    C = (a,0.0)
    I = (d1, r)
    # F and E:
    Fx = d1 * xA / c
    Fy = d1 * yA / c
    Ex = a + (xA - a) * d2 / b
    Ey = yA * d2 / b
    # line EF
    # foot N from D onto EF:
    D = (d1, 0.0)
    # vector EF
    vEF = (Ex - Fx, Ey - Fy)
    vFD = (D[0] - Fx, D[1] - Fy)
    # projection parameter t
    dot = vFD[0]*vEF[0] + vFD[1]*vEF[1]
    len2 = vEF[0]**2 + vEF[1]**2
    if len2 == 0:
        return None
    t_proj = dot / len2
    Nx = Fx + t_proj * vEF[0]
    Ny = Fy + t_proj * vEF[1]
    # NB, NE
    NB = math.hypot(Nx, Ny)
    NE = math.hypot(Nx - Ex, Ny - Ey)
    # Circumcenter of ABC:
    ox = a / 2.0
    # compute oy:
    # oy = (-2*ox*xA + xA^2 + yA^2) / (2*yA)
    # If yA=0 (degenerate), skip.
    if abs(yA) < 1e-12:
        return None
    oy = (-2*ox*xA + xA*xA + yA*yA) / (2*yA)
    O_ABC = (ox, oy)
    R_ABC = math.hypot(ox, oy)  # distance to B
    # Circumcenter of AEF:
    O_AEF = circumcenter((xA, yA), (Ex, Ey), (Fx, Fy))
    if O_AEF is None:
        return None
    ox2, oy2 = O_AEF
    R_AEF = math.hypot(ox2 - xA, oy2 - yA)
    # Intersection of circles O_ABC,R_ABC and O_AEF,R_AEF
    inters = circle_intersections(O_ABC, R_ABC, O_AEF, R_AEF)
    # Choose K not equal to A (tolerance)
    K = None
    for pt in inters:
        if math.hypot(pt[0] - xA, pt[1] - yA) > 1e-6:
            K = pt
            break
    if K is None:
        return None
    Kx,Ky = K
    # K' reflection across EF
    Kp = reflection_point((Kx,Ky), (Fx,Fy), (Ex,Ey))
    Kpx, Kpy = Kp
    # cyclic condition determinant
    det_val = cyclic_det(B, (Nx, Ny), (Kx,Ky), (Kpx, Kpy))
    # compute O_T (center of circle through B and K tangent to BN)
    # line BN direction: N - B = (Nx, Ny)
    # w1 = (-Ny, Nx)
    # midpoint of BK: (Kx/2, Ky/2)
    # w2 = (-Ky, Kx)
    # Solve t*(-Ny) = Kx/2 + s*(-Ky); t*Nx = Ky/2 + s*Kx
    A_mat = np.array([[-Ny, Ky],
                      [Nx, -Kx]], dtype=float)
    b_vec = np.array([Kx/2.0, Ky/2.0], dtype=float)
    try:
        sol = np.linalg.solve(A_mat, b_vec)
    except np.linalg.LinAlgError:
        return None
    t, s = sol
    OTx = -Ny * t
    OTy = Nx * t
    BT = 2 * OTx
    CT = a - BT
    if BT <= 0 or BT >= a:
        # invalid intersection maybe
        return None
    # compute ratio
    ratio = (CT * NB) / (BT * NE)
    return {
        'ratio': ratio,
        'det': det_val,
        'c': c,
        'NB': NB,
        'NE': NE,
        'BT': BT,
        'CT': CT,
        'K': (Kx,Ky),
        'Kp': (Kpx,Kpy),
        'N': (Nx,Ny),
        'O_T': (OTx, OTy)
    }

def find_c_roots_for_n(n, c_min_factor=1.01, c_max_factor=5.0, steps=400):
    d1 = fibonacci(n)
    # c must be > d1
    c_min = d1 * c_min_factor
    c_max = d1 * c_max_factor
    prev_val = None
    prev_c = None
    roots = []
    for i in range(steps+1):
        c = c_min + (c_max - c_min) * i / steps
        res = compute_ratio_and_condition_numeric(n, c)
        if res is None:
            continue
        det = res['det']
        # Look for sign changes
        if prev_val is not None and det * prev_val < 0:
            # root between prev_c and c
            lo = prev_c
            hi = c
            # bisect refine
            for _ in range(50):
                mid = (lo + hi) / 2
                res_mid = compute_ratio_and_condition_numeric(n, mid)
                if res_mid is None:
                    lo = mid
                    continue
                dmid = res_mid['det']
                if abs(dmid) < 1e-12:
                    lo = hi = mid
                    break
                if det * dmid <= 0:
                    hi = mid
                    det = dmid
                else:
                    lo = mid
            root_c = (lo + hi) / 2
            roots.append(root_c)
        prev_val = det
        prev_c = c
    # compute ratios at roots
    solutions = []
    for rc in roots:
        res = compute_ratio_and_condition_numeric(n, rc)
        if res:
            solutions.append((rc, res['ratio']))
    return solutions

    # compute coordinates B(0,0), C(a,0), find A coordinates using sides lengths
    B = sp.Point(0,0)
    C = sp.Point(a,0)
    # Solve for A coordinates (x,y) such that distances to B is c, to C is b.
    x,y = sp.symbols('x y', real=True)
    solutions = sp.solve([ (x-0)**2 + (y-0)**2 - c**2,
                          (x-a)**2 + y**2 - b**2 ], [x,y])
    # there should be two solutions symmetric in y (positive and negative)
    # choose positive y.
    sol = None
    for s_ in solutions:
        if s_[1] > 0:
            sol = s_
            break
    if sol is None:
        sol = solutions[0]
    A = sp.Point(sol[0], sol[1])
    # Compute incircle radius and center: need incenter I.
    # Incenter can be computed via weighted average of vertices by side lengths a,b,c.
    # coordinates: (ax_A + bx_B + cx_C) / (a+b+c)? Actually formula: I = (a*A + b*B + c*C) / (a+b+c) where a=BC, etc.
    # Using standard notation a=BC, b=CA, c=AB.
    a_len = a
    b_len = b
    c_len = c
    # incenter coordinates:
    I = sp.Point((a_len*A.x + b_len*B.x + c_len*C.x) / (a_len+b_len+c_len),
                (a_len*A.y + b_len*B.y + c_len*C.y) / (a_len+b_len+c_len))
    # incircle radius: distance from I to BC (x-axis) = |I.y|
    r = abs(I.y)
    # D is foot on BC at x-coordinate equal to distance BD = s - b =? Actually BD = s - b = s - b_len =?
    # But from incircle contact property: BD = s - b_len = s - b_len = F_n by definition.
    # But we know BD = F_n, which is a-b? Actually BD = s - b. Compute BD = s - b_len
    BD = s - b_len
    D = sp.Point(BD, 0)
    # Compute E and F: using distances CE = s - a = t, BF = t.
    t = s - a_len
    # F on AB: param t/c
    # vector AB = A - B
    AB = A - B
    F = sp.Point(B.x + (t/c_len)*AB.x, B.y + (t/c_len)*AB.y)
    # E on AC: param t/b
    AC = A - C
    E = sp.Point(C.x + (t/b_len)*AC.x, C.y + (t/b_len)*AC.y)
    # compute circles etc.
    circ_ABC = sp.Circle(A,B,C)
    circ_AFE = sp.Circle(A,F,E)
    inter = circ_ABC.intersection(circ_AFE)
    K = None
    for pt in inter:
        if not pt.equals(A):
            K = pt
    line_EF = sp.Line(E,F)
    Kprime = K.reflect(line_EF)
    N = line_EF.projection(D)
    # cyclic condition check:
    circ_BKN = sp.Circle(B,K,N)
    cyclic_val = sp.simplify(Kprime.distance(circ_BKN.center) - circ_BKN.radius)
    # circle tangent at B to BN through B and K
    line_BN = sp.Line(B,N)
    perp_B = line_BN.perpendicular_line(B)
    mid_BK = sp.Point((B.x+K.x)/2, (B.y+K.y)/2)
    vec_BK = sp.Point(K.x - B.x, K.y - B.y)
    perp_dir = sp.Point(-vec_BK.y, vec_BK.x)
    perp_mid_BK = sp.Line(mid_BK, mid_BK + perp_dir)
    Oprime = perp_B.intersection(perp_mid_BK)[0]
    rprime = Oprime.distance(B)
    circle_Oprime = sp.Circle(Oprime, rprime)
    line_BC = sp.Line(B,C)
    inter_T = circle_Oprime.intersection(line_BC)
    T = None
    for pt in inter_T:
        if not pt.equals(B):
            T = pt
    # compute ratio
    CT = C.distance(T)
    BT = B.distance(T)
    NB = N.distance(B)
    NE = N.distance(E)
    R = sp.simplify(CT*NB/(BT*NE))
    return {
        "A":A,"B":B,"C":C,"I":I,"D":D,"E":E,"F":F,"K":K,"Kprime":Kprime,"N":N,"T":T,
        "cyclic_val": cyclic_val,
        "R":sp.nsimplify(R, [sp.sqrt(5)]),
        "R_approx": float(R.evalf())
    }

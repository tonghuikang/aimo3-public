from typing import Tuple, List

def circle_center_radius(p1: Tuple[float,float], p2: Tuple[float,float], p3: Tuple[float,float]) -> Tuple[Tuple[float,float], float]:
    # Compute circumcenter of three points
    (x1,y1),(x2,y2),(x3,y3)=p1,p2,p3
    # Using perpendicular bisector intersection
    d = 2*(x1*(y2-y3) + x2*(y3-y1) + x3*(y1-y2))
    if abs(d) < 1e-12:
        raise ValueError("Collinear points")
    ux = ((x1**2+y1**2)*(y2-y3) + (x2**2+y2**2)*(y3-y1) + (x3**2+y3**2)*(y1-y2))/d
    uy = ((x1**2+y1**2)*(x3-x2) + (x2**2+y2**2)*(x1-x3) + (x3**2+y3**2)*(x2-x1))/d
    r = math.hypot(ux-x1, uy-y1)
    return (ux, uy), r

def circle_intersections(c1: Tuple[float,float], r1: float, c2: Tuple[float,float], r2: float) -> List[Tuple[float,float]]:
    (x0,y0), (x1,y1) = c1, c2
    dx = x1-x0
    dy = y1-y0
    d = math.hypot(dx,dy)
    # No intersections
    if d > r1 + r2 + 1e-12 or d < abs(r1 - r2) - 1e-12:
        return []
    # Single tangent
    if d < 1e-12 and abs(r1 - r2) < 1e-12:
        return []  # infinite intersections
    # Intersection points
    a = (r1**2 - r2**2 + d**2) / (2*d)
    h_sq = r1**2 - a**2
    if h_sq < -1e-12:
        return []
    h = math.sqrt(max(h_sq,0.0))
    xm = x0 + a*dx/d
    ym = y0 + a*dy/d
    # Intersection points
    rx = -dy * (h/d)
    ry = dx * (h/d)
    p1 = (xm + rx, ym + ry)
    p2 = (xm - rx, ym - ry)
    if h < 1e-12:
        return [p1]  # single point
    else:
        return [p1,p2]

def foot_of_perpendicular(point: Tuple[float,float], line_pt1: Tuple[float,float], line_pt2: Tuple[float,float]) -> Tuple[float,float]:
    # foot from point to line defined by two points
    (x0,y0) = point
    (x1,y1) = line_pt1
    (x2,y2) = line_pt2
    # line direction
    dx = x2-x1
    dy = y2-y1
    # compute projection
    t = ((x0-x1)*dx + (y0-y1)*dy) / (dx*dx + dy*dy)
    # foot
    xf = x1 + t*dx
    yf = y1 + t*dy
    return (xf,yf)

def reflect_point_across_line(P: Tuple[float,float], A: Tuple[float,float], B: Tuple[float,float]) -> Tuple[float,float]:
    # reflect point P across line AB
    # compute foot of perpendicular from P to line AB
    foot = foot_of_perpendicular(P, A, B)
    # reflection: P' = 2*foot - P
    return (2*foot[0] - P[0], 2*foot[1] - P[1])

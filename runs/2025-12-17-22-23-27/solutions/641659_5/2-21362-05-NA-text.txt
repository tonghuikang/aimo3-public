    x, y = sp.symbols('x y')
    sol_F = sp.solve([x**2 + y**2 - d**2, d*x + r*y - d**2], [x, y])
    # Choose solution with y>0 presumably.
    sol_F = [s for s in sol_F if s[1] > 0][0]  # take positive y
    F = sp.Point(sol_F[0], sol_F[1])

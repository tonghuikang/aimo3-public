def circumcenter(p1: Tuple[float,float], p2: Tuple[float,float], p3: Tuple[float,float]):
    x1,y1 = p1
    x2,y2 = p2
    x3,y3 = p3
    D = 2*(x1*(y2 - y3) + x2*(y3 - y1) + x3*(y1 - y2))
    if abs(D) < 1e-12:
        raise ValueError("Collinear points")
    Ux = ((x1**2 + y1**2)*(y2 - y3) + (x2**2 + y2**2)*(y3 - y1) + (x3**2 + y3**2)*(y1 - y2)) / D
    Uy = ((x1**2 + y1**2)*(x3 - x2) + (x2**2 + y2**2)*(x1 - x3) + (x3**2 + y3**2)*(x2 - x1)) / D
    R = math.hypot(Ux - x1, Uy - y1)
    return (Ux, Uy), R

def circle_intersections(c1, r1, c2, r2):
    # Returns list of intersection points (0,1,2).
    (x1,y1) = c1
    (x2,y2) = c2
    dx = x2 - x1
    dy = y2 - y1
    d = math.hypot(dx, dy)
    # No intersection
    if d > r1 + r2 + 1e-12: # too far apart
        return []
    if d < abs(r1 - r2) - 1e-12: # one inside the other
        return []
    if d == 0 and abs(r1 - r2) < 1e-12:
        # circles coincident
        return []
    # distance from c1 to point2 (midpoint base)
    a = (r1**2 - r2**2 + d**2) / (2*d)
    # height from base to intersection points
    h_sq = max(r1**2 - a**2, 0.0)
    h = math.sqrt(h_sq)
    # base point
    xm = x1 + a * dx / d
    ym = y1 + a * dy / d
    # intersection points
    rx = -dy * (h / d)
    ry = dx * (h / d)
    p1 = (xm + rx, ym + ry)
    p2 = (xm - rx, ym - ry)
    if h < 1e-12:
        return [p1]
    else:
        return [p1, p2]

def foot_of_perpendicular(P, A, B):
    # foot of perpendicular from P to line AB
    # A,B points
    (x0, y0) = P
    (x1, y1) = A
    (x2, y2) = B
    # line AB vector
    vx = x2 - x1
    vy = y2 - y1
    # compute projection scalar t = ((P-A)·(B-A))/|B-A|^2
    t = ((x0 - x1)*vx + (y0 - y1)*vy) / (vx*vx + vy*vy)
    # foot
    xf = x1 + t*vx
    yf = y1 + t*vy
    return (xf, yf)

def reflect_point_across_line(P, A, B):
    # reflect point P across line AB
    foot = foot_of_perpendicular(P, A, B)
    # P' = 2*foot - P
    return (2*foot[0] - P[0], 2*foot[1] - P[1])

def find_roots_for_d(d, num_samples=1000):
    lo = 1e-9
    hi = min(d, 1-d) - 1e-9
    # sample points
    rs = [lo + (hi - lo) * i / num_samples for i in range(num_samples + 1)]
    vals = [residual_for_d_r(d, r) for r in rs]
    roots = []
    for i in range(num_samples):
        v1, v2 = vals[i], vals[i+1]
        if v1 == 0:
            roots.append(rs[i])
        elif v1*v2 < 0:
            # bracket found
            a_br, b_br = rs[i], rs[i+1]
            fa, fb = v1, v2
            # bisection
            for _ in range(60):
                mid = (a_br + b_br) / 2
                fm = residual_for_d_r(d, mid)
                if fm == 0:
                    a_br = b_br = mid
                    break
                if fa*fm < 0:
                    b_br = mid
                    fb = fm
                else:
                    a_br = mid
                    fa = fm
            roots.append((a_br+b_br)/2)
    return roots

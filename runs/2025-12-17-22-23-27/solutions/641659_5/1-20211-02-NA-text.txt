def circumcenter(p1, p2, p3):
    # returns circumcenter as tuple (x,y)
    # using perpendicular bisectors intersection
    x1, y1 = p1
    x2, y2 = p2
    x3, y3 = p3
    # Solve linear system:
    # (x - x1)^2 + (y - y1)^2 = (x - x2)^2 + (y - y2)^2
    # (x - x1)^2 + (y - y1)^2 = (x - x3)^2 + (y - y3)^2
    A = sp.Matrix([[2*(x2 - x1), 2*(y2 - y1)],
                   [2*(x3 - x1), 2*(y3 - y1)]])
    b = sp.Matrix([x2**2 + y2**2 - x1**2 - y1**2,
                   x3**2 + y3**2 - x1**2 - y1**2])
    sol = A.LUsolve(b)
    return (sol[0], sol[1])

def line_eq(p1, p2):
    # returns A,B,C for line equation Ax + By + C = 0
    x1,y1 = p1; x2,y2 = p2
    A = y1 - y2
    B = x2 - x1
    C = x1*y2 - x2*y1
    return (A,B,C)

def intersect_circles(c1, r1, c2, r2):

    dx, dy = x1-x0, y1-y0

def incircle_center_and_radius(A,B,C):
    # compute incenter and inradius
    # using formula: incenter is weighted by side lengths
    # side lengths opposite A,B,C:
    a = math.dist(B,C)
    b = math.dist(A,C)
    c = math.dist(A,B)
    # incenter coordinates
    x = (a*A[0] + b*B[0] + c*C[0])/(a+b+c)
    y = (a*A[1] + b*B[1] + c*C[1])/(a+b+c)
    # radius = area / s
    s = (a+b+c)/2
    # area via Heron
    area = math.sqrt(s*(s-a)*(s-b)*(s-c))
    r = area/s
    return (x,y,r)

def foot_of_perpendicular(P, Q, R):
    # foot from P to line QR
    # compute projection of vector (P-Q) onto direction (R-Q)
    Qx,Qy = Q
    Rx,Ry = R
    Px,Py = P
    vx = Rx - Qx
    vy = Ry - Qy
    # projection of (P-Q) onto v
    t = ((Px - Qx)*vx + (Py - Qy)*vy)/(vx*vx + vy*vy)
    # foot point = Q + t * v
    return (Qx + t*vx, Qy + t*vy)

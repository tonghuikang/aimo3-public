    min_l = None
    best = None
    # brute a<b<c
    for a in range(1, S-1):
        for b in range(a+1, S):
            c = S - a - b
            if c <= b or c <= 0:
                continue
            # distinct: a<b<c automatically
            from math import gcd
            # compute lcm
            def lcm(x,y):
                return x // gcd(x,y) * y
            l = lcm(lcm(a,b),c)
            if min_l is None or l < min_l:
                min_l = l
                best = (a,b,c)
    return min_l, best

    # naive search over L:
    for L in range((S+2)//3 + 1, S): # L > S/3

        # For small S it's okay.
        divisors = set()
        for d in range(1, isqrt(L)+1):

                divisors.add(d)

        # need distinct a,b in divisors with a+b = S - L.

        for a in divisors:
            b = target - a
            if b != a and b in divisors:
                # found
                return L
    return None

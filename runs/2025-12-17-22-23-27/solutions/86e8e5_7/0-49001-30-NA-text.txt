def best_ratio_for_sum(s):
    # compute minimal ratio for sum s (requires distinct positive ints a<b<c, a|c,b|c)
    best = None
    best_trip = None
    for a in range(1, s):
        for b in range(a+1, s):
            c = s - a - b
            if c <= b:
                continue
            if c % a == 0 and c % b == 0:
                ratio = c / s
                if best is None or ratio < best:
                    best = ratio
                    best_trip = (a,b,c)
    return best, best_trip

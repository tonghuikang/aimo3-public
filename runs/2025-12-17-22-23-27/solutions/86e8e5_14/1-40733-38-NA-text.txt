def smallest_divisor_ge3_not_multiple_of_3(n):
    # return smallest divisor d >= 3, d not multiple of 3, such that d|n
    # if none found (i.e., n=1 or 2), return None
    for d in range(3, int(math.sqrt(n))+1):
        if n % d == 0:
            if d % 3 != 0:
                return d
            # also if d multiple of 3, not allowed; but maybe quotient also qualifies
            # but we need smallest divisor >=3 overall, not necessarily prime.
    # Check if n itself qualifies
    if n >= 3 and n % 3 != 0:
        return n
    else:
        return None

        r_num = S - 1 - d

    def M_mod_S(S):
        # find a: largest exponent such that 3^a divides S
        a = 0
        tmp = S
        while tmp % 3 == 0:

            a += 1
        s1 = 3**a

        inv = pow(s1, -1, s2)  # using Python 3.8's pow with mod inverse
        x = (s1 * inv) % S

        return x

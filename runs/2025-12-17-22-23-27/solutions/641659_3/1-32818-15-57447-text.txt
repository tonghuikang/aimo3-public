def circumcenter(p1, p2, p3):
    # using perpendicular bisector intersection
    (x1, y1), (x2, y2), (x3, y3) = p1, p2, p3
    d = 2*(x1*(y2-y3) + x2*(y3-y1) + x3*(y1-y2))
    if abs(d) < 1e-12:
        return None
    ux = ((x1**2 + y1**2)*(y2-y3) + (x2**2 + y2**2)*(y3-y1) + (x3**2 + y3**2)*(y1-y2)) / d
    uy = ((x1**2 + y1**2)*(x3-x2) + (x2**2 + y2**2)*(x1-x3) + (x3**2 + y3**2)*(x2-x1)) / d
    return (ux, uy)

def circle_center_radius(p1, p2, p3):
    cen = circumcenter(p1, p2, p3)
    if cen is None:
        return None, None
    r = math.hypot(p1[0]-cen[0], p1[1]-cen[1])
    return cen, r

def intersect_circles(c1, r1, c2, r2):
    # return list of intersection points (0,1,2)
    x0, y0 = c1
    x1, y1 = c2
    dx, dy = x1 - x0, y1 - y0
    d = math.hypot(dx, dy)
    # no intersection
    if d > r1 + r2 + 1e-12 or d < abs(r1 - r2) - 1e-12:
        return []
    # find a = (r1^2 - r2^2 + d^2) / (2d)
    a = (r1*r1 - r2*r2 + d*d) / (2*d)
    # h = sqrt(r1^2 - a^2)
    h2 = r1*r1 - a*a
    if h2 < -1e-12:
        return []
    h = math.sqrt(max(0.0, h2))
    # point p2 = p0 + a*(p1-p0)/d
    xm = x0 + a * dx / d
    ym = y0 + a * dy / d
    # intersection points
    rx = -dy * (h / d)
    ry = dx * (h / d)
    p_int1 = (xm + rx, ym + ry)
    p_int2 = (xm - rx, ym - ry)
    if h < 1e-12:
        return [p_int1]
    else:
        return [p_int1, p_int2]

def reflect_point_across_line(P, A, B):
    # reflect point P across line through A,B
    # using vector projection
    # line direction vector
    vx, vy = B[0] - A[0], B[1] - A[1]
    # unit direction?
    # projection of AP onto line
    # Compute param t for projection of P onto line
    # formula: t = ((P-A).v) / (v.v)
    PAx, PAy = P[0] - A[0], P[1] - A[1]
    vv = vx*vx + vy*vy
    if vv == 0:
        return None
    t = (PAx*vx + PAy*vy) / vv
    # projection point Q = A + t v
    Qx = A[0] + t*vx
    Qy = A[1] + t*vy
    # reflection: P' = 2Q - P
    Rx = 2*Qx - P[0]
    Ry = 2*Qy - P[1]
    return (Rx, Ry)

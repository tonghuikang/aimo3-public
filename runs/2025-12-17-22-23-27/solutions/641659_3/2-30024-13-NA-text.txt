def incircle_and_contacts(A,B,C):
    # Returns incenter I, radius r, and contact points D,E,F
    A = sp.Point(A); B = sp.Point(B); C = sp.Point(C)
    # side lengths
    a = B.distance(C)  # BC
    b = C.distance(A)  # CA
    c = A.distance(B)  # AB
    # incenter
    I = sp.Point((a*A.x + b*B.x + c*C.x)/(a+b+c), (a*A.y + b*B.y + c*C.y)/(a+b+c))
    # radius = distance from I to line BC
    r = sp.Line(B, C).distance(I)
    # contact points: foot of perpendicular from I to each side
    D = sp.Line(B, C).projection(I)
    E = sp.Line(C, A).projection(I)
    F = sp.Line(A, B).projection(I)
    return I, r, D, E, F

    # Solve using sympy geometry if possible
    c1 = sp.Point(center1)
    c2 = sp.Point(center2)
    # using sympy geometry circles:
    circ1 = sp.Circle(c1, r1)
    circ2 = sp.Circle(c2, r2)
    pts = list(circ1.intersection(circ2))
    # return points
    return pts

def reflect_point_across_line(P, line):
    # reflect P across line (sympy)
    return line.reflect(P)

def is_concyclic(points, tol=1e-7):
    # Check if 4 points are concyclic: there exists a circle passing through all (coincident within tolerance)
    # Use determinant condition: for 4 points (x_i,y_i)
    # | x_i^2 + y_i^2, x_i, y_i, 1 | = 0 for each point, determinant of 5x5 matrix = 0?
    # Actually for 4 points, we can compute circle from first three and check fourth.
    P1,P2,P3,P4 = points
    # find circle through first three (if non-collinear)
    try:
        circ = sp.Circle(P1,P2,P3)
    except Exception as e:
        # points collinear
        return False
    # check if P4 lies on circ
    return abs(circ.distance(P4)) < tol

def circumcenter(p1, p2, p3):
    # p are tuples (x,y)
    x1,y1 = p1
    x2,y2 = p2
    x3,y3 = p3
    # Solve linear equations:
    # 2*(x2 - x1)*ox + 2*(y2 - y1)*oy = x2^2 + y2^2 - x1^2 - y1^2
    # 2*(x3 - x1)*ox + 2*(y3 - y1)*oy = x3^2 + y3^2 - x1^2 - y1^2
    A = [[2*(x2 - x1), 2*(y2 - y1)],
         [2*(x3 - x1), 2*(y3 - y1)]]
    B = [x2*x2 + y2*y2 - x1*x1 - y1*y1,
         x3*x3 + y3*y3 - x1*x1 - y1*y1]
    # Solve linear system
    det = A[0][0]*A[1][1] - A[0][1]*A[1][0]
    if abs(det) < 1e-12:
        # points are collinear or nearly
        return None, None
    ox = (B[0]*A[1][1] - B[1]*A[0][1]) / det
    oy = (A[0][0]*B[1] - A[1][0]*B[0]) / det
    R = math.hypot(ox - x1, oy - y1)
    return (ox, oy), R

def circle_intersections_center_radius(c1, r1, c2, r2, known_point=None):
    # compute intersection points of circles (c1,r1) and (c2,r2)
    # known_point: optional point to exclude (like A)
    # Returns list of points (x,y)
    x1,y1 = c1
    x2,y2 = c2
    # distance between centers
    d = math.hypot(x2 - x1, y2 - y1)
    if d > r1 + r2 + 1e-12 or d < abs(r1 - r2) - 1e-12:
        # no intersection
        return []
    # a = (r1^2 - r2^2 + d^2) / (2d)
    a = (r1*r1 - r2*r2 + d*d) / (2*d)
    # h = sqrt(r1^2 - a^2)
    h_sq = max(r1*r1 - a*a, 0.0)
    h = math.sqrt(h_sq)
    # point P2 = c1 + a*(c2-c1)/d
    x3 = x1 + a * (x2 - x1) / d
    y3 = y1 + a * (y2 - y1) / d
    # intersection points:
    # (x3 +- h*(y2-y1)/d, y3 -+ h*(x2-x1)/d)
    rx = -(y2 - y1) * (h / d)
    ry = (x2 - x1) * (h / d)
    p_int1 = (x3 + rx, y3 + ry)
    p_int2 = (x3 - rx, y3 - ry)
    # if known_point given, exclude it
    res = []
    for pt in [p_int1, p_int2]:
        if known_point is not None:
            if math.hypot(pt[0] - known_point[0], pt[1] - known_point[1]) < 1e-9:
                continue
        res.append(pt)
    return res

def foot_of_perpendicular(P, A, B):
    # foot from point P onto line AB
    # returns point Q
    # A,B as tuples
    x0,y0 = P
    x1,y1 = A
    x2,y2 = B
    # line direction vector
    dx = x2 - x1
    dy = y2 - y1
    # projection factor t = ((P-A)·(B-A)) / |B-A|^2
    t = ((x0 - x1)*dx + (y0 - y1)*dy) / (dx*dx + dy*dy)
    xq = x1 + t*dx
    yq = y1 + t*dy
    return (xq, yq)

def reflect_point_across_line(P, A, B):
    # reflect point P across line through A,B
    # compute foot Q and then reflect
    Q = foot_of_perpendicular(P, A, B)
    # reflection P' = 2Q - P
    return (2*Q[0] - P[0], 2*Q[1] - P[1])

    perp1 = (-vBN[1], vBN[0])
    # midpoint of BK
    M_BK = ((B[0] + K[0]) / 2.0, (B[1] + K[1]) / 2.0)
    # direction of perpendicular bisector of BK: perp2 = (-(K[1] - B[1]), K[0] - B[0])
    perp2 = (-(K[1] - B[1]), K[0] - B[0])
    # solve for intersection O_T: B + u*perp1 = M_BK + v*perp2
    # => u*perp1 - v*perp2 = M_BK - B
    # Write linear equations for u,v:
    # u*perp1_x - v*perp2_x = Mx
    # u*perp1_y - v*perp2_y = My
    Mx = M_BK[0] - B[0]
    My = M_BK[1] - B[1]
    # Solve 2x2 for u,v
    # [perp1_x, -perp2_x][u; v] = [Mx]
    # [perp1_y, -perp2_y][u; v] = [My]
    A_mat = [[perp1[0], -perp2[0]],
             [perp1[1], -perp2[1]]]
    detA = A_mat[0][0]*A_mat[1][1] - A_mat[0][1]*A_mat[1][0]
    if abs(detA) < 1e-12:
        return None
    u = (Mx * A_mat[1][1] - My * A_mat[0][1]) / detA
    # v = (A_mat[0][0]*My - A_mat[1][0]*Mx) / detA  # not needed
    O_T = (B[0] + u * perp1[0], B[1] + u * perp1[1])
    R_T = math.hypot(O_T[0] - B[0], O_T[1] - B[1])
    # Intersection with BC (y=0)
    # Solve (x - Ox)^2 + (0 - Oy)^2 = R_T^2
    # => (x - Ox)^2 = R_T^2 - Oy^2
    rhs = R_T*R_T - O_T[1]*O_T[1]
    if rhs < -1e-9:
        # no real intersection
        return None
    sqrt_rhs = math.sqrt(max(rhs, 0.0))
    # solutions
    x_sol1 = O_T[0] + sqrt_rhs
    x_sol2 = O_T[0] - sqrt_rhs
    # choose the one not near B (0)
    # B is (0,0). So if one solution is close to 0, that's B.
    if abs(x_sol1) < 1e-6:
        t = x_sol2
    elif abs(x_sol2) < 1e-6:
        t = x_sol1
    else:
        # choose the solution between 0 and 1 (since T is on segment BC)
        # we expect 0 < t < 1
        # pick whichever is within (0,1)
        if 0 < x_sol1 < 1:
            t = x_sol1
        elif 0 < x_sol2 < 1:
            t = x_sol2
        else:
            # no appropriate solution
            return None
    # compute lengths
    CT = 1 - t
    BT = t
    NB = math.hypot(N[0] - B[0], N[1] - B[1])
    NE = math.hypot(N[0] - E[0], N[1] - E[1])
    Rratio = (CT * NB) / (BT * NE)
    return Rratio, residual, A, B, C, D, I, E, F, K, Kp, N, t

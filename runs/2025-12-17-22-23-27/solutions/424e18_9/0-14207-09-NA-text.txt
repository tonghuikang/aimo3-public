def generate_pairings(players):
    # returns list of pairings (list of tuples) that partition players into pairs
    # players is a tuple/list
    # generate recursively
    if not players:
        return [[]]
    # pick first player
    first = players[0]
    rest = players[1:]
    pairings = []
    for i, other in enumerate(rest):
        # pair first with other
        new_rest = rest[:i] + rest[i+1:]
        for subpair in generate_pairings(new_rest):
            pairings.append([(first, other)] + subpair)
    return pairings

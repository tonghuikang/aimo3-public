def all_matchings(elements):
    """Return all possible perfect matchings (unordered pairs) of a list of elements (even length). 
    Each matching is a list of tuples (a,b) where a<b (or order not important). 
    Returns list of list of pairs."""
    # recursion: pick first element, pair with any other, then recursively match the rest
    if not elements:
        return [[]]
    first = elements[0]
    rest = elements[1:]
    matchings = []
    for i, other in enumerate(rest):
        remaining = rest[:i] + rest[i+1:]
        for submatching in all_matchings(remaining):
            matchings.append([(first, other)] + submatching)
    return matchings

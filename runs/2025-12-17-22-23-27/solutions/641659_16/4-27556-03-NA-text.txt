def compute_K(A,B,C):
    # coordinates as tuples
    A = sp.Point(A); B = sp.Point(B); C = sp.Point(C)
    # side lengths
    a = B.distance(C); b = C.distance(A); c = A.distance(B)
    # incircle
    s = (a+b+c)/2
    # incenter coordinates:
    I = sp.Point((a*A.x + b*B.x + c*C.x)/(a+b+c), (a*A.y + b*B.y + c*C.y)/(a+b+c))
    # incircle radius
    # area via Heron
    area = sp.sqrt(s*(s-a)*(s-b)*(s-c))
    r = area/s
    # foot points D, E, F on BC, CA, AB (incircle tangency)
    # D: foot from I to BC
    BC_line = sp.Line(B,C)
    D = BC_line.projection(I)
    # E: projection onto CA
    CA_line = sp.Line(C,A)
    E = CA_line.projection(I)
    # F: projection onto AB
    AB_line = sp.Line(A,B)
    F = AB_line.projection(I)
    # circumcircle of ABC
    circ_ABC = sp.Circle(A,B,C)
    # circumcircle of AEF
    circ_AEF = sp.Circle(A,E,F)
    # intersection
    inters = circ_ABC.intersection(circ_AEF)
    # filter distinct points
    K_candidates = [pt for pt in inters if not pt.equals(A)]
    if not K_candidates:
        return None
    K = K_candidates[0]
    # compute intersection of line AI with circumcircle (other than A)
    lineAI = sp.Line(A,I)
    inter2 = circ_ABC.intersection(lineAI)
    inter2_candidates = [pt for pt in inter2 if not pt.equals(A)]
    # might be same as K
    return K, inter2_candidates[0] if inter2_candidates else None

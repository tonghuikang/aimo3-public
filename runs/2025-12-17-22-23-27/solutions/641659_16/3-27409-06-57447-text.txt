def reflect_point_across_line(P, A, B):
    # reflect point P across line AB (given points A,B)
    # compute projection of P onto line AB
    # vector AB
    ABx = B[0] - A[0]
    ABy = B[1] - A[1]
    # vector AP
    APx = P[0] - A[0]
    APy = P[1] - A[1]
    # dot product
    dot = APx*ABx + APy*ABy
    len2 = ABx*ABx + ABy*ABy
    if len2 == 0:
        # degenerate line
        raise ValueError("A and B are same point")
    # projection scalar t
    t = dot/len2
    # foot Q
    Qx = A[0] + t*ABx
    Qy = A[1] + t*ABy
    # reflection: R = 2Q - P
    Rx = 2*Qx - P[0]
    Ry = 2*Qy - P[1]
    return (Rx, Ry)

def foot_point(P, A, B):
    ABx = B[0] - A[0]
    ABy = B[1] - A[1]
    APx = P[0] - A[0]
    APy = P[1] - A[1]
    dot = APx*ABx + APy*ABy
    len2 = ABx*ABx + ABy*ABy
    t = dot/len2
    Qx = A[0] + t*ABx
    Qy = A[1] + t*ABy
    return (Qx, Qy)

def circle_center_from_three_points(P1, P2, P3):
    # compute circumcenter of triangle P1,P2,P3 using perpendicular bisectors
    # Solve using linear algebra.
    x1,y1 = P1
    x2,y2 = P2
    x3,y3 = P3
    # Compute the perpendicular bisector of P1P2 and P1P3
    # midpoints
    mx12 = (x1 + x2)/2
    my12 = (y1 + y2)/2
    mx13 = (x1 + x3)/2
    my13 = (y1 + y3)/2
    # direction vectors of original segments
    dx12 = x2 - x1
    dy12 = y2 - y1
    dx13 = x3 - x1
    dy13 = y3 - y1
    # Perp bisector slopes: direction vector for bisector is (-dy,dx)
    # Solve for intersection of lines:
    # (mx12, my12) + s1 * (-dy12, dx12) = (mx13, my13) + s2 * (-dy13, dx13)
    # Solve for s1, s2
    # Write as system: mx12 - mx13 + s1*(-dy12) - s2*(-dy13) = 0 (x)
    # similarly for y.
    # Solve linear system for s1 and s2.
    A = [[-dy12, dy13], [dx12, -dx13]]
    b = [mx13 - mx12, my13 - my12]
    # Solve linear equations: A*[s1,s2] = b
    det = A[0][0]*A[1][1] - A[0][1]*A[1][0]
    if abs(det) < 1e-12:
        # collinear
        return None
    s1 = (b[0]*A[1][1] - b[1]*A[0][1]) / det
    #s2 = (A[0][0]*b[1] - A[1][0]*b[0]) / det # not needed
    # center:
    cx = mx12 + s1*(-dy12)
    cy = my12 + s1*(dx12)
    return (cx, cy)

def circle_center_and_radius(P1, P2, P3):
    O = circle_center_from_three_points(P1, P2, P3)
    if O is None:
        return None, None
    cx,cy = O
    R = math.hypot(P1[0]-cx, P1[1]-cy)
    return O, R

def intersect_two_circles(O0,R0, O1,R1):
    # returns two intersection points (as tuples)
    x0,y0 = O0
    x1,y1 = O1
    dx = x1 - x0
    dy = y1 - y0
    d = math.hypot(dx, dy)
    # no intersection?
    if d > R0 + R1 or d < abs(R0 - R1):
        return []  # none or one
    # a = (R0^2 - R1^2 + d^2) / (2d)
    a = (R0*R0 - R1*R1 + d*d) / (2*d)
    # h^2 = R0^2 - a^2
    h_sq = R0*R0 - a*a
    if h_sq < 0:
        h = 0.0
    else:
        h = math.sqrt(h_sq)
    # point P2 = O0 + a*(O1-O0)/d
    x2 = x0 + a * dx / d
    y2 = y0 + a * dy / d
    # offset vector (perp)
    rx = -dy * (h/d)
    ry = dx * (h/d)
    # two intersections
    P_int1 = (x2 + rx, y2 + ry)
    P_int2 = (x2 - rx, y2 - ry)
    return [P_int1, P_int2]

def is_point_equal(P, Q, eps=1e-9):
    return math.hypot(P[0]-Q[0], P[1]-Q[1]) < eps

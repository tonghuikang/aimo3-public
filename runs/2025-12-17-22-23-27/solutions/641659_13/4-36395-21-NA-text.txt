def circumcenter(p1, p2, p3):
    # p1,p2,p3 each (x,y) tuple
    x1, y1 = p1
    x2, y2 = p2
    x3, y3 = p3
    # Solve linear equations:
    # 2*(x2 - x1) * X + 2*(y2 - y1) * Y = x2^2 + y2^2 - x1^2 - y1^2
    # 2*(x3 - x1) * X + 2*(y3 - y1) * Y = x3^2 + y3^2 - x1^2 - y1^2
    A = np.array([[2*(x2 - x1), 2*(y2 - y1)],
                  [2*(x3 - x1), 2*(y3 - y1)]], dtype=float)
    b = np.array([x2**2 + y2**2 - x1**2 - y1**2,
                  x3**2 + y3**2 - x1**2 - y1**2], dtype=float)
    # Solve A X = b
    sol = np.linalg.solve(A, b)
    return (sol[0], sol[1])

def circle_intersections(o1, r1, o2, r2):
    # returns two intersection points as (x,y) tuples (order not guaranteed)
    (x1,y1) = o1
    (x2,y2) = o2
    dx = x2 - x1
    dy = y2 - y1
    d = np.hypot(dx, dy)
    if d == 0:
        return [] # coincident centers
    # Check for intersection existence (should intersect)
    # Compute a
    a = (r1**2 - r2**2 + d**2) / (2*d)
    # height from line between centers to intersection points
    h_sq = r1**2 - a**2
    if h_sq < -1e-9:
        return [] # no intersection
    h = np.sqrt(max(h_sq,0))
    # point P2 = O1 + a*(O2-O1)/d
    xm = x1 + a * dx / d
    ym = y1 + a * dy / d
    # offset vector for intersection points
    rx = -dy * (h/d)
    ry = dx * (h/d)
    p_int1 = (xm + rx, ym + ry)
    p_int2 = (xm - rx, ym - ry)
    return [p_int1, p_int2]

def reflect_point_across_line(P, A, B):
    # reflect point P across line AB (given by two points A,B)
    # using vector formula
    # Compute projection of (P-A) onto direction d = B-A
    Ax,Ay = A
    Bx,By = B
    Px,Py = P
    dx = Bx - Ax
    dy = By - Ay
    # Normalize direction vector (not needed)
    # Compute projection factor t = ((P-A)·d) / (d·d)
    t = ((Px - Ax)*dx + (Py - Ay)*dy) / (dx*dx + dy*dy)
    # Foot of perpendicular (projection) on line AB:
    proj_x = Ax + t*dx
    proj_y = Ay + t*dy
    # Reflection: P' = 2*proj - P
    Rx = 2*proj_x - Px
    Ry = 2*proj_y - Py
    return (Rx, Ry)

def cyclic_det(B, N, K, Kp):
    # determinant of matrix rows [x^2+y^2, x, y, 1]
    M = np.array([
        [B[0]**2 + B[1]**2, B[0], B[1], 1.0],
        [N[0]**2 + N[1]**2, N[0], N[1], 1.0],
        [K[0]**2 + K[1]**2, K[0], K[1], 1.0],
        [Kp[0]**2 + Kp[1]**2, Kp[0], Kp[1], 1.0]
    ], dtype=float)
    return np.linalg.det(M)

def compute_V(lam, u):
    # r = 1
    r = 1.0
    # points
    B = (-u, -r)
    C = (lam*u, -r)
    D = (0.0, -r)
    # E,F
    v = lam*u
    E = (2*v/(1+v**2), -(1 - v**2)/(1+v**2))
    F = (-2*u/(1+u**2), -(1 - u**2)/(1+u**2))
    # A
    denom = lam*u**2 - 1.0
    if abs(denom) < 1e-12:
        return None, None  # invalid
    A = (u*(1 - lam)/denom, (lam*u**2 + 1)/denom)
    # N: foot from D onto line EF
    # using projection onto line EF
    # direction vector d = E - F
    dEF = (E[0] - F[0], E[1] - F[1])
    # projection of D-F onto dEF
    t = ((D[0] - F[0])*dEF[0] + (D[1] - F[1])*dEF[1]) / (dEF[0]**2 + dEF[1]**2)
    N = (F[0] + t*dEF[0], F[1] + t*dEF[1])
    # Circumcenters
    O1 = circumcenter(A, B, C)
    R1 = np.hypot(A[0] - O1[0], A[1] - O1[1])
    O2 = circumcenter(A, E, F)
    R2 = np.hypot(A[0] - O2[0], A[1] - O2[1])
    # Intersections of circles
    inters = circle_intersections(O1, R1, O2, R2)
    # Choose K as point not equal to A
    K = None
    for pt in inters:
        # compare to A
        if np.hypot(pt[0]-A[0], pt[1]-A[1]) > 1e-6:
            K = pt
            break
    if K is None:
        # maybe only A (degenerate)
        return None, None
    # K' is reflection of K across line EF
    Kp = reflect_point_across_line(K, E, F)
    # compute cyclic determinant
    det_val = cyclic_det(B, N, K, Kp)
    # compute T: circle through B and K tangent to line BN at B
    # line BN direction vector
    BN = (N[0] - B[0], N[1] - B[1])
    # line perpendicular to BN through B: direction vector perp = (-BN[1], BN[0])
    # center O must lie on this perpendicular line: O = B + s * perp for some s
    # Also O must be equidistant to B and K: |O-B| = |O-K|
    # Solve for s: let O = B + s * perp
    perp = (-BN[1], BN[0])
    # unknown s
    # compute equation: |O-B|^2 = |O-K|^2
    # |s*perp|^2 = |B + s*perp - K|^2
    # s^2 * (perp·perp) = (B - K + s*perp)·(B - K + s*perp)
    # Expand: s^2*pp = (B-K)·(B-K) + 2 s * perp·(B-K) + s^2*pp
    # Cancel s^2*pp both sides => 0 = (B-K)·(B-K) + 2 s * perp·(B-K)
    # => s = - (|B-K|^2) / (2 * perp·(B-K))
    BK = (B[0] - K[0], B[1] - K[1])
    BK_sq = BK[0]**2 + BK[1]**2
    dot_perp_BK = perp[0]*BK[0] + perp[1]*BK[1]
    if abs(dot_perp_BK) < 1e-12:
        # line is parallel? 
        return None, None
    s_val = - BK_sq / (2.0 * dot_perp_BK)
    O = (B[0] + s_val*perp[0], B[1] + s_val*perp[1])
    R = np.hypot(O[0] - B[0], O[1] - B[1])
    # Intersection of circle (center O,R) with line y = -r (BC)
    # Solve (x - O_x)^2 + (-r - O_y)^2 = R^2 => (x - O_x)^2 = R^2 - ( -r - O_y )^2
    rhs = R**2 - ((-r) - O[1])**2
    if rhs < -1e-12:
        return None, None
    # two solutions: x = O_x ± sqrt(rhs)
    sqrt_rhs = np.sqrt(max(rhs,0.0))
    # one solution is B_x = -u; pick other
    x1 = O[0] + sqrt_rhs
    x2 = O[0] - sqrt_rhs
    # choose T_x different from B_x
    if abs(x1 - B[0]) > 1e-6:
        T_x = x1
    else:
        T_x = x2
    T = (T_x, -r)
    # Compute distances for V
    # CT = C_x - T_x (should be positive if T between B and C)
    CT = C[0] - T_x
    BT = T_x - B[0]
    NB = np.hypot(N[0] - B[0], N[1] - B[1])
    NE = np.hypot(N[0] - E[0], N[1] - E[1])
    V = (CT * NB) / (BT * NE)
    return V, det_val

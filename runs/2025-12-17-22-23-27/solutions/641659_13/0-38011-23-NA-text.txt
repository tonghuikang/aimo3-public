    p_int1 = (xm + rx, ym + ry)
    p_int2 = (xm - rx, ym - ry)
    if h < 1e-12:
        return [p_int1]
    return [p_int1, p_int2]

def foot_point(P, A, B):
    # foot of perpendicular from P to line AB
    # A, B points
    # vector AB
    ax, ay = A
    bx, by = B
    # projection param t = ((P-A)·(B-A))/|B-A|^2
    dx = bx - ax
    dy = by - ay
    if dx==0 and dy==0:
        return A
    t = ((P[0]-ax)*dx + (P[1]-ay)*dy) / (dx*dx + dy*dy)
    # foot point Q = A + t*(B-A)
    qx = ax + t*dx
    qy = ay + t*dy
    return (qx, qy)

def reflect_point_across_line(P, A, B):
    # reflect point P across line AB
    # compute foot Q of P onto line AB, then P' = 2Q - P
    Q = foot_point(P, A, B)
    return (2*Q[0] - P[0], 2*Q[1] - P[1])

def is_cyclic(p1,p2,p3,p4):
    # Check if four points are concyclic using determinant
    # Return determinant value
    def det4(m):
        # compute determinant of 4x4
        return (
            m[0][0]*(
                m[1][1]*(
                    m[2][2]*m[3][3] - m[2][3]*m[3][2]
                ) - m[1][2]*(
                    m[2][1]*m[3][3] - m[2][3]*m[3][1]
                ) + m[1][3]*(
                    m[2][1]*m[3][2] - m[2][2]*m[3][1]
                )
            )
            - m[0][1]*(
                m[1][0]*(
                    m[2][2]*m[3][3] - m[2][3]*m[3][2]
                ) - m[1][2]*(
                    m[2][0]*m[3][3] - m[2][3]*m[3][0]
                ) + m[1][3]*(
                    m[2][0]*m[3][2] - m[2][2]*m[3][0]
                )
            )
            + m[0][2]*(
                m[1][0]*(
                    m[2][1]*m[3][3] - m[2][3]*m[3][1]
                ) - m[1][1]*(
                    m[2][0]*m[3][3] - m[2][3]*m[3][0]
                ) + m[1][3]*(
                    m[2][0]*m[3][1] - m[2][1]*m[3][0]
                )
            )
            - m[0][3]*(
                m[1][0]*(
                    m[2][1]*m[3][2] - m[2][2]*m[3][1]
                ) - m[1][1]*(
                    m[2][0]*m[3][2] - m[2][2]*m[3][0]
                ) + m[1][2]*(
                    m[2][0]*m[3][1] - m[2][1]*m[3][0]
                )
            )
        )
    mat = []
    for P in [p1,p2,p3,p4]:
        x,y = P
        mat.append([x, y, x*x + y*y, 1])
    return det4(mat)

def compute_ratio(u, v, r):
    # compute ratio CT*NB/(BT*NE)
    # geometry steps
    # Points
    B = (0.0, 0.0)
    C = (u+v, 0.0)
    D = (u, 0.0)
    # slopes
    if abs(u*u - r*r) < 1e-12 or abs(v*v - r*r) < 1e-12:
        return None, None
    mB = 2*u*r/(u*u - r*r)
    mC = -2*v*r/(v*v - r*r)
    # A intersection
    xA = -mC*(u+v)/(mB - mC)
    yA = mB*xA
    A = (xA, yA)
    # foot F: projection onto line through origin with slope mB
    lambdaF = (u + r*mB)/(1 + mB*mB)
    F = (lambdaF, lambdaF*mB)
    # foot E: t = (-v + r*mC)/(1 + mC*mC)
    t = (-v + r*mC)/(1 + mC*mC)

    R1 = circ_ABC[2]

    inters = intersect_circles(O1, R1, O2, R2)
    # Find K (not A)
    K = None
    for P in inters:
        # compare with A

            K = P

    if K is None:

    O_BNK = (O_BNK[0], O_BNK[1])

    # Not needed now.
    # Compute T: circle through B and K tangent to BN at B. Center O_T is intersection of perp bisector of BK and perpendicular to BN at B.
    # Perp bisector of BK:
    M_BK = ((B[0]+K[0])/2.0, (B[1]+K[1])/2.0)
    # direction of BK
    dx_BK = K[0] - B[0]
    dy_BK = K[1] - B[1]
    # line perp to BK: direction (-dy_BK, dx_BK)
    # line perp to BN at B: direction (-dy_BN, dx_BN)
    dx_BN = N[0] - B[0]
    dy_BN = N[1] - B[1]
    # Solve intersection of two lines:
    # Param t for line1: O = M_BK + s1*(-dy_BK, dx_BK)
    # Param t for line2: O = B + s2*(-dy_BN, dx_BN)
    # Solve for s1,s2.
    # Write as M_BK + s1*v1 = B + s2*v2
    # Solve linear equations for s1,s2
    v1 = (-dy_BK, dx_BK)
    v2 = (-dy_BN, dx_BN)
    # Solve 2x2 system: s1*v1 - s2*v2 = B - M_BK
    # Let unknowns s1, s2.
    A_mat = [[v1[0], -v2[0]],
             [v1[1], -v2[1]]]
    b_vec = [B[0] - M_BK[0], B[1] - M_BK[1]]
    # Solve linear equations
    # Compute determinant
    detA = A_mat[0][0]*A_mat[1][1] - A_mat[0][1]*A_mat[1][0]
    if abs(detA) < 1e-12:
        return None, None
    s1 = (b_vec[0]*A_mat[1][1] - b_vec[1]*A_mat[0][1]) / detA
    # compute O_T
    O_T = (M_BK[0] + s1*v1[0], M_BK[1] + s1*v1[1])
    # radius
    R_T = math.hypot(O_T[0] - B[0], O_T[1] - B[1])
    # Intersection with BC (y=0)
    # solve (x - O_T.x)^2 + (O_T.y)^2 = R_T^2
    # => (x - O_T.x)^2 = R_T^2 - O_T.y^2
    disc = R_T*R_T - O_T[1]*O_T[1]
    if disc < -1e-9:
        return None, None
    disc = max(disc,0)
    sqrt_disc = math.sqrt(disc)
    x_candidate1 = O_T[0] + sqrt_disc
    x_candidate2 = O_T[0] - sqrt_disc
    # Choose T that is not B (0)
    eps = 1e-6
    T = None
    for x in [x_candidate1, x_candidate2]:
        if abs(x) > eps:
            T = (x, 0.0)
            break
    if T is None:
        # maybe both close to zero? improbable
        return None, None
    # compute lengths
    NB = math.hypot(N[0] - B[0], N[1] - B[1])
    NE = math.hypot(N[0] - E[0], N[1] - E[1])
    BT = abs(T[0] - B[0])
    CT = abs(C[0] - T[0])
    ratio = (CT * NB) / (BT * NE)
    return ratio, det

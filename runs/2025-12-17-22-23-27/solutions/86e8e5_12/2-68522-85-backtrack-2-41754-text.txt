    # Compute minimal LCM of three distinct positive integers summing to N.
    # brute force over triples (a,b,c) distinct, a<b<c, a+b+c=N.
    best = None
    for a in range(1, N-2):
        for b in range(a+1, N-a-1):
            c = N - a - b
            if c <= b:
                continue
            # a,b,c distinct
            l = math.lcm(a,b,c)
            if best is None or l < best:
                best = l
    return best

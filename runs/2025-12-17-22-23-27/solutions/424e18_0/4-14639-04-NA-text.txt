def all_pairings(lst):
    # lst is list of indices; returns list of list of pairs (tuple of two)
    # generate all perfect matchings
    if not lst:
        return [[]]
    # pick first element, pair with each other
    first = lst[0]
    res = []
    for i in range(1, len(lst)):
        second = lst[i]
        rest = lst[1:i] + lst[i+1:]
        for sub in all_pairings(rest):
            res.append([(first, second)] + sub)
    return res

            return

            # use all_pairings recursively
            pairings = all_pairings(idxs)
            group_pairings[score] = pairings
        
        # Now iterate over product of pairings across groups
        # We'll use recursion to combine.

def all_pairings(indices):
    # indices is tuple/list of indices; return list of list of pairs (tuples)
    if not indices:
        return [[]]
    first = indices[0]
    res = []
    for i in range(1, len(indices)):

        rest = indices[1:i] + indices[i+1:]
        for sub in all_pairings(rest):

            ranking = tuple([s for sc,s in sorted_state])

        # combine across groups
        scores = list(group_pairings.keys())
        def combine(idx, current_pairs):
            if idx == len(scores):
                # apply matches
                inc = 2**(k-round_i)

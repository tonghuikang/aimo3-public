def all_pairings(players):
    # players is a tuple/list of player ids; length must be even
    # generate all perfect matchings as list of list of pairs (tuples)
    if not players:
        yield []
        return
    # pick first player
    first = players[0]
    for i in range(1, len(players)):
        second = players[i]
        rest = players[1:i] + players[i+1:]
        for sub in all_pairings(rest):
            yield [(first, second)] + sub

    while queue:

            # tournament done; compute ordering

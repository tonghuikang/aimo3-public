def pairings(group):
    # returns list of list of pairs (tuples) where group is tuple of participants
    # generate all perfect matchings recursively
    if not group:
        return [[]]
    first = group[0]
    rest = group[1:]
    matchings = []
    for i, other in enumerate(rest):
        # pair first with other
        new_group = rest[:i] + rest[i+1:]
        for sub in pairings(new_group):
            matchings.append([(first, other)] + sub)
    return matchings

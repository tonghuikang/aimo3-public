def prime_factors(n):
    i = 2
    factors = {}
    while i*i <= n:
        while n%i == 0:
            factors[i] = factors.get(i,0) + 1
            n //= i
        i += 1
    if n>1:
        factors[n] = factors.get(n,0) + 1
    return factors

def allowed_D(D):
    # D must be of form 3^a * 5^b with b <= 1

        if p == 3:
            continue
        elif p == 5:
            if e > 1:
                return False
        else:
            return False
    return True

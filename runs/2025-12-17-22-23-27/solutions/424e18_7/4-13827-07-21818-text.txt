def generate_matchings(players):
    # returns list of list of pairs representing perfect matchings
    # players list length must be even
    n = len(players)
    if n == 0:
        return [[]]
    # fix first player, pair with each other, then recursively match rest
    first = players[0]
    matchings = []
    for i in range(1, n):
        second = players[i]
        rest = players[1:i] + players[i+1:]
        for sub in generate_matchings(rest):
            matchings.append([(first, second)] + sub)
    return matchings

<|end|><|start|>assistant<|channel|>analysis to=python code<|message|>def reflect_point_across_line(P, A, B):
    # Reflect point P across line AB.
    # Using vector projection.
    # line defined by points A, B
    # Compute projection of vector AP onto AB, then get foot H, then reflection = 2H - P.
    # If A==B, undefined.
    Ax,Ay = A
    Bx,By = B
    Px,Py = P
    # direction vector of line AB
    dx = Bx - Ax
    dy = By - Ay
    # normalize direction
    # compute projection factor t = dot(AP, AB)/|AB|^2
    APx = Px - Ax
    APy = Py - Ay
    norm2 = dx*dx + dy*dy
    if norm2 == 0:
        raise ValueError("Degenerate line")
    t = (APx*dx + APy*dy) / norm2
    # foot H = A + t*AB
    Hx = Ax + t*dx
    Hy = Ay + t*dy
    # reflection: K' = 2H - P
    Rx = 2*Hx - Px
    Ry = 2*Hy - Py
    return (Rx, Ry)

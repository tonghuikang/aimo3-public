    # Compute F and E
    BD = BD_frac

    # F coordinates (tangent from B)
    xF = -2*BD * h**2 / (h**2 + BD**2)
    yF = -h - (BD/h) * xF
    # E coordinates (tangent from C)
    xE = 2*CD * h**2 / (h**2 + CD**2)
    yE = (CD*xE - h**2) / h

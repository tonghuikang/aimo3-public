def compute_geometry(n, s):
    # Fibonacci numbers
    def fib(k):
        if k < 0:
            raise ValueError
        a,b = 0,1
        for _ in range(k):
            a,b = b,a+b
        return a
    F = [fib(i) for i in range(n+3)] # ensure up to n+2
    d = F[n]  # BD
    a_len = F[n+2]  # BC
    # side lengths
    b_len = s - d  # CA

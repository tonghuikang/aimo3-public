def min_lcm_for_sum_bruteforce(S):
    # brute force with loops up to S
    import math
    minL = None
    best = None
    for a in range(1, S-1):
        for b in range(a+1, S):
            c = S - a - b
            if c <= b:
                continue
            # distinct
            L = a*b//math.gcd(a,b)
            L = L*c//math.gcd(L,c)
            if minL is None or L < minL:
                minL = L
                best = (a,b,c)
    return minL, best

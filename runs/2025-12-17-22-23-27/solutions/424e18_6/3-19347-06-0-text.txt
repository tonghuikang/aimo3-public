    N = 2**k
    # Leaves indexed by binary path bits from most significant (round) to least
    # Let's represent leaf index from 0 to N-1, where binary representation of index gives path bits: 
    # 0 = left child, 1 = right child at each level. Let's generate all permutations of 1..N, assign to leaves.
    # We'll test condition: for each internal node, the min label in its right subtree must be less than min label in left subtree.
    N = 2**k
    leaves = list(range(N))
    valid = []
    for perm in itertools.permutations(range(1,N+1)):
        # mapping leaf->label
        # we can store in array labels[leaf] = perm[leaf]
        labels = perm
        ok = True
        # iterate over all internal nodes: there are N-1 internal nodes.
        # Represent nodes by (depth, start) where depth from 0 (root) to k-1, start index of leftmost leaf of that node's subtree (size 2^(k-depth)).
        # For each node, we need to compute min of left child subtree and right child subtree.
        def check_node(start, size):
            nonlocal ok
            if size == 1:
                return
            half = size // 2
            # left subtree leaves indices start..start+half-1
            # right subtree leaves indices start+half..start+size-1
            left_min = min(labels[i] for i in range(start, start+half))

                ok = False
                return
            # recursively check children
            check_node(start, half)
            check_node(start+half, half)
        check_node(0, N)
        if ok:
            valid.append(labels)
    return valid

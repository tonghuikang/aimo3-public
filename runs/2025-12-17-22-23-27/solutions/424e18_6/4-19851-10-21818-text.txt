    """players: list of player ids, assume they need to be paired arbitrarily.
    Returns list of list of pairs (tuple) of pairings (order doesn't matter)."""
    if not players:
        return [[]]
    # choose first player
    first=players[0]
    rest=players[1:]
    matchings=[]
    # pair first with any other player
    for i,other in enumerate(rest):
        pair=(first, other)
        remaining=rest[:i] + rest[i+1:]

    return matchings

from collections import defaultdict

def all_matchings(players):
    """players list; return all possible pairings as list of list of pairs (tuples)"""
    # return list of matchings (list of pairs)
    if not players:
        return [[]]
    # pick first player and pair with any other
    first = players[0]
    matchings = []
    for i in range(1, len(players)):
        second = players[i]
        rest = players[1:i] + players[i+1:]
        for sub in all_matchings(rest):
            matchings.append([(first, second)] + sub)

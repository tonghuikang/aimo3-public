def find_all_roots_for_n(n, t_min_factor=1.0001, t_max_factor=30, steps=2000):
    a = mp.mpf(fibonacci(n+2))
    t_min = a * t_min_factor
    t_max = a * t_max_factor
    # sample values
    ts = [t_min + (t_max - t_min) * i/(steps-1) for i in range(steps)]
    vals = [D_of_t_numeric(n, t) for t in ts]
    # find sign changes
    roots = []
    for i in range(len(ts)-1):
        v1 = vals[i]
        v2 = vals[i+1]
        if mp.isnan(v1) or mp.isnan(v2):
            continue
        if v1 == 0:
            roots.append(ts[i])
        elif v1 * v2 < 0:
            # bracket between ts[i] and ts[i+1]; use bisection to find root
            lo, hi = ts[i], ts[i+1]
            # refine with bisection
            for _ in range(50):
                mid = (lo + hi)/2
                vmid = D_of_t_numeric(n, mid)
                if vmid == 0:
                    lo = hi = mid
                    break
                if v1 * vmid < 0:
                    hi = mid
                    v2 = vmid
                else:
                    lo = mid
                    v1 = vmid
            roots.append((lo+hi)/2)
    return roots

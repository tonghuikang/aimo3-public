def geometry_data(u, v, r, h):
    # returns dict of points: A, B, C, D, E, F, K, Kprime, N, residual
    # using numpy for floats, but for intersection find K via solving circles analytically using sympy (but convert to float)
    # Let's implement using sympy for robust intersection.
    import sympy as sp
    # define points as sympy Points
    I = sp.Point(0, h)
    D = sp.Point(r, h)
    B = sp.Point(r, h + u)
    C = sp.Point(r, h - v)
    # angles
    phi = 2*sp.atan(u/r)
    beta = 2*sp.atan(v/r)
    # points F, E
    F = sp.Point(r*sp.cos(phi), h + r*sp.sin(phi))
    E = sp.Point(r*sp.cos(beta), h - r*sp.sin(beta))
    # direction vectors
    v_AB = sp.Point(-sp.sin(phi), sp.cos(phi))
    v_AC = sp.Point(sp.sin(beta), sp.cos(beta))
    # solve for t
    t, s = sp.symbols('t s')
    sol = sp.solve((sp.Eq(B.x + t*v_AB.x, C.x + s*v_AC.x),
                    sp.Eq(B.y + t*v_AB.y, C.y + s*v_AC.y)), (t, s), dict=True)
    sol = sol[0]
    t_val = sol[t]
    A = sp.Point(B.x + t_val*v_AB.x, B.y + t_val*v_AB.y)
    # circumcircles
    circ_ABC = sp.Circle(A, B, C)
    circ_AFE = sp.Circle(A, F, E)
    # intersection:
    inter = circ_ABC.intersection(circ_AFE)
    # find K not equal to A
    K = None
    for p in inter:
        if not p.equals(A):
            K = p
    # N foot from D onto EF
    line_EF = sp.Line(E, F)
    N = line_EF.projection(D)
    # K'
    Kprime = K.reflect(line_EF)
    # compute residual: distance of Kprime to circumcircle of K,N,B minus radius
    circ_KNB = sp.Circle(K, N, B)
    residual = Kprime.distance(circ_KNB.center) - circ_KNB.radius
    return {
        'A':A, 'B':B, 'C':C, 'D':D, 'E':E, 'F':F,
        'K':K, 'Kprime':Kprime, 'N':N,
        'residual':sp.N(residual, 12)
    }

    for _ in range(60):

            f_high = f_mid
        else:
            h_low = h_mid
            f_low = f_mid

def circumcenter(P1, P2, P3):
    # returns center (x,y) and radius
    x1,y1 = P1
    x2,y2 = P2
    x3,y3 = P3
    # Using formula
    D = 2*(x1*(y2-y3) + x2*(y3-y1) + x3*(y1-y2))
    if abs(D) < 1e-12:
        raise ValueError("Collinear points")
    Ux = ((x1**2+y1**2)*(y2-y3) + (x2**2+y2**2)*(y3-y1) + (x3**2+y3**2)*(y1-y2)) / D
    Uy = ((x1**2+y1**2)*(x3-x2) + (x2**2+y2**2)*(x1-x3) + (x3**2+y3**2)*(x2-x1)) / D
    O = np.array([Ux, Uy])
    R = np.linalg.norm(O - np.array(P1))
    return O, R

def intersect_two_circles(O1, R1, O2, R2):
    # returns two intersection points as arrays (list)
    # Compute distance between centers
    d = np.linalg.norm(O2 - O1)
    if d > R1 + R2 + 1e-12 or d < abs(R1 - R2) - 1e-12:
        return []  # no intersection
    # a = (R1^2 - R2^2 + d^2) / (2d)
    a = (R1**2 - R2**2 + d**2) / (2*d)
    # h = sqrt(R1^2 - a^2)
    h_sq = max(R1**2 - a**2, 0.0)
    h = np.sqrt(h_sq)
    # point P2 = O1 + a*(O2-O1)/d
    P2 = O1 + a * (O2 - O1) / d
    # offset vector
    offset = h * np.array([-(O2[1] - O1[1]) / d, (O2[0] - O1[0]) / d])
    X1 = P2 + offset
    X2 = P2 - offset
    return [X1, X2]

def foot_of_perpendicular(P, A, B):
    # foot from point P to line AB
    A = np.array(A)
    B = np.array(B)
    P = np.array(P)
    AB = B - A
    t = np.dot(P - A, AB) / np.dot(AB, AB)
    proj = A + t * AB
    return proj

def reflect_point_across_line(P, A, B):
    # reflect point P across line AB
    # projection onto line AB:
    proj = foot_of_perpendicular(P, A, B)
    # reflect: P' = 2*proj - P
    return 2*proj - P

def cyclic_residual(u, v, r):
    # h = 0
    # define points
    D = np.array([r, 0.0])
    B = np.array([r, u])
    C = np.array([r, -v])
    # angles
    phi = 2*np.arctan(u/r)
    beta = 2*np.arctan(v/r)
    # Points E,F
    F = np.array([r*np.cos(phi), r*np.sin(phi)])
    E = np.array([r*np.cos(beta), -r*np.sin(beta)])
    # Compute A: intersection of AB and AC
    # directions
    v_AB = np.array([-np.sin(phi), np.cos(phi)])
    v_AC = np.array([np.sin(beta), np.cos(beta)])
    # Solve B + t*v_AB = C + s*v_AC
    # Solve linear system for t,s
    # 2 equations: B + t*v_AB - C = s*v_AC
    # Represent as: t*v_AB - s*v_AC = C - B
    A_mat = np.column_stack((v_AB, -v_AC))
    rhs = C - B
    # solve for t,s
    try:
        sol = np.linalg.solve(A_mat, rhs)
    except np.linalg.LinAlgError:
        # maybe degenerate
        return None
    t_val = sol[0]
    # s_val = sol[1] not needed
    A_pt = B + t_val * v_AB
    # circumcircles
    O1, R1 = circumcenter(A_pt, B, C)
    O2, R2 = circumcenter(A_pt, F, E)
    # Intersections of circles:
    inter_pts = intersect_two_circles(O1, R1, O2, R2)
    if len(inter_pts) == 0:
        return None
    # Choose K not equal to A_pt
    K = None
    for pt in inter_pts:
        if np.linalg.norm(pt - A_pt) > 1e-8:
            K = pt
            break
    if K is None:
        return None
    # N foot from D onto line EF
    N = foot_of_perpendicular(D, E, F)
    # K' reflection across line EF
    Kprime = reflect_point_across_line(K, E, F)
    # Determine cyclic determinant
    # Build matrix
    pts = np.array([K, N, Kprime, B])
    M = np.column_stack([pts[:,0], pts[:,1], np.sum(pts**2, axis=1), np.ones(4)])
    det = np.linalg.det(M)
    return det

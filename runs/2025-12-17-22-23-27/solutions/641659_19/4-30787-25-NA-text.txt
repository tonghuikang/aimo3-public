def incircle_center(A,B,C):
    # returns incenter coordinates (as complex)
    # using formula: intersection of angle bisectors => weighted sum of vertices by side lengths
    a = abs(B-C)
    b = abs(C-A)
    c = abs(A-B)
    P = a*A + b*B + c*C
    I = P/(a+b+c)
    return I

def foot_point(P, X, Y):

    # compute BN vector

    ratio = CT * NB / (BT * NE)
    # cyclic residual: determinant
    detval = det_cyclic(K, N, Kprime, B)

        if np.isnan(v1) or np.isnan(v2):
            continue
        if v1 == 0:

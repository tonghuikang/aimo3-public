        x2,y2 = P2[0], P2[1]
        x3,y3 = P3[0], P3[1]

        if abs(d) < 1e-30:
            return None
        ux = ((x1**2+y1**2)*(y2-y3)+(x2**2+y2**2)*(y3-y1)+(x3**2+y3**2)*(y1-y2))/d
        uy = ((x1**2+y1**2)*(x3-x2)+(x2**2+y2**2)*(x1-x3)+(x3**2+y3**2)*(x2-x1))/d
        return mp.matrix([ux, uy])
    O_bkn = circumcenter(B, K, N)
    if O_bkn is None:
        return mp.nan
    R_bkn = mp.sqrt((B[0]-O_bkn[0])**2 + (B[1]-O_bkn[1])**2)
    dist_Kp = mp.sqrt((Kp[0]-O_bkn[0])**2 + (Kp[1]-O_bkn[1])**2)
    return dist_Kp - R_bkn

    # foot from P onto line QR
    # line defined by Q,R
    # compute projection of vector PQ onto QR
    Q = sp.Point(Q)
    R = sp.Point(R)
    P = sp.Point(P)
    # param t such that foot = Q + t*(R-Q)
    v = R - Q
    t = ((P - Q).dot(v)) / (v.dot(v))
    foot = Q + t * v
    return sp.Point(foot)

def incircle_center(A,B,C):
    # side lengths opposite A: a = BC, b = CA, c = AB
    a = sp.Point(B).distance(sp.Point(C))
    b = sp.Point(C).distance(sp.Point(A))
    c = sp.Point(A).distance(sp.Point(B))
    # coordinates weighted by side lengths
    I = (a*sp.Point(A) + b*sp.Point(B) + c*sp.Point(C)) / (a+b+c)
    return sp.Point(I)

def circumcircle_center_and_radius(P1,P2,P3):
    # use sympy's Circle
    C = sp.Circle(sp.Point(P1), sp.Point(P2), sp.Point(P3))
    return C.center, C.radius

def second_intersection_of_circles(circle1, circle2, P):
    # circle1 and circle2 are sympy Circle objects
    # find intersection points, return the one not equal to P
    pts = list(circle1.intersection(circle2))
    # pts may be two points (maybe duplicates)
    # Choose point not approximately equal to P
    for pt in pts:
        if pt.distance(sp.Point(P)) > 1e-6:
            return sp.Point(pt)
    return None

def reflect_across_line(P, L1, L2):
    # reflect point P across line through L1, L2
    # Use sympy geometry: line, then projection
    line = sp.Line(sp.Point(L1), sp.Point(L2))
    foot = line.projection(sp.Point(P))
    # reflection = 2*foot - P
    return sp.Point(2*foot.x - P.x, 2*foot.y - P.y)

def cyclic_check(B, K, N, Kp):
    # compute circle through B, K, N and see if Kp lies on it
    try:
        circle = sp.Circle(sp.Point(B), sp.Point(K), sp.Point(N))
        # check distance
        return abs(circle.center.distance(sp.Point(Kp)) - circle.radius) < 1e-6
    except Exception as e:
        return False

def triangle_geometry(p,q,r):
    # p = BD, q = CD, r = incircle radius
    # Coordinates as per earlier
    B = (-p, -r)
    C = (q, -r)
    D = (0.0, -r)
    # Compute F, E
    denom_p = p**2 + r**2
    xF = -2*p*r**2 / denom_p
    yF = r*(p**2 - r**2) / denom_p
    F = (xF, yF)
    denom_q = q**2 + r**2
    xE = 2*q*r**2 / denom_q
    yE = r*(q**2 - r**2) / denom_q
    E = (xE, yE)
    # Intersection of lines BF and CE: find A
    # Solve B + t*(F - B) = C + s*(E - C)
    xb, yb = B
    xf, yf = F
    xc, yc = C
    xe, ye = E
    # direction vectors
    v1 = (xf - xb, yf - yb)
    v2 = (xe - xc, ye - yc)
    # Solve xb + t*v1x = xc + s*v2x
    #      yb + t*v1y = yc + s*v2y
    # => t*v1x - s*v2x = xc - xb
    #    t*v1y - s*v2y = yc - yb
    # Solve 2x2 linear system for t,s.
    # Write matrix [[v1x, -v2x],[v1y, -v2y]] * [t, s]^T = [xc-xb, yc-yb]
    a11 = v1[0]; a12 = -v2[0]
    a21 = v1[1]; a22 = -v2[1]
    det = a11*a22 - a12*a21
    if abs(det) < 1e-12:
        # lines parallel? shouldn't happen
        return None
    rhs1 = xc - xb
    rhs2 = yc - yb
    t = (rhs1*a22 - a12*rhs2) / det
    # compute A
    Ax = xb + t*v1[0]

def circumcenter_and_radius(P1, P2, P3):
    # Compute circumcenter of triangle given points as tuples
    # Using formula based on perpendicular bisectors.
    (x1,y1), (x2,y2), (x3,y3) = P1, P2, P3
    # Compute midpoints and slopes
    # Solve linear equations for circumcenter (ox,oy):
    d = 2*(x1*(y2 - y3) + x2*(y3 - y1) + x3*(y1 - y2))
    if abs(d) < 1e-12:
        # points collinear
        return (float('nan'), float('nan'))
    ux = ((x1**2 + y1**2)*(y2 - y3) + (x2**2 + y2**2)*(y3 - y1) + (x3**2 + y3**2)*(y1 - y2)) / d
    uy = ((x1**2 + y1**2)*(x3 - x2) + (x2**2 + y2**2)*(x1 - x3) + (x3**2 + y3**2)*(x2 - x1)) / d
    O = (ux, uy)
    R = math.hypot(x1-ux, y1-uy)
    return O, R

def intersect_two_circles(O1,R1,O2,R2):
    (x0,y0) = O1
    (x1,y1) = O2
    dx = x1 - x0
    dy = y1 - y0
    d = math.hypot(dx, dy)
    if d == 0:
        # same center
        return []
    # Check for existence
    if d > R1 + R2 + 1e-12 or d < abs(R1 - R2) - 1e-12:
        # no intersection
        return []
    # distance from O1 to line of intersection points
    a = (R1**2 - R2**2 + d**2) / (2*d)
    # height from that line to the intersection points
    h_sq = R1**2 - a**2
    if h_sq < -1e-12:
        return []
    h = math.sqrt(max(h_sq, 0.0))
    # point P = O1 + a*(O2-O1)/d
    xm = x0 + a*dx/d
    ym = y0 + a*dy/d
    # offset vector
    rx = -dy * (h/d)
    ry = dx * (h/d)
    p1 = (xm + rx, ym + ry)
    p2 = (xm - rx, ym - ry)
    return [p1, p2]

def reflect_point_across_line(P, A, B):
    # reflect P across line AB (points given as tuples)
    # Compute projection onto line
    # vector AB
    (x1,y1) = A
    (x2,y2) = B
    (xp,yp) = P
    # Compute projection of P onto line AB:
    # t = ((P-A)·(B-A))/|B-A|^2
    dx = x2 - x1
    dy = y2 - y1
    if dx == 0 and dy == 0:
        return None
    t = ((xp - x1)*dx + (yp - y1)*dy) / (dx*dx + dy*dy)
    projx = x1 + t*dx
    projy = y1 + t*dy
    # Reflection point = 2*proj - P
    return (2*projx - xp, 2*projy - yp)

def projection_point_to_line(P, A, B):
    # foot of perpendicular from P to line AB
    (x1,y1) = A
    (x2,y2) = B
    (xp,yp) = P
    dx = x2 - x1
    dy = y2 - y1
    if dx == 0 and dy == 0:
        return None
    t = ((xp - x1)*dx + (yp - y1)*dy) / (dx*dx + dy*dy)
    projx = x1 + t*dx
    projy = y1 + t*dy
    return (projx, projy)

def distance(P,Q):
    return math.hypot(P[0]-Q[0], P[1]-Q[1])

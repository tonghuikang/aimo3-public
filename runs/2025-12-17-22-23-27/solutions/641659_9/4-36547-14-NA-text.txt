    # foot from point P to line Q1Q2
    # Using projection formula
    # vector Q1Q2
    v = sp.Matrix([Q2.x - Q1.x, Q2.y - Q1.y])
    w = sp.Matrix([P.x - Q1.x, P.y - Q1.y])
    # projection scalar t = (v·w) / (v·v)
    t = (v.dot(w)) / (v.dot(v))
    # foot point = Q1 + t * v
    foot = sp.Point(Q1.x + t * v[0], Q1.y + t * v[1])
    return foot

    pts = []

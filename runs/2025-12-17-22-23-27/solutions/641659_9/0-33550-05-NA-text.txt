    # we want to find a root where det = 0. Might be multiple roots. We'll assume sign change.
    if det_low == 0:
        return low
    if det_high == 0:
        return high
    # Try to find sign change by scanning
    # If same sign, we might not have root.
    # Let's just try to locate root by scanning many s.
    # We'll do a simple approach: sample many s values and look for sign changes.
    import numpy as np
    # sample 200 points between low and high
    ss = np.linspace(low, high, 401)
    dets = [compute_triangle_coords(n, sv)["cyclic_det"] for sv in ss]
    # find intervals where sign changes
    intervals = []
    for i in range(len(ss)-1):
        if dets[i]==0:
            intervals.append((ss[i],ss[i]))
        elif dets[i]*dets[i+1] < 0:
            intervals.append((ss[i], ss[i+1]))
    # If no interval found, enlarge high
    if not intervals:
        # try larger high
        for factor in [2,5,10]:
            new_high = high*factor
            ss = np.linspace(low, new_high, 401)
            dets = [compute_triangle_coords(n, sv)["cyclic_det"] for sv in ss]
            for i in range(len(ss)-1):
                if dets[i]*dets[i+1] < 0:
                    intervals.append((ss[i], ss[i+1]))
            if intervals:
                break
    if not intervals:
        return None
    # Take first interval
    lo, hi = intervals[0]
    # refine by bisection
    for _ in range(50):
        mid = (lo+hi)/2
        det_mid = compute_triangle_coords(n, mid)["cyclic_det"]
        if det_mid == 0:
            return mid
        if det_low * det_mid <= 0:
            hi = mid
            det_high = det_mid
        else:
            lo = mid
            det_low = det_mid
    return (lo+hi)/2

def circumcenter(P1, P2, P3):
    # returns circumcenter of triangle (P1,P2,P3)
    x1,y1 = P1
    x2,y2 = P2
    x3,y3 = P3
    # Solve linear system: 2*(x2-x1)*Ux + 2*(y2-y1)*Uy = x2^2 + y2^2 - x1^2 - y1^2
    #                     2*(x3-x1)*Ux + 2*(y3-y1)*Uy = x3^2 + y3^2 - x1^2 - y1^2
    A = np.array([[2*(x2-x1), 2*(y2-y1)], [2*(x3-x1), 2*(y3-y1)]], dtype=float)
    B = np.array([x2*x2 + y2*y2 - x1*x1 - y1*y1, x3*x3 + y3*y3 - x1*x1 - y1*y1], dtype=float)
    # Solve; if determinant zero, points are collinear
    det = np.linalg.det(A)
    if abs(det) < 1e-12:
        return None
    sol = np.linalg.solve(A, B)
    return (sol[0], sol[1])

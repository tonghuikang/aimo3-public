def perfect_matchings(lst):
    # generate all perfect matchings of lst (list of elements)
    # Use recursion: pick first element, pair it with any other, then recursively match rest.
    if not lst:
        yield []
        return
    a = lst[0]
    for i in range(1, len(lst)):
        b = lst[i]
        rest = lst[1:i] + lst[i+1:]
        for submatching in perfect_matchings(rest):
            yield [(a,b)] + submatching

def generate_orders(players):
    # returns set of tuples representing final ordering (higher scores first)
    players = list(players)
    n = len(players)
    if n == 1:
        return {tuple(players)}
    result = set()
    # generate all perfect matchings
    for matching in perfect_matchings(players):
        # compute winners and losers
        winners = []
        losers = []
        for a,b in matching:
            if a > b:
                winners.append(a)
                losers.append(b)
            else:
                winners.append(b)
                losers.append(a)
        # recursively generate orders for losers and winners groups
        for lo in generate_orders(losers):
            for wi in generate_orders(winners):
                # final ordering: losers first then winners
                result.add(lo + wi)
    return result

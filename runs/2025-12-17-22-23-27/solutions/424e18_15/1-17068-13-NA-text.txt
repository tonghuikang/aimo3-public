def pairings(participants):
    # participants is list of identifiers; we need to generate all possible perfect matchings (unordered pairs) for them
    # number of participants is even.
    # naive recursion:
    if not participants:
        yield []
        return
    first = participants[0]
    for i in range(1, len(participants)):
        second = participants[i]
        rest = participants[1:i] + participants[i+1:]
        for sub in pairings(rest):
            yield [(first, second)] + sub

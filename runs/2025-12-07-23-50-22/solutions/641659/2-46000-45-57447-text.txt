One solution is λ=0 (since A lies on both circles). Indeed plugging λ=0 gives |A-O1|^2 = R1^2 because A is on circle ABC. So constant term is zero. So quadratic reduces: λ*( (u·u) λ + 2 u·(A - O1) ) = 0. So λ=0 or λ = -2 u·(A - O1) / (u·u). That gives K.

Thus K = A + λ * u, with λ computed.

Thus we can compute K easily.

Then K' is reflection of K across EF: reflect K across line defined by points E and F.

Reflection formula: For point P, line given by normalized direction, can compute projection onto line, then symmetric.

Alternatively, use vector projection: If line defined by point Q and unit direction d, then projection of P onto line is Q + ((P - Q)·d) d. Then reflection: P' = 2*proj - P.

def cyclic_condition(n, t):
    pts = compute_triangle_points(n, t)

def ratio_CT_NB_BT_NE(n, t):

    B = pts['B']; C = pts['C']; N = pts['N']; E = pts['E']

    # Center O lies on line perpendicular to BN at B and equidistant to B and K
    K = pts['K']
    # compute line BN direction
    BN_dir = (N[0]-B[0], N[1]-B[1])
    # O must satisfy dot((O-B), BN_dir) = 0
    # And distance(O,B) = distance(O,K)
    # Let O = (ox, oy)
    # Condition1: ox*BN_dir[0] + oy*BN_dir[1] = 0
    # Condition2: ox**2 + oy**2 = (ox-Kx)**2 + (oy-Ky)**2
    # Expand condition2: ox**2 + oy**2 = ox**2 - 2 ox Kx + Kx**2 + oy**2 - 2 oy Ky + Ky**2
    # Cancel ox**2, oy**2 => 0 = -2 ox Kx + Kx**2 - 2 oy Ky + Ky**2
    # => 2 ox Kx + 2 oy Ky = Kx**2 + Ky**2
    # So linear: Kx*ox + Ky*oy = (Kx**2 + Ky**2)/2
    # Together with condition1, we can solve linear 2x2.
    Kx, Ky = K
    # Solve linear equations:
    # [BN_dir[0], BN_dir[1]] . [ox, oy] = 0
    # [Kx, Ky] . [ox, oy] = (Kx**2 + Ky**2)/2
    a1,b1 = BN_dir
    a2,b2 = Kx, Ky
    c1 = 0.0
    c2 = (Kx*Kx + Ky*Ky)/2.0
    # Solve for ox, oy:
    det = a1*b2 - a2*b1
    if abs(det) < 1e-12:
        # degenerate; maybe BN_dir perpendicular to K? But unlikely.
        raise ValueError("Degenerate linear system")
    ox = (c1*b2 - c2*b1) / det
    oy = (a1*c2 - a2*c1) / det
    O = (ox, oy)
    # radius r = distance O to B
    r = math.hypot(ox - B[0], oy - B[1])
    # Intersection with line BC (x-axis y=0):
    # Solve (x-ox)^2 + (0-oy)^2 = r^2 => (x-ox)^2 = r^2 - oy^2
    # Compute rad2 = r^2 - oy^2
    rad2 = r*r - oy*oy
    if rad2 < -1e-12:
        # No intersection due to numeric error
        rad2 = 0.0
    # sqrt
    sqrt_val = math.sqrt(max(rad2, 0.0))
    # Solutions: x = ox ± sqrt_val
    # One solution is x = 0 (point B). So check which root corresponds to zero:
    # Determine sign: if ox - sqrt_val approx 0, then other root is ox + sqrt_val = T_x.
    # else if ox + sqrt_val approx 0, then T_x = ox - sqrt_val.
    if abs(ox - sqrt_val) < 1e-6:
        T_x = ox + sqrt_val
    elif abs(ox + sqrt_val) < 1e-6:
        T_x = ox - sqrt_val
    else:
        # choose root not zero
        # pick the one with larger magnitude (should be >0)
        candidates = [ox - sqrt_val, ox + sqrt_val]
        # choose one not close to zero
        T_x = candidates[0] if abs(candidates[0]) > 1e-6 else candidates[1]
    T = (T_x, 0.0)
    # lengths:
    CT = math.hypot(C[0] - T[0], C[1] - T[1])
    NB = math.hypot(N[0] - B[0], N[1] - B[1])
    BT = math.hypot(B[0] - T[0], B[1] - T[1])
    NE = math.hypot(N[0] - E[0], N[1] - E[1])
    ratio = (CT * NB) / (BT * NE)
    return ratio

def ratio_CT_NB_BT_NE(n, t):
    pts = compute_triangle_points(n, t)
    B = pts['B']; C = pts['C']; N = pts['N']; E = pts['E']; K = pts['K']
    # Center O of circle tangent to BN at B and passing through B,K
    # Solve linear system for O = (ox, oy)
    BN_dir = (N[0] - B[0], N[1] - B[1])
    # eq1: ox * BN_dir[0] + oy * BN_dir[1] = 0
    # eq2: Kx*ox + Ky*oy = (Kx^2 + Ky^2)/2
    Kx, Ky = K
    a1, b1 = BN_dir
    a2, b2 = Kx, Ky
    c1 = 0.0
    c2 = (Kx*Kx + Ky*Ky)/2.0
    det = a1*b2 - a2*b1
    if abs(det) < 1e-12:
        # degenerate, maybe BN_dir is zero? Should not happen.
        raise ValueError("Degenerate linear system")
    ox = (c1*b2 - c2*b1) / det
    oy = (a1*c2 - a2*c1) / det
    O = (ox, oy)
    r = math.hypot(ox - B[0], oy - B[1])
    # Intersection of circle with BC (y=0)
    # (x - ox)^2 + (0 - oy)^2 = r^2 => (x - ox)^2 = r^2 - oy^2
    rad2 = r*r - oy*oy
    if rad2 < 0:
        rad2 = 0.0
    sqrt_val = math.sqrt(rad2)
    # solutions: x = ox ± sqrt_val
    # one solution is x=0 (B). Determine which.
    # Choose root not close to 0.
    x1 = ox + sqrt_val
    x2 = ox - sqrt_val
    # Choose T as the one not approx 0.
    if abs(x1) > 1e-6:
        T_x = x1
    else:
        T_x = x2
    T = (T_x, 0.0)
    # lengths
    CT = math.hypot(C[0] - T[0], C[1] - T[1])
    NB = math.hypot(N[0] - B[0], N[1] - B[1])
    BT = math.hypot(B[0] - T[0], B[1] - T[1])
    NE = math.hypot(N[0] - E[0], N[1] - E[1])
    ratio = (CT * NB) / (BT * NE)
    return ratio

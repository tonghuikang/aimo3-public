    # residual1: I.x - BD
    res1 = vals["I"][0] - BD
    # compute O' = intersection of EF and perp bisector of BN
    # line EF: given two points E, F. Represent as (point, direction).
    E = vals["E"]
    F = vals["F"]
    # direction vector for EF:
    vEF = F - E
    # perpendicular bisector of BN: compute midpoint M_BN and direction perp to BN.
    B = np.array([0.0,0.0])
    N = vals["N"]
    M_BN = (B + N) / 2.0
    dirBN = N - B
    # direction of perp bisector = perpendicular to BN (i.e., rotate 90)
    perpBN = np.array([-dirBN[1], dirBN[0]])
    # Solve intersection of line EF (E + s*vEF) and line through M_BN with direction perpBN: M_BN + t*perpBN
    # Solve for s,t: E + s*vEF = M_BN + t*perpBN
    # 2 equations:
    A_mat = np.column_stack((vEF, -perpBN))

    try:

        OT = (M_BK[0] + s_val2*perpBK[0], M_BK[1] + s_val2*perpBK[1])
    R_T = mp.sqrt((OT[0] - B[0])**2 + (OT[1] - B[1])**2)
    disc = R_T**2 - OT[1]**2
    if disc < 0:
        disc = mp.mpf('0')
    sqrt_disc = mp.sqrt(disc)
    x_T1 = OT[0] + sqrt_disc
    x_T2 = OT[0] - sqrt_disc
    # choose T not at B
    if mp.fabs(x_T1) < mp.fabs(x_T2):
        x_T = x_T2
    else:
        x_T = x_T1
    T = (x_T, mp.mpf('0'))
    # lengths for ratio
    CT = mp.fabs(C[0] - T[0])
    BT = mp.fabs(T[0] - B[0])
    NB_len = mp.sqrt((N[0] - B[0])**2 + (N[1] - B[1])**2)
    NE_len = mp.sqrt((N[0] - E[0])**2 + (N[1] - E[1])**2)
    ratio = (CT * NB_len) / (BT * NE_len) if (BT != 0 and NE_len != 0) else mp.nan
    return {
        "I":I, "D":D, "F":F, "E":E, "N":N,
        "K":K, "Kprime":Kprime, "Oprime":Oprime,
        "cyclic_res":cyclic_res,
        "ratio":ratio,
        "BD_val":BD_val, "L":L, "b":b, "c":c
    }

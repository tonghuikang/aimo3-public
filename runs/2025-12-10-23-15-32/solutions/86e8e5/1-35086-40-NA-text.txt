    # returns sorted list of divisors
    divs = []
    i = 1
    while i*i <= n:
        if n % i == 0:
            divs.append(i)
            if i != n//i:
                divs.append(n//i)
        i += 1
    return sorted(divs)

def f_bruteforce(n, limit=10000):

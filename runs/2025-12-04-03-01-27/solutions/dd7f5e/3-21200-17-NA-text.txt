    expr = 1 + x**d

    # Now generate all combinations of subsets of factors (with exponent 1 each)

    divs=set()

            # product of selected polys
            prod=sp.Poly(1, x)  # start with 1

            # ensure degree <=8
            if prod.degree() <= 8:
                # record polynomial coefficients as tuple up to deg<=8

                divs.add(tuple(coeffs_full))

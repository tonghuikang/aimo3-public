    # returns list of divisors sorted
    divs = set()
    for i in range(1, int(math.isqrt(n))+1):
        if n % i == 0:
            divs.add(i)
            divs.add(n//i)
    return sorted(divs)

def f_by_search(N):
    # find minimal x such that exists a,b distinct proper divisors of x with a+b = N-x
    # a,b < x, distinct
    # we also need that a,b are divisors of x
    # The sum N-x must be positive? Yes, need a,b >0 => S>0 => x < N.
    # So we search x from 1 to N-1.
    # Minimal x, but x must be >=? We search increasing x.
    for x in range(1, N):

        # we need S > 0
        if S <= 0:
            continue
        # proper divisors of x
        divs = divisors(x)
        # remove x
        if x in divs:
            divs.remove(x)

                return x

                i += 1
            else:
                j -= 1
    # if not found, maybe need x=N? But then S=0 impossible
    # return None
    return None

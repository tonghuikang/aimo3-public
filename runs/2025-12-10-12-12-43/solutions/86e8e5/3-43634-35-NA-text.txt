    # return one triple achieving min lcm
    min_l = None
    best = None
    for a in range(1, n):
        for b in range(a+1, n):
            c = n - a - b
            if c <= b:
                continue
            l = sp.ilcm(a,b,c)
            if min_l is None or l < min_l:
                min_l = l
                best = (a,b,c)
    return min_l, best

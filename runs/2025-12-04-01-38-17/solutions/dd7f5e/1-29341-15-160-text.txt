            q, r = sp.div(1 + t**d, D_poly, domain=sp.ZZ)
            if r != 0:
                print("Division not exact for subset", subset)
                valid=False
                break
            # Verify product
            A_poly = sp.expand(t**a * D_poly)
            prod = sp.expand(A_poly * B_poly)
            # Expected product is t^a + t**(a+d)
            expected = sp.expand(t**a + t**(a+d))
            if prod != expected:
                print("Product mismatch", a, subset)
                valid=False
                break

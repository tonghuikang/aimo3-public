def dot(u, v):
    return u[0]*v[0] + u[1]*v[1]

def sub(u, v):
    return (u[0] - v[0], u[1] - v[1])
def add(u, v):
    return (u[0] + v[0], u[1] + v[1])
def mul_scalar(u, t):
    return (u[0]*t, u[1]*t)

def length(u):
    return mp.sqrt(dot(u,u))

def project_point(P, Q, X):
    # project X onto line PQ
    v = sub(Q, P)
    t = dot(sub(X, P), v) / dot(v, v)
    return add(P, mul_scalar(v, t))

def circumcenter(P, Q, R):
    # returns center and radius
    # Using perpendicular bisector intersection
    # Compute midpoints
    mPQ = mul_scalar(add(P, Q), 0.5)
    mPR = mul_scalar(add(P, R), 0.5)
    # direction vectors
    dPQ = sub(Q, P)
    dPR = sub(R, P)
    # perpendicular vectors
    pPQ = (-dPQ[1], dPQ[0])
    pPR = (-dPR[1], dPR[0])
    # Solve mPQ + t1 pPQ = mPR + t2 pPR
    # => t1 pPQ - t2 pPR = mPR - mPQ
    A_mat = mp.matrix([[pPQ[0], -pPR[0]], [pPQ[1], -pPR[1]]])
    b_vec = (mPR[0] - mPQ[0], mPR[1] - mPQ[1])
    # Solve linear equations
    sol = mp.lu_solve(A_mat, mp.matrix(b_vec))
    t1 = sol[0]
    center = (mPQ[0] + pPQ[0]*t1, mPQ[1] + pPQ[1]*t1)
    radius = length(sub(center, P))
    return center, radius

def circle_intersections(O1, r1, O2, r2):
    # return two intersection points (as tuples). if no intersection returns None.
    # using standard formulas.
    d = length(sub(O2, O1))
    if d == 0:
        return None
    # check for existence
    if d > r1 + r2 + 1e-12:
        return None
    if d < abs(r1 - r2) - 1e-12:
        return None
    a = (r1**2 - r2**2 + d**2) / (2*d)
    h_sq = r1**2 - a**2
    if h_sq < 0:
        h_sq = 0
    h = mp.sqrt(h_sq)
    # point P2
    O1_to_O2 = sub(O2, O1)
    P2 = add(O1, mul_scalar(O1_to_O2, a/d))
    # offset vector (perpendicular)
    offset = (-O1_to_O2[1] * (h/d), O1_to_O2[0] * (h/d))
    I1 = add(P2, offset)
    I2 = (P2[0] - offset[0], P2[1] - offset[1])
    return I1, I2

def reflect_point_across_line(P, Q, X):
    proj = project_point(P, Q, X)
    reflected = (2*proj[0] - X[0], 2*proj[1] - X[1])
    return reflected

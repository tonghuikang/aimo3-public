def compute_Qs_upto_K_optimized(K, max_T):
    # generate class candidates
    cand1 = gen_class_candidates(primes_mod1, max_T)
    cand3 = gen_class_candidates(primes_mod3, max_T)
    best1 = best_per_T(cand1)  # dict T1 -> (log2_O, mod_O, int_O)
    best3 = best_per_T(cand3)  # dict T3 -> ...
    t1_vals = sorted(best1.keys())
    t3_vals = sorted(best3.keys())
    best_log2N = [float('inf')]*(K+1)
    best_N_mod = [None]*(K+1)
    best_e = [None]*(K+1)
    best_O_int = [None]*(K+1)
    pow10 = [10**i for i in range(K+1)]
    for t1 in t1_vals:
        H = t1
        numerator_base = [2*pow10[k] - H + 2 for k in range(K+1)]
        # limit t3 such that product <= max_T
        max_t3_allowed = max_T // t1
        # iterate t3 values up to this limit
        for t3 in t3_vals:
            if t3 > max_t3_allowed:
                break
            T = t1 * t3
            log2_O1, mod_O1, int_O1 = best1[t1]
            log2_O3, mod_O3, int_O3 = best3[t3]
            log2_O = log2_O1 + log2_O3
            O_mod = (mod_O1 * mod_O3) % M_mod
            O_int = int_O1 * int_O3
            for k in range(1, K+1):
                numerator = numerator_base[k]
                if numerator % T != 0:
                    continue
                q = numerator // T
                if q % 2 == 0:
                    continue
                e = (q + 1)//2
                log2_N = e + log2_O
                if log2_N < best_log2N[k]:
                    best_log2N[k] = log2_N
                    best_N_mod[k] = pow(2, e, M_mod) * O_mod % M_mod
                    best_e[k] = e
                    best_O_int[k] = O_int
    return best_N_mod[1:], best_log2N[1:], best_e[1:], best_O_int[1:]

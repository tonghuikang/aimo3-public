def intersect_circles(O1, R1, O2, R2):
    # O1, O2 are mp.matrix of length 2
    dx = O2[0] - O1[0]
    dy = O2[1] - O1[1]
    d = mp.sqrt(dx*dx + dy*dy)
    # No intersection if d > R1+R2 or d < |R1 - R2|
    if d == 0:
        return []  # coincident
    # ex, ey unit vectors
    ex = mp.matrix([dx/d, dy/d])
    # compute x (distance from O1 to point P)
    x = (R1**2 - R2**2 + d**2) / (2*d)
    # compute height
    # due to numeric errors, ensure non-negative
    val = R1**2 - x**2
    if val < 0:
        # maybe due to tiny negative
        if val > -1e-12:
            val = 0
        else:
            return []  # no intersect
    h = mp.sqrt(val)
    # point P
    P = mp.matrix([O1[0] + x*ex[0], O1[1] + x*ex[1]])
    # orthogonal unit vector ey (rotate ex by 90 deg)
    ey = mp.matrix([-ex[1], ex[0]])
    # intersection points
    I1 = mp.matrix([P[0] + h*ey[0], P[1] + h*ey[1]])
    I2 = mp.matrix([P[0] - h*ey[0], P[1] - h*ey[1]])
    return [I1, I2]

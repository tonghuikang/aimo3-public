def get_t_and_mlist(d_factors):
    # Input: list of n indices (even numbers) used in product d
    # Compute t = v2(n) (should be same for all)
    t_vals = {v2(n) for n in d_factors}
    if len(t_vals) != 1:

    t = t_vals.pop()
    # odd parts
    mlist = [n // (2**t) for n in d_factors]  # n = 2^t * m
    return t, mlist

def compute_p_from_factors(d_factors):
    # d_factors is list of n even numbers

    # q = lcm of odd parts
    q = 1
    for m in mlist:
        q = lcm(q, m)
    p = (2**(t-1)) * q

    return p

    divs = []
    i = 1
    while i*i <= n:
        if n % i == 0:
            divs.append(i)
            if i != n//i:
                divs.append(n//i)
        i+=1
    return sorted(divs)

def is_norwegian(N, target):
    # check if N has three distinct positive divisors summing to target
    divs = divisors(N)

    L = len(divs)
    if L < 3:

    # try triples
    # For speed, we could use combos
    import itertools
    for a,b,c in itertools.combinations(divs,3):
        if a + b + c == target:
            return True
    return False

    # compute M = 3^K mod mod
    if mod == 1:
        return 0
    m = mod
    power3 = 1
    while m % 3 == 0:
        m //= 3
        power3 *= 3
    if m == 1:
        # mod is power of 3 only

    # compute inv of power3 modulo m
    # ensure power3 and m are coprime
    # find inv of power3 mod m
    # using extended Euclid
    def extended_gcd(a,b):
        if b==0:

        else:
            g,x1,y1 = extended_gcd(b, a%b)
            return (g, y1, x1 - (a//b)*y1)
    g, inv, _ = extended_gcd(power3 % m, m)
    inv = inv % m
    x = (power3 * inv) % mod
    return x

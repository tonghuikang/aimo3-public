    I = (a*A + b*B + c*C) / (a+b+c)
    # semiperimeter
    s = (a+b+c)/2
    # area using cross product
    area = abs(np.cross(B-A, C-A))/2
    r = area / s
    return I, r

def foot_point(P, line_point1, line_point2):
    # projection of P onto line through A,B
    A = np.array(line_point1); B = np.array(line_point2); P = np.array(P)
    AB = B - A
    t = np.dot(P-A, AB) / np.dot(AB, AB)
    proj = A + t * AB
    return proj

def circle_center_radius_from_three_points(P1,P2,P3):
    # returns center, radius
    P1 = np.array(P1); P2 = np.array(P2); P3 = np.array(P3)
    # solve perpendicular bisectors
    # using linear algebra
    # Compute midpoints and direction vectors
    d12 = P2-P1
    d13 = P3-P1
    # matrix for linear equations: 2*(P2-P1) . O = |P2|^2 - |P1|^2 etc.
    A = np.vstack([2*d12, 2*d13])
    B = np.array([np.dot(P2,P2) - np.dot(P1,P1), np.dot(P3,P3) - np.dot(P1,P1)])
    # Solve for center O
    try:
        O = np.linalg.solve(A, B)
    except np.linalg.LinAlgError:
        # degenerate
        return None, None
    r = np.linalg.norm(O - P1)
    return O, r

def circle_intersections_center_radius(O1,R1,O2,R2):
    # returns list of intersection points
    O1 = np.array(O1); O2 = np.array(O2)
    d = np.linalg.norm(O2-O1)
    if d > R1+R2 or d < abs(R1-R2):
        return [] # no intersections
    # find a = (R1^2 - R2^2 + d^2) / (2d)
    a = (R1**2 - R2**2 + d**2) / (2*d)
    h2 = R1**2 - a**2
    if h2 < -1e-12:
        return []
    h = math.sqrt(max(0.0, h2))
    # point P2 = O1 + a*(O2-O1)/d
    P2 = O1 + a*(O2-O1)/d
    # Intersection points
    offset = h * np.array([-(O2[1]-O1[1])/d, (O2[0]-O1[0])/d])
    inter1 = P2 + offset
    inter2 = P2 - offset
    return [inter1, inter2]

def reflect_point_across_line(P, A, B):
    # reflect P across line AB
    P = np.array(P); A = np.array(A); B = np.array(B)
    # compute foot Q of P onto line AB
    Q = foot_point(P, A, B)
    # reflected point = Q + (Q - P) = 2Q - P
    return 2*Q - P

def is_concyclic(points, tol=1e-6):
    # Given list of 4 points, check if they lie on same circle within tolerance
    # Compute circle from three points, check distance of fourth
    O, R = circle_center_radius_from_three_points(points[0], points[1], points[2])
    if O is None:
        return False
    return abs(np.linalg.norm(points[3]-O)-R) < tol

    F = fibonacci_upto(n+2)

    d = BD
    # slopes
    if r <= 0 or r >= d:
        return None
    m_B = 2 * d * r / (d**2 - r**2)
    # solve t for line from C
    t = r / CD if CD != 0 else 0.1
    for _ in range(30):
        g = t*CD - r*(math.sqrt(t*t+1)+1)
        dg = CD - r*(t / math.sqrt(t*t+1))
        if dg == 0:

    A_x = -s * a / (m_B - s)
    A_y = m_B * A_x
    # incircle foot points
    I = np.array([d, r])
    v = np.array([A_x, A_y])
    t_proj = np.dot(I, v) / np.dot(v, v)
    Fp = t_proj * v
    w = np.array([A_x - a, A_y])
    t2 = np.dot(I - np.array([a,0]), w) / np.dot(w,w)
    Ep = np.array([a,0]) + t2 * w
    # circles
    def circle_from_three(P1,P2,P3):
        A1 = P2 - P1
        B1 = P3 - P1
        M = np.vstack([2*A1, 2*B1])
        rhs = np.array([np.dot(P2,P2) - np.dot(P1,P1), np.dot(P3,P3) - np.dot(P1,P1)])
        O = np.linalg.solve(M, rhs)
        R = np.linalg.norm(O - P1)
        return O,R
    O_ABC,R_ABC = circle_from_three(np.array([A_x,A_y]), np.array([0.0,0.0]), np.array([a,0.0]))

        return A1 + t*u

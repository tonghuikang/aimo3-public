    if d > r1 + r2 + 1e-12 or d < abs(r1 - r2) - 1e-12:
        return []
    # a distance from c1 to point p2 where line connecting intersection points cross the line connecting centers
    a = (r1**2 - r2**2 + d**2) / (2*d) if d != 0 else 0
    # height of intersection from p2
    h2 = r1**2 - a**2
    if h2 < -1e-12:
        return []
    h = math.sqrt(max(0.0, h2))
    # point p2:
    xm = x1 + a * dx / d if d != 0 else x1
    ym = y1 + a * dy / d if d != 0 else y1
    # intersection points:
    if h < 1e-12:
        return [(xm, ym)]
    else:
        rx = -dy * (h / d)
        ry = dx * (h / d)
        p1 = (xm + rx, ym + ry)
        p2 = (xm - rx, ym - ry)
        return [p1, p2]

def foot_of_perpendicular(point, line_pt1, line_pt2):
    # foot of perpendicular from point to line through line_pt1, line_pt2
    x0,y0 = point
    x1,y1 = line_pt1
    x2,y2 = line_pt2
    # line direction vector
    dx = x2 - x1
    dy = y2 - y1
    # compute projection t of (point - line_pt1) onto direction
    t = ((x0 - x1)*dx + (y0 - y1)*dy) / (dx*dx + dy*dy)
    # foot:
    xf = x1 + t*dx
    yf = y1 + t*dy
    return (xf, yf)

def reflect_point_across_line(P, A, B):
    # reflect point P across line AB
    # Use formula: find foot Q then P' = 2Q - P
    Q = foot_of_perpendicular(P, A, B)
    return (2*Q[0] - P[0], 2*Q[1] - P[1])

def det4(M):
    # M is list of 4 rows, each length 4
    import numpy as np
    return float(np.linalg.det(np.array(M, dtype=float)))

def cyclic_condition(B, N, K, Kprime):
    # returns value of determinant for cyclic condition
    # Points B,N,K,Kprime
    # matrix rows: [x^2+y^2, x, y, 1]
    M = [
        [B[0]**2 + B[1]**2, B[0], B[1], 1.0],
        [N[0]**2 + N[1]**2, N[0], N[1], 1.0],
        [K[0]**2 + K[1]**2, K[0], K[1], 1.0],
        [Kprime[0]**2 + Kprime[1]**2, Kprime[0], Kprime[1], 1.0],
    ]
    return det4(M)

def compute_ratio_given_t(n, t):
    B0 = fibonacci_int(n)
    C0 = fibonacci_int(n+1)
    a_len = B0 + C0
    r = math.sqrt(t * B0 * C0 / (a_len + t))
    B_pt = (0.0, 0.0)
    C_pt = (float(a_len), 0.0)
    D_pt = (float(B0), 0.0)
    # incircle center I not needed directly now
    # F
    B0_sq = B0*B0
    r_sq = r*r
    denom_F = B0_sq + r_sq
    xF = B0 * (B0_sq - r_sq) / denom_F
    yF = (2 * B0_sq * r) / denom_F
    # E
    C0_sq = C0*C0
    denom_E = C0_sq + r_sq
    xE = B0 + (2 * C0 * r_sq) / denom_E
    yE = (2 * C0_sq * r) / denom_E
    F_pt = (xF, yF)
    E_pt = (xE, yE)
    # Compute A via line intersection as before
    # Solve s*F = C + t*(E - C)
    # Using s as scalar for B->F param.
    # use earlier formulas
    if abs(yE) < 1e-12:
        raise ValueError("yE zero")
    denominator = (xF - yF * (xE - a_len) / yE)
    if abs(denominator) < 1e-12:
        raise ValueError("Denominator zero")
    s = a_len / denominator
    t_param = s * yF / yE
    A_pt = (s * xF, s * yF)
    # circumcenters
    O_ABC = circumcenter(A_pt, B_pt, C_pt)
    O_AFE = circumcenter(A_pt, F_pt, E_pt)
    R_ABC = distance(O_ABC, B_pt)
    R_AFE = distance(O_AFE, A_pt)
    # intersection of circles -> K
    inters = circle_intersections(O_ABC, R_ABC, O_AFE, R_AFE)
    # choose K not equal to A
    K_pt = None
    for pt in inters:
        if distance(pt, A_pt) > 1e-6:
            K_pt = pt
            break
    if K_pt is None:
        raise ValueError("K not found")
    # N foot from D onto EF
    N_pt = foot_of_perpendicular(D_pt, E_pt, F_pt)
    # K' reflection across EF
    Kprime_pt = reflect_point_across_line(K_pt, E_pt, F_pt)
    # circle tangent to BN at B passing through K
    # line BN
    lineBN = (B_pt, N_pt)
    # direction of BN vector
    dx_BN = N_pt[0] - B_pt[0]
    dy_BN = N_pt[1] - B_pt[1]
    # line perp through B: direction (-dy, dx)
    # center O_T intersection of this line and perp bisector of BK
    # define line1: B + s1 * (-dy_BN, dx_BN)
    # define line2: midpoint of BK + s2 * (-dy_BK, dx_BK)
    dx_BK = K_pt[0] - B_pt[0]
    dy_BK = K_pt[1] - B_pt[1]
    # compute intersection
    # solve B + s1 * v1 = mid + s2 * v2
    # v1 = (-dy_BN, dx_BN)
    # v2 = (-dy_BK, dx_BK)
    v1x, v1y = -dy_BN, dx_BN
    v2x, v2y = -dy_BK, dx_BK
    # unknown s1, s2
    # B = (0,0)
    # mid = (K.x/2, K.y/2)
    # Equation:
    # s1*v1x = Kx/2 + s2*v2x
    # s1*v1y = Ky/2 + s2*v2y
    # Solve linear system for s1, s2
    # Write as:
    # s1*v1x - s2*v2x = Kx/2
    # s1*v1y - s2*v2y = Ky/2
    A_mat = [[v1x, -v2x], [v1y, -v2y]]
    b_vec = [K_pt[0]/2.0, K_pt[1]/2.0]
    # solve 2x2
    detA = A_mat[0][0]*A_mat[1][1] - A_mat[0][1]*A_mat[1][0]
    if abs(detA) < 1e-12:
        raise ValueError("det zero for O_T")
    s1 = (b_vec[0]*A_mat[1][1] - b_vec[1]*A_mat[0][1]) / detA
    # compute O_T:
    OT_x = s1 * v1x
    OT_y = s1 * v1y
    OT_pt = (OT_x, OT_y)
    # radius
    r_T = math.hypot(OT_x, OT_y)  # distance to B
    # intersection of circle centered at OT with BC (y=0)
    # (x - OT_x)^2 + OT_y^2 = r_T^2
    # => (x - OT_x)^2 = r_T^2 - OT_y^2
    disc = r_T**2 - OT_y**2
    if disc < -1e-12:
        raise ValueError("disc negative")
    disc = max(disc, 0.0)
    sqrt_disc = math.sqrt(disc)
    x_candidates = [OT_x + sqrt_disc, OT_x - sqrt_disc]
    # choose the one not equal to B.x (0) within tolerance
    T_pt = None
    for xc in x_candidates:
        if abs(xc) > 1e-6:
            # also must lie on BC segment between B and C? maybe not necessarily inside segment, but T is on line BC.
            # We'll accept any other point.
            T_pt = (xc, 0.0)
            break
    if T_pt is None:
        # maybe both zero? improbable.
        T_pt = (x_candidates[0], 0.0)
    # compute distances:
    CT = distance(C_pt, T_pt)
    BT = distance(B_pt, T_pt)
    NB = distance(N_pt, B_pt)
    NE = distance(N_pt, E_pt)
    ratio = (CT * NB) / (BT * NE)
    return ratio

def compute_ratio_for_n(n):
    # find t root then compute ratio
    t_root = solve_t_for_n(n)
    if t_root is None:
        raise RuntimeError(f"t not found for n={n}")
    ratio = compute_ratio_given_t(n, t_root)
    return ratio, t_root

def circumcenter(p1, p2, p3):
    (x1,y1),(x2,y2),(x3,y3) = p1,p2,p3
    d = 2*(x1*(y2 - y3) + x2*(y3 - y1) + x3*(y1 - y2))

        return None

    uy = ((x1**2 + y1**2)*(x3 - x2) + (x2**2 + y2**2)*(x1 - x3) + (x3**2 + y3**2)*(x2 - x1)) / d
    return (ux, uy)

def distance(p,q):

def circle_intersections(c1,R1,c2,R2):
    x1,y1 = c1
    x2,y2 = c2

    dy = y2 - y1
    d = math.hypot(dx, dy)

def foot_of_perpendicular(point, line_pt1, line_pt2):
    x0,y0 = point
    x1,y1 = line_pt1
    x2,y2 = line_pt2
    dx = x2 - x1
    dy = y2 - y1
    if dx==0 and dy==0:
        return point
    t = ((x0 - x1)*dx + (y0 - y1)*dy) / (dx*dx + dy*dy)
    xf = x1 + t*dx
    yf = y1 + t*dy
    return (xf, yf)

def reflect_point_across_line(P, A, B):
    # reflect P across line AB
    # compute foot Q then P' = 2Q - P
    Q = foot_of_perpendicular(P, A, B)
    return (2*Q[0] - P[0], 2*Q[1] - P[1])

def cyclic_condition_val(B, N, K, Kprime):
    import numpy as np
    M = np.array([

def f_cyclic_numeric(n, t):
    B0 = fibonacci_int(n)
    C0 = fibonacci_int(n+1)
    a_len = B0 + C0
    if t <= 0:
        return None

def compute_cycle_det(r_val, d, L):
    # numeric compute determinant for given r
    # using mp.mpf for high precision
    d = mp.mpf(d)
    L = mp.mpf(L)
    r = mp.mpf(r_val)
    # compute slopes
    # check denominator not zero
    if abs(d**2 - r**2) < 1e-12 or abs((L-d)**2 - r**2) < 1e-12:
        return None
    m_AB = 2*d*r/(d**2 - r**2)
    m_CA = -2*(L-d)*r/(((L-d)**2) - r**2)
    # intersection for A
    xA = -m_CA*L/(m_AB - m_CA)
    yA = m_AB * xA
    # compute foot points F and E
    # s = (d*xA + r*yA)/(xA**2 + yA**2)
    s = (d*xA + r*yA)/(xA**2 + yA**2)
    Fx = s*xA
    Fy = s*yA
    # t for foot onto AC
    t = ((d-L)*(xA-L) + r*yA)/((xA-L)**2 + yA**2)
    Ex = L + t*(xA-L)
    Ey = t*yA
    # foot N from D onto EF
    Dx = d
    Dy = 0
    # vector EF
    EFx = Fx - Ex
    EFy = Fy - Ey
    # vector D-E
    DEx = Dx - Ex
    DEy = Dy - Ey
    # projection scalar onto EF
    tN = (DEx*EFx + DEy*EFy) / (EFx**2 + EFy**2)
    Nx = Ex + tN*EFx
    Ny = Ey + tN*EFy
    # compute K: intersection of circumcircle of ABC and circle AEF
    # circumcircle of ABC center O1 and radius R1
    # using standard formula for circumcenter given coordinates.
    # Use function to compute circle center and radius given three points.
    # We'll use linear algebra approach.
    def circumcenter(p1, p2, p3):
        # return center (ox, oy) as mp.mpf
        x1, y1 = p1
        x2, y2 = p2
        x3, y3 = p3
        d = 2*(x1*(y2 - y3) + x2*(y3 - y1) + x3*(y1 - y2))
        if abs(d) < 1e-12:
            return None
        ux = ((x1**2 + y1**2)*(y2 - y3) + (x2**2 + y2**2)*(y3 - y1) + (x3**2 + y3**2)*(y1 - y2))/d
        uy = ((x1**2 + y1**2)*(x3 - x2) + (x2**2 + y2**2)*(x1 - x3) + (x3**2 + y3**2)*(x2 - x1))/d
        return (ux, uy)
    O1 = circumcenter((xA, yA), (0,0), (L,0))
    if O1 is None:
        return None
    O1x, O1y = O1
    R1 = mp.sqrt((xA-O1x)**2 + (yA-O1y)**2)
    # circle AEF
    O2 = circumcenter((xA, yA), (Ex, Ey), (Fx, Fy))
    O2x, O2y = O2
    R2 = mp.sqrt((xA-O2x)**2 + (yA-O2y)**2)
    # Intersections of two circles
    # Compute distance between centers
    d_centers = mp.sqrt((O1x-O2x)**2 + (O1y-O2y)**2)
    # If circles intersect, there are two points.
    # Use formula for intersection points.
    # a = (R1^2 - R2^2 + d^2) / (2d)
    a = (R1**2 - R2**2 + d_centers**2) / (2*d_centers)
    # height h
    h_sq = R1**2 - a**2
    if h_sq < -1e-12:
        # no intersection
        return None
    h = mp.sqrt(max(h_sq, 0))
    # point P2 = O1 + a*(O2-O1)/d
    Px = O1x + a*(O2x - O1x)/d_centers
    Py = O1y + a*(O2y - O1y)/d_centers
    # Intersection points:
    # offset vector (rx, ry) = h * (-(O2y - O1y)/d, (O2x - O1x)/d)
    rx = -h * (O2y - O1y)/d_centers
    ry = h * (O2x - O1x)/d_centers
    K1x = Px + rx
    K1y = Py + ry
    K2x = Px - rx
    K2y = Py - ry
    # One of these points is A (should be close to (xA,yA)). Choose the other as K.
    # Determine which is not near A
    if mp.hypot(K1x - xA, K1y - yA) < 1e-8:
        Kx, Ky = K2x, K2y
    else:
        Kx, Ky = K1x, K1y
    # Compute K' reflection across line EF
    # line EF: points E(Ex,Ey), F(Fx,Fy)
    # reflection formula: reflect point across line.
    # Use vector approach.
    # Compute unit direction vector of EF.
    # Compute projection of point onto line, then reflect.
    # param t_line for projection of K onto line EF: t_line = ((K-E)·(F-E))/|F-E|^2
    vEFx = Fx - Ex
    vEFy = Fy - Ey
    # projection scalar
    t_line = ((Kx-Ex)*vEFx + (Ky-Ey)*vEFy)/(vEFx**2 + vEFy**2)
    projx = Ex + t_line * vEFx
    projy = Ey + t_line * vEFy
    # reflection: K' = 2*proj - K
    Kpx = 2*projx - Kx
    Kpy = 2*projy - Ky
    # Compute determinant for cyclic condition (B,N,K,K')
    # determinant = |x y x^2+y^2 1|
    Bx, By = 0, 0
    # define function to compute determinant
    def det4(P1, P2, P3, P4):
        x1,y1 = P1
        x2,y2 = P2
        x3,y3 = P3
        x4,y4 = P4
        M = mp.matrix([[x1, y1, x1**2 + y1**2, 1],
                       [x2, y2, x2**2 + y2**2, 1],
                       [x3, y3, x3**2 + y3**2, 1],
                       [x4, y4, x4**2 + y4**2, 1]])
        return mp.det(M)
    det_val = det4((Kx,Ky),(Nx,Ny),(Kpx,Kpy),(Bx,By))
    return det_val, {
        'A': (xA,yA), 'K': (Kx,Ky), 'Kp': (Kpx,Kpy), 'N': (Nx,Ny),
        'E': (Ex,Ey), 'F': (Fx,Fy)
    }

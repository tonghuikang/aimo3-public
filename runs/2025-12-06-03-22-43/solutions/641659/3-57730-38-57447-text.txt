def solve_center_parabola(B, N, K):
    # B, N, K are sympy points (2d vectors)
    x, y = sp.symbols('x y', real=True)
    # radius squared r2 = (x - B.x)^2 + (y - B.y)^2 = x^2 + y^2
    r2 = x**2 + y**2
    # distance to line BN (through B and N)
    # line BN direction vector v = N - B
    v = (N[0] - B[0], N[1] - B[1])
    # cross product magnitude |v_x*y - v_y*x|
    cross = v[0]*y - v[1]*x
    # distance squared = cross^2 / (v·v)
    dist2 = cross**2 / (v[0]**2 + v[1]**2)

    eq1 = sp.Eq(dist2, r2)

    eq2 = sp.Eq((x - K[0])**2 + (y - K[1])**2, r2)
    sol = sp.solve((eq1, eq2), (x, y), dict=True)
    return sol

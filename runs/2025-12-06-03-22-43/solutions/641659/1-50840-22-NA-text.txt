def get_fibonacci(n):
    # returns F_n with F_0=0, F_1=1
    # use fast method
    if n < 0:
        raise ValueError
    a, b = 0, 1
    for _ in range(n):
        a, b = b, a+b
    return a

def triangle_coords_from_sides(a,b,c):
    # B=(0,0), C=(a,0), A=(x_A, y_A)
    # Use law of cos for x_A
    x_A = (c**2 - b**2 + a**2) / (2*a)
    # y_A positive sqrt
    # ensure positivity
    val = c**2 - x_A**2
    if val < 0:
        raise ValueError("Invalid triangle")
    y_A = sp.sqrt(val)
    return sp.Point(0,0), sp.Point(a,0), sp.Point(x_A, y_A)

def incircle_center_and_radius(A,B,C):
    # B(0,0), C(a,0), A(x_A,y_A)

def circumcircle_center_and_radius(P1, P2, P3):
    # Compute circle center using sympy geometry Circle
    circ = sp.Circle(P1, P2, P3)
    return circ.center, circ.radius

def second_intersection_of_circles(circ1, circ2, common_pt):
    # Return the other intersection point of two circles (excluding common_pt)
    inter = circ1.intersection(circ2)
    # intersection returns list of Points or empty
    # Might be 0,1,2 intersection points
    # Return the one not equal to common_pt (within tolerance)
    # Convert to set
    pts = inter
    # For numeric, we can evaluate approximations
    for pt in pts:
        if not pt.equals(common_pt):
            return pt
    # if both equal, return None
    return None

def reflect_point_across_line(P, L1, L2):
    # reflect point P across line through L1-L2
    line = sp.Line(L1, L2)
    # reflect using sympy geometry
    return P.reflect(line)

def foot_of_perpendicular(P, L1, L2):
    line = sp.Line(L1, L2)
    return line.projection(P)

def cyclic_condition(B,N,K,Kp):
    # Determine if points B,N,K,Kp are concyclic using determinant test
    # Compute matrix determinant
    pts = [B,N,K,Kp]
    M = sp.Matrix([[pt.x**2 + pt.y**2, pt.x, pt.y, 1] for pt in pts])
    return sp.simplify(M.det())

def compute_Q_for_c(n, c_val):
    # returns Q and cyclic condition determinant
    # n integer >= 2
    # c_val length AB
    # Compute a = F_{n+2}, b = c_val + F_{n-1}
    # ensure c_val satisfies triangle inequality 2c > F_{n+2} - F_{n-1}
    F_n = get_fibonacci(n)
    F_np1 = get_fibonacci(n+1)
    F_np2 = get_fibonacci(n+2)
    F_nm1 = get_fibonacci(n-1)
    a = F_np2
    c = sp.nsimplify(c_val) # AB
    b = c + F_nm1
    # Compute coordinates
    B, C, A = triangle_coords_from_sides(a,b,c)
    # Incenter and incircle radius
    I, r = incircle_center_and_radius(A,B,C)
    # Contact points
    D, E, F_pt = contact_points(A,B,C, I, r, F_n, F_np1)
    # circumcircles
    # A,F,E circle
    circ_AFE_center, circ_AFE_r = circumcircle_center_and_radius(A, F_pt, E)
    circ_AFE = sp.Circle(A, F_pt, E)
    # circumcircle of ABC
    circ_ABC_center, circ_ABC_r = circumcircle_center_and_radius(A,B,C)
    circ_ABC = sp.Circle(A,B,C)
    # Intersection other than A => K
    K = second_intersection_of_circles(circ_AFE, circ_ABC, A)
    if K is None:
        # maybe degenerate
        return None, None
    # reflect K across EF to get K'
    Kp = reflect_point_across_line(K, E, F_pt)
    # foot from D to EF => N
    N = foot_of_perpendicular(D, E, F_pt)
    # cyclic condition determinant
    det = cyclic_condition(B,N,K,Kp)
    # Now compute T: circle through B,K tangent to BN at B
    # find center O' as intersection of perpendicular to BN at B and perpendicular bisector of BK
    line_perp_BN = sp.Line(B,N).perpendicular_line(B)
    # Perp bisector of BK
    mid_BK = sp.Point((B.x + K.x)/2, (B.y + K.y)/2)
    perp_bis_BK = sp.Line(B, K).perpendicular_line(mid_BK)
    # Intersection gives center O'
    Oprime = line_perp_BN.intersection(perp_bis_BK)[0]
    rprime = Oprime.distance(B) # radius
    # Intersection of this circle with BC (line y=0)
    line_BC = sp.Line(B,C)
    # Solve for intersection
    # Equation: (x - Oprime.x)^2 + (0 - Oprime.y)^2 = rprime**2
    x = sp.symbols('x', real=True)
    eq = (x - Oprime.x)**2 + ( - Oprime.y)**2 - rprime**2
    sol = sp.solve(eq, x)
    # sol contains B.x = 0 and T.x = something else
    # Filter out 0
    T_x = None
    for s in sol:
        s_simp = sp.simplify(s)
        if not sp.simplify(s_simp - B.x).equals(0):
            T_x = s_simp
    if T_x is None:
        # might be degenerate or both solutions same
        return None, det
    T = sp.Point(T_x, 0)
    # lengths
    CT = C.distance(T)
    BT = B.distance(T)
    NB = N.distance(B)
    NE = N.distance(E)
    Q = (CT * NB) / (BT * NE)
    # simplify Q maybe
    Q = sp.simplify(Q)
    return Q, sp.simplify(det)

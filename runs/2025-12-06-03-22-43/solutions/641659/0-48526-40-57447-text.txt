        if prev_f == 0:
            # root exactly at prev_x
            return prev_x

            return x

            try:

                    f_mid = compute_f_c(mid, n)

                    if prev_f * f_mid < 0:
                        b0 = mid
                        f_val = f_mid
                    else:
                        a0 = mid
                        prev_f = f_mid
                return (a0+b0)/2
        prev_f = f_val
        prev_x = x
    return None

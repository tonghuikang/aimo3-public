def compute_triangle_data(n, s):

    # n integer, s numeric (float or sympy)
    # compute fibonacci
    F = [0,1]
    for i in range(2, n+5):
        F.append(F[-1] + F[-2])
    a = F[n+2]  # BC
    d = F[n]    # BD
    # s must be > a
    # side lengths
    b = s - d
    c = s - a + d
    # r
    r = sp.sqrt((s - a) * d * (a - d) / s)
    # points
    B = sp.Point(0,0)
    C = sp.Point(a,0)
    D = sp.Point(d,0)
    I = sp.Point(d, r)
    # A coordinate
    xA = (c**2 - b**2 + a**2) / (2*a)
    # ensure positivity inside sqrt
    yA_sq = c**2 - xA**2
    # If negative due to numeric errors, clamp to 0
    # Use sp.sqrt, maybe yields I* if negative
    yA = sp.sqrt(yA_sq)
    A = sp.Point(xA, yA)
    # foot to AB
    BA = A - B
    t = ((I - B).dot(BA)) / (BA.dot(BA))
    Fpt = B + t * BA
    # foot to AC
    CA = A - C
    u = ((I - C).dot(CA)) / (CA.dot(CA))
    Ept = C + u * CA

    circ_AFE = sp.Circle(A, Fpt, Ept)
    circ_ABC = sp.Circle(A, B, C)
    inter = circ_AFE.intersection(circ_ABC)
    # Identify K: not equal A
    K = None
    for p in inter:
        if not p.equals(A):
            K = p
    if K is None:

        return None

    line_EF = sp.Line(Ept, Fpt)

    # N foot from D onto EF

def circumcenter(p1,p2,p3):

def foot_point(P, A, B):
    # foot of perpendicular from P to line AB
    # Return point coordinates
    (x0,y0)=P
    (x1,y1)=A
    (x2,y2)=B
    # vector AB
    vx = x2 - x1
    vy = y2 - y1
    # vector AP
    wx = x0 - x1
    wy = y0 - y1
    # projection scalar t = (AP·AB) / (AB·AB)
    t = (wx*vx + wy*vy) / (vx*vx + vy*vy)
    # foot = A + t*AB
    return (x1 + t*vx, y1 + t*vy)

def reflect_point_across_line(P, A, B):
    # reflect point P across line AB
    # Find foot H of P onto AB, then reflect
    H = foot_point(P, A, B)
    # vector PH
    vx = H[0] - P[0]
    vy = H[1] - P[1]
    # reflected point = P + 2*PH
    return (P[0] + 2*vx, P[1] + 2*vy)

def distance(P,Q):
    return math.hypot(P[0]-Q[0], P[1]-Q[1])

# Define function to compute minimal r for given c (c >=0).
def best_r_for_c(c, max_d=5000):
    # Pre-calc sorted pairs up to max_d
    pairs = []
    for d1 in range(2, max_d+1):
        for d2 in range(d1+1, max_d+1):

            D = d1*d2 + d1 + d2
            g = math.gcd(d1,d2)

            r = Fraction(d1*d2, D)

    pairs.sort(key=lambda x: x[0])  # Fraction sort works

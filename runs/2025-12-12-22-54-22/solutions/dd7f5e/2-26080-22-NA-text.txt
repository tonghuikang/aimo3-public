def poly_to_dict(poly):
    # poly is sympy Poly
    coeff_dict = {}
    for exp, coeff in zip(poly.monoms(), poly.coeffs()):

        coeff_dict[exp[0]] = int(coeff)

def dot_product(alpha_dict, beta_dict):
    # sum over all integer t: alpha(t) * beta(t)
    # Since dict only finite entries, sum over intersection keys
    total = 0

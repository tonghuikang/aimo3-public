def poly_to_str(p):
    # sort exponents
    terms = []
    for exp in sorted(p):
        coeff = p[exp]
        if coeff == 0: continue
        term = ''
        if coeff == -1:
            term = '-'
        elif coeff != 1:
            term = str(coeff)
        if exp == 0:
            if term == '' or term == '-':
                term += '1'
        else:
            if term == '' or term == '-':
                term += ''
            term += 'x'
            if exp != 1:
                term += '^' + str(exp)
        terms.append(term)
    return ' + '.join(terms).replace('+ -','- ')

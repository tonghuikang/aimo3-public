    fac = {}
    d = 2
    while d*d <= n:
        while n % d == 0:
            fac[d] = fac.get(d,0)+1

def generate_divisors(pf):
    # pf dict prime->exp
    primes = list(pf.items())
    divs = [1]
    for p, e in primes:
        cur = []
        for d in divs:
            mul = 1
            for i in range(e+1):
                cur.append(d*mul)
                mul*=p
        divs = cur
    return divs

def omega_distinct(num):
    # number of distinct prime factors (any)
    return len(factorize(num))

def omega1(num):
    # number of distinct primes p ≡1 mod4 dividing num
    return sum(1 for p in factorize(num) if p%4==1)

def has_odd_exp_3mod4(num):
    # returns True if there exists prime ≡3 mod4 with odd exponent
    for p,e in factorize(num).items():
        if p%4==3 and e%2==1:
            return True
    return False

def f_formula(N):
    pf = factorize(N)
    divs = generate_divisors(pf)
    total=0
    # odd leg contributions
    for d in divs:

            total += 2**(omega1(d)-1)
    return total

def triangle_points_bc(b, c, r):
    a = b + c
    B = mp.matrix([0,0])
    C = mp.matrix([a,0])
    D = mp.matrix([b,0])
    I = mp.matrix([b, r])
    # slopes

    mC = -2*c*r/(c**2 - r**2)

def line_coeff(P, Q):
    a = P[1] - Q[1]
    b = Q[0] - P[0]
    c = P[0]*Q[1] - Q[0]*P[1]
    return a,b,c

def foot_point(P, line):
    a,b,c = line
    denom = a*a + b*b
    x0,y0 = P[0], P[1]
    factor = (a*x0 + b*y0 + c) / denom
    x = x0 - a*factor
    y = y0 - b*factor
    return mp.matrix([x,y])

def circumcircle(P, Q, R):
    x1,y1 = P[0], P[1]
    x2,y2 = Q[0], Q[1]
    x3,y3 = R[0], R[1]

def vp_factorial(n, p):
    # Legendre's formula
    exp = 0
    power = p
    while power <= n:
        exp += n // power
        power *= p
    return exp

def vp_catalan(n, p):
    # n is integer
    return vp_factorial(2*n, p) - vp_factorial(n+1, p) - vp_factorial(n, p)

def compute_vp_N(p, max_t=19):
    total = 0
    for t in range(max_t+1):
        n = 2**t
        vp = vp_catalan(n, p)
        weight = 2**(max_t - t)
        total += weight * vp
    return total

def all_matchings(group):
    # group is list of player IDs
    # returns list of list of pairs (tuple)
    if not group:
        return [[]]
    # pick first element
    first = group[0]
    matchings = []
    for i in range(1, len(group)):
        second = group[i]
        rest = group[1:i] + group[i+1:]
        for sub in all_matchings(rest):
            matchings.append([(first, second)] + sub)
    return matchings

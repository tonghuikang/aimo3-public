    poly = sum(alpha[i] * z**i for i in range(9))
    return sp.expand(poly)

def divides_binomial(poly):
    # check if there exists d>0 such that poly divides 1+z^d in Z[z] (allow units)
    # We'll check d up to, say, 30
    for d in range(1,31):
        # compute remainder of (z**d+1) divided by poly
        # use polynomial division over integers: sympy div
        q, r = sp.div(z**d + 1, poly, domain=sp.ZZ)
        if r == 0:
            return True
    return False

        # divisibility ensures poly divides integer binomial => content must be 1 or -1.

    # returns list of intersection points (as np arrays)
    d = np.linalg.norm(O2 - O1)
    if d > r1 + r2 or d < abs(r1 - r2):
        return []  # no intersection
    # distance from O1 to line connecting intersection points
    a = (r1**2 - r2**2 + d**2) / (2*d)
    h2 = r1**2 - a**2
    if h2 < -1e-12:  # numerical issues

def geometry_for_xy(r, x, y):
    # base length a
    a = x + y
    # points
    B = mp.mpc(0,0)
    C = mp.mpc(a, 0)
    # incircle center I at (x, r)
    I = mp.mpc(x, r)
    # angles
    B_angle = 2*mp.atan(r/x)  # B angle at vertex B
    C_angle = 2*mp.atan(r/y)  # C angle at vertex C
    # point F: B + x * unit direction at angle B_angle
    F = mp.mpc(x*mp.cos(B_angle), x*mp.sin(B_angle))
    # side length c = AB = distance from B to A (unknown). But we can compute A as intersection of lines from B and C.
    # line from B: direction (cos B_angle, sin B_angle)
    # line from C: direction (-cos C_angle, sin C_angle)
    # solve for intersection point A:
    # B + t1 * (cosB, sinB) = C + t2 * (-cosC, sinC)
    # Solve for t1, t2:
    # two equations:
    # t1*cosB = a - t2*cosC
    # t1*sinB = t2*sinC
    # from second: t1 = t2*sinC/sinB
    # substitute in first:
    # (t2*sinC/sinB)*cosB = a - t2*cosC
    # t2*(sinC*cosB/sinB + cosC) = a
    # t2 = a / (sinC*cosB/sinB + cosC)
    sinB = mp.sin(B_angle)
    cosB = mp.cos(B_angle)
    sinC = mp.sin(C_angle)
    cosC = mp.cos(C_angle)
    denom = sinC*cosB/sinB + cosC
    if abs(denom) < 1e-15:
        return None
    t2 = a / denom
    t1 = t2*sinC/sinB
    A = mp.mpc(t1*cosB, t1*sinB)  # also equals C + t2*(-cosC, sinC)
    # point E: from C moving distance y along direction (-cosC, sinC)
    E = mp.mpc(a - y*cosC, y*sinC)
    # line EF
    line_EF = (E, F)
    # foot N from D=(x,0) to line EF
    D = mp.mpc(x, 0)
    # projection
    FE = F - E
    DE = D - E
    t_proj = (DE.real*FE.real + DE.imag*FE.imag) / (FE.real**2 + FE.imag**2)
    N = E + t_proj * FE
    # circumcenter O1 of ABC
    def circumcenter(P1, P2, P3):
        x1,y1 = P1.real, P1.imag
        x2,y2 = P2.real, P2.imag
        x3,y3 = P3.real, P3.imag
        d = 2*(x1*(y2-y3) + x2*(y3-y1) + x3*(y1-y2))

        return mp.mpc(ux, uy)

    # circumcenter O2 of AFE
    O2 = circumcenter(A, F, E)
    # line through A perpendicular to O1O2:
    O1O2 = O2 - O1
    # direction perpendicular to O1O2
    perp = mp.mpc(-O1O2.imag, O1O2.real)  # rotate 90 deg CCW

    R1 = abs(A - O1)
    # Solve |A + s*perp - O1| = R1
    w = A - O1

            ratio = (CT*NB)/(BT*NE)

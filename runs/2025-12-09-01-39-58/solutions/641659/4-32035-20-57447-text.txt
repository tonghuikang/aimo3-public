def circumcenter(p1, p2, p3):
    # p are numpy arrays [x,y]
    # Using formula from perpendicular bisectors: solve linear system
    A = np.array([
        [2*(p2[0] - p1[0]), 2*(p2[1] - p1[1])],
        [2*(p3[0] - p1[0]), 2*(p3[1] - p1[1])]
    ])
    B = np.array([
        p2[0]**2 + p2[1]**2 - p1[0]**2 - p1[1]**2,
        p3[0]**2 + p3[1]**2 - p1[0]**2 - p1[1]**2
    ])
    # Solve A * O = B
    # If points collinear, determinant zero; but here not
    O = np.linalg.solve(A, B)
    return O

def circle_intersections(O1, R1, O2, R2):
    # return intersection points (list of np arrays) of two circles
    # compute distance between centers
    d = np.linalg.norm(O2 - O1)
    if d > R1+R2 or d < abs(R1-R2):
        return []  # no intersection
    # point P2 = point along line O1->O2 where line connecting centers meets chord
    # from formula:
    a = (R1**2 - R2**2 + d**2) / (2*d)
    h_sq = max(R1**2 - a**2, 0)
    h = np.sqrt(h_sq)
    # base point:
    P2 = O1 + a * (O2 - O1) / d
    # intersection points:
    # offset vector perpendicular to line O1O2
    offset = h * np.array([-(O2[1]-O1[1])/d, (O2[0]-O1[0])/d])
    inter1 = P2 + offset
    inter2 = P2 - offset
    return [inter1, inter2]

def incircle_contact_points(A, B, C):
    # compute side lengths
    a = B.distance(C) # BC
    b = C.distance(A) # CA
    c = A.distance(B) # AB
    # semiperimeter
    s = (a+b+c)/2

    I = sp.Point((a*A.x + b*B.x + c*C.x)/(a+b+c),

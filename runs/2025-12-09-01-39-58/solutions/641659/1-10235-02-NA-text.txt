def find_s_numeric(n):
    # compute Fibonacci numbers
    F = [0, 1]
    for i in range(2, n+6):
        F.append(F[-1]+F[-2])
    a = F[n+2]
    Fn = F[n]
    Fn1 = F[n+1]
    # define function f(s) = condition value (determinant of cyclic) 
    def cyclic_condition(s):
        b = s - Fn
        c = s - Fn1
        # area
        # using Heron: s*(s-a)*(s-b)*(s-c) must be positive
        area_sq = s*(s-a)*(s-b)*(s-c)
        if area_sq <= 0:
            return None
        delta = mp.sqrt(area_sq)
        # incircle radius
        r = delta / s
        # coordinates
        B = (0.0, 0.0)
        C = (a, 0.0)
        xA = (a**2 + c**2 - b**2) / (2*a)
        yA = 2*delta / a
        A = (xA, yA)
        # points F, E
        BF = Fn
        # length of AB = c
        F = (BF/c * xA, BF/c * yA)
        CE = Fn1
        # point E = C + (CE / b)*(A - C)
        E = (C[0] + CE/b * (xA - C[0]), C[1] + CE/b * (yA - C[1]))
        # function to compute circle center and radius from 3 points
        def circle_from_3(p1,p2,p3):
            # using circumcenter formula
            (x1,y1),(x2,y2),(x3,y3) = p1,p2,p3
            # compute determinants
            d = 2*(x1*(y2-y3) + x2*(y3-y1) + x3*(y1-y2))
            if abs(d) < 1e-12:
                return None
            ux = ((x1**2+y1**2)*(y2-y3) + (x2**2+y2**2)*(y3-y1) + (x3**2+y3**2)*(y1-y2))/d
            uy = ((x1**2+y1**2)*(x3-x2) + (x2**2+y2**2)*(x1-x3) + (x3**2+y3**2)*(x2-x1))/d
            R = mp.sqrt((ux-x1)**2+(uy-y1)**2)
            return (ux,uy,R)
        circ_AFE = circle_from_3(A,F,E)
        circ_ABC = circle_from_3(A,B,C)
        if circ_AFE is None or circ_ABC is None:
            return None
        # find intersection points of two circles: solve for K
        (x0,y0,R0) = circ_AFE
        (x1,y1,R1) = circ_ABC
        # distance between centers
        d = mp.sqrt((x1-x0)**2 + (y1-y0)**2)
        # check for intersection
        if d > R0+R1 or d < abs(R0-R1):
            return None
        # compute intersection points:
        # from standard formula
        a_ = (R0**2 - R1**2 + d**2) / (2*d)
        h = mp.sqrt(max(R0**2 - a_**2, 0))
        # point P2 = center0 + a_/d * (center1 - center0)
        x2 = x0 + a_ * (x1 - x0) / d
        y2 = y0 + a_ * (y1 - y0) / d
        # intersection points:
        # offset vector perpendicular
        rx = -(y1 - y0) * (h/d)
        ry = (x1 - x0) * (h/d)
        # two solutions:
        K1 = (x2 + rx, y2 + ry)
        K2 = (x2 - rx, y2 - ry)
        # Determine which is A (common point). Should be one of them.
        # Compare distances to A
        def dist(p,q):
            return mp.sqrt((p[0]-q[0])**2 + (p[1]-q[1])**2)
        if dist(K1, A) < 1e-9:
            K = K2
        elif dist(K2, A) < 1e-9:
            K = K1
        else:
            # might be small numeric error, pick one far from A
            if dist(K1, A) > dist(K2, A):
                K = K1
            else:
                K = K2
        # reflection K' across line EF
        # line EF: points E and F
        # compute reflection of point K across line defined by points E,F
        def reflect_point_across_line(P, L1, L2):
            # using vector projection
            x0,y0 = P
            x1,y1 = L1
            x2,y2 = L2
            # line direction
            dx = x2 - x1
            dy = y2 - y1
            # normalize direction
            # compute projection of (P-L1) onto line direction
            t = ((x0 - x1)*dx + (y0 - y1)*dy) / (dx*dx + dy*dy)
            # foot of perpendicular:
            x_proj = x1 + t*dx
            y_proj = y1 + t*dy
            # reflection:
            xr = 2*x_proj - x0
            yr = 2*y_proj - y0
            return (xr, yr)
        Kp = reflect_point_across_line(K, E, F)
        # N foot from D to EF
        N = reflect_point_across_line(D, E, F) # actually reflection gives foot? Not exactly.
        # Wait N is foot of perpendicular from D to EF, not reflection.
        # So we need foot of perpendicular, which is projection of D onto EF.
        def foot_of_perp(P, L1, L2):
            x0,y0 = P
            x1,y1 = L1
            x2,y2 = L2
            dx = x2-x1
            dy = y2-y1
            t = ((x0 - x1)*dx + (y0 - y1)*dy) / (dx*dx + dy*dy)
            xp = x1 + t*dx
            yp = y1 + t*dy
            return (xp, yp)
        N = foot_of_perp(D, E, F)
        # Now compute cyclic condition for points K, N, K', B.
        # Points lie on same circle iff cross ratio? Let's use determinant method: 
        # For points (x_i, y_i), they are concyclic if determinant of matrix 
        # |x_i^2 + y_i^2, x_i, y_i, 1| for i=1..4 equals 0.
        import numpy as np
        def concyclic_det(Ps):
            M = []
            for (x,y) in Ps:
                M.append([x*x + y*y, x, y, 1])
            return np.linalg.det(np.array(M, dtype=float))
        det = concyclic_det([K, N, Kp, B])
        return det
    # Search for root s > a.
    # Provide initial guess e.g., s = a*2
    def find_root():
        # Use findroot from mpmath
        try:
            # Starting guesses
            # Use bracket method scanning
            # We'll try to find root via secant
            # We'll find sign changes
            s_vals = [a + 0.1 + i*0.5 for i in range(1,200)]
            f_vals = []
            for sv in s_vals:
                f = cyclic_condition(sv)
                if f is None:
                    continue
                f_vals.append((sv, f))
            # Find sign change intervals
            for i in range(len(f_vals)-1):
                s1,f1 = f_vals[i]
                s2,f2 = f_vals[i+1]
                if f1 == 0:
                    return s1
                if f1*f2 < 0:
                    # root between s1 and s2
                    try:
                        root = mp.findroot(cyclic_condition, (s1, s2))
                        return root
                    except Exception as e:
                        # fallback
                        pass
            return None
        except Exception as e:
            return None
    root = find_root()
    return root

def circumcenter(p1, p2, p3):
    # p1, p2, p3 are np arrays shape (2,)
    # Solve linear equations for O: (p2 - p1)·O = (|p2|^2 - |p1|^2)/2 ; similarly for p3
    A = np.vstack([p2 - p1, p3 - p1])
    b = np.array([(np.dot(p2, p2) - np.dot(p1, p1))/2,
                  (np.dot(p3, p3) - np.dot(p1, p1))/2])
    # Solve A O = b
    O = np.linalg.solve(A, b)
    return O

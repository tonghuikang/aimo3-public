def cyclic_error_limit(x):
    d = mp.mpf(1)
    phi_val = (1+mp.sqrt(5))/2
    c = phi_val
    r = mp.sqrt((x-1)*d*c/x)
    B = mp.matrix([-d, -r])
    C = mp.matrix([c, -r])
    D = mp.matrix([0, -r])
    F = mp.matrix([-2*d*r**2/(d**2 + r**2), r*(d**2 - r**2)/(d**2 + r**2)])
    E = mp.matrix([2*c*r**2/(c**2 + r**2), r*(r**2 - c**2)/(c**2 + r**2)])
    # Find A intersection BF and CE
    v1 = F - B
    v2 = -(E - C)
    M = [[v1[0], v2[0]],[v1[1], v2[1]]]
    rhs = [C[0]-B[0], C[1]-B[1]]
    det = M[0][0]*M[1][1] - M[0][1]*M[1][0]
    t = (rhs[0]*M[1][1] - M[0][1]*rhs[1]) / det
    A = B + t * v1
    # circumcenters
    O1 = circumcenter_mp(A,B,C)
    R1 = mp.norm(A - O1)
    O2 = circumcenter_mp(A,F,E)
    R2 = mp.norm(A - O2)
    d_cent = mp.norm(O2 - O1)
    a_val = (R1**2 - R2**2 + d_cent**2) / (2 * d_cent)
    h_sq = R1**2 - a_val**2
    if h_sq < 0:
        return None
    h = mp.sqrt(h_sq)
    P2 = O1 + a_val * (O2 - O1) / d_cent
    perp = mp.matrix([-(O2[1]-O1[1]), O2[0]-O1[0]]) / d_cent
    inter1 = P2 + h * perp
    inter2 = P2 - h * perp
    # Choose K not equal A
    if mp.norm(inter1 - A) < 1e-10:
        K = inter2
    else:
        K = inter1
    # N projection of D onto EF
    vEF = F - E
    t_proj = dot(D - E, vEF) / dot(vEF, vEF)
    N = E + t_proj * vEF
    # K' reflection
    tK = dot(K - E, vEF) / dot(vEF, vEF)
    projK = E + tK * vEF

    # circumcenter of B,K,N
    O3 = circumcenter_mp(B, K, N)
    R3_sq = dot(B - O3, B - O3)
    err = dot(Kprime - O3, Kprime - O3) - R3_sq
    return err

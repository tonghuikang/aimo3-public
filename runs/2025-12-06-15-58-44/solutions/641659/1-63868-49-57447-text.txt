def circumcenter(p1, p2, p3):
    # p1,p2,p3 are tuples (x,y)
    (x1,y1), (x2,y2), (x3,y3) = p1,p2,p3
    # Compute using perpendicular bisector intersection formula
    d = 2*(x1*(y2-y3) + x2*(y3-y1) + x3*(y1-y2))
    if abs(d) < 1e-12:
        raise ValueError("Collinear points")
    ux = ((x1**2 + y1**2)*(y2 - y3) + (x2**2 + y2**2)*(y3 - y1) + (x3**2 + y3**2)*(y1 - y2)) / d
    uy = ((x1**2 + y1**2)*(x3 - x2) + (x2**2 + y2**2)*(x1 - x3) + (x3**2 + y3**2)*(x2 - x1)) / d
    return (ux, uy)

def circle_intersections(O1, R1, O2, R2):
    # Returns up to two intersection points
    x1,y1 = O1
    x2,y2 = O2
    dx = x2 - x1
    dy = y2 - y1
    d = mp.sqrt(dx*dx + dy*dy)
    # No intersection if d > R1+R2 or d < |R1-R2|
    if d == 0:
        return []  # concentric
    # Compute a
    a = (R1**2 - R2**2 + d**2) / (2*d)
    # Compute h^2
    h_sq = R1**2 - a**2
    if h_sq < -1e-12:
        # No intersection
        return []
    h = mp.sqrt(max(h_sq, 0))
    # base point
    xm = x1 + a * dx / d
    ym = y1 + a * dy / d
    # offset for intersection points
    rx = -dy * (h / d)
    ry = dx * (h / d)
    p1 = (xm + rx, ym + ry)
    p2 = (xm - rx, ym - ry)
    return [p1, p2]

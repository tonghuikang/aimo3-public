# phi function
def phi(n):
    result = n
    p=2
    nn=n
    while p*p <= nn:
        if nn%p==0:
            while nn%p==0:
                nn//=p
            result -= result//p
        p+=1
    if nn>1:
        result -= result//nn
    return result

def subsets(lst):
    # return list of subsets as list of tuples
    res = []
    n=len(lst)
    for r in range(0,n+1):
        for combo in combinations(lst, r):
            res.append(combo)
    return res

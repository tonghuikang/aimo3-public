def line_projection(point, line_pt1, line_pt2):
    # project point onto line defined by line_pt1,line_pt2
    # returns projection point
    x0,y0 = point[0], point[1]
    x1,y1 = line_pt1[0], line_pt1[1]

    # vector along line
    vx, vy = x2 - x1, y2 - y1

    wx, wy = x0 - x1, y0 - y1

    dot_wv = wx*vx + wy*vy

def circumcenter(P1, P2, P3):
    # P1,P2,P3 are tuples (x,y)
    x1,y1 = P1
    x2,y2 = P2
    x3,y3 = P3
    # Using formula based on determinant:
    # compute using solving linear equations:
    # (x - x1)^2 + (y - y1)^2 = (x - x2)^2 + (y - y2)^2
    # (x - x1)^2 + (y - y1)^2 = (x - x3)^2 + (y - y3)^2
    # Simplify to linear equations:
    A = 2*(x2 - x1)
    B = 2*(y2 - y1)
    C = x2**2 + y2**2 - x1**2 - y1**2
    D = 2*(x3 - x1)
    E = 2*(y3 - y1)
    F = x3**2 + y3**2 - x1**2 - y1**2
    # Solve linear system:
    # A x + B y = C
    # D x + E y = F
    det = A*E - B*D
    if abs(det) < mp.mpf('1e-30'):
        raise ValueError('Points are collinear')
    x = (C*E - B*F) / det
    y = (A*F - C*D) / det
    R = mp.sqrt((x - x1)**2 + (y - y1)**2)
    return (x, y, R)

def project_point_on_line(P, Q, R):
    # Project point P onto line QR
    # Q,R define line, with direction vector d = R - Q
    d = (R[0] - Q[0], R[1] - Q[1])
    # compute dot product of (P-Q) with d
    PQ = (P[0] - Q[0], P[1] - Q[1])
    d_norm2 = d[0]**2 + d[1]**2
    if d_norm2 == 0:
        raise ValueError('Q and R are the same point')
    t = (PQ[0]*d[0] + PQ[1]*d[1]) / d_norm2
    # projection point
    proj = (Q[0] + t*d[0], Q[1] + t*d[1])
    return proj

def reflect_point_across_line(P, Q, R):
    proj = project_point_on_line(P, Q, R)
    # reflected point: 2*proj - P
    return (2*proj[0] - P[0], 2*proj[1] - P[1])

    D = 2*(x3-x1); E = 2*(y3-y1); Ff = x3**2+y3**2 - x1**2 - y1**2
    det = A*E - B*D
    if abs(det) < mp.mpf('1e-30'):
        return None
    xc = (Cc*E - B*Ff)/det
    yc = (A*Ff - Cc*D)/det
    R = mp.sqrt((xc-x1)**2 + (yc-y1)**2)
    return (xc,yc,R)

def foot_point(P, Q, R):
    # foot from P to line QR
    # projection of vector PQ onto QR
    v = (R[0]-Q[0], R[1]-Q[1])
    w = (P[0]-Q[0], P[1]-Q[1])
    t = (w[0]*v[0] + w[1]*v[1])/(v[0]**2+v[1]**2)
    return (Q[0] + t*v[0], Q[1] + t*v[1])

def reflect_point_across_line(P, Q, R):
    proj = foot_point(P, Q, R)
    return (2*proj[0]-P[0], 2*proj[1]-P[1])

def delta_for_d(n, d):
    D, Fp, Ep, B, C = unit_incircle_data(n, d)
    # compute A = intersection of tangents at F and E
    # tangent at F: line through F with direction perpendicular to radius OF
    # radius OF is vector F (since incircle centre is origin). So tangent direction = (-Fp[1], Fp[0])
    # similarly for E.
    # Represent lines in param form and solve intersection.
    # line1: F + s * tF, where tF = (-Fp[1], Fp[0])
    # line2: E + t * tE, where tE = (-Ep[1], Ep[0])
    tF = (-Fp[1], Fp[0])
    tE = (-Ep[1], Ep[0])
    # Solve F + s*tF = E + t*tE
    # two equations:
    # F[0] + s*tF[0] = E[0] + t*tE[0]
    # F[1] + s*tF[1] = E[1] + t*tE[1]
    # solve for s and t
    A_mat = mp.matrix([[tF[0], -tE[0]],
                       [tF[1], -tE[1]]])
    b_vec = mp.matrix([E[0]-Fp[0], E[1]-Fp[1]])
    sol = mp.lu_solve(A_mat, b_vec)
    s = sol[0]
    A_pt = (Fp[0] + s*tF[0], Fp[1] + s*tF[1])
    # Now K = second intersection of circles (A,F,E) and (A,B,C)
    circ_AFE = circumcenter(A_pt, Fp, Ep)
    circ_ABC = circumcenter(A_pt, B, C)
    if circ_AFE is None or circ_ABC is None:
        return None
    # intersect circles
    def intersect_circles(c1, c2):
        x1,y1,R1 = c1
        x2,y2,R2 = c2
        d = mp.sqrt((x2-x1)**2 + (y2-y1)**2)
        if d == 0:
            return None
        a = (R1**2 - R2**2 + d**2) / (2*d)
        h2 = R1**2 - a**2
        if h2 < 0:
            h = 0
        else:
            h = mp.sqrt(h2)
        # point P2
        x3 = x1 + a*(x2-x1)/d
        y3 = y1 + a*(y2-y1)/d
        rx = -(y2-y1)*(h/d)
        ry =  (x2-x1)*(h/d)
        return ( (x3+rx, y3+ry), (x3-rx, y3-ry) )
    inter = intersect_circles(circ_AFE, circ_ABC)
    if inter is None:
        return None
    # choose K that is not A_pt
    Kcands = inter
    # compute distances to A_pt
    d0 = mp.sqrt((Kcands[0][0]-A_pt[0])**2 + (Kcands[0][1]-A_pt[1])**2)
    d1 = mp.sqrt((Kcands[1][0]-A_pt[0])**2 + (Kcands[1][1]-A_pt[1])**2)
    # one of them will be (practically) zero distance (A itself)
    if d0 < mp.mpf('1e-12'):
        K = Kcands[1]
    elif d1 < mp.mpf('1e-12'):
        K = Kcands[0]
    else:
        # choose the one farther from A
        K = Kcands[0] if d0 > d1 else Kcands[1]
    # K'
    Kprime = reflect_point_across_line(K, Fp, Ep)
    # N = foot from D onto EF
    N = foot_point(D, Fp, Ep)
    # circumcenter of B, K, K'
    circ_BKK = circumcenter(B, K, Kprime)
    if circ_BKK is None:
        return None
    xb,yb,Rb = circ_BKK
    dist_N = mp.sqrt((N[0]-xb)**2 + (N[1]-yb)**2)
    delta = dist_N - Rb
    return delta

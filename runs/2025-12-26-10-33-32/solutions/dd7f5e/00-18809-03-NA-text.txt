def phi(n):
    result = n
    p = 2
    nn = n
    while p*p <= nn:
        if nn % p == 0:
            while nn % p == 0:
                nn //= p

def count_subsets_with_sum_limit(items, limit):
    # items is list of (index, weight)
    # count subsets with total weight <= limit, including empty?
    # We'll count nonempty separately.
    total = 0
    for r in range(1, len(items)+1):
        for combo in combinations(items, r):
            if sum(w for _,w in combo) <= limit:
                total += 1
    return total

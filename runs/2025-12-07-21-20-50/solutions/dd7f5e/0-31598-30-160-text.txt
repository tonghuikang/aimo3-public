def is_shifty_alpha(alpha_coeffs, max_d=200):
    # alpha_coeffs is tuple of 9 ints (coeff for x^0..x^8)
    # find minimal index c with non-zero
    # ignore all-zero case
    if all(c==0 for c in alpha_coeffs):
        return False
    # find c
    c = next(i for i,coeff in enumerate(alpha_coeffs) if coeff != 0)
    # construct polynomial alpha(x)
    poly = sum(alpha_coeffs[i]*x**i for i in range(len(alpha_coeffs)))
    # factor out x^c
    poly_prime = sp.expand(poly / x**c)  # polynomial division (since c minimal)
    # ensure integer coefficients
    poly_prime = sp.Poly(poly_prime, x)
    # Check constant term is ±1
    const_term = poly_prime.nth(0)
    if abs(const_term) != 1:
        return False
    # Check leading coefficient is ±1
    lead_deg = poly_prime.degree()
    lead_coeff = poly_prime.nth(lead_deg)
    if abs(lead_coeff) != 1:
        return False
    # now test divisibility of 1+x^d for d up to max_d
    for d in range(1, max_d+1):
        # compute (1 + x^d) modulo poly_prime
        # using sp.rem?
        # Use polynomial division: remainder = (1 + x**d).rem(poly_prime)
        rem = sp.rem(1 + x**d, poly_prime, domain='ZZ')
        if rem == 0:
            # found divisor
            return True
    return False

    # We'll generate a up to some bound where T_a <= N^2 maybe.
    # Since c^2 = T_a T_b <= N^2, we need T_a <= N^2 and T_b <= N^2.
    # So we can bound a up to amax such that T_a <= N^2.
    # T_a approx a^2/2, so amax ≈ sqrt(2) N. For N up to 10^5, amax ≈ 141421.
    amax = int(math.sqrt(2)*N)+10
    # Compute T values.
    T = [0]*(amax+1)
    for i in range(amax+1):
        T[i] = i*(i+1)//2
    total = 0
    for a in range(1, amax+1):
        Ta = T[a]
        # b must be > a, and T_a*T_b <= N^2
        # So T_b <= N^2 // Ta
        max_Tb = N*N // Ta
        # find max b such that T_b <= max_Tb
        # approximate b_max
        # we can binary search on T to find max b.
        lo, hi = a+1, amax
        while lo <= hi:
            mid = (lo+hi)//2
            if T[mid] <= max_Tb:
                lo = mid+1
            else:
                hi = mid-1
        bmax = hi
        for b in range(a+1, bmax+1):
            prod = Ta * T[b]
            c = int(math.isqrt(prod))
            if c*c == prod:
                if c <= N:
                    total += c
    return total

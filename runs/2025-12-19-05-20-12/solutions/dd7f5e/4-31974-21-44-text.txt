def get_factor_t_list(A0):
    t_factors = []
    # copy of A0 to divide out factors
    remainder = sp.Poly(A0.as_expr(), x)  # copy
    for t, poly in [(t, phis[2*t]) for t in t_list]:
        # while poly divides remainder
        while sp.rem(remainder, poly) == 0:
            t_factors.append(t)
            remainder = sp.quo(remainder, poly)
    # after dividing all possible, remainder should be constant ±1
    # if remainder not constant, something's wrong
    if remainder.degree() > 0:
        # not fully factored
        raise ValueError("Remaining polynomial not constant")
    return t_factors

def construct_beta(alpha_poly):
    # compute p (valuation)
    coeffs = alpha_poly.all_coeffs()
    # find smallest exponent with non-zero coefficient
    # use as_dict for exact exponents
    terms = alpha_poly.as_dict()
    if not terms:
        return None
    exps = [e[0] for e,c in terms.items() if c != 0]
    p = min(exps)
    # compute A0
    A0_expr = sp.expand(alpha_poly.as_expr() * x**(-p))
    A0 = sp.Poly(A0_expr, x)
    # find t's
    t_factors = get_factor_t_list(A0)
    # if no t_factors (i.e., A0 = ±1)
    if not t_factors:
        # choose m = 1 (any odd maybe)
        m = 1
    else:
        # compute m = lcm of t_factors
        m = 1
        for t in t_factors:
            m = m * t // math.gcd(m, t)
    # compute D = (1 + x^m) / A0
    numerator = sp.Poly(1 + x**m, x)
    D, rem = sp.div(numerator, A0)
    if rem != 0:
        raise ValueError("Division not exact")
    # D is polynomial; construct beta as reverse of D
    # beta(t) = coefficient of D at x^{-t} => D's coefficient at x^{e} maps to beta(t) where t = -e
    # So beta will have support at non-positive indices
    beta = {}
    for exp, coeff in D.as_dict().items():
        e = exp[0]
        beta[-e] = int(coeff)
    return beta, p, p+m

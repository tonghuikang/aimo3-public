# define cyclotomic data: n, phi, v2
import math

def phi(n):
    # Euler totient
    result = n
    p=2
    nn=n
    while p*p <= nn:
        if nn%p==0:
            while nn%p==0:
                nn//=p
            result -= result//p
        p+=1
    if nn>1:
        result -= result//nn
    return result

def v2(n):
    cnt=0
    while n%2==0:
        n//=2
        cnt+=1
    return cnt

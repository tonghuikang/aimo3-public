def solve_for_n(n, guess_vals):
    # Returns list of solutions (xA,yA,r) satisfying constraints.
    # guess_vals: list of initial guesses for nsolve.
    Fn = sp.fibonacci(n)
    Fn1 = sp.fibonacci(n+1)
    Fn2 = sp.fibonacci(n+2)
    a = Fn2
    x0 = Fn
    # define symbols
    xA, yA, r = sp.symbols('xA yA r', real=True)
    # distance equations (squared) for AB and AC
    eq1 = (yA*x0 - xA*r)**2 - r**2 * (yA**2 + xA**2)
    eq2 = (yA*x0 - (xA - a)*r - yA*a)**2 - r**2 * (yA**2 + (xA - a)**2)
    # K,N,K',B cyclic condition: determinant=0
    # define points
    B = sp.Point(0,0)
    D = sp.Point(x0, 0)
    I = sp.Point(x0, r)
    A = sp.Point(xA, yA)
    # Compute foot of perpendicular from I to AB (point F)
    # line AB:
    line_AB = sp.Line(B, A)
    F = line_AB.projection(I)
    # line AC:
    C = sp.Point(a, 0)
    line_AC = sp.Line(C, A)
    E = line_AC.projection(I)
    # Compute N: foot from D to line EF
    line_EF = sp.Line(E, F)
    N = line_EF.projection(D)
    # Compute K: second intersection of circles (A,F,E) and (A,B,C)
    circle_AFE = sp.Circle(A, F, E)
    circle_ABC = sp.Circle(A, B, C)

            P1.x**2 + P1.y**2 + D*P1.x + E*P1.y + Fcoeff,
            P2.x**2 + P2.y**2 + D*P2.x + E*P2.y + Fcoeff,

        return sol[D], sol[E], sol[Fcoeff]
    D1, E1, F1 = circle_eq_from_three_points(A, F, E)

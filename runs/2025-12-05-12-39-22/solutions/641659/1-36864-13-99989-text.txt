def circumcenter(p1, p2, p3):
    # return center (x,y) of circle passing through p1,p2,p3
    # using perpendicular bisectors approach
    # Solve linear equations
    x1,y1 = p1
    x2,y2 = p2
    x3,y3 = p3
    # compute determinants
    d = 2*(x1*(y2-y3) + x2*(y3-y1) + x3*(y1-y2))
    if abs(d) < 1e-12:
        return None
    ux = ((x1**2+y1**2)*(y2-y3) + (x2**2+y2**2)*(y3-y1) + (x3**2+y3**2)*(y1-y2)) / d
    uy = ((x1**2+y1**2)*(x3-x2) + (x2**2+y2**2)*(x1-x3) + (x3**2+y3**2)*(x2-x1)) / d

def circle_intersections(O0, R0, O1, R1):

def project_point_to_line(P, A, B):

    return (Qx, Qy)

def reflect_point_across_line(P, A, B):
    Q = project_point_to_line(P, A, B)

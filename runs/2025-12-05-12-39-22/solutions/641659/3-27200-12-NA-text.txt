def circle_intersection(c0, r0, c1, r1):
    # c0, c1 are (x,y) tuples
    x0, y0 = c0
    x1, y1 = c1
    dx = x1 - x0
    dy = y1 - y0
    d = math.hypot(dx, dy)
    if d == 0:
        return []
    # check for intersection existence
    if d > r0 + r1 + 1e-12 or d < abs(r0 - r1) - 1e-12:
        return []
    # distance from c0 to point p2 where line through intersections meets line between centers
    a = (r0**2 - r1**2 + d**2) / (2*d)
    # point p2
    x2 = x0 + a * dx / d
    y2 = y0 + a * dy / d
    # distance from p2 to each intersection point
    h_sq = r0**2 - a**2
    if h_sq < 0:
        h = 0.0
    else:
        h = math.sqrt(h_sq)
    # offset vector perpendicular to (dx,dy)
    rx = -dy * (h / d)
    ry = dx * (h / d)
    # two intersection points
    pA = (x2 + rx, y2 + ry)
    pB = (x2 - rx, y2 - ry)
    return [pA, pB]

def foot_of_perpendicular(P, Q1, Q2):
    # foot of perpendicular from point P to line through Q1,Q2
    # returns point
    x0,y0 = P
    x1,y1 = Q1
    x2,y2 = Q2

    dx = x2 - x1
    dy = y2 - y1

        return Q1

    # foot
    xf = x1 + t*dx
    yf = y1 + t*dy
    return (xf, yf)

def reflect_point_across_line(P, Q1, Q2):

    xr = 2*proj[0] - P[0]
    yr = 2*proj[1] - P[1]
    return (xr, yr)

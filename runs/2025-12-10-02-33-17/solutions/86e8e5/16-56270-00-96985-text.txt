def factorial_mod(n, mod):
    res = 1
    for i in range(2, n+1):
        res = (res * i) % mod
    return res

# function to compute M mod m = 3^K mod m, where K = 2025!
def M_mod(m):
    # factor out powers of 3
    e = 0
    temp = m
    while temp % 3 == 0:
        e += 1
        temp //= 3
    r = temp  # r coprime to 3
    # part modulo 3^e
    if e == 0:
        a_mod = None
    else:
        # M = 3^K, K >= e, thus M ≡ 0 mod 3^e
        a_mod = 0
        mod1 = 3**e
    # modulo r
    if r == 1:
        b_mod = 0
        mod2 = 1
    else:

        K_mod_phi = factorial_mod(2025, phi_r)

        b_mod = pow(3, K_mod_phi, r)
        mod2 = r
    # combine via CRT
    if e == 0:
        # only mod r
        return b_mod % m
    if r == 1:
        return a_mod % m
    # use sympy's crt
    res, lcm_mod = sp.crt([mod1, mod2], [a_mod, b_mod])
    return int(res % m)

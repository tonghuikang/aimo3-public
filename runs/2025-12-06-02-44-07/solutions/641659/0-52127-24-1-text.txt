def find_root_r(a,b):
    # find r in (0, min(a,b)) where determinant changes sign (positive to negative)
    lo = 1e-8
    hi = min(a,b) - 1e-8
    f_lo = cyclic_det(lo, a, b)
    f_hi = cyclic_det(hi, a, b)
    # evaluate sign at some points
    N=200
    xs = np.linspace(lo, hi, N)
    f_vals = [cyclic_det(x, a, b) for x in xs]
    # find sign change interval
    interval = None
    for i in range(N-1):
        if f_vals[i] is None or f_vals[i+1] is None:
            continue
        if f_vals[i]*f_vals[i+1] < 0:
            interval = (xs[i], xs[i+1])
            break
    if interval is None:
        # maybe sign change near hi
        # Try using bisection between lo and hi if signs differ
        if f_lo is not None and f_hi is not None and f_lo*f_hi < 0:
            interval = (lo, hi)
        else:
            # no sign change found; return None
            return None
    a0,b0 = interval
    f_a = cyclic_det(a0, a, b)
    f_b = cyclic_det(b0, a, b)
    for _ in range(60):
        mid = (a0+b0)/2.0
        f_mid = cyclic_det(mid, a, b)
        if f_mid == 0:
            return mid
        # decide which subinterval contains root
        if f_a * f_mid <= 0:
            b0 = mid
            f_b = f_mid
        else:
            a0 = mid
            f_a = f_mid
    return (a0+b0)/2.0

    for k in range(1,limit+1):
        # compute divisors
        divisors = []
        i = 1
        while i*i <= k:
            if k%i==0:
                divisors.append(i)
                if i != k//i:
                    divisors.append(k//i)
            i+=1
        divisors = sorted(set(divisors))
        if len(divisors) < 3: continue
        # check triples distinct
        from itertools import combinations
        for a,b,c in combinations(divisors,3):
            if a+b+c == N:
                return k
    return None

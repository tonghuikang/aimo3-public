def compute_small_K(K):
    M = 3**K
    # p0 smallest odd prime factor of M-2
    D0 = M - 2
    # factor small to find p0
    p0 = None
    for p in sp.primerange(3, 50000):
        if D0 % p == 0:
            p0 = p
            break
    # p1 smallest odd prime factor of 5M-2
    D1 = 5*M - 2
    p1 = None
    for p in sp.primerange(3, 50000):
        if D1 % p == 0:
            p1 = p
            break
    # compute m,n
    m = K // p0
    n = (5*K) // p1
    N2 = (26*K)//3 - m - n - 2
    from math import gcd
    g = math.gcd(N2, K)
    num = N2 // g
    den = K // g
    # compute p+q mod some prime e.g. 99991
    mod = 99991
    result = (num + den) % mod
    return {
        "K":K,
        "p0":p0,"p1":p1,
        "m":m,"n":n,
        "N2":N2,
        "gcd":g,
        "num":num,"den":den,
        "p+q_mod": result
    }

    alphas=set()

        poly = 1 + x**d
        # get list of cyclotomic factor indices
        # sympy factor can factor into cyclotomics? Let's factor and get irreducible factors
        fl = sp.factor_list(poly)[1]  # list of (factor,exp)

        # We'll get their indices by matching known cyclotomics
        # Let's compute list of divisors by generating all subsets of these irreducible factors
        factors = [fac for fac,exp in fl if exp==1]
        # generate subsets

<|end|><|start|>assistant<|channel|>analysis to=python code<|message|>def reflect_point_across_line(P, A, B):
    # reflect point P across line AB
    # Using vector projection formula.
    # line defined by A and B
    # Compute projection of vector AP onto AB, then compute reflected point.
    # Actually reflection: P' = 2*proj_line(P) - P, where proj_line(P) is projection onto line AB.
    # Let AB = B - A, AP = P - A.
    import numpy as np
    A_arr = np.array(A)
    B_arr = np.array(B)
    P_arr = np.array(P)
    AB = B_arr - A_arr
    AP = P_arr - A_arr
    # projection scalar t = (AP·AB) / (AB·AB)
    t = np.dot(AP, AB) / np.dot(AB, AB)
    proj = A_arr + t * AB
    P_ref = 2*proj - P_arr
    return (float(P_ref[0]), float(P_ref[1]))

def foot_of_perpendicular(P, A, B):
    # foot of perpendicular from point P onto line AB
    import numpy as np
    A_arr = np.array(A)
    B_arr = np.array(B)
    P_arr = np.array(P)
    AB = B_arr - A_arr
    # projection scalar from A onto line AB of AP
    AP = P_arr - A_arr
    t = np.dot(AP, AB) / np.dot(AB, AB)
    foot = A_arr + t * AB
    return (float(foot[0]), float(foot[1]))

def cyclic_det(B_pt, N_pt, K_pt, Kp_pt):
    # compute determinant for concyclicity

    M = []

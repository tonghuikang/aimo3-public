    a1 = perp_dir[0]; b1 = -vEF[0]
    a2 = perp_dir[1]; b2 = -vEF[1]
    rhs0 = E[0] - mid_BK[0]; rhs1 = E[1] - mid_BK[1]
    det = a1*b2 - a2*b1
    if abs(det) < 1e-14:
        return None
    s1 = (rhs0*b2 - rhs1*b1) / det
    O_center = mid_BK + s1*perp_dir
    # compute squared distances
    ON2 = (O_center[0]-N[0])**2 + (O_center[1]-N[1])**2
    OK2 = (O_center[0]-K[0])**2 + (O_center[1]-K[1])**2
    return ON2 - OK2

def ratio_R(F_n, F_n1, t):
    pts = geometry_values_correct(F_n, F_n1, t)
    B = pts['B']; C = pts['C']; N = pts['N']; E = pts['E']; K = pts['K']
    # compute circle tangent to BN at B and passing through K: find intersection with BC line to get T.
    # Circle passes through B and K, tangent to line BN at B => center O_T lies on line perpendicular to BN at B.
    # Let BN direction vector = N - B
    BN = N - B
    # Perpendicular direction at B = rotate BN by 90 degrees: (-BN_y, BN_x)
    perp_BN = mp.matrix([-BN[1], BN[0]])
    # Center O_T = B + u * perp_BN (for some u)
    # O_T must satisfy distance to B = distance to K.
    # |O_T - B| = |O_T - K| => distance from B to O_T = distance from O_T to K.
    # Let O_T = B + u * perp_BN = u * perp_BN (since B is origin)
    # Then squared distances: |u * perp_BN|^2 = |u*perp_BN - K|^2
    # Solve for u.
    # Let p = perp_BN, so O_T = u p.
    # Equation: u^2 * |p|^2 = (u p - K)·(u p - K) = u^2 * |p|^2 - 2u (p·K) + |K|^2
    # Cancel u^2|p|^2 both sides => 0 = -2u (p·K) + |K|^2 => u = |K|^2 / (2 * p·K)
    p = perp_BN
    p_dot_K = p[0]*K[0] + p[1]*K[1]
    if abs(p_dot_K) < 1e-14:
        # degenerate case?
        return None
    u = (K[0]**2 + K[1]**2) / (2 * p_dot_K)
    O_T = u * p
    # radius r_T = distance from O_T to B = |O_T|
    r_T = mp.sqrt(O_T[0]**2 + O_T[1]**2)
    # Now intersect this circle with BC (x-axis) again.
    # Circle equation: (x - O_T_x)^2 + (y - O_T_y)^2 = r_T^2. For y=0 (BC), we solve for x.
    # (x - O_T_x)^2 + (0 - O_T_y)^2 = r_T^2 => (x - O_T_x)^2 = r_T^2 - O_T_y^2.
    # Compute RHS = r_T^2 - O_T_y^2 = O_T_x^2 + O_T_y^2 - O_T_y^2 = O_T_x^2.
    # Indeed r_T^2 = O_T_x^2 + O_T_y^2. So RHS = O_T_x^2.
    # Therefore (x - O_T_x)^2 = O_T_x^2 => x - O_T_x = ± O_T_x => x = O_T_x ± O_T_x = 0 or 2*O_T_x.
    # Solutions: x=0 (point B) and x=2*O_T_x (the other intersection T).
    # So T = (2*O_T_x, 0)
    T = mp.matrix([2*O_T[0], 0])
    # Compute ratio R = (CT * NB) / (BT * NE)
    # CT = distance from C to T = abs(C.x - T.x)
    CT = abs(C[0] - T[0])
    BT = abs(T[0] - B[0])  # should be T.x
    NB = mp.sqrt((N[0] - B[0])**2 + (N[1] - B[1])**2)
    NE = mp.sqrt((N[0] - E[0])**2 + (N[1] - E[1])**2)
    if BT == 0:
        return None
    R_val = (CT * NB) / (BT * NE)
    return R_val

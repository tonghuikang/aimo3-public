    # v = Q-P
    vx = Q[0] - P[0]
    vy = Q[1] - P[1]

        rad2_minus_y2 = R_t*R_t - O_t[1]*O_t[1]
        if rad2_minus_y2 < -1e-9:
            BT = None
            CT = None
        else:
            # root(s)
            rad2_minus_y2 = max(rad2_minus_y2, 0.0)
            sqrt_val = math.sqrt(rad2_minus_y2)

            x2 = O_t[0] - sqrt_val
            # B is at x=0, one root should be zero (or near)
            # pick the other root
            if abs(x1) < 1e-6:
                x_T = x2
            elif abs(x2) < 1e-6:
                x_T = x1
            else:
                # pick the positive root maybe >0?
                # choose the one larger than 0
                # but if both non-zero, choose the one that is not near zero
                if abs(x1) > abs(x2):
                    x_T = x1
                else:
                    x_T = x2
            BT = abs(x_T - B[0])
            CT = abs(C[0] - x_T)
    # NB and NE
    NB = math.dist(N, B)
    NE = math.dist(N, E)
    # ratio
    if BT is None or CT is None or NB == 0 or NE == 0:
        ratio = None
    else:
        ratio = (CT * NB) / (BT * NE)

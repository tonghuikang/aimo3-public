from math import sqrt

def circumcenter(p1, p2, p3):
    # returns center (ox,oy) and radius
    # Using perpendicular bisector method
    (x1, y1), (x2, y2), (x3, y3) = p1, p2, p3
    # Compute using formula from coordinates
    d = 2*(x1*(y2 - y3) + x2*(y3 - y1) + x3*(y1 - y2))
    if abs(d) < 1e-12:
        raise ValueError("Collinear points")
    ux = ((x1**2 + y1**2)*(y2 - y3) + (x2**2 + y2**2)*(y3 - y1) + (x3**2 + y3**2)*(y1 - y2)) / d
    uy = ((x1**2 + y1**2)*(x3 - x2) + (x2**2 + y2**2)*(x1 - x3) + (x3**2 + y3**2)*(x2 - x1)) / d
    R = sqrt((ux - x1)**2 + (uy - y1)**2)
    return (ux, uy), R

def intersect_circles(O1, R1, O2, R2):
    (x1, y1), (x2, y2) = O1, O2
    dx = x2 - x1
    dy = y2 - y1
    d = sqrt(dx*dx + dy*dy)
    # No intersection
    if d > R1 + R2 + 1e-12:
        return []
    if d < abs(R1 - R2) - 1e-12:
        return []
    # a = distance from O1 to chord midpoint
    a = (R1*R1 - R2*R2 + d*d) / (2*d)
    # height of intersection from chord midpoint
    h_sq = R1*R1 - a*a
    if h_sq < -1e-12:
        return []
    h = sqrt(max(0.0, h_sq))
    # midpoint
    xm = x1 + a * dx / d
    ym = y1 + a * dy / d
    # intersection points
    rx = -dy * (h / d)
    ry = dx * (h / d)
    p1 = (xm + rx, ym + ry)
    p2 = (xm - rx, ym - ry)
    return [p1, p2]

def foot_of_perpendicular(P, A, B):
    # foot of perpendicular from point P onto line AB
    # line AB: vector v = B - A
    (x0, y0) = P
    (x1, y1) = A
    (x2, y2) = B
    vx = x2 - x1
    vy = y2 - y1
    # projection parameter t = ((P-A)·v) / (v·v)
    t = ((x0 - x1)*vx + (y0 - y1)*vy) / (vx*vx + vy*vy)
    fx = x1 + t * vx
    fy = y1 + t * vy
    return (fx, fy)

def reflect_point_across_line(P, A, B):
    # reflect point P across line AB
    # Compute foot H of perpendicular from P onto line AB, then reflect: P' = 2H - P
    H = foot_of_perpendicular(P, A, B)
    Px, Py = P
    Hx, Hy = H
    return (2*Hx - Px, 2*Hy - Py)

    # Represent as matrix [[-dy_bn, dy_bk], [dx_bn, -dx_bk]] * [t1,t2]^T = RHS

    RHS = [mid_BK[0] - B[0], mid_BK[1] - B[1]]
    try:
        t1, t2 = sp.Matrix(A_mat).LUsolve(sp.Matrix(RHS))
    except Exception as e:
        # fallback to numpy
        import numpy.linalg as la
        try:
            t1, t2 = la.solve(A_mat, RHS)
        except Exception as ee:
            return None
    # center O2
    O2 = (B[0] + t1*(-dy_bn), B[1] + t1*dx_bn)
    # radius
    R2 = math.hypot(O2[0] - B[0], O2[1] - B[1])
    # Intersection with BC (y=0)
    ox, oy = O2
    disc = R2*R2 - oy*oy
    if disc < -1e-9:
        return None
    sqrt_disc = math.sqrt(max(0.0, disc))
    x_cand1 = ox + sqrt_disc
    x_cand2 = ox - sqrt_disc
    # choose x != 0
    if abs(x_cand1) > 1e-6:
        x_T = x_cand1
    else:
        x_T = x_cand2
    T = (x_T, 0.0)
    # lengths
    CT = math.hypot(C[0] - T[0], C[1] - T[1])
    BT = math.hypot(T[0] - B[0], T[1] - B[1])
    NB = math.hypot(N0[0] - B[0], N0[1] - B[1])
    NE = math.hypot(N0[0] - E[0], N0[1] - E[1])
    # avoid division by zero
    if BT == 0 or NE == 0:
        return None
    ratio = (CT * NB) / (BT * NE)
    return ratio

            return None

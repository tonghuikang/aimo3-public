def angle(p, q, r):
    # angle at q formed by p-q-r
    # vectors qp = p - q, qr = r - q
    qp = (p[0]-q[0], p[1]-q[1])
    qr = (r[0]-q[0], r[1]-q[1])
    # cross product (2D) qp x qr = qp_x * qr_y - qp_y * qr_x
    cross = qp[0]*qr[1] - qp[1]*qr[0]
    dot = qp[0]*qr[0] + qp[1]*qr[1]
    return mp.atan2(cross, dot)

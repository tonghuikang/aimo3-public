    BT = abs(xT); CT = abs(1.0 - xT)
    NB = np.linalg.norm(N-B); NE = np.linalg.norm(N-E)
    return (CT*NB)/(BT*NE)

def a_even(n):
    d = fib(n)/fib(n+2)   # scaled BD/BC
    roots = find_r(d)
    ratios = [ratio_for_root(d,r) for r in roots]
    return max(ratios)

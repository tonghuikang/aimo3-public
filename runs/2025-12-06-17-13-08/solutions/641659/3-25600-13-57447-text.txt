def triangle_data(n, c):
    # compute geometric data for given n and side c (>F_n)
    Fn = fib_seq(n)
    a = fib_seq(n+2)  # BC
    # sides
    b = c + fib_seq(n-1)
    # Coordinates
    B = (0.0, 0.0)
    C = (float(a), 0.0)
    # compute u,v for A

    # ensure discriminant positive

        return None

    D = (float(Fn), 0.0)

    F = (u*lam, v*lam)
    # E: AE = c - Fn
    mu = (c - Fn) / b
    E = (u + mu*(a - u), v*(1 - mu))
    # return all points
    return {'A':A,'B':B,'C':C,'D':D,'E':E,'F':F,'a':a,'b':b,'c':c,'Fn':Fn}
def circumcenter(p1,p2,p3):
    # compute circumcenter of triangle points
    # using perpendicular bisector intersection
    # returns (x,y) as float
    (x1,y1),(x2,y2),(x3,y3)=p1,p2,p3
    d = 2*(x1*(y2 - y3) + x2*(y3 - y1) + x3*(y1 - y2))
    if abs(d) < 1e-12:
        return None
    ux = ((x1**2 + y1**2)*(y2 - y3) + (x2**2 + y2**2)*(y3 - y1) + (x3**2 + y3**2)*(y1 - y2))/d
    uy = ((x1**2 + y1**2)*(x3 - x2) + (x2**2 + y2**2)*(x1 - x3) + (x3**2 + y3**2)*(x2 - x1))/d
    return (ux, uy)

def circle_center_radius(p1,p2,p3):
    cen = circumcenter(p1,p2,p3)
    if cen is None:
        return None
    r = math.hypot(p1[0]-cen[0], p1[1]-cen[1])
    return (cen, r)

def intersect_circles(c1_center, r1, c2_center, r2):
    # return intersection points (two)
    x0,y0 = c1_center
    x1,y1 = c2_center
    dx = x1 - x0
    dy = y1 - y0
    d = math.hypot(dx,dy)
    if d > r1+r2+1e-9 or d < abs(r1-r2)-1e-9:
        return [] # no intersection
    # find a = (r1^2 - r2^2 + d^2) / (2d)
    a = (r1*r1 - r2*r2 + d*d) / (2*d)
    h_sq = r1*r1 - a*a
    if h_sq < -1e-9:
        return []
    h = math.sqrt(max(h_sq,0.0))
    xm = x0 + a*dx/d
    ym = y0 + a*dy/d
    # intersection points
    xs1 = xm + h*dy/d
    ys1 = ym - h*dx/d
    xs2 = xm - h*dy/d
    ys2 = ym + h*dx/d
    return [(xs1,ys1),(xs2,ys2)]

def foot_of_perpendicular(P, Q, R):
    # foot from point P to line QR
    # Q,R are points on line
    x0,y0 = P
    x1,y1 = Q
    x2,y2 = R
    # line QR param t: Q + t*(R-Q)
    vx = x2 - x1
    vy = y2 - y1
    # projection t = ((P-Q)·v)/|v|^2
    t = ((x0-x1)*vx + (y0-y1)*vy) / (vx*vx + vy*vy)
    foot = (x1 + t*vx, y1 + t*vy)
    return foot

def reflect_point_across_line(P, Q, R):
    # reflect point P across line QR
    # Compute foot of perpendicular from P to line QR, then reflect
    foot = foot_of_perpendicular(P, Q, R)
    # reflect: P' = 2*foot - P
    return (2*foot[0] - P[0], 2*foot[1] - P[1])

def det4(p1,p2,p3,p4):
    # compute determinant for concyc condition: 
    # |x^2+y^2  x y 1|
    M = []
    for (x,y) in [p1,p2,p3,p4]:
        M.append([x*x + y*y, x, y, 1])
    return sp.Matrix(M).det()

    # t = 2025!
    # Compute t factorial modulo (p-1) for each p
    # Precompute t_mod for all p up to maxp
    # We'll compute t modulo (p-1) by iterating product 1..2025 modulo (p-1)
    for p in range(3, maxp+1, 2):
        if not sp.isprime(p):
            continue
        if p == 3:
            # M mod 3 = 0
            N_mod_p = (c - 1) % 3
        else:
            # compute t mod (p-1)
            mod_phi = p-1
            # compute 2025! mod (p-1)
            t_mod = 1
            for i in range(1,2026):
                t_mod = (t_mod * i) % mod_phi
            # compute M mod p
            M_mod_p = pow(3, t_mod, p)
            N_mod_p = (M_mod_p + (c - 1)) % p
        if N_mod_p == 0:
            return p
    return None

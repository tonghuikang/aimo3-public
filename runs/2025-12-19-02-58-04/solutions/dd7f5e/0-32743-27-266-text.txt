def convolve(a,b):
    # convolution a*b
    res = [0]*(len(a)+len(b)-1)
    for i,ai in enumerate(a):
        if ai==0: continue
        for j,bj in enumerate(b):
            if bj==0: continue
            res[i+j] += ai * bj
    return res

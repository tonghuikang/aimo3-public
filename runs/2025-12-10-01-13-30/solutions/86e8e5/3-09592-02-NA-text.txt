    # find minimal integer x that has three distinct divisors sum to n
    # brute force up to limit
    best = None
    for x in range(1, limit+1):
        divisors = [d for d in range(1, x+1) if x%d==0]
        # pick distinct triple sum n
        for i in range(len(divisors)):
            for j in range(i+1, len(divisors)):
                for k in range(j+1, len(divisors)):
                    if divisors[i]+divisors[j]+divisors[k] == n:
                        best = x
                        return best
    return None

for n in range(6, 31):

def simulate_tournament(order):
    # order: list of player ids sorted from fastest to slowest? Actually we provide speed ranking: 
    # higher speed => smaller index in list? Let's define speeds decreasing: speed higher = faster.
    # We'll assign each player a "speed rank", lower rank = faster.
    # We'll simulate pairings arbitrarily: At each round, pair players with same score arbitrarily.
    n = len(order)
    # Build speed dict: speed rank (lower is faster)
    speed_rank = {player: i for i, player in enumerate(order)}  # smaller i faster
    # Initialize all scores 0
    scores = {player: 0 for player in order}
    # For each round i from 1..log2(n)
    for i in range(1, int(math.log2(n)) + 1):
        weight = 2**(int(math.log2(n)) - i)
        # group by current score
        groups = {}
        for player in order:
            groups.setdefault(scores[player], []).append(player)
        # for each group, pair arbitrarily
        for score, group in groups.items():
            # group size must be power of 2, else error
            assert len(group) % 2 == 0
            random.shuffle(group)
            for j in range(0, len(group), 2):
                a, b = group[j], group[j+1]
                # winner is faster (lower rank)
                if speed_rank[a] < speed_rank[b]:
                    winner, loser = a, b
                else:
                    winner, loser = b, a
                scores[winner] += weight
                # loser gets nothing
    # ranking by score descending
    ranking = sorted(order, key=lambda p: scores[p], reverse=True)
    return tuple(ranking)

def all_matchings(players):
    if not players:
        return [[]]
    first = players[0]
    res = []
    for i in range(1, len(players)):
        second = players[i]
        rest = players[1:i] + players[i+1:]
        for matching in all_matchings(rest):
            res.append([(first, second)] + matching)
    return res

def simulate_one_pairings(initial_players, pairings_by_round):
    # initial_players: list of player IDs; assume speed order is identity (lower ID faster)
    n=len(initial_players)
    logn=int(math.log2(n))
    # scores dict
    scores={p:0 for p in initial_players}
    # speed rank dict: lower ID = faster
    speed_rank={p:p for p in initial_players}
    # pairings_by_round: list of list of pairs for each round
    for i in range(logn):
        weight=2**(logn-1-i) # i from 0..logn-1; weight = 2^{logn-i-1}
        # groups determined by scores before this round, but pairings_by_round are already grouped accordingly.
        # We'll assume they follow grouping.
        for a,b in pairings_by_round[i]:
            # winner based on speed
            if speed_rank[a] < speed_rank[b]:
                winner=a
                loser=b
            else:
                winner=b
                loser=a
            scores[winner]+=weight
    # final ranking
    ranking=sorted(initial_players, key=lambda p:scores[p], reverse=True)
    return tuple(ranking)

def all_rankings_n4():
    players=list(range(4))
    n=4
    logn=2
    # all possible first round matchings
    first_matchings=all_matchings(players)
    rankings=set()
    for m1 in first_matchings:
        # compute scores after round1
        scores={p:0 for p in players}
        weight1=2**(logn-1-0) # =2
        # determine winners after round1
        winners=[]; losers=[]
        for a,b in m1:
            winner = a if a<b else b # smaller index faster
            loser = b if winner==a else a
            scores[winner] += weight1
            winners.append(winner)
            losers.append(loser)
        # round2 groups: winners and losers each size 2
        # possible pairings within each group:
        group_pairs_w = all_matchings(winners)
        group_pairs_l = all_matchings(losers)
        for m2w in group_pairs_w:
            for m2l in group_pairs_l:
                # combine matchings for round2
                m2 = m2w + m2l
                # simulate
                ranking = simulate_one_pairings(players, [m1, m2])
                rankings.add(ranking)
    return rankings

def simulate_random_tournament(order, seed=None):
    random.seed(seed)
    n=len(order)
    logn=int(math.log2(n))
    # scores dict
    scores={p:0 for p in order}
    # speed rank: order index: position in order list; faster= earlier index (lower)
    speed_index={player: i for i, player in enumerate(order)}
    # group players by scores at each round
    for i in range(logn):
        weight=2**(logn-1-i)
        # group players by current score
        groups = {}
        for p in order:
            groups.setdefault(scores[p], []).append(p)
        # for each group, random shuffle and pair
        for grp in groups.values():
            random.shuffle(grp)
            for j in range(0, len(grp), 2):
                a,b=grp[j], grp[j+1]
                if speed_index[a] < speed_index[b]:
                    winner, loser = a, b
                else:
                    winner, loser = b, a
                scores[winner] += weight
    # ranking by score desc
    ranking = sorted(order, key=lambda p: scores[p], reverse=True)
    return tuple(ranking)

def all_matchings(participants):
    # participants is a tuple sorted
    n = len(participants)
    if n == 0:
        return [()]
    # recursion: pick first element, pair with each of the others
    res = []
    first = participants[0]
    others = participants[1:]
    for i, second in enumerate(others):
        # create remaining tuple after removing second
        remaining = others[:i] + others[i+1:]
        for sub in all_matchings(remaining):
            res.append(((first, second),) + sub)

def simulate(participants, round_idx, r):
    # participants: tuple of player indices
    # round_idx: 1-indexed round number
    # returns a list of dicts mapping player->score after completing all rounds from this round onward, assuming players start with current scores stored elsewhere?
    # Actually we maintain scores dict passed as mutable? Instead we can compute final scores at the end by adding points each round.
    # Simpler: recursion returns list of assignments of total added points per player from round_idx onward.
    if round_idx > r:
        # no more rounds, return one assignment: zero points for each participant
        return [dict()]  # empty dict
    point = 2**(r - round_idx)
    results = []
    # generate all matchings among participants
    for matching in all_matchings(participants):
        # Determine winners and losers
        winners = []
        losers = []
        for a,b in matching:
            if a < b:
                winners.append(a)
                losers.append(b)
            else:
                winners.append(b)
                losers.append(a)
        # Recursively simulate for winners and losers groups for next round
        # both groups have same size (half)
        sub_winners = simulate(tuple(sorted(winners)), round_idx+1, r)
        sub_losers = simulate(tuple(sorted(losers)), round_idx+1, r)
        # combine results
        for sw in sub_winners:
            for sl in sub_losers:

                combined = {}

                for w in winners:

                for d in (sw, sl):
                    for k,v in d.items():

                results.append(combined)
    return results

def final_orderings(r):

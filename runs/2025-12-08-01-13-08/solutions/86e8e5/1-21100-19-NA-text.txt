def min_lcm_for_sum(s):
    min_l = None
    best = None
    # iterate over triples a<b<c positive ints summing s
    # naive O(s^2). s small maybe up to some limit.
    for a in range(1, s-2):
        for b in range(a+1, s-a-1):
            c = s - a - b
            if c <= b: continue
            # distinct positive automatically
            l = sp.ilcm(a,b,c)
            if min_l is None or l < min_l:
                min_l = l
                best = (a,b,c)
    return min_l, best

    l,_=min_lcm_for_sum(s)

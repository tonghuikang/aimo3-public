def triangle_data(d, cd, r):
    # d = BD, cd = CD
    a = d + cd  # BC length
    # coordinates
    B = sp.Point(0, 0)
    C = sp.Point(a, 0)
    D = sp.Point(d, 0)
    I = sp.Point(d, r)
    # slopes of tangents from B and C
    # AB line: y = mB x
    # mB formula:
    mB = 2*d*r/(d**2 - r**2)
    # AC line: y = mC (x-a)
    # mC = -2*cd*r/(cd**2 - r**2)
    mC = -2*cd*r/(cd**2 - r**2)
    # compute intersection point A
    # solve mB*x = mC*(x-a)
    xA = -mC*a/(mB - mC)
    yA = mB*xA
    A = sp.Point(xA, yA)
    # foot of perpendicular from I to line AB (point F)
    # line AB: y - mB x = 0
    # projection formula: foot = I - ((I dot n)/||n||^2) * n, where n is normal vector of line AB
    # line AB: n = (mB, -1) (since line equation mB*x - y = 0)
    n_AB = sp.Matrix([mB, -1])
    I_vec = sp.Matrix([I.x, I.y])
    # foot = I - (n·I)/||n||^2 * n
    foot_F_vec = I_vec - (n_AB.dot(I_vec))/ (n_AB.dot(n_AB)) * n_AB
    F = sp.Point(foot_F_vec[0], foot_F_vec[1])
    # foot from I to line AC (point E)
    # line AC: y - mC*(x-a) = 0 => y - mC*x + mC*a = 0 => mC*x - y - mC*a =0
    n_AC = sp.Matrix([mC, -1])
    # line constant term = -mC*a
    # distance from point to line: Ax+By+C, but foot formula with offset:
    # For line Ax+By+C = 0, normal vector n = (A,B), foot = P - (A*x0 + B*y0 + C)/(A^2+B^2) * (A,B)
    A_line = mC
    B_line = -1
    C_line = -mC*a

    foot_E_vec = sp.Matrix([I.x, I.y]) - (A_line*I.x + B_line*I.y + C_line)/(A_line**2 + B_line**2) * sp.Matrix([A_line, B_line])
    E = sp.Point(foot_E_vec[0], foot_E_vec[1])
    # circumcircle of A, F, E: find circle center and radius via sympy
    circle_AFE = sp.Circle(A, F, E)

    NB = N.distance(B)

        "mB": mB, "mC": mC,
        "r": r
    }

    # Compute points in numpy for speed

        a,b = 0,1
        for i in range(k):
            a,b = b,a+b
        return a

    CD = fib(n+1)
    a = fib(n+2)

    ox, oy = O_T

    ratio = CT * NB / (BT * NE) if BT !=0 and NE !=0 else None
    return {'ratio': ratio, 'cyclic_err': cyclic_err, 's': s_val, 'K_pt':K_pt, 'K_ref':K_ref}

def poly_div(poly_num, poly_den):
    # divide integer polynomials, exact division assumed, return dict.
    # Use sympy for simplicity
    x=sp.Symbol('x')
    pnum = sum(c*x**e for e,c in poly_num.items())
    pden = sum(c*x**e for e,c in poly_den.items())
    quotient = sp.simplify(pnum / pden)
    # quotient should be polynomial
    # convert to dict
    qpoly = sp.Poly(quotient, x)
    coeffs = {i:int(qpoly.coeff_monomial(x**i)) for i in range(qpoly.degree()+1) if qpoly.coeff_monomial(x**i)!=0}
    return coeffs

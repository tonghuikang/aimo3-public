def min_lcm_for_S(S):
    min_lcm = None
    min_triple = None
    # find all distinct triples a<b<c positive integers with sum S
    # Since S is not huge (up to maybe 30), brute force is fine.
    for a in range(1, S-1):
        for b in range(a+1, S):
            c = S - a - b
            if c <= b: # ensure distinct and increasing
                continue
            # distinct positive ints
            # compute lcm
            import math
            def lcm(x,y):
                return x*y // math.gcd(x,y)
            L = lcm(lcm(a,b), c)
            if min_lcm is None or L < min_lcm:
                min_lcm = L
                min_triple = (a,b,c)
    return min_lcm, min_triple

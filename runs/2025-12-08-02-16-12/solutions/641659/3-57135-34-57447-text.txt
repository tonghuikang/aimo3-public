def cyclic_det_precise(n, r):
    Fn = fib(n); Fn1 = fib(n+1)
    B = mp.matrix([ -Fn, -r ])
    C = mp.matrix([ Fn1, -r ])
    # compute F,E
    F = mp.matrix([ -2 * r**2 * Fn/(Fn**2 + r**2), r*(Fn**2 - r**2)/(Fn**2 + r**2) ])
    E = mp.matrix([ 2 * r**2 * Fn1/(Fn1**2 + r**2), r*(Fn1**2 - r**2)/(Fn1**2 + r**2) ])
    # compute A
    dir_AB = mp.matrix([ -F[1], F[0] ])
    dir_AC = mp.matrix([ -E[1], E[0] ])
    dx1, dy1 = dir_AB[0], dir_AB[1]
    dx2, dy2 = dir_AC[0], dir_AC[1]
    A_mat = mp.matrix([[dx1, -dx2],[dy1, -dy2]])
    rhs = mp.matrix([C[0] - B[0], C[1] - B[1]])
    t = mp.lu_solve(A_mat, rhs)
    t1 = t[0]
    A = mp.matrix([ B[0] + t1*dx1, B[1] + t1*dy1 ])
    # circumcenters
    def circumcenter(P1,P2,P3):
        x1,y1=P1[0],P1[1]
        x2,y2=P2[0],P2[1]
        x3,y3=P3[0],P3[1]
        M=mp.matrix([[2*(x2-x1),2*(y2-y1)],[2*(x3-x1),2*(y3-y1)]])
        rhs2=mp.matrix([x2**2+y2**2 - x1**2 - y1**2, x3**2+y3**2 - x1**2 - y1**2])
        O=mp.lu_solve(M,rhs2)
        return O
    O = circumcenter(A,B,C)
    O2 = circumcenter(A,F,E)
    # K second intersection
    # line from subtract circles
    R2 = (O2[0]-A[0])**2 + (O2[1]-A[1])**2
    R = (O[0]-A[0])**2 + (O[1]-A[1])**2
    a_lin = 2*(O2[0] - O[0])
    b_lin = 2*(O2[1] - O[1])
    c_lin = (O2[0]**2+O2[1]**2 - R2)-(O[0]**2+O[1]**2 - R)
    # solve line eq and circle
    if abs(b_lin) > mp.mpf('1e-30'):
        m = -a_lin/b_lin
        k = c_lin/b_lin
        Acoeff = 1 + m**2
        Bcoeff = -2*O[0] + 2*m*(k - O[1])
        Ccoeff = O[0]**2 + (k - O[1])**2 - R
        disc = Bcoeff**2 - 4*Acoeff*Ccoeff
        sqrt_disc = mp.sqrt(disc)
        x1 = (-Bcoeff + sqrt_disc)/(2*Acoeff)
        x2 = (-Bcoeff - sqrt_disc)/(2*Acoeff)
        y1 = m*x1 + k
        y2 = m*x2 + k
        # choose K not A
        if mp.sqrt((x1-A[0])**2 + (y1-A[1])**2) > mp.mpf('1e-30'):
            K = mp.matrix([x1, y1])
        else:
            K = mp.matrix([x2, y2])
    else:
        # fallback
        return None

    vEF_norm = mp.sqrt(vEF[0]**2 + vEF[1]**2)
    u = vEF / vEF_norm

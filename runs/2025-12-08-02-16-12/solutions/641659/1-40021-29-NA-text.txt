def triangle_geometry(n, x):
    # Compute geometry data for triangle given n and x (semiperimeter ratio)
    # returns a dict with points etc.
    # Compute Fibonacci numbers
    F = [0,1]
    for i in range(2, n+3):
        F.append(F[-1] + F[-2])
    Fn = F[n]
    Fn1 = F[n+1]
    a = F[n+2]  # BC
    s = x * a
    r2 = Fn * Fn1 * (s - a) / s
    if r2 <= 0:
        return None
    r = math.sqrt(r2)
    # slopes
    mB = 2*Fn*r/(Fn**2 - r2)
    mC = -2*Fn1*r/(Fn1**2 - r2)  # a-Fn=Fn1
    # points
    B = np.array([0.0,0.0])
    C = np.array([a,0.0])
    D = np.array([Fn,0.0])
    I = np.array([Fn, r])
    # A
    xA = Fn*x - Fn1*(x - 1)
    yA = mB * xA
    A = np.array([xA, yA])
    # foot F: projection of I onto AB (through origin)
    v = np.array([1.0, mB])
    t = (I @ v) / (v @ v)
    Fpt = np.array([t, t*mB])
    # foot E: projection onto AC (through C)
    w = np.array([1.0, mC])
    IminusC = I - C
    tE = (IminusC @ w) / (w @ w)
    Ept = C + np.array([tE, tE*mC])
    # circumcenter of ABC: O1 = (a/2, yO)
    # yO = (xA**2 - a*xA + yA**2) / (2*yA)   if yA !=0
    if abs(yA) < 1e-12:
        # degenerate
        return None
    yO = (xA**2 - a*xA + yA**2) / (2*yA)
    O1 = np.array([a/2.0, yO])
    R1 = np.linalg.norm(O1 - B)  # distance to B
    # circumcenter of AFE: solve linear system
    # Solve for center (u,v) such that distances to A and F equal, and to A and E equal
    # (u - xA)^2 + (v - yA)^2 = (u - Fx)^2 + (v - Fy)^2
    # (u - xA)^2 + (v - yA)^2 = (u - Ex)^2 + (v - Ey)^2
    # Subtract to get linear equations:
    # 2*(Fx - xA)*u + 2*(Fy - yA)*v = Fx^2 + Fy^2 - xA^2 - yA^2
    # 2*(Ex - xA)*u + 2*(Ey - yA)*v = Ex^2 + Ey^2 - xA^2 - yA^2
    A_mat = np.array([[2*(Fpt[0] - xA), 2*(Fpt[1] - yA)],
                      [2*(Ept[0] - xA), 2*(Ept[1] - yA)]])
    b_vec = np.array([Fpt[0]**2 + Fpt[1]**2 - xA**2 - yA**2,
                      Ept[0]**2 + Ept[1]**2 - xA**2 - yA**2])
    # Solve linear system
    try:
        u, v = np.linalg.solve(A_mat, b_vec)
    except np.linalg.LinAlgError:
        return None
    O2 = np.array([u, v])
    R2 = np.linalg.norm(O2 - A)
    # Intersection of circles: compute K (second intersection)
    # Use standard formula:
    # Let d = distance between centers O1,O2
    d = np.linalg.norm(O2 - O1)
    # If circles intersect, compute a = (R1^2 - R2^2 + d^2) / (2d)
    # height h = sqrt(R1^2 - a^2)
    # point P2 = O1 + a*(O2-O1)/d (point along line between centers)
    # intersection points = P2 +/- h * (perp unit vector)
    if d == 0:
        return None
    a_len = (R1**2 - R2**2 + d**2) / (2*d)
    # due to numeric error, ensure positive
    temp = R1**2 - a_len**2
    if temp < 0:
        # No real intersection (should not happen)
        # return None
        # but maybe due to rounding small negative; set to zero
        temp = 0.0
    h = math.sqrt(temp)
    # unit vector from O1 to O2
    ex = (O2 - O1) / d
    # perp vector (rotate 90 deg)
    ey = np.array([-ex[1], ex[0]])
    P2 = O1 + a_len * ex
    # two intersection points:
    I1 = P2 + h * ey
    I2 = P2 - h * ey
    # Determine which is A (should be approx coordinates of A)
    # Choose K as the one not near A
    if np.linalg.norm(I1 - A) < 1e-6:
        K = I2
    elif np.linalg.norm(I2 - A) < 1e-6:
        K = I1
    else:
        # Both not exactly A: pick the one with larger distance from A?
        # Choose whichever farther? but should be one close.
        # We'll pick the one not equal to A within tolerance
        dist1 = np.linalg.norm(I1 - A)
        dist2 = np.linalg.norm(I2 - A)
        K = I1 if dist1 > dist2 else I2
    # Reflect K across line EF
    # line EF: points Ept, Fpt
    # line equation ax + by + c = 0
    x1,y1 = Ept
    x2,y2 = Fpt
    a_line = y1 - y2
    b_line = x2 - x1
    c_line = x1*y2 - x2*y1
    # reflect point K
    xk, yk = K
    denom = a_line**2 + b_line**2
    xk_ref = xk - 2*a_line*(a_line*xk + b_line*yk + c_line)/denom
    yk_ref = yk - 2*b_line*(a_line*xk + b_line*yk + c_line)/denom
    Kprime = np.array([xk_ref, yk_ref])
    # N: foot from D to EF (projection)
    # projection formula using line: param t = ((D-P1)·(P2-P1))/|P2-P1|^2, then point = P1 + t*(P2-P1)
    P1 = np.array([x1,y1])
    P2 = np.array([x2,y2])
    t_proj = ((D - P1) @ (P2 - P1)) / np.dot(P2 - P1, P2 - P1)
    N = P1 + t_proj * (P2 - P1)

    # BN line direction: (N - B)
    BN = N - B
    # Perpendicular direction at B: rotate BN by 90 deg CCW: (-BN[1], BN[0])
    dir_perp = np.array([-BN[1], BN[0]])
    # Line for center: passes through B and direction dir_perp: param s * dir_perp
    # Perpendicular bisector of BK: midpoint M = (B+K)/2, direction perp to BK: rotate BK 90 deg: (-K[1], K[0])
    BK = K - B
    dir_perp_BK = np.array([-BK[1], BK[0]])
    # Intersection of the two lines: solve B + t*dir_perp = M + u*dir_perp_BK
    # Since B=0, we have t*dir_perp = M + u*dir_perp_BK
    # Solve for t,u (2 eq)
    # Build linear system: [dir_perp, -dir_perp_BK] [t, u]^T = M
    # Actually we can solve using matrix:
    M = (B + K) / 2.0
    A_mat = np.column_stack((dir_perp, -dir_perp_BK))  # 2x2 matrix
    try:
        sol = np.linalg.solve(A_mat, M)
        t_val = sol[0]
        # center O_T = t_val * dir_perp (since B=origin)
        O_T = t_val * dir_perp
    except np.linalg.LinAlgError:
        # degenerate case maybe BN direction zero?
        O_T = None
    # T: second intersection of circle O_T with BC (x-axis)
    # Circle radius = distance O_T to B = norm(O_T)
    # Intersection with x-axis y=0 solves (x - O_T.x)^2 + (0 - O_T.y)^2 = R^2 = O_T.x^2 + O_T.y^2
    # => (x - O_T.x)^2 + O_T.y^2 = O_T.x^2 + O_T.y^2 => (x - O_T.x)^2 = O_T.x^2 => x - O_T.x = ± O_T.x => x = 0 or x = 2*O_T.x
    # T is x=2*O_T.x, y=0
    if O_T is None:
        T = None
    else:
        T = np.array([2*O_T[0], 0.0])
    # compute distances needed for ratio
    # CT, NB, BT, NE
    CT = np.linalg.norm(C - T) if T is not None else None
    BT = np.linalg.norm(B - T) if T is not None else None
    NB = np.linalg.norm(B - N)
    NE = np.linalg.norm(N - Ept)
    # ratio
    ratio = (CT * NB) / (BT * NE) if all(v is not None for v in [CT,BT]) else None
    # cyclic condition: determinant for points B,K,K',N
    # compute matrix for concyclicity: |x y x^2+y^2 1| = 0
    def cyclic_det(P1,P2,P3,P4):
        M = np.array([[P1[0], P1[1], P1[0]**2+P1[1]**2, 1.0],
                      [P2[0], P2[1], P2[0]**2+P2[1]**2, 1.0],
                      [P3[0], P3[1], P3[0]**2+P3[1]**2, 1.0],
                      [P4[0], P4[1], P4[0]**2+P4[1]**2, 1.0]])
        return np.linalg.det(M)
    det_cyc = cyclic_det(B, K, Kprime, N)
    return {
        'A': A, 'B': B, 'C': C, 'D': D, 'I': I,
        'E': Ept, 'F': Fpt,
        'K': K, 'Kprime': Kprime, 'N': N, 'T': T,
        'ratio': ratio,
        'det_cyc': det_cyc,
        'x': x,
        'Fn': Fn, 'Fn1': Fn1, 'a': a, 's': s, 'r': r, 'xA': xA, 'yA': yA,
        'mB': mB, 'mC': mC
    }

        sign = np.sign(vals[0])

                high = xs[i]

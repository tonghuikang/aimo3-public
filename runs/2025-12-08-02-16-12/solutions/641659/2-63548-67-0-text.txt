def compute_triangle_params(d, e, r):
    a = d + e
    # slopes
    mB = 2*d*r/(d**2 - r**2)
    mC = -2*e*r/(e**2 - r**2)
    # intersection A
    xA = -mC * a / (mB - mC)
    yA = mB * xA
    A = (xA, yA)
    B = (0.0, 0.0)
    C = (a, 0.0)
    I = (d, r)
    # foot points F, E using projection formula
    # line AB: direction vector vAB = (xA, yA)
    # projection of I onto line through origin B:
    # t = dot(I, vAB) / |vAB|^2
    vAB = (xA, yA)
    vAB_norm2 = xA**2 + yA**2
    tF = (I[0]*xA + I[1]*yA) / vAB_norm2
    F = (tF*xA, tF*yA)
    # line AC: direction vector vAC = (xA - a, yA)
    vAC = (xA - a, yA)
    vAC_norm2 = (xA - a)**2 + yA**2
    # line passes through C at (a,0). projection of I onto line
    # vector from C to I = (d - a, r)
    w = (I[0] - a, I[1] - 0)
    tE = (w[0]*vAC[0] + w[1]*vAC[1]) / vAC_norm2
    # point on line from C: C + tE * vAC
    E = (a + tE*vAC[0], 0 + tE*vAC[1])
    D = (d, 0.0)
    return {
        'A':A,'B':B,'C':C,'I':I,'F':F,'E':E,'D':D,'mB':mB,'mC':mC
    }

def circumcenter(p1, p2, p3):
    # using formula from coordinates
    (x1,y1),(x2,y2),(x3,y3)=p1,p2,p3
    d = 2*(x1*(y2-y3)+x2*(y3-y1)+x3*(y1-y2))
    if abs(d)<1e-12:
        return None
    ux = ((x1**2+y1**2)*(y2-y3)+(x2**2+y2**2)*(y3-y1)+(x3**2+y3**2)*(y1-y2))/d
    uy = ((x1**2+y1**2)*(x3-x2)+(x2**2+y2**2)*(x1-x3)+(x3**2+y3**2)*(x2-x1))/d
    return (ux, uy)

def circle_center_radius(p1,p2,p3):
    O = circumcenter(p1,p2,p3)
    if O is None:
        return None,None
    R = math.hypot(p1[0]-O[0], p1[1]-O[1])
    return O,R

def intersect_circles(O1,R1,O2,R2, known_pt):
    # returns the other intersection point
    # compute distance between centers
    dx = O2[0]-O1[0]
    dy = O2[1]-O1[1]
    d = math.hypot(dx, dy)
    if d==0:
        return None
    # ex unit vector
    ex = (dx/d, dy/d)
    # distance from O1 to line of centers intersection point P
    # formula: a = (R1^2 - R2^2 + d^2) / (2d)
    a = (R1**2 - R2**2 + d**2) / (2*d)
    # point P
    Px = O1[0] + ex[0]*a
    Py = O1[1] + ex[1]*a
    # height h
    h_sq = R1**2 - a**2
    if h_sq < 0:
        h_sq = 0
    h = math.sqrt(h_sq)
    # perpendicular vector
    perp = (-ex[1], ex[0])
    # two intersection points
    int1 = (Px + perp[0]*h, Py + perp[1]*h)
    int2 = (Px - perp[0]*h, Py - perp[1]*h)
    # choose the one not equal to known_pt
    # compute distances
    if known_pt is not None:
        d1 = math.hypot(int1[0]-known_pt[0], int1[1]-known_pt[1])
        d2 = math.hypot(int2[0]-known_pt[0], int2[1]-known_pt[1])
        # choose the larger distance (since known_pt should be A, intersection at A is one of them)
        if d1 > d2:
            return int1
        else:
            return int2
    else:
        return int1

def reflect_point_across_line(P, L1, L2):
    # reflect point P across line passing through L1 and L2
    # using projection
    # line direction vector
    dx = L2[0] - L1[0]
    dy = L2[1] - L1[1]
    # normalize
    mag2 = dx*dx + dy*dy
    if mag2 == 0:
        return None
    # projection of P onto line
    t = ((P[0]-L1[0])*dx + (P[1]-L1[1])*dy) / mag2
    proj = (L1[0] + t*dx, L1[1] + t*dy)
    # reflect: P' = 2*proj - P
    Rx = 2*proj[0] - P[0]
    Ry = 2*proj[1] - P[1]
    return (Rx, Ry)

def foot_point(P, L1, L2):

    return reflect_point_across_line(P, L1, L2)  # Wait reflect across line gives mirror; foot is projection; need to compute projection

def foot_point_proj(P, L1, L2):
    dx = L2[0] - L1[0]
    dy = L2[1] - L1[1]
    mag2 = dx*dx + dy*dy
    if mag2 == 0:
        return L1
    t = ((P[0]-L1[0])*dx + (P[1]-L1[1])*dy)/mag2
    proj = (L1[0] + t*dx, L1[1] + t*dy)
    return proj

def circle_tangent_through_B_and_K(line_BN, B, K):
    # return circle center (O) and radius (R) for circle passing through B,K and tangent to line BN at B (i.e., BN is tangent at B)
    # Center must lie on line perpendicular to BN at B.
    # Let BN direction vector v = (N - B)
    v = (line_BN[1][0] - line_BN[0][0], line_BN[1][1] - line_BN[0][1])
    # perpendicular direction u = (-v_y, v_x)
    u = (-v[1], v[0])
    # unit vector
    norm = math.hypot(u[0], u[1])
    if norm == 0:
        return None,None
    ux, uy = u[0]/norm, u[1]/norm
    # Two possible centers: B + R*ux, B - R*ux where R = distance BK.
    BK = math.hypot(K[0]-B[0], K[1]-B[1])
    O1 = (B[0] + BK*ux, B[1] + BK*uy)
    O2 = (B[0] - BK*ux, B[1] - BK*uy)
    return (O1,O2), BK

def intersect_circle_line(O,R,line_pt1,line_pt2):
    # intersect circle center O radius R with line passing through two points.
    # return intersection points.
    # param equation: P = P0 + t * d
    x0,y0 = O
    x1,y1 = line_pt1
    x2,y2 = line_pt2
    dx = x2 - x1
    dy = y2 - y1
    # solve (x1 + t*dx - x0)^2 + (y1 + t*dy - y0)^2 = R^2
    # quadratic in t
    a = dx*dx + dy*dy
    b = 2*(dx*(x1-x0) + dy*(y1-y0))
    c = (x1-x0)**2 + (y1-y0)**2 - R*R
    disc = b*b - 4*a*c
    if disc < 0:
        disc = 0.0
    sqrt_disc = math.sqrt(disc)
    t1 = (-b + sqrt_disc) / (2*a)
    t2 = (-b - sqrt_disc) / (2*a)
    P1 = (x1 + t1*dx, y1 + t1*dy)
    P2 = (x1 + t2*dx, y1 + t2*dy)
    return P1, P2

    Kprime = reflect_point_across_line(K, E, F)

    # check cyclic condition: B, K, N, Kprime concyclic?
    O_KNKp,R_KNKp = circle_center_radius(K, N, Kprime)
    # distance from O to B minus radius
    cyclic_val = math.hypot(B[0]-O_KNKp[0], B[1]-O_KNKp[1]) - R_KNKp
    # compute T as other intersection of the circle (through B,K tangent to BN) with BC
    # line BN
    lineBN = (B, N)
    # find centers
    centers,_ = circle_tangent_through_B_and_K(lineBN, B, K)
    # Try both centers to find intersection with BC (y=0) besides B
    T=None
    for O in centers:
        # find intersections with line BC (y=0)
        # param line BC from B (0,0) to C (a,0) -> horizontal line
        # Intersection of circle (O,R=BK) with y=0: solve (x - O_x)^2 + (0-O_y)^2 = R^2
        # So (x - O_x)^2 = R^2 - O_y^2
        R = math.hypot(K[0]-B[0], K[1]-B[1])
        rad_sq = R*R - O[1]*O[1]
        if rad_sq < 0:
            continue
        dx = math.sqrt(rad_sq)
        # two x solutions:
        x1 = O[0] + dx
        x2 = O[0] - dx
        # choose solution not close to B.x (0)
        if abs(x1) > 1e-6:
            T_candidate = (x1, 0.0)
        else:
            T_candidate = (x2, 0.0)
        # ensure within BC range (0, a)
        if 0 < T_candidate[0] < C[0]:
            T = T_candidate
            break
    if T is None:
        return None, None, cyclic_val
    # compute ratio X = (CT*NB)/(BT*NE)
    CT = math.hypot(C[0]-T[0], C[1]-T[1])
    NB = math.hypot(N[0]-B[0], N[1]-B[1])
    BT = math.hypot(B[0]-T[0], B[1]-T[1])
    NE = math.hypot(N[0]-E[0], N[1]-E[1])
    X = (CT*NB)/(BT*NE)
    return X, cyclic_val, (A,B,C,D,E,F,N,K,Kprime,T)

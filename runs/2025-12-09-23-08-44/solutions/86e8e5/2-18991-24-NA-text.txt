def divisors(n):
    divs = set()
    for i in range(1, int(math.isqrt(n))+1):
        if n%i==0:
            divs.add(i)
            divs.add(n//i)
    return sorted(divs)

def is_norwegian(num,n):
    divs = divisors(num)
    # need three distinct positive divisors with sum n
    # brute force combos
    dlen=len(divs)
    for i in range(dlen):
        for j in range(i+1,dlen):
            for k in range(j+1,dlen):
                if divs[i]+divs[j]+divs[k]==n:
                    return True
    return False

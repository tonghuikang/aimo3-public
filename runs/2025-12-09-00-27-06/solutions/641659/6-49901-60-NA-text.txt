def circumcenter(p1, p2, p3):
    # returns circumcenter (ox, oy)
    (x1,y1),(x2,y2),(x3,y3) = p1, p2, p3
    d = 2*(x1*(y2 - y3) + x2*(y3 - y1) + x3*(y1 - y2))
    if abs(d) < 1e-12:
        return None
    ux = ((x1**2 + y1**2)*(y2 - y3) + (x2**2 + y2**2)*(y3 - y1) + (x3**2 + y3**2)*(y1 - y2)) / d
    uy = ((x1**2 + y1**2)*(x3 - x2) + (x2**2 + y2**2)*(x1 - x3) + (x3**2 + y3**2)*(x2 - x1)) / d
    return (ux, uy)

def intersect_two_circles(O1, R1, O2, R2):
    # returns two intersection points (x,y) as list of tuples
    (x0,y0) = O1
    (x1,y1) = O2
    dx = x1 - x0
    dy = y1 - y0
    d = math.hypot(dx, dy)
    if d > R1 + R2 + 1e-12 or d < abs(R1 - R2) - 1e-12:
        return []  # no intersection
    # a = (R1^2 - R2^2 + d^2) / (2d)
    a = (R1**2 - R2**2 + d**2) / (2*d)
    # h = sqrt(R1^2 - a^2)
    tmp = R1**2 - a**2
    if tmp < 0:
        tmp = 0
    h = math.sqrt(tmp)
    # midpoint M
    xm = x0 + a * dx / d
    ym = y0 + a * dy / d
    # offset vector (perp)
    rx = -dy * (h / d)
    ry = dx * (h / d)
    p1 = (xm + rx, ym + ry)
    p2 = (xm - rx, ym - ry)
    return [p1, p2]

def foot_point_to_line(P, A, B):
    # return projection of P onto line AB
    # line defined by points A and B
    (x0,y0) = P
    (x1,y1) = A
    (x2,y2) = B
    # vector AB
    vx = x2 - x1
    vy = y2 - y1
    # parameter t
    if vx==0 and vy==0:
        return A
    t = ((x0 - x1)*vx + (y0 - y1)*vy) / (vx*vx + vy*vy)
    proj = (x1 + t*vx, y1 + t*vy)
    return proj

def reflect_point_across_line(P, A, B):
    # reflect point P across line AB
    H = foot_point_to_line(P, A, B)
    # reflection: K' = 2H - P
    return (2*H[0] - P[0], 2*H[1] - P[1])

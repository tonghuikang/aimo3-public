    # return sorted list of positive divisors of n
    divs = []
    for i in range(1, int(n**0.5)+1):
        if n % i == 0:
            divs.append(i)
            if i != n//i:
                divs.append(n//i)
    return sorted(divs)

def has_three_distinct_div_sum(x, target):
    divs = divisors(x)
    # we need any three distinct divisors with sum = target
    # we can brute triple loops if few divisors
    L = len(divs)
    # try combinations
    # naive O(L^3)

        for j in range(i+1, L):
            for k in range(j+1, L):
                if divs[i] + divs[j] + divs[k] == target:
                    return True
    return False

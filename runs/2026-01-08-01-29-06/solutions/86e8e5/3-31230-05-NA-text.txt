    for K in range(1, limit+1):
        # get divisors
        divs = []
        for d in range(1, int(K**0.5)+1):

                divs.append(d)
                if d != K//d:

        if len(divs) < 3:

        import itertools
        for a,b,c in itertools.combinations(divs,3):
            if a+b+c == n:
                return K
    return None

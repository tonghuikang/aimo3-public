    # Fibonacci numbers up to n+2
    F = [0,1]
    for i in range(2, n+3):
        F.append(F[-1] + F[-2])
    a = sp.Integer(F[n+2])
    d = sp.Integer(F[n])
    s = a + t_val
    b = s - d
    c = s - a + d
    
    # Coordinates
    xA = a - d - s + 2*d*s/a
    yA = 2*sp.sqrt(d*s*(s - a)*(a - d))/a
    A = sp.Point(xA, yA)
    B = sp.Point(0,0)
    C = sp.Point(a,0)
    
    # points E, F
    tE = (a - d)/b
    E = sp.Point(a,0) + tE * (A - sp.Point(a,0))
    u = (s - a)/c
    Fp = A + u * (B - A)
    
    # circles
    circ_ABC = sp.Circle(A,B,C)
    circ_AEF = sp.Circle(A,E,Fp)
    # intersection points
    inters = circ_ABC.intersection(circ_AEF)
    # pick K that is not A
    K_candidates = [pt for pt in inters if not pt.equals(A)]
    if len(K_candidates)==0:
        raise ValueError("No K")

    line_EF = sp.Line(E, Fp)
    # reflection of K across line EF
    Kp = K.reflect(line_EF)

    D = sp.Point(d,0)
    N = line_EF.projection(D)  # projection point
    # compute circle through B,K tangent to BN at B
    # line BN:
    line_BN = sp.Line(B, N)
    # direction perpendicular to BN:
    # line through B perpendicular to BN:

    O = perp_at_B.intersection(line_perp_BK)[0]
    # radius

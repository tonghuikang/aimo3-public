def triangle_params(n, s):
    # returns side lengths a,b,c given n and semiperimeter s (must be > F_{n+2})
    # Fibonacci numbers
    F = [0,1]
    for i in range(2,n+3):
        F.append(F[-1]+F[-2])
    # a = BC
    a = F[n+2]
    # b = CA = s - F_n
    b = s - F[n]
    # c = AB = s - F[n+1]
    c = s - F[n+1]
    return a,b,c, F

def triangle_coords(a,b,c):
    # Place B at (0,0), C at (a,0)
    B = sp.Point(0,0)
    C = sp.Point(a,0)

    # find smallest N such that there exist three distinct divisors a,b,c of N with sum = n
    # brute force up to limit for N
    for N in range(1, limit+1):
        # compute divisors of N
        divs = [d for d in range(1, N+1) if N % d == 0]
        # need three distinct
        if len(divs) < 3:
            continue
        # find any triple distinct sum = n
        # use combinations
        from itertools import combinations
        for a,b,c in combinations(divs, 3):
            if a+b+c == n:
                return N
    return None

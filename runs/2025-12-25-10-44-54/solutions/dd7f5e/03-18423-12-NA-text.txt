def compute_f(alpha, beta):
    # alpha, beta dict exponent->coeff
    # compute f(n) = sum_m alpha(m) * beta(m-n)
    # We can compute for n in some range covering support union.
    # Determine min_n, max_n such that any term could be non-zero:
    alist = list(alpha.keys())
    blist = list(beta.keys())
    if not alist or not blist:
        return {}
    min_n = min(alist) - max(blist)
    max_n = max(alist) - min(blist)
    f = {}
    for n in range(min_n, max_n+1):
        s = 0
        for m,am in alpha.items():
            bm = beta.get(m - n, 0)
            if bm:
                s += am * bm
        if s != 0:
            f[n] = s
    return f

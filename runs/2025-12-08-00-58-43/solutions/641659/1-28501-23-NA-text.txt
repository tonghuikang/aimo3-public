def compute_ratio(n, s_val):
    # Compute ratio for given n and s (float)
    # returns ratio value
    # n: integer
    # s_val: float > F_{n+2}
    # We'll compute geometry as described.
    # First compute Fibonacci numbers
    F = [0, 1]
    for i in range(2, n+5):
        F.append(F[-1] + F[-2])
    # Now use appropriate values
    F_n = F[n]
    F_np1 = F[n+1]
    F_np2 = F[n+2]
    a = F_np2
    s = s_val
    b = s - F_n # CA
    c = s - F_np1 # AB
    # Coordinates:
    B = sp.Point(0,0)
    C = sp.Point(a,0)
    # x_A = (a^2 + c^2 - b^2) / (2a)
    xA = (a**2 + c**2 - b**2) / (2*a)
    # yA = sqrt(c^2 - xA^2)
    # ensure positivity
    yA_sq = c**2 - xA**2
    if yA_sq < 0:
        return None
    yA = sp.sqrt(yA_sq)
    A = sp.Point(xA, yA)
    # incircle center:
    d = F_n # BD = s - b = F_n
    # incircle radius r = sqrt((s-a)*F_n*F_np1 / s)
    r = sp.sqrt((s - a) * F_n * F_np1 / s)
    I = sp.Point(d, r)
    # Points F and E
    # BF = F_n
    # AF = s - a = s - F_np2
    # Ratio BF/AB = F_n / c
    F_pt = sp.Point(F_n / c * xA, F_n / c * yA)  # B + (F_n/c)*(A-B)
    # CE = F_np1, CA length = b
    # E = C + (F_np1 / b)*(A - C)
    E_pt = sp.Point(a + (F_np1 / b)*(xA - a), (F_np1 / b)*(yA))
    # Circle through A,F,E: get circumcenter O1
    circle_AFE = sp.Circle(A, F_pt, E_pt)
    # Circumcircle of triangle ABC
    circle_ABC = sp.Circle(A, B, C)
    # Intersection of circles: they intersect at A and K
    inters = circle_AFE.intersection(circle_ABC)
    # sympy returns Points maybe duplicates; need K distinct from A
    K_pt = None
    for p in inters:
        if not p.equals(A):
            K_pt = p
    if K_pt is None:
        # maybe intersection returns only A? Shouldn't
        return None
    # Compute K' reflection across line EF
    line_EF = sp.Line(E_pt, F_pt)
    K_prime = K_pt.reflect(line_EF)
    # Compute foot N from D = (F_n,0) onto line EF
    D_pt = sp.Point(F_n, 0)
    N_pt = line_EF.projection(D_pt)
    # Now compute circumcenter of triangle B,K,K' (circle through them)
    # check collinearity:
    if sp.Line(B, K_pt).is_parallel(sp.Line(K_pt, K_prime)):
        # collinear? unlikely
        return None
    circle_BKKp = sp.Circle(B, K_pt, K_prime)
    # Check if N lies on this circle:
    # If not exactly due to numeric approximations, we can compute error.
    # But for our given s we want this to hold exactly. 
    # So we compute difference:
    error = float(N_pt.distance(circle_BKKp.center) - circle_BKKp.radius)
    # compute T: circle through B and K tangent to BN at B
    # line BN:
    line_BN = sp.Line(B, N_pt)
    # line perpendicular to BN through B:
    # direction vector of BN is (N.x, N.y). Perpendicular slope = -dx/dy maybe.
    # Compute center O_T as intersection of perpendicular bisector of BK with this line.
    # perpendicular bisector of BK:
    mid_BK = sp.Point( (B.x+K_pt.x)/2, (B.y+K_pt.y)/2)
    # slope of BK: (K.y - B.y)/(K.x - B.x) (but line direction)

    # sympy can give perpendicular bisector as line:

def circumcenter(p1, p2, p3):
    # Returns circumcenter of three points (x,y) as tuple (x,y)
    # Using perpendicular bisectors solution.
    (x1,y1) = p1
    (x2,y2) = p2
    (x3,y3) = p3
    d = 2*(x1*(y2-y3) + x2*(y3-y1) + x3*(y1-y2))
    if abs(d) < 1e-12:
        return None
    ux = ((x1**2 + y1**2)*(y2-y3) + (x2**2 + y2**2)*(y3-y1) + (x3**2 + y3**2)*(y1-y2)) / d
    uy = ((x1**2 + y1**2)*(x3-x2) + (x2**2 + y2**2)*(x1-x3) + (x3**2 + y3**2)*(x2-x1)) / d
    return (ux, uy)

def circle_intersections(O1, R1, O2, R2):
    # Returns intersection points (list of up to 2 points)
    x1,y1 = O1; x2,y2 = O2
    dx = x2-x1; dy = y2-y1
    d = math.hypot(dx,dy)
    if d > R1 + R2 + 1e-12 or d < abs(R1 - R2) - 1e-12:
        return [] # no intersect
    # distance from O1 to line of intersection points along line connecting centers
    a = (R1**2 - R2**2 + d**2) / (2*d)
    h_sq = R1**2 - a**2
    if h_sq < -1e-12:
        return []
    h = math.sqrt(max(0.0, h_sq))
    # point P2 = O1 + a*(O2-O1)/d
    xm = x1 + a*dx/d
    ym = y1 + a*dy/d
    # intersection points
    rx = -dy * (h/d)
    ry = dx * (h/d)
    p1 = (xm + rx, ym + ry)
    p2 = (xm - rx, ym - ry)
    if h < 1e-12:
        return [p1]
    else:
        return [p1,p2]

def reflect_point_across_line(P, A, B):
    # reflect point P across line AB
    # Compute projection onto line, then reflect
    # line vector
    (x0,y0)=P
    (x1,y1)=A
    (x2,y2)=B
    # line direction
    dx = x2 - x1
    dy = y2 - y1
    # if line is a point, reflection undefined.
    # projection formula: param t = ((P-A).(B-A))/|B-A|^2
    denom = dx*dx + dy*dy
    if denom == 0:
        return (2*x1 - x0, 2*y1 - y0) # reflect across point?
    t = ((x0 - x1)*dx + (y0 - y1)*dy) / denom
    # projection point:
    xp = x1 + t*dx
    yp = y1 + t*dy
    # reflection: P' = (2*proj - P)
    xr = 2*xp - x0
    yr = 2*yp - y0
    return (xr, yr)

def projection_point_to_line(P, A, B):
    (x0,y0)=P
    (x1,y1)=A
    (x2,y2)=B
    dx = x2 - x1
    dy = y2 - y1
    denom = dx*dx + dy*dy
    if denom == 0:
        return (x1,y1)
    t = ((x0 - x1)*dx + (y0 - y1)*dy) / denom
    xp = x1 + t*dx
    yp = y1 + t*dy
    return (xp,yp)

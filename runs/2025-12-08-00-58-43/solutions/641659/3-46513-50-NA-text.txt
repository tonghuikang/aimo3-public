def circumcenter(P, Q, R):
    # Returns center (x,y) and radius of circumcircle of triangle PQR
    # Using formula based on perpendicular bisectors
    (x1,y1), (x2,y2), (x3,y3) = P, Q, R
    d = 2*(x1*(y2-y3) + x2*(y3-y1) + x3*(y1-y2))
    if abs(d) < 1e-12:
        raise ValueError("Collinear")
    ux = ((x1**2 + y1**2)*(y2-y3) + (x2**2 + y2**2)*(y3-y1) + (x3**2 + y3**2)*(y1-y2)) / d
    uy = ((x1**2 + y1**2)*(x3-x2) + (x2**2 + y2**2)*(x1-x3) + (x3**2 + y3**2)*(x2-x1)) / d
    R = math.hypot(ux - x1, uy - y1)
    return (ux, uy), R

def intersect_circles(O1, r1, O2, r2):
    # Return two intersection points as tuples. May be 0 or 1 solutions.
    x1, y1 = O1
    x2, y2 = O2
    dx = x2 - x1
    dy = y2 - y1
    d = math.hypot(dx, dy)
    if d > r1 + r2 + 1e-12 or d < abs(r1 - r2) - 1e-12 or d == 0:
        return [] # no intersection or infinite
    # a = (r1^2 - r2^2 + d^2) / (2d)
    a = (r1**2 - r2**2 + d**2) / (2*d)
    # h^2 = r1^2 - a^2
    h_sq = r1**2 - a**2
    if h_sq < -1e-12:
        return []
    h = math.sqrt(max(0, h_sq))
    # point P2 = O1 + a*(dx,dy)/d
    xm = x1 + a*dx/d
    ym = y1 + a*dy/d
    # intersection points:
    # offset = (-dy, dx)*(h/d)
    rx = -dy * (h/d)
    ry = dx * (h/d)
    p1 = (xm + rx, ym + ry)
    p2 = (xm - rx, ym - ry)
    return [p1, p2]

def foot_of_perpendicular(P, A, B):
    # foot from point P onto line AB
    # A, B as tuples
    # compute projection of vector AP onto AB
    (px, py) = P
    (ax, ay) = A
    (bx, by) = B
    # line AB direction
    dx = bx - ax
    dy = by - ay
    # if line degenerate
    if abs(dx) < 1e-12 and abs(dy) < 1e-12:
        return (ax, ay)
    # parameter t such that projection = A + t*(dx,dy)
    t = ((px - ax)*dx + (py - ay)*dy) / (dx*dx + dy*dy)
    # foot:
    fx = ax + t*dx
    fy = ay + t*dy
    return (fx, fy)

def reflect_point_across_line(P, A, B):
    # reflect P across line AB
    # compute foot H of P onto line AB, then reflect: P' = 2*H - P
    H = foot_of_perpendicular(P, A, B)
    return (2*H[0] - P[0], 2*H[1] - P[1])

def cyclic_det(B,N,K,Kp):
    # Compute determinant for cyclic condition
    # Build matrix M = [[x,y,x^2+y^2,1], ...] and compute det
    import numpy as np
    M = np.array([[B[0], B[1], B[0]**2 + B[1]**2, 1.0],
                  [N[0], N[1], N[0]**2 + N[1]**2, 1.0],
                  [K[0], K[1], K[0]**2 + K[1]**2, 1.0],
                  [Kp[0], Kp[1], Kp[0]**2 + Kp[1]**2, 1.0]])
    return np.linalg.det(M)

    # Use standard indexing: F0=0, F1=1
    # Compute F_n up to needed

    while len(fib) <= n+2:
        fib.append(fib[-1] + fib[-2])
    F = fib # list
    a = F[n+2] # BC length
    d = F[n]   # BD

    # Unknown r > 0
    # Compute slopes
    r = sp.symbols('r', positive=True, real=True)

    m_AB = 2*d*r/(d**2 - r**2)
    m_AC = 2*(d - a)*r/(((d - a)**2) - r**2)
    # Intersection A
    xA = - m_AC * a / (m_AB - m_AC)
    yA = m_AB * xA
    # Points B(0,0), C(a,0)
    # Compute foot from I to line AB: point F

def count_subsets_of_group(deg_list):
    count = 0
    subsets = []
    n = len(deg_list)
    for r in range(1, n+1):
        for combo in itertools.combinations(range(n), r):
            s = sum(deg_list[i] for i in combo)
            if s <= 8:
                count += 1
                subsets.append(combo)
    return count, subsets

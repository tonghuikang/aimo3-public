    F_n = fibonacci(n)
    F_np1 = fibonacci(n+1)
    L = F_n + F_np1
    B = np.array([0.0,0.0])
    C = np.array([L,0.0])
    D = np.array([F_n,0.0])
    I = np.array([F_n, r])
    # slopes
    mB = 2*F_n*r/(F_n**2 - r**2)
    mC = -2*F_np1*r/(F_np1**2 - r**2)
    # Intersection for A: solve mB*x = mC*(x - L)
    # => x*(mB - mC) = -mC*L
    xA = - mC * L / (mB - mC)
    yA = mB * xA
    A = np.array([xA, yA])
    # direction vectors for lines
    vB = np.array([1.0, mB])
    vC = np.array([1.0, mC])
    # foot from I to BF
    # foot = B + vB * ((I-B)·vB)/|vB|^2
    F_pt = B + vB * np.dot(I - B, vB) / np.dot(vB, vB)
    # foot from I to CE (line through C)
    E_pt = C + vC * np.dot(I - C, vC) / np.dot(vC, vC)
    # circumcenter of triangle ABC
    def circumcenter(P1,P2,P3):
        # using formula
        x1,y1 = P1
        x2,y2 = P2
        x3,y3 = P3
        D = 2*(x1*(y2 - y3) + x2*(y3 - y1) + x3*(y1 - y2))
        if abs(D) < 1e-12:
            raise ValueError("collinear")
        Ux = ((x1**2 + y1**2)*(y2 - y3) + (x2**2 + y2**2)*(y3 - y1) + (x3**2 + y3**2)*(y1 - y2)) / D
        Uy = ((x1**2 + y1**2)*(x3 - x2) + (x2**2 + y2**2)*(x1 - x3) + (x3**2 + y3**2)*(x2 - x1)) / D
        R = np.hypot(Ux - x1, Uy - y1)
        return np.array([Ux, Uy]), R
    O_ABC, R_ABC = circumcenter(A,B,C)
    O_AFE, R_AFE = circumcenter(A,F_pt,E_pt)
    # intersect circles
    # Compute distance between centers
    d = np.linalg.norm(O_ABC - O_AFE)
    # check for intersection
    if d > R_ABC + R_AFE or d < abs(R_ABC - R_AFE):
        # no intersection
        raise ValueError("no intersect")
    a = (R_ABC**2 - R_AFE**2 + d**2) / (2*d)
    h2 = R_ABC**2 - a**2
    if h2 < 0:
        h2 = 0
    h = np.sqrt(h2)
    # unit vector from O_ABC to O_AFE
    ex = (O_AFE - O_ABC) / d
    P0 = O_ABC + a*ex
    ey = np.array([-ex[1], ex[0]])
    inter1 = P0 + h*ey
    inter2 = P0 - h*ey
    # choose K as the intersection not equal to A
    if np.linalg.norm(inter1 - A) > 1e-8:
        K_pt = inter1
    else:
        K_pt = inter2
    # reflection of K across line EF
    # line through E_pt, F_pt
    def reflect_point(P, L1, L2):
        # foot of P onto line L1L2
        v = L2 - L1
        foot = L1 + v * np.dot(P - L1, v) / np.dot(v, v)
        return 2*foot - P
    Kp_pt = reflect_point(K_pt, E_pt, F_pt)
    # foot N from D onto line EF
    N_pt = reflect_point(D, E_pt, F_pt)  # actually foot from D onto line EF: same method but without reflection: foot = L1 + v * ((D-L1)·v)/|v|^2
    vEF = F_pt - E_pt
    N_pt = E_pt + vEF * np.dot(D - E_pt, vEF) / np.dot(vEF, vEF)
    # compute T: circle tangent to line BN at B and passing through K
    # line BN direction
    BN_dir = N_pt - B
    # perpendicular direction to BN (center lies on this line)
    perp_dir = np.array([-BN_dir[1], BN_dir[0]])
    # perpendicular bisector of BK
    M_mid = (B + K_pt)/2
    BK_dir = K_pt - B
    perp_BK_dir = np.array([-BK_dir[1], BK_dir[0]])
    # Solve for intersection O_center: B + s*perp_dir = M_mid + t*perp_BK_dir
    # 2 equations in s,t
    A_mat = np.column_stack((perp_dir, -perp_BK_dir))
    rhs = M_mid - B
    try:
        sol = np.linalg.solve(A_mat, rhs)
    except np.linalg.LinAlgError:
        raise ValueError("singular")
    s = sol[0]
    O_center = B + s*perp_dir
    R_T = np.linalg.norm(O_center - B)
    # Intersection of circle with line y=0 (BC) other than B
    # Solve (x - O_center.x)^2 + (0 - O_center.y)^2 = R_T^2
    # => (x - xc)^2 = R_T^2 - yc^2
    disc = R_T**2 - O_center[1]**2
    if disc < 0:
        disc = 0
    sqrt_disc = np.sqrt(disc)
    x1 = O_center[0] + sqrt_disc
    x2 = O_center[0] - sqrt_disc
    # choose the one not zero (or not near B.x=0)
    if abs(x1) > 1e-8:
        T_pt = np.array([x1, 0.0])
    else:
        T_pt = np.array([x2, 0.0])
    # compute ratio CT * NB / (BT * NE)
    CT = np.linalg.norm(C - T_pt)
    NB = np.linalg.norm(N_pt - B)
    BT = np.linalg.norm(B - T_pt)
    NE = np.linalg.norm(N_pt - E_pt)
    ratio = (CT * NB) / (BT * NE)
    # compute cyclic residual determinant
    def det_cyclic(PB,PK,PKp,PN):
        M = np.array([[PB[0], PB[1], PB[0]**2 + PB[1]**2, 1.0],
                      [PK[0], PK[1], PK[0]**2 + PK[1]**2, 1.0],
                      [PKp[0], PKp[1], PKp[0]**2 + PKp[1]**2, 1.0],
                      [PN[0], PN[1], PN[0]**2 + PN[1]**2, 1.0]])
        return np.linalg.det(M)
    residual = det_cyclic(B, K_pt, Kp_pt, N_pt)
    return {
        'B': B, 'C': C, 'D': D, 'I': I,
        'A': A, 'F': F_pt, 'E': E_pt,
        'K': K_pt, 'Kp': Kp_pt, 'N': N_pt,
        'T': T_pt,
        'ratio': ratio,
        'residual': residual
    }

    M = (B + K) / 2
    BK = K - B

    # Equation: distance from O to line BN = |O - B|

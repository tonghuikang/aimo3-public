    # brute compute minimal N up to limit
    # return minimal N such that exists three distinct divisors summing to n
    # naive approach
    for N in range(1, limit+1):
        # list divisors
        divs = [d for d in range(1, N+1) if N % d == 0]
        if len(divs) < 3:

        # check combos of 3 distinct
        from itertools import combinations
        for a,b,c in combinations(divs,3):

    return None

def f_bruteforce2(n, maxN=5000):
    # more efficient: compute divisors and combos for each N up to maxN
    # returns minimal N
    # Might be slow for large n, but test small.
    for N in range(1, maxN+1):
        # get divisors
        divs = []
        for d in range(1,int(N**0.5)+1):
            if N % d == 0:
                divs.append(d)
                if d != N//d:
                    divs.append(N//d)
        if len(divs) < 3:
            continue
        divs.sort()
        for i in range(len(divs)):
            a = divs[i]
            for j in range(i+1,len(divs)):
                b = divs[j]
                need = n - a - b
                if need <= b:
                    continue
                # need > b to be distinct and larger
                if need in divs and need != a and need != b:
                    return N
    return None

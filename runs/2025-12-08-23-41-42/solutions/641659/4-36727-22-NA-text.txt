def triangle_from_angles(d, e, f, r=1.0):
    # incircle radius r, center at origin
    D = np.array([r*math.cos(d), r*math.sin(d)])
    E = np.array([r*math.cos(e), r*math.sin(e)])
    F = np.array([r*math.cos(f), r*math.sin(f)])
    # line coefficients for tangent lines: x cosθ + y sinθ = r
    # Represent line as (A,B,C) for equation A x + B y = C
    # For BC line (tangent at D): (cos d, sin d, r)
    # AB line (tangent at F): (cos f, sin f, r)
    # AC line (tangent at E): (cos e, sin e, r)
    # Solve for intersections:
    # B = intersection of BC and AB
    # Solve matrix [[cos d, sin d], [cos f, sin f]] * [x,y] = [r,r]
    M_B = np.array([[math.cos(d), math.sin(d)],
                    [math.cos(f), math.sin(f)]])
    Cvec = np.array([r, r])
    B = np.linalg.solve(M_B, Cvec)
    # C = intersection of BC and AC
    M_C = np.array([[math.cos(d), math.sin(d)],
                    [math.cos(e), math.sin(e)]])
    C = np.linalg.solve(M_C, Cvec)
    # A = intersection of AB and AC
    M_A = np.array([[math.cos(f), math.sin(f)],
                    [math.cos(e), math.sin(e)]])
    A = np.linalg.solve(M_A, Cvec)
    return A, B, C, D, E, F

def circumcircle_center_radius(P, Q, R):
    # compute circumcenter of triangle PQR
    # using perpendicular bisector intersection
    # return center and radius
    # Using formula via linear algebra
    # Solve for circumcenter O = (ox, oy) such that |O-P|^2 = |O-Q|^2 = |O-R|^2
    A = np.array([[2*(Q[0]-P[0]), 2*(Q[1]-P[1])],
                  [2*(R[0]-P[0]), 2*(R[1]-P[1])]])
    b = np.array([Q[0]**2 + Q[1]**2 - P[0]**2 - P[1]**2,
                  R[0]**2 + R[1]**2 - P[0]**2 - P[1]**2])
    O = np.linalg.solve(A, b)
    R0 = np.linalg.norm(O-P)
    return O, R0

def circle_intersections(O1, r1, O2, r2):
    # return intersection points of circles (if any), assume they intersect
    # returns list of points
    d = np.linalg.norm(O2-O1)
    if d > r1 + r2 or d < abs(r1-r2):
        return []  # no intersection
    # distance from O1 to line between intersection points along the line connecting centers
    a = (r1**2 - r2**2 + d**2) / (2*d)
    h = math.sqrt(max(r1**2 - a**2,0))
    # point P2 = O1 + a*(O2-O1)/d
    P2 = O1 + a*(O2-O1)/d
    # intersection points
    # offset vector perpendicular to line O1-O2
    offset = h * np.array([-(O2[1]-O1[1])/d, (O2[0]-O1[0])/d])
    I1 = P2 + offset
    I2 = P2 - offset
    return [I1, I2]

def foot_of_perpendicular(P, A, B):
    # foot from point P onto line AB
    AB = B - A
    t = np.dot(P - A, AB) / np.dot(AB, AB)
    foot = A + t * AB
    return foot

def cyclic_residual_mp(e, f):
    # compute cyclic residual (N distance to circumcenter of B,K,Kp - radius)
    d = -mp.pi/2
    # points
    D = mp.matrix([0, -1])
    E = mp.matrix([mp.cos(e), mp.sin(e)])
    F = mp.matrix([mp.cos(f), mp.sin(f)])
    # B, C
    B = mp.matrix([(1+mp.sin(f))/mp.cos(f), -1])
    C = mp.matrix([(1+mp.sin(e))/mp.cos(e), -1])
    # A
    det = mp.sin(e - f)
    # if det nearly zero, may cause errors
    A = mp.matrix([(mp.sin(e) - mp.sin(f))/det,
                   (mp.cos(f) - mp.cos(e))/det])
    # circumcenter O_ABC, O_AEF
    # We'll compute using formula for circumcenter from coordinates: solve system of perpendicular bisectors
    def circumcenter(P, Q, R):
        # Using linear algebra: solve (O - P)·(Q-P) = 0? Actually we can solve using matrix.
        # We'll compute using sympy style: solve for O such that |O-P|^2 = |O-Q|^2 and |O-P|^2 = |O-R|^2.
        # Solve linear equations:
        # 2*(Q-P).dot(O) = |Q|^2 - |P|^2
        # 2*(R-P).dot(O) = |R|^2 - |P|^2
        A_mat = mp.matrix([[2*(Q[0]-P[0]), 2*(Q[1]-P[1])],
                           [2*(R[0]-P[0]), 2*(R[1]-P[1])]])
        B_vec = mp.matrix([Q[0]**2 + Q[1]**2 - P[0]**2 - P[1]**2,
                           R[0]**2 + R[1]**2 - P[0]**2 - P[1]**2])
        O = mp.lu_solve(A_mat, B_vec)
        r = mp.sqrt((O[0]-P[0])**2 + (O[1]-P[1])**2)
        return O, r
    O_ABC, R_ABC = circumcenter(A, B, C)
    O_AEF, R_AEF = circumcenter(A, E, F)
    # intersect circles O_ABC,R_ABC and O_AEF,R_AEF
    # compute intersection of two circles
    # using standard formula
    # d = distance between centers
    d_cent = mp.sqrt((O_AEF[0]-O_ABC[0])**2 + (O_AEF[1]-O_ABC[1])**2)
    if d_cent == 0:
        return mp.mpf('nan')
    # a = (r0^2 - r1^2 + d^2)/(2d)
    a = (R_ABC**2 - R_AEF**2 + d_cent**2) / (2*d_cent)
    # h = sqrt(r0^2 - a^2)
    h_sq = R_ABC**2 - a**2
    if h_sq < 0:
        # no intersection
        h = mp.mpc(0)
    else:
        h = mp.sqrt(h_sq)
    # point P2 = O_ABC + a*(O_AEF - O_ABC)/d
    P2 = mp.matrix([O_ABC[0] + a*(O_AEF[0]-O_ABC[0])/d_cent,
                    O_ABC[1] + a*(O_AEF[1]-O_ABC[1])/d_cent])
    # offset vector perpendicular to O_AEF-O_ABC
    # (dx, dy) = (-(y2-y1), x2-x1)
    dx = O_AEF[0] - O_ABC[0]
    dy = O_AEF[1] - O_ABC[1]
    # Perp vector normalized:
    # offset = h * ( -dy/d_cent, dx/d_cent )
    offset = mp.matrix([-dy * h / d_cent, dx * h / d_cent])
    I1 = P2 + offset
    I2 = P2 - offset
    # Choose K as point not equal to A (within tolerance)
    # compute distances
    d1 = mp.sqrt((I1[0]-A[0])**2 + (I1[1]-A[1])**2)
    d2 = mp.sqrt((I2[0]-A[0])**2 + (I2[1]-A[1])**2)
    # whichever is larger distance from A likely K (since intersection other than A)
    if d1 > d2:
        K = I1
    else:
        K = I2
    # K' reflection across line EF
    # reflect point across line defined by points E and F
    # formula: reflect K across line EF: find projection onto line and reflect
    # param t = ((K-E).dot(F-E))/|F-E|^2
    FE = F - E
    t_proj = ((K - E).dot(FE)) / (FE.dot(FE))
    proj = E + t_proj * FE
    Kp = proj*2 - K
    # N foot from D onto EF
    # foot of perpendicular from D to line EF: compute projection similarly
    t_N = ((D - E).dot(FE)) / (FE.dot(FE))
    N = E + t_N * FE
    # circumcenter of B, K, Kp
    O_BKKp, R_BKKp = circumcenter(B, K, Kp)

    residual = mp.sqrt((N[0]-O_BKKp[0])**2 + (N[1]-O_BKKp[1])**2) - R_BKKp
    return residual

    # => (x - O_tan_x)^2 = r_tan^2 - ( -1 - O_tan_y )^2

        return None

    x2 = O_tan[0] - sqrt_val
    x_B = B[0]
    # Choose the intersection not equal to B's x
    # tolerance for equality
    if mp.almosteq(x1, x_B):
        x_T = x2
    else:
        x_T = x1
    T = mp.matrix([x_T, -1])
    # distances
    CT = mp.sqrt((C[0]-T[0])**2 + (C[1]-T[1])**2)
    BT = mp.sqrt((B[0]-T[0])**2 + (B[1]-T[1])**2)
    NB = mp.sqrt((N[0]-B[0])**2 + (N[1]-B[1])**2)
    NE = mp.sqrt((N[0]-E[0])**2 + (N[1]-E[1])**2)
    Rratio = (CT * NB) / (BT * NE)
    return Rratio

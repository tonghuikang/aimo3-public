def geometry_ratio_mp(u, v, t):
    mp.mp.dps = 50
    a = mp.mpf(1)
    b = mp.mpf(1) + t - u
    c = mp.mpf(1) + t - v
    r = mp.sqrt(t * u * v / (1 + t))
    # points
    B = mp.matrix([0,0])
    C = mp.matrix([a,0])
    D = mp.matrix([u,0])
    I = mp.matrix([u, r])
    xA = (c*c - b*b + a*a) / (2*a)
    yA2 = c*c - xA*xA
    if yA2 <= 0:
        return None
    yA = mp.sqrt(yA2)
    A = mp.matrix([xA, yA])
    # projection onto AB
    vAB = A - B
    IB = I - B
    t_AB = (IB[0]*vAB[0] + IB[1]*vAB[1]) / (vAB[0]*vAB[0] + vAB[1]*vAB[1])
    F = B + t_AB * vAB
    # projection onto AC
    vAC = A - C
    IC = I - C
    t_AC = (IC[0]*vAC[0] + IC[1]*vAC[1]) / (vAC[0]*vAC[0] + vAC[1]*vAC[1])
    E = C + t_AC * vAC
    # circumcenter O of ABC (B(0,0),C(1,0))
    if abs(yA) < mp.mpf('1e-30'):
        return None
    y0 = ((xA - mp.mpf('0.5'))**2 + yA**2 - mp.mpf('0.25')) / (2*yA)
    O = mp.matrix([mp.mpf('0.5'), y0])
    R = mp.norm(O - B)
    # circumcenter O1 of A,F,E
    def circumcenter(P,Q,Rr):
        ax, ay = P[0], P[1]
        bx, by = Q[0], Q[1]
        cx, cy = Rr[0], Rr[1]
        d = 2*(ax*(by - cy) + bx*(cy - ay) + cx*(ay - by))
        ux = ((ax*ax + ay*ay)*(by - cy) + (bx*bx + by*by)*(cy - ay) + (cx*cx + cy*cy)*(ay - by)) / d
        uy = ((ax*ax + ay*ay)*(cx - bx) + (bx*bx + by*by)*(ax - cx) + (cx*cx + cy*cy)*(bx - ax)) / d
        return mp.matrix([ux, uy])
    O1 = circumcenter(A, F, E)
    r1 = mp.norm(O1 - A)
    # intersect circles O,R and O1,r1
    d0 = mp.norm(O1 - O)
    a_len = (R**2 - r1**2 + d0**2) / (2*d0)
    h_sq = R**2 - a_len**2
    if h_sq < 0:

    else:
        h = mp.sqrt(h_sq)
    x2 = O[0] + a_len * (O1[0] - O[0]) / d0

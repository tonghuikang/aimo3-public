def circumcenter(p1,p2,p3):
    # return circumcenter of triangle p1,p2,p3 using linear algebra
    # solve perpendicular bisector intersection
    # p1,p2,p3 are 2d arrays
    # formula using circumcenter = intersection of perpendicular bisectors
    # Use linear equations: (c - p1)·(p2-p1) = (|c|^2 - |p1|^2 - (|p2|^2 - |p1|^2))/2? Actually easier: use matrix solve.
    A = np.vstack([2*(p2 - p1), 2*(p3 - p1)])
    b = np.array([np.dot(p2,p2) - np.dot(p1,p1), np.dot(p3,p3) - np.dot(p1,p1)])
    # Solve A * c = b
    try:
        c = np.linalg.solve(A, b)
    except np.linalg.LinAlgError:
        # degenerate case
        c = np.array([np.nan, np.nan])
    return c

def circle_intersections(c1,r1,c2,r2):
    # return intersection points of two circles
    # based on geometry
    # c1,c2 arrays
    d = np.linalg.norm(c2 - c1)
    if d > r1 + r2 or d < abs(r1 - r2):
        return []  # no intersection
    # find a = (r1^2 - r2^2 + d^2) / (2d)
    a = (r1**2 - r2**2 + d**2) / (2*d)
    # height h = sqrt(r1^2 - a^2)
    h_sq = r1**2 - a**2
    if h_sq < 0: h_sq = 0
    h = np.sqrt(h_sq)
    # point P2 = c1 + a*(c2 - c1)/d
    P2 = c1 + a * (c2 - c1) / d
    # intersection points:
    # offset vector = h * perp(c2-c1)/d
    offset = h * np.array([-(c2[1] - c1[1]), c2[0] - c1[0]]) / d
    inter1 = P2 + offset
    inter2 = P2 - offset
    return [inter1, inter2]

def reflect_point_across_line(P, A, B):
    # reflect point P across line through A,B
    # compute projection onto line, then reflect
    # line direction unit vector
    AB = B - A
    AB_norm2 = np.dot(AB, AB)
    if AB_norm2 == 0:
        return None
    # projection of (P-A) onto AB
    proj = A + np.dot(P - A, AB) / AB_norm2 * AB
    # reflection: P' = 2*proj - P
    return 2*proj - P

def foot_point_to_line(P, A, B):
    # foot from P to line AB
    AB = B - A
    AB_norm2 = np.dot(AB, AB)
    if AB_norm2 == 0:
        return None
    t = np.dot(P - A, AB) / AB_norm2
    proj = A + t * AB
    return proj

def is_cyclic_quad(P1,P2,P3,P4, eps=1e-8):
    # check if 4 points cyclic by computing circle through first 3 and checking distance of fourth
    O = circumcenter(P1,P2,P3)
    R = np.linalg.norm(P1 - O)
    return abs(np.linalg.norm(P4 - O) - R) < eps

    B, C, D, A, F, E, I = pts['B'], pts['C'], pts['D'], pts['A'], pts['F'], pts['E'], pts['I']

    cyclic = is_cyclic_quad(B,K,N,Kprime, eps=1e-6)

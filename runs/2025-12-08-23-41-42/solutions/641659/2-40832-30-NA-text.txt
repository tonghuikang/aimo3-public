def circumcenter(P, Q, R):
    # returns center (ox, oy) of circumcircle of triangle PQR
    (x1,y1) = P
    (x2,y2) = Q
    (x3,y3) = R
    # Using formula from coordinates
    d = 2 * (x1*(y2 - y3) + x2*(y3 - y1) + x3*(y1 - y2))
    if d == 0:
        raise ValueError("collinear")
    ux = ((x1**2 + y1**2)*(y2 - y3) + (x2**2 + y2**2)*(y3 - y1) + (x3**2 + y3**2)*(y1 - y2)) / d
    uy = ((x1**2 + y1**2)*(x3 - x2) + (x2**2 + y2**2)*(x1 - x3) + (x3**2 + y3**2)*(x2 - x1)) / d
    return (ux, uy)

def circle_intersections(O1, R1, O2, R2):
    # returns two intersection points of circles centered at O1,O2 with radii R1,R2
    (x0,y0) = O1
    (x1,y1) = O2
    dx = x1 - x0
    dy = y1 - y0
    d = mp.sqrt(dx*dx + dy*dy)
    if d == 0:
        return []
    # check for intersections
    # a = (R1^2 - R2^2 + d^2) / (2d)
    a = (R1**2 - R2**2 + d**2) / (2*d)
    # h^2 = R1^2 - a^2
    h_sq = R1**2 - a**2
    if h_sq < -1e-30:
        return []  # no intersection
    h = mp.sqrt(max(h_sq, 0))
    # point P2 = O1 + a*(O2-O1)/d
    x2 = x0 + a * dx / d
    y2 = y0 + a * dy / d
    # intersection points:
    rx = -dy * (h / d)
    ry = dx * (h / d)
    p_int1 = (x2 + rx, y2 + ry)
    p_int2 = (x2 - rx, y2 - ry)
    return [p_int1, p_int2]

def foot_of_perpendicular(P, Q, R):
    # foot of perpendicular from P to line QR
    # compute projection of vector PQ onto line direction
    # formula: projection of vector (P - Q) onto direction (R - Q)
    (px,py) = P
    (qx,qy) = Q
    (rx,ry) = R
    # line direction
    dx = rx - qx
    dy = ry - qy
    # if line direction zero?
    if dx == 0 and dy == 0:
        raise ValueError("zero direction")
    # parameter t for projection of QP onto QR: t = ((P-Q)·(R-Q)) / |R-Q|^2
    t = ((px - qx)*dx + (py - qy)*dy) / (dx*dx + dy*dy)
    # foot = Q + t*(R-Q)
    return (qx + t*dx, qy + t*dy)

def reflect_point_across_line(P, Q, R):
    # reflect point P across line QR
    # Compute foot of perpendicular from P to line QR, then reflect
    foot = foot_of_perpendicular(P, Q, R)
    # vector from P to foot
    vx = foot[0] - P[0]
    vy = foot[1] - P[1]
    # reflected point = P + 2 * (vector to foot)
    return (P[0] + 2*vx, P[1] + 2*vy)

    d = 2*(Ax*(By - Cy) + Bx*(Cy - Ay) + Cx*(Ay - By))

def circle_intersections_center_radius(O1, R1, O2, R2):
    # returns list of intersection points (tuples)
    (x0,y0) = O1
    (x1,y1) = O2
    dx = x1 - x0
    dy = y1 - y0
    d = mp.sqrt(dx*dx + dy*dy)
    # if no intersection, return empty list
    if d > R1 + R2 + 1e-20 or d < abs(R1 - R2) - 1e-20:
        return []
    # compute a
    a = (R1**2 - R2**2 + d**2) / (2*d)
    # height
    h_sq = R1**2 - a**2
    if h_sq < 0:

    h = mp.sqrt(h_sq)
    # point P2
    x2 = x0 + a*dx/d
    y2 = y0 + a*dy/d
    # intersection points
    rx = -dy * (h/d)
    ry = dx * (h/d)
    p_int1 = (x2 + rx, y2 + ry)
    p_int2 = (x2 - rx, y2 - ry)
    return [p_int1, p_int2]

def foot_of_perpendicular(P, Q, R):
    # projection of P onto line QR
    (px,py) = P
    (qx,qy) = Q
    (rx,ry) = R
    dx = rx - qx
    dy = ry - qy
    # avoid zero length
    if dx == 0 and dy == 0:
        raise ValueError("Q and R same")
    t = ((px - qx)*dx + (py - qy)*dy) / (dx*dx + dy*dy)
    return (qx + t*dx, qy + t*dy)

def reflect_point_across_line(P, Q, R):
    # reflect P across line QR
    foot = foot_of_perpendicular(P, Q, R)
    vx = foot[0] - P[0]
    vy = foot[1] - P[1]
    return (P[0] + 2*vx, P[1] + 2*vy)

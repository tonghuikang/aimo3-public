def count_subsets(weights, max_sum):
    # count nonempty subsets where sum <= max_sum
    n = len(weights)
    count = 0
    subsets = []
    for r in range(1, n+1):
        for comb in combinations(range(n), r):
            s = sum(weights[i] for i in comb)
            if s <= max_sum:
                count += 1
                subsets.append(comb)
    return count, subsets

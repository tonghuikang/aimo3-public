    # compute minimal LCM for sum S via brute up to maybe S<=200
    import math
    best = None
    for a in range(1, S-1):
        for b in range(a+1, S-a):
            c = S - a - b
            if c <= b:
                continue
            l = math.lcm(a,b,c)
            if best is None or l < best:
                best = l
    return best

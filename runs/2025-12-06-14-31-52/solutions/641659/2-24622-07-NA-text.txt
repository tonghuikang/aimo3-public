    # Use bisection or findroot on each interval

        # bisection
        fl = f(low)
        fh = f(high)

        for _ in range(60):
            mid = (low+high)/2
            fm = f(mid)

                low = mid
                continue
            if fl*fm <= 0:
                high = mid
                fh = fm
            else:
                low = mid
                fl = fm
        roots.append((low+high)/2)

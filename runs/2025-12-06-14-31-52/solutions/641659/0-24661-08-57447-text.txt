    # BN direction:
    BNx = N[0]; BNy = N[1]
    # center O_T is intersection of line perpendicular to BN through B (direction (-BNy, BNx)) and perpendicular bisector of BK.
    # Solve linear equations:
    # O_T = s * (-BNy, BNx)
    # Midpoint of BK: M = (K[0]/2, K[1]/2)
    # perpendicular bisector direction: (-K[1], K[0]) passes through M.
    # So O_T = M + t * (-K[1], K[0])
    # Solve: s*(-BNy) = Mx + t*(-K[1])
    #       s*(BNx) = My + t*(K[0])
    # Solve for s,t.
    Mx = K[0]/2
    My = K[1]/2
    # Solve linear system:
    # -BNy * s + K[1] * t = Mx
    # BNx * s - K[0] * t = My
    # Represent as matrix:
    A_mat = mp.matrix([[-BNy, K[1]],
                       [BNx, -K[0]]])
    b_vec = mp.matrix([Mx, My])
    sol = mp.lu_solve(A_mat, b_vec)
    s_param = sol[0]
    # O_T coordinates
    O_T = (s_param * (-BNy), s_param * BNx)
    # radius
    R_T = mp.sqrt(O_T[0]**2 + O_T[1]**2)
    # Intersection with y=0: (x - O_Tx)^2 + O_Ty^2 = R_T^2 => (x - O_Tx)^2 = R_T^2 - O_Ty^2
    rhs = R_T**2 - O_T[1]**2
    if rhs < 0:
        rhs = mp.mpf('0')
    sqrt_rhs = mp.sqrt(rhs)
    xT1 = O_T[0] + sqrt_rhs
    xT2 = O_T[0] - sqrt_rhs
    # Choose the one not zero (B is at x=0)
    if mp.fabs(xT1) > mp.mpf('1e-12'):
        xT = xT1
    else:
        xT = xT2
    # compute lengths
    CT = mp.fabs(1 - xT)
    BT = mp.fabs(xT)
    NB = mp.sqrt(N[0]**2 + N[1]**2)
    NE = mp.sqrt((N[0]-E[0])**2 + (N[1]-E[1])**2)
    R_ratio = (CT * NB) / (BT * NE)
    return R_ratio

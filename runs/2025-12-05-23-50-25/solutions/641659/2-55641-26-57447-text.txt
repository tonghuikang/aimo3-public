def f_t(t):
    # compute determinant for given t (s)
    # t > 1
    # set s = t
    s = t
    # side lengths b = s - BD, c = s - CD
    b = s - BD_const
    c = s - CD_const
    # coordinates
    xA = (c**2 - b**2 + a_const**2) / (2*a_const)
    yA_sq = c**2 - xA**2
    if yA_sq <= 0:
        return mp.mpf('nan')
    yA = mp.sqrt(yA_sq)
    # points
    B = (mp.mpf(0), mp.mpf(0))
    C = (a_const, mp.mpf(0))
    A = (xA, yA)
    # D
    D = (BD_const, mp.mpf(0))
    # F: BF = BD_const
    BF = BD_const
    # AB length is c
    # F = B + (BF/c) * (A - B) = (BF/c * xA, BF/c * yA)
    F = (BF/c * xA, BF/c * yA)
    # E: CE = CD_const
    CE = CD_const
    # AC length is b
    # E = C + (CE/b) * (A - C)
    E = (a_const + (CE/b)*(xA - a_const), (CE/b)*yA)
    # compute circumcircle of ABC using sympy
    # Convert to sympy Points with high precision floats
    A_sym = sp.Point(float(A[0]), float(A[1]))
    B_sym = sp.Point(0.0, 0.0)

    F_sym = sp.Point(float(F[0]), float(F[1]))
    E_sym = sp.Point(float(E[0]), float(E[1]))

        if not pt.equals(A_sym):
            K_pt = pt
            break

        return mp.mpf('nan')

    # return determinant (should be zero when cyclic condition holds)
    # Returns mp.mpf
    # Compute Fibonacci numbers
    F = [0,1]
    for i in range(2, n+3):
        F.append(F[-1] + F[-2])
    d = F[n]  # BD
    a = F[n+2]  # BC length
    # slopes
    mB = 2 * r * d / (d**2 - r**2)
    mC = -2 * r * (a - d) / ((a - d)**2 - r**2)
    # intersection for A
    X_A = (-mC * a) / (mB - mC)
    Y_A = mB * X_A
    # Points
    # incircle center I
    I = mp.matrix([d, r])
    # foot F onto AB
    tF = (I[0] + I[1] * mB) / (1 + mB**2)
    F_pt = mp.matrix([tF, mB * tF])
    # foot E onto AC
    # compute using formula: projection onto line direction (1, mC) from point C
    # vector from C to I = (d - a, r)

y_center = Y_A/2 + (-1/mB)*(a/2 - X_A/2)
O_ABC = sp.Point(a/2, y_center)
# R_ABC2
R_ABC2 = O_ABC.x**2 + O_ABC.y**2

# circumcenter of AEF: solve linear system
# M * O = RHS
M11 = 2 * (E_pt.x - X_A)
M12 = 2 * (E_pt.y - Y_A)
M21 = 2 * (F_pt.x - X_A)
M22 = 2 * (F_pt.y - Y_A)
RHS1 = E_pt.x**2 + E_pt.y**2 - X_A**2 - Y_A**2
RHS2 = F_pt.x**2 + F_pt.y**2 - X_A**2 - Y_A**2

detM = M11 * M22 - M12 * M21
ox = (RHS1 * M22 - M12 * RHS2) / detM
oy = (M11 * RHS2 - RHS1 * M21) / detM
O_AEF = sp.Point(ox, oy)
R_AEF2 = (ox - X_A)**2 + (oy - Y_A)**2

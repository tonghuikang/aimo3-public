def circumcenter(p1, p2, p3):
    # p1,p2,p3 are (x,y) tuples
    x1,y1 = p1
    x2,y2 = p2
    x3,y3 = p3
    # Compute using perpendicular bisector intersection
    d = 2*(x1*(y2 - y3) + x2*(y3 - y1) + x3*(y1 - y2))
    if abs(d) < 1e-12:
        return None  # collinear or degenerate
    ux = ((x1**2 + y1**2)*(y2 - y3) + (x2**2 + y2**2)*(y3 - y1) + (x3**2 + y3**2)*(y1 - y2)) / d
    uy = ((x1**2 + y1**2)*(x3 - x2) + (x2**2 + y2**2)*(x1 - x3) + (x3**2 + y3**2)*(x2 - x1)) / d
    return (ux, uy)

def circle_intersections(O1,R1,O2,R2):
    # Returns two intersection points (x,y) as tuples
    x1,y1 = O1
    x2,y2 = O2
    dx = x2 - x1
    dy = y2 - y1
    d = math.hypot(dx, dy)
    if d == 0:
        return []  # concentric circles
    # check for intersection
    if d > R1 + R2 + 1e-12 or d < abs(R1 - R2) - 1e-12:
        return []  # no intersection
    # a distance from O1 to point P2 along line O1O2
    a = (R1**2 - R2**2 + d**2) / (2*d)
    # height of intersection from line
    h_sq = R1**2 - a**2
    if h_sq < -1e-12:
        # no intersection due to numeric
        return []
    h = math.sqrt(max(h_sq, 0.0))
    # point P2
    xm = x1 + a * dx / d
    ym = y1 + a * dy / d
    # offset vector orthogonal to (dx,dy)
    rx = -dy * (h / d)
    ry = dx * (h / d)
    p_int1 = (xm + rx, ym + ry)
    p_int2 = (xm - rx, ym - ry)
    return [p_int1, p_int2]

def foot_of_perpendicular(P, A, B):
    # foot of perpendicular from point P to line AB
    # returns projection point
    # line AB param A + t*(B-A)
    Ax,Ay = A
    Bx,By = B
    Px,Py = P
    ABx = Bx - Ax
    ABy = By - Ay
    # t = ((P-A)·AB) / |AB|^2
    t = ((Px - Ax)*ABx + (Py - Ay)*ABy) / (ABx*ABx + ABy*ABy)
    proj = (Ax + t*ABx, Ay + t*ABy)
    return proj

def reflect_point_across_line(P, A, B):
    # reflect point P across line AB
    # projection
    proj = foot_of_perpendicular(P, A, B)
    # reflection = 2*proj - P
    return (2*proj[0] - P[0], 2*proj[1] - P[1])

def triangle_points_from_sides(a,b,c):
    # BC = a on x-axis from B(0,0) to C(a,0)
    # side lengths: AB=c, AC=b
    # compute A coordinates
    xA = (a*a + c*c - b*b) / (2*a)
    y_sq = c*c - xA*xA
    if y_sq <= 0:
        # degenerate
        return None
    yA = math.sqrt(y_sq)
    return (xA, yA)

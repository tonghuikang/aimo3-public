<|end|><|start|>assistant<|channel|>analysis to=python code<|message|>import mpmath as mp

mp.mp.dps = 50  # high precision

def circumcenter_coords(P, Q, R):
    # P,Q,R are (x,y) tuples of mp.mpf
    x1,y1 = P
    x2,y2 = Q
    x3,y3 = R
    d = 2*(x1*(y2 - y3) + x2*(y3 - y1) + x3*(y1 - y2))
    # If d zero, points collinear; but should not.
    ux = ((x1**2 + y1**2)*(y2 - y3) + (x2**2 + y2**2)*(y3 - y1) + (x3**2 + y3**2)*(y1 - y2)) / d
    uy = ((x1**2 + y1**2)*(x3 - x2) + (x2**2 + y2**2)*(x1 - x3) + (x3**2 + y3**2)*(x2 - x1)) / d
    return (ux, uy)

def geometry_compute(phi, psi):
    # incircle radius = 1, center at (0,0)
    D = (mp.mpf('0'), mp.mpf('-1'))  # (0,-1)
    # points E,F on unit circle
    E = (mp.cos(phi), mp.sin(phi))
    F = (mp.cos(psi), mp.sin(psi))
    # B, C points
    Bx = (1 + mp.sin(psi)) / mp.cos(psi)
    By = mp.mpf('-1')
    B = (Bx, By)
    Cx = (1 + mp.sin(phi)) / mp.cos(phi)
    Cy = mp.mpf('-1')
    C = (Cx, Cy)
    # Solve for A: intersection of tangents at E and F (line equations)
    # System: cos(phi)*x + sin(phi)*y = 1; cos(psi)*x + sin(psi)*y = 1
    # Solve via linear algebra: M*[x,y] = [1,1]
    a11 = mp.cos(phi); a12 = mp.sin(phi)
    a21 = mp.cos(psi); a22 = mp.sin(psi)
    detM = a11*a22 - a12*a21
    # Inverse M = (1/detM) * [[a22, -a12], [-a21, a11]]
    Ax = ( (a22*1 - a12*1) ) / detM
    Ay = ( (-a21*1 + a11*1) ) / detM
    A = (Ax, Ay)
    # Circumcenters
    O_ABC = circumcenter_coords(A, B, C)
    O_AFE = circumcenter_coords(A, F, E)
    # Compute vector v = O_ABC - O_AFE
    v = (O_ABC[0] - O_AFE[0], O_ABC[1] - O_AFE[1])
    # perpendicular vector w = (v_y, -v_x)
    w = (v[1], -v[0])
    # Compute K: second intersection of line through A with direction w and circle circ_ABC
    # u = A - O_ABC
    u = (A[0] - O_ABC[0], A[1] - O_ABC[1])
    # dot u·w
    dot_uw = u[0]*w[0] + u[1]*w[1]
    w2 = w[0]**2 + w[1]**2
    tK = -2 * dot_uw / w2
    K = (A[0] + tK*w[0], A[1] + tK*w[1])
    # Compute line EF direction vector
    vEF = (F[0] - E[0], F[1] - E[1])
    # projection of K onto line EF: compute scalar t_proj_K = ((K-E)·vEF) / (vEF·vEF)
    tKproj = ((K[0] - E[0])*vEF[0] + (K[1] - E[1])*vEF[1]) / (vEF[0]**2 + vEF[1]**2)
    projK = (E[0] + tKproj*vEF[0], E[1] + tKproj*vEF[1])
    # K' = 2*projK - K
    Kprime = (2*projK[0] - K[0], 2*projK[1] - K[1])
    # Projection of D onto EF: N
    tNproj = ((D[0] - E[0])*vEF[0] + (D[1] - E[1])*vEF[1]) / (vEF[0]**2 + vEF[1]**2)
    N = (E[0] + tNproj*vEF[0], E[1] + tNproj*vEF[1])
    # Compute circumcenter of B,K,N
    O_BKN = circumcenter_coords(B, K, N)
    # Radius:
    r_BKN = mp.sqrt((B[0] - O_BKN[0])**2 + (B[1] - O_BKN[1])**2)
    # Distance from K' to O_BKN
    d_Kp = mp.sqrt((Kprime[0] - O_BKN[0])**2 + (Kprime[1] - O_BKN[1])**2)
    # cyclic condition diff:
    diff_cyclic = d_Kp - r_BKN
    # Compute T: circle passing through B and K tangent to line BN at B.
    # Compute BN direction vector
    BN = (N[0] - B[0], N[1] - B[1])
    # Center O_center lies on line through B perpendicular to BN: O_center = B + s * perp(BN)
    # perp(BN) = (BN_y, -BN_x) or (-BN_y, BN_x). Choose any; direction vector perpendicular.
    perp_BN = (BN[1], -BN[0])  # this is perpendicular
    # also must be equidistant to B and K: |O_center - B| = |O_center - K|
    # Let O_center = B + u * perp_BN.
    # Then distance squared to B = u^2 * |perp_BN|^2.
    # Distance squared to K: |(B + u*perp) - K|^2 = (B-K + u*perp)^2.
    # Solve for u.
    # Let dx = Bx - Kx, dy = By - Ky.
    dx = B[0] - K[0]
    dy = B[1] - K[1]
    # Solve (dx + u*perp_BN[0])^2 + (dy + u*perp_BN[1])^2 = u^2 * (perp_BN[0]^2 + perp_BN[1]^2)
    # Expand: (dx^2 + 2*dx*u*perp_x + u^2*perp_x^2) + (dy^2 + 2*dy*u*perp_y + u^2*perp_y^2) = u^2*(perp_x^2 + perp_y^2)
    # Left u^2 terms cancel with right side, leaving:
    # dx^2 + dy^2 + 2*u*(dx*perp_x + dy*perp_y) = 0
    # Solve for u: u = -(dx^2 + dy^2) / (2*(dx*perp_x + dy*perp_y))
    perp_x, perp_y = perp_BN
    denominator = 2*(dx*perp_x + dy*perp_y)
    if denominator == 0:
        # degenerate case; return None
        O_center = None
    else:
        u_center = -(dx**2 + dy**2) / denominator
        O_center = (B[0] + u_center*perp_x, B[1] + u_center*perp_y)
    # Compute radius of this circle
    if O_center is not None:
        r_circle = mp.sqrt((B[0] - O_center[0])**2 + (B[1] - O_center[1])**2)
        # Intersection with line BC (y = -1): solve (x - O_center.x)^2 + (-1 - O_center.y)^2 = r_circle^2
        # => (x - O_center.x)^2 = r_circle^2 - (-1 - O_center.y)^2
        radicand = r_circle**2 - (-1 - O_center[1])**2
        if radicand < 0:
            # no real intersection
            T = None
        else:
            # x = O_center.x ± sqrt(radicand)
            sqrt_radicand = mp.sqrt(radicand)
            # choose the root not equal to B.x
            x1 = O_center[0] + sqrt_radicand
            x2 = O_center[0] - sqrt_radicand
            # B.x known, select one not close to B.x
            if mp.almosteq(x1, B[0], rel_eps=1e-12):
                Tx = x2
            elif mp.almosteq(x2, B[0], rel_eps=1e-12):
                Tx = x1
            else:
                # Choose root farther from B.x? Actually one is B itself (should be exact).
                # Use whichever is farther from B.x
                if abs(x1 - B[0]) > abs(x2 - B[0]):
                    Tx = x1
                else:
                    Tx = x2
            T = (Tx, mp.mpf('-1'))
    else:
        T = None
    # Compute distances for ratio: CT, BT, NB, NE
    # Compute NE = distance N to E
    NB = mp.sqrt((N[0] - B[0])**2 + (N[1] - B[1])**2)
    NE = mp.sqrt((N[0] - E[0])**2 + (N[1] - E[1])**2)
    if T is not None:
        CT = mp.sqrt((C[0] - T[0])**2 + (C[1] - T[1])**2)
        BT = mp.sqrt((B[0] - T[0])**2 + (B[1] - T[1])**2)
    else:
        CT = BT = None
    # Return relevant values
    return {
        'A':A, 'B':B, 'C':C, 'D':D, 'E':E, 'F':F,
        'K':K, 'Kprime':Kprime, 'N':N, 'O_center':O_center,
        'diff_cyclic': diff_cyclic,
        'CT':CT, 'BT':BT, 'NB':NB, 'NE':NE,
        'T': T
    }

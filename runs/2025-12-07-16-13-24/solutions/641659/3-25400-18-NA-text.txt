def circumcenter(P1, P2, P3):
    # returns center (x,y) and radius
    # using perpendicular bisector intersection.
    (x1, y1), (x2, y2), (x3, y3) = (P1, P2, P3)
    # Compute using formula
    d = 2*(x1*(y2 - y3) + x2*(y3 - y1) + x3*(y1 - y2))
    if abs(d) < 1e-12:
        raise ValueError("Collinear points")
    ux = ((x1**2 + y1**2)*(y2 - y3) + (x2**2 + y2**2)*(y3 - y1) + (x3**2 + y3**2)*(y1 - y2)) / d
    uy = ((x1**2 + y1**2)*(x3 - x2) + (x2**2 + y2**2)*(x1 - x3) + (x3**2 + y3**2)*(x2 - x1)) / d
    R = math.hypot(ux - x1, uy - y1)
    return (ux, uy), R

def intersect_circles(O1, R1, O2, R2):
    # returns list of intersection points (x,y), possibly empty or 1 or 2.
    x1,y1 = O1
    x2,y2 = O2
    dx = x2 - x1
    dy = y2 - y1
    d = math.hypot(dx, dy)
    if d > R1 + R2 + 1e-12 or d < abs(R1 - R2) - 1e-12:
        return []  # no intersection
    # distance from O1 to line connecting intersection points
    # a = (R1^2 - R2^2 + d^2) / (2d)
    a = (R1**2 - R2**2 + d**2) / (2*d) if d != 0 else 0
    # height from line to intersection points
    h_sq = max(R1**2 - a**2, 0.0)
    h = math.sqrt(h_sq)
    # point P2 = O1 + a*(O2-O1)/d
    xm = x1 + a*dx/d if d != 0 else x1
    ym = y1 + a*dy/d if d != 0 else y1
    # intersection points
    # offset perpendicular by h*( -dy/d, dx/d )
    rx = -dy * (h/d) if d != 0 else 0
    ry = dx * (h/d) if d != 0 else 0
    p1 = (xm + rx, ym + ry)
    p2 = (xm - rx, ym - ry)
    if h < 1e-12:
        return [p1]
    else:
        return [p1, p2]

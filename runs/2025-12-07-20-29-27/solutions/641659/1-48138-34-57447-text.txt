# Define a function to reflect point P across line defined by points Q,R.
def reflect_point_across_line(P, Q, R):
    # Vector from Q to R
    v = R - Q
    # Unit direction
    v_norm2 = sp.simplify(v.dot(v))
    # Projection of (P-Q) onto v
    proj = ((P - Q).dot(v) / v_norm2) * v
    # Perpendicular component:
    perp = (P - Q) - proj
    # Reflection: Q + proj - perp
    return sp.simplify(Q + proj - perp)

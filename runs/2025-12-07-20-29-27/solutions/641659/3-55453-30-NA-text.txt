def solve_triangle_n_numeric2(n, guess=None):
    Fn, Fn1, a = get_fib_seq(n)
    x, y, r, Kx, Ky = sp.symbols('x y r Kx Ky')
    # Points
    B = sp.Point(0,0)
    C = sp.Point(a,0)
    D = sp.Point(Fn,0)
    I = sp.Point(Fn, r)
    A = sp.Point(x,y)
    # foot F
    t = (x*Fn + y*r) / (x**2 + y**2)
    F = sp.Point(x*t, y*t)
    # foot E
    t2 = ((Fn - a)*(x - a) + r*y) / ((x - a)**2 + y**2)
    E = sp.Point(a + (x - a)*t2, y*t2)
    # eq1, eq2
    eq1 = (x*r - y*Fn)**2 - r**2 * (x**2 + y**2)
    eq2 = ((x - a)*r - y*(Fn - a))**2 - r**2 * ((x - a)**2 + y**2)
    # circumcenter O of ABC
    # Dden = 2*a*y
    Dden = 2*a*y
    O_x = (a**2 * y) / Dden  # since only term remains (x^2+y^2)*(0) =0, plus a^2*y
    O_y = ((x**2 + y**2) * a - a**2 * x) / Dden
    # radius squared R2
    R2 = (O_x - x)**2 + (O_y - y)**2
    
    # circumcenter O_AFE
    fx, fy = F.x, F.y
    ex, ey = E.x, E.y
    D_AFE = 2*(x*(fy - ey) + fx*(ey - y) + ex*(y - fy))
    O_AFE_x = ((x**2+y**2)*(fy - ey) + (fx**2+fy**2)*(ey - y) + (ex**2+ey**2)*(y - fy)) / D_AFE
    O_AFE_y = ((x**2+y**2)*(ex - fx) + (fx**2+fy**2)*(x - ex) + (ex**2+ey**2)*(fx - x)) / D_AFE
    # radius squared R_AFE2
    R_AFE2 = (O_AFE_x - x)**2 + (O_AFE_y - y)**2
    
    # K is (Kx,Ky)
    # eq3: K on circumcircle ABC
    eq3 = (Kx - O_x)**2 + (Ky - O_y)**2 - R2
    # eq4: K on circle AFE
    eq4 = (Kx - O_AFE_x)**2 + (Ky - O_AFE_y)**2 - R_AFE2
    
    # N foot from D onto EF
    vx = fx - ex
    vy = fy - ey
    dot_DE_v = (Fn - ex)*vx + (-ey)*vy
    denom_v2 = vx**2 + vy**2
    Nx = ex + dot_DE_v * vx / denom_v2
    Ny = ey + dot_DE_v * vy / denom_v2
    # reflection K' across EF
    # projection of K onto EF:
    dot_KE_v = (Kx - ex)*vx + (Ky - ey)*vy
    proj_Kx = ex + dot_KE_v * vx / denom_v2
    proj_Ky = ey + dot_KE_v * vy / denom_v2
    Kp_x = 2*proj_Kx - Kx
    Kp_y = 2*proj_Ky - Ky
    # Compute u,v for circle through B,K,N: w=0
    # det = Kx*Ny - Ky*Nx
    det = Kx*Ny - Ky*Nx
    # Avoid zero det; but for solution, should be non-zero.
    u = ( - (Kx**2 + Ky**2) * Ny + (Nx**2 + Ny**2) * Ky ) / det
    v = ( (Kx**2 + Ky**2) * Nx - (Nx**2 + Ny**2) * Kx ) / det
    # eq5: K' on same circle: Kp_x^2 + Kp_y^2 + u*Kp_x + v*Kp_y = 0
    eq5 = Kp_x**2 + Kp_y**2 + u*Kp_x + v*Kp_y
    
    eqs = [eq1, eq2, eq3, eq4, eq5]

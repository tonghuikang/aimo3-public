    # Compute intersections of circumcircles of triangles p1p2p3 and q1q2q3
    O1 = circumcenter(p1, p2, p3)
    R1 = distance(O1, p1)
    O2 = circumcenter(q1, q2, q3)
    R2 = distance(O2, q1)

    a = -2*(O1[0] - O2[0])
    b = -2*(O1[1] - O2[1])

    # Solve intersection of line with circle1

def foot_of_perpendicular(P, Q, R):
    # foot from point P to line through Q,R
    # returns projection point H
    # vector QR = R - Q
    # project PQ onto QR: H = Q + ((P-Q)·(R-Q) / |R-Q|^2)*(R-Q)
    vx = R[0] - Q[0]
    vy = R[1] - Q[1]
    # compute dot product
    wx = P[0] - Q[0]
    wy = P[1] - Q[1]
    dot = wx*vx + wy*vy
    norm2 = vx*vx + vy*vy
    t = dot / norm2
    Hx = Q[0] + t*vx
    Hy = Q[1] + t*vy
    return (Hx, Hy)

def reflect_point_across_line(P, Q, R):
    # reflect point P across line through Q,R
    H = foot_of_perpendicular(P, Q, R)  # foot
    # reflection: P' = 2H - P
    return (2*H[0] - P[0], 2*H[1] - P[1])

    # Input n integer, r numeric
    d = fib(n)  # BD
    cd = fib(n+1)  # CD
    a = fib(n+2)  # BC
    # coordinates
    B = (mp.mpf('0'), mp.mpf('0'))
    C = (mp.mpf(a), mp.mpf('0'))
    D = (mp.mpf(d), mp.mpf('0'))
    I = (mp.mpf(d), mp.mpf(r))
    # slopes
    # m_B = 2*d*r/(d**2 - r**2)
    mB = 2*mp.mpf(d)*mp.mpf(r) / (mp.mpf(d)**2 - mp.mpf(r)**2)

    # projection of vector I onto direction (1,mB): t = (I·u) / (u·u), where u = (1,mB)
    # I = (d, r)
    # u·u = 1 + mB**2

    F = (tF, mB*tF)  # since line passes through origin

    tE = ((mp.mpf(d) - mp.mpf(a)) + mC*mp.mpf(r)) / (1 + mC**2)  # parameter along direction vector from C

    # BC is x-axis (y=0). Solve intersection of circle (center O = (ox, oy), radius R) with y=0.
    ox, oy = Obn
    # Equation: (x - ox)^2 + (0 - oy)^2 = R^2

    xT1 = ox + sqrt_rad

    # One of these is B at x=0 (since B is intersection). Choose the one not near 0.
    # Check which is approx 0
    if abs(xT1) < abs(xT2):

        xT_other = xT2

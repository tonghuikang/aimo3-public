def poly_mul(p, q):
    # p and q are lists of ints, p[i] coeff of x^i.
    r = [0] * (len(p) + len(q) - 1)
    for i, a in enumerate(p):
        if a == 0: continue
        for j, b in enumerate(q):
            if b == 0: continue
            r[i+j] += a * b
    return r

def poly_sub(p, q):
    # p - q
    n = max(len(p), len(q))
    r = [0] * n
    for i in range(len(p)):
        r[i] += p[i]
    for i in range(len(q)):
        r[i] -= q[i]
    # remove trailing zeros
    while len(r) > 0 and r[-1] == 0:
        r.pop()
    if not r:
        r = [0]
    return r

def poly_div_exact(num, den):
    # exact division of integer polynomials assuming divisible.
    # Return quotient.
    # Use naive long division.
    num = num[:]  # copy
    deg_num = len(num) - 1
    deg_den = len(den) - 1
    lc_den = den[-1]
    if lc_den == 0:
        raise ValueError("Denominator leading coeff zero")
    q = [0] * (max(0, deg_num - deg_den) + 1)
    while deg_num >= deg_den and any(num):
        coeff = num[-1] // lc_den
        if num[-1] % lc_den != 0:
            raise ValueError("non exact division")
        shift = deg_num - deg_den
        q[shift] = coeff
        # subtract coeff * den shifted
        for i in range(len(den)):
            num[shift + i] -= coeff * den[i]
        # trim trailing zeros
        while len(num) > 0 and num[-1] == 0:
            num.pop()
        deg_num = len(num) - 1
    # after division, remainder should be zero.
    # ensure remainder zero.
    if any(num):
        raise ValueError("Non zero remainder")
    # remove trailing zeros in q? keep as list.
    while len(q) > 0 and q[-1] == 0:
        q.pop()
    if not q:
        q = [0]
    return q

def poly_pow_x_minus_one(n):
    # polynomial x^n - 1
    p = [ -1 ] + [0]* (n-1) + [1]
    return p

@lru_cache(None)
def cyclotomic(n):
    # compute cyclotomic polynomial Phi_n
    # n >=1
    if n == 1:
        return [-1, 1]  # x - 1
    # compute product of (x^n - 1) divided by product of Phi_d for proper divisors d|n
    # compute numerator
    num = poly_pow_x_minus_one(n)
    # divide by each divisor's cyclotomic
    for d in range(1, n):
        if n % d == 0:
            # proper divisor
            # multiply divisor's phi polynomial
            # Actually we need to divide by Phi_d
            phi_d = cyclotomic(d)
            num = poly_div_exact(num, phi_d)
    # after division, num is Phi_n
    # ensure leading coefficient is 1 (monic)
    # The result should be monic.
    return num

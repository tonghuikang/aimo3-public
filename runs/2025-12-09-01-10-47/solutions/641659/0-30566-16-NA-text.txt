def foot_point(P0, P1, P2):
    # foot from P0 to line through P1-P2
    x0,y0 = P0
    x1,y1 = P1
    x2,y2 = P2
    # direction vector v = P2-P1
    vx = x2 - x1
    vy = y2 - y1
    # w = P0 - P1
    wx = x0 - x1
    wy = y0 - y1
    # t = (v·w) / (v·v)
    denom = vx*vx + vy*vy
    if denom == 0:
        return (x1,y1)  # P1==P2
    t = (vx*wx + vy*wy) / denom
    xf = x1 + t*vx
    yf = y1 + t*vy
    return (xf, yf)

def second_intersection_of_circles(circle1, circle2, ptA):
    # Return intersection points of two circles other than ptA
    inter = circle1.intersection(circle2)
    # inter is list of 0/1/2 points
    # remove ptA
    pts = []
    for p in inter:
        if not p.equals(ptA):
            pts.append(p)
    return pts[0] if pts else None

def compute_K(n, t):
    # Compute point K
    tri = construct_triangle(n, t)
    B = tri['B']
    C = tri['C']
    A = tri['A']
    E = tri['E']
    F = tri['F']
    # circles: circumcircle of ABC, circumcircle of AEF
    circ_ABC = sp.Circle(A, B, C)
    circ_AEF = sp.Circle(A, E, F)
    # find K (second intersection)
    K = second_intersection_of_circles(circ_ABC, circ_AEF, A)
    return K

def reflect_point_across_line(P, line):
    # reflection of point P across line
    return P.reflect(line)

def compute_T(n, t):
    # Compute T, intersection of circle through B,K tangent to line BN at B with BC
    tri = construct_triangle(n, t)
    B = tri['B']
    N = tri['N']
    # compute K
    K = compute_K(n, t)
    # circle passing through B and K and tangent to line BN at B
    # condition: line BN is tangent at B, so radius line O->B perpendicular to BN.
    # We can find circle center O such that O lies on line perpendicular to BN passing through B (i.e., line through B with direction BN rotated 90 deg)
    # and distance from O to B equals distance from O to K.
    # Let O = B + u * v where v is unit vector perpendicular to BN.
    # Solve for u such that |O - B| = |O - K|
    # Let direction perpendicular to BN be p = BN rotated 90 deg.
    # Let's compute O param: O = B + t0 * p (since B is on circle and O-B = t0 * p)
    # Then radius = |O - B| = |t0|*|p| = |t0|*|BN|? Actually p is vector perpendicular to BN; length of p = |BN| (if we rotate BN vector by 90 deg we keep magnitude). So p = sp.Point(- (N.y - B.y), N.x - B.x). But we can treat p as vector.
    # Then we need |O - K|^2 = |O - B|^2 = t0^2 * |p|^2.
    # Solve for t0.
    # Then circle intersection with BC (line y=0) aside from B gives T.
    BN = sp.Line(B, N)
    # Vector BN:
    BN_vec = sp.Point(N.x - B.x, N.y - B.y)
    # Perpendicular vector:
    perp = sp.Point(-BN_vec.y, BN_vec.x)  # rotate by +90 degrees
    # Parameter t0:
    t0 = sp.symbols('t0', real=True)
    O = B + t0 * perp  # sympy addition: B + t0*perp yields a Point with symbolic
    # Solve |O-K|^2 = |O-B|^2
    eq = (O.distance(K)**2 - O.distance(B)**2).simplify()
    sol_t0 = sp.solve(eq, t0)
    # choose nonzero solution
    # sol_t0 may be [0, ...], ignore 0
    t0_sol = None
    for s in sol_t0:
        if not s.is_zero:
            t0_sol = s
            break
    # compute center O
    O = B + t0_sol * perp
    # Circle center O radius = |O - B|
    circle_BK = sp.Circle(O, B)
    # intersect with BC line (y=0)
    line_BC = sp.Line(B, C)
    inters = circle_BK.intersection(line_BC)
    # pick point not B
    T = None
    for pt in inters:
        if not pt.equals(B):
            T = pt
    return T

def compute_ratio_expr(n, t):
    # Compute ratio CT * NB / (BT * NE)
    tri = construct_triangle(n, t)
    B = tri['B']
    C = tri['C']
    N = tri['N']
    E = tri['E']
    # compute T
    T = compute_T(n, t)
    # compute distances
    CT = C.distance(T)
    NB = N.distance(B)
    BT = B.distance(T)
    NE = N.distance(E)
    # ratio
    return sp.simplify((CT * NB) / (BT * NE))

def cyclic_condition_det(n, t):
    # Return determinant of B,N,K,K'
    tri = construct_triangle(n, t)
    B = tri['B']
    N = tri['N']
    K = compute_K(n, t)
    # line EF
    line_EF = sp.Line(tri['E'], tri['F'])
    K_ref = reflect_point_across_line(K, line_EF)
    # points B,N,K,K_ref
    # Compute determinant:
    # For point (x,y), row = [x, y, x**2 + y**2, 1]
    def row(P):
        return [P.x, P.y, P.x**2 + P.y**2, 1]
    M = sp.Matrix([row(B), row(N), row(K), row(K_ref)])
    det = M.det()
    return sp.simplify(det)

def find_root_by_bisection(n, t_low, t_high, max_iter=50):
    f_low = det_val(n, t_low)
    f_high = det_val(n, t_high)
    if f_low == 0:
        return t_low
    if f_high == 0:
        return t_high
    if f_low*f_high > 0:
        raise ValueError("No sign change")
    for _ in range(max_iter):
        t_mid = (t_low + t_high) / 2
        f_mid = det_val(n, t_mid)
        if abs(f_mid) < 1e-12:
            return t_mid
        if f_low * f_mid < 0:
            t_high = t_mid
            f_high = f_mid
        else:
            t_low = t_mid
            f_low = f_mid
    return (t_low + t_high)/2

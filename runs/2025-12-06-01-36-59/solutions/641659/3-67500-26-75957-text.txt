def triangle_points():
    # random triangle not isosceles
    A = sp.Point(0,0)
    B = sp.Point(5,0)
    C = sp.Point(2,4)
    return A,B,C

def incircle_and_contacts(A,B,C):
    # compute incircle centre and radius
    # incircle center is intersection of angle bisectors
    # Using sympy's incenter?
    incenter = sp.geometry.Triangle(A,B,C).incenter
    # contact points: D on BC, E on CA, F on AB
    # D is foot from incenter to BC
    lineBC = sp.Line(B,C)
    D = lineBC.projection(incenter)
    lineCA = sp.Line(C,A)
    E = lineCA.projection(incenter)
    lineAB = sp.Line(A,B)
    F = lineAB.projection(incenter)
    return incenter, D, E, F

def mixtilinear_touch_point(A,B,C):
    # compute A-mixtilinear incircle touch point with circumcircle.
    # Known property: point of intersection of circumcircle of AEF with circumcircle of ABC (other than A)
    # So compute circumcircle of AEF and intersect with circumcircle of ABC.
    circABC = sp.geometry.Circle(A,B,C)
    # compute incircle contacts:
    _, D, E, F = incircle_and_contacts(A,B,C)
    circAEF = sp.geometry.Circle(A,E,F)
    # intersection of two circles
    inter = circABC.intersection(circAEF)
    # inter contains A and another point K
    K = inter[0] if inter[0] != A else inter[1]
    return K

def reflect_across_line(P, line):
    # reflect point P across line
    return P.reflect(line)

def foot_of_perpendicular(P, line):
    # foot of perpendicular from P to line
    return line.projection(P)

def is_cyclic(P1,P2,P3,P4):
    # returns True if points are concyclic (within tolerance)
    # Compute circle through P1,P2,P3; then check if P4 lies on same circle
    try:
        circ = sp.geometry.Circle(P1,P2,P3)
    except Exception as e:
        return False
    return sp.simplify(circ.distance(P4)) == 0

def compute_ratio_geometry(n):
    # Fibonacci numbers
    x = fibonacci(n)  # F_n
    y = fibonacci(n+1) # F_{n+1}
    a = x + y  # BC = F_{n+2}
    # compute s
    s = (a + sp.sqrt(3*a**2 - 2*x*y))/2
    # compute incircle radius r
    r_sq = (s - a) * x * y / s
    r = sp.sqrt(r_sq)
    # coordinates
    B = sp.Point(0,0)
    C = sp.Point(a,0)
    I = sp.Point(x, r)
    # compute slopes for AB and AC using tangents
    # slope AB:
    m_AB = 2*x*r / (x**2 - r**2)
    m_AC = -2*y*r / (y**2 - r**2)
    # compute intersection point A
    X = -m_AC * a / (m_AB - m_AC)
    Y = m_AB * X
    A = sp.Point(sp.simplify(X), sp.simplify(Y))
    # Compute contact points
    line_AB = sp.Line(A, B)
    line_AC = sp.Line(A, C)
    D = sp.Point(x, 0)
    E = line_AC.projection(I)
    F = line_AB.projection(I)
    # N foot from D to EF
    N = sp.Line(E, F).projection(D)
    # K = A-mixtilinear touch point: intersection of circumcircles of ABC and AEF other than A
    circ_ABC = sp.Circle(A,B,C)
    circ_AEF = sp.Circle(A,E,F)
    inter = circ_ABC.intersection(circ_AEF)
    # pick K not equal A
    K = inter[0] if not inter[0].equals(A) else inter[1]
    # K' reflection across EF
    Kprime = K.reflect(sp.Line(E,F))
    # Check cyclic condition (should hold)
    # Compute circle passing through B,K,N
    circ_BKN = sp.Circle(B, K, N)
    # compute distance of Kprime to that circle
    cyclic = sp.simplify(Kprime.distance(circ_BKN.center)**2 - circ_BKN.radius**2)
    # Compute T: circle tangent to line BN at B, passing through K
    line_BN = sp.Line(B,N)
    # center O is intersection of line BN's perpendicular at B and perp bisector of BK
    perp_at_B = line_BN.perpendicular_line(B)
    mid_BK = sp.Point( (B.x+K.x)/2, (B.y+K.y)/2 )
    perp_bis_BK = sp.Line(B,K).perpendicular_line(mid_BK)
    O = perp_at_B.intersection(perp_bis_BK)[0]
    R = O.distance(B)

    inter_T = circ_tan.intersection(sp.Line(B,C))

    # compute lengths
    CT = C.distance(T)
    NB = N.distance(B)
    BT = B.distance(T)
    NE = N.distance(E)
    ratio = sp.simplify(CT * NB / (BT * NE))
    # simplify ratio to numeric
    ratio_val = sp.N(ratio, 30)
    return float(ratio_val), float(cyclic.evalf())

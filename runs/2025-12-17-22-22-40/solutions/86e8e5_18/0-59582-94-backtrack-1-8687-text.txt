# Precompute powers of 3 factor for M modulo small d
def M_mod_d(d):
    # compute 3^K mod d efficiently using CRT
    if d == 1:
        return 0
    if math.gcd(3, d) == 1:
        # Since K is multiple of lambda(d), exponent reduces to 0 mod lambda(d)
        # So result is 1 mod d
        return 1 % d
    # else d has factor 3
    # factor out 3^e
    e = 0
    tmp = d
    while tmp % 3 == 0:
        tmp //= 3
        e += 1
    # now d = 3^e * t, t = tmp
    t = tmp
    # compute k such that 3^e * k ≡ 1 (mod t)
    # i.e., k ≡ inv(3^e mod t) (mod t)
    mod_t = t
    if mod_t == 1:
        # t=1, then condition is just x ≡ 0 mod 3^e, and mod 1 always satisfied, so x = 0.
        return 0
    # compute inv of 3^e modulo t
    # use pow with -1 for inverse in Python 3.8+
    inv = pow(pow(3, e, mod_t), -1, mod_t)  # compute 3^e mod t then invert
    k = inv % mod_t
    x = (pow(3, e, d) * k) % d
    return x

    best = None

    # naive loops a from 1 to m-3, b from a+1 to m-2-a, c = m-a-b, ensure c > b
    for a in range(1, m-2):

            # distinctness guaranteed
            l = (a*b)//math.gcd(a,b)

            # l is lcm

                best_trip = (a,b,c)
    return best, best_trip

def min_ratio(limit_sum=200):
    best_ratio = 10.0
    best_trip = None
    for a in range(1, limit_sum):
        for b in range(a+1, limit_sum):
            # compute minimal c such that sum <= limit_sum
            # c must be > b, and sum <= limit_sum
            # Let c_max = limit_sum - a - b
            max_c = limit_sum - a - b
            if max_c <= b:
                continue
            for c in range(b+1, max_c+1):
                # compute ratio
                # skip if gcd(a,b,c) != 1? Let's check scaling invariance.
                # We can include all, ratio unchanged by scaling, but to reduce search we can only consider primitive.
                if math.gcd(a, math.gcd(b, c)) != 1:
                    continue
                l = a

def h_bruteforce_limit(m):
    best = None
    for a in range(1, m-2):
        for b in range(a+1, m-a-1):
            c = m - a - b
            if c <= b:
                continue
            # compute lcm
            l = a
            l = (l*b)//math.gcd(l,b)
            l = (l*c)//math.gcd(l,c)
            if best is None or l < best:
                best = l
    return best

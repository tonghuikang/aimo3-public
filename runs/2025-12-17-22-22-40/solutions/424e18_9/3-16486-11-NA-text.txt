def all_matchings(players):
    # returns list of list of pairs (tuple) covering all players exactly once, unordered pairs.
    # players is a list of distinct items.
    if not players:
        return [[]]
    # pick first player
    first = players[0]
    rest = players[1:]
    matchings = []
    for i, p2 in enumerate(rest):
        pair = (first, p2)
        remaining = rest[:i] + rest[i+1:]
        for sub in all_matchings(remaining):
            matchings.append([pair] + sub)
    return matchings

def winner_subsets(n):
    # n must be even. participants labeled 1..n (higher = faster)
    # generate all perfect matchings and get winner sets (set of max of each pair)
    # Since number of matchings grows quickly, for n=8 it's manageable.
    # We'll generate recursively.
    def rec(remaining):
        if not remaining:
            yield []
            return
        a = remaining[0]
        for i in range(1, len(remaining)):
            b = remaining[i]
            # create a pair (a,b)
            rest = remaining[1:i] + remaining[i+1:]
            for sub in rec(rest):
                yield [(a,b)] + sub
    all_sets = set()
    for m in rec(list(range(1, n+1))):
        winners = frozenset(max(a,b) for a,b in m)
        all_sets.add(winners)
    return all_sets

    # Legendre's formula
    cnt = 0

    while power <= n:
        cnt += n // power
        power *= p
    return cnt

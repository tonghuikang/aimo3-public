def all_matchings(players):
    # Return list of list of pairs (tuple) for perfect matchings of players list (sorted)
    # Players list length even
    if not players:
        return [[]]
    # Take first element, pair with any other
    res=[]
    a = players[0]
    for i in range(1,len(players)):
        b = players[i]
        rest = players[1:i] + players[i+1:]
        for m in all_matchings(rest):
            res.append([(a,b)] + m)
    return res

def tournament_orders(n):
    # n = exponent, number of rounds = n, players = 2^n
    N = 2**n
    players = list(range(1,N+1)) # speed: 1 fastest
    # We'll recursively generate all possible pairings each round.
    def recurse(scores, round_num):
        # scores dict of player->score after previous rounds
        # round_num from 1 to n (inclusive)
        # return set of final ordering tuples from this state onward
        if round_num > n:
            # compute final ordering by descending score
            order = tuple(sorted(players, key=lambda p: scores[p], reverse=True))
            return {order}
        # Determine groups of players with same score
        # Group by score value
        score_to_players = {}
        for p in players:
            s = scores[p]
            score_to_players.setdefault(s, []).append(p)
        # For each group, generate all pairings
        # We'll need to consider Cartesian product of pairings for each group
        group_matchings_lists = [] # list of list of possible matchings per group
        for grp in score_to_players.values():
            # generate all matchings for this group (size always power of 2)
            matchings = all_matchings(sorted(grp))
            group_matchings_lists.append(matchings)
        # Now cartesian product over groups
        final_orders=set()
        # For each combination of matchings across groups
        for combo in itertools.product(*group_matchings_lists):
            # combo is tuple of matchings per group; combine them into a single matching list
            combined = []
            for m in combo:
                combined.extend(m)
            # compute new scores after this round
            new_scores = dict(scores)
            points = 2**(n - round_num) # points awarded for win
            for a,b in combined:
                # winner faster = smaller number
                winner = a if a<b else b
                new_scores[winner] += points
                # loser unchanged
            # Recurse to next round
            final_orders.update(recurse(new_scores, round_num+1))
        return final_orders

    # initial scores all 0
    init_scores = {p:0 for p in players}
    return recurse(init_scores, 1)

But what constraints? After n rounds, each player ends with a unique binary string of length n, representing whether they won each round (1) or lost (0). Alternatively, each player's binary string must correspond to some leaf in a perfect binary tree of depth n.

def pairings(lst):
    # generate all perfect matchings of the list (list of elements)
    # recursion: pick first element, pair with any other, then generate matchings of rest
    if not lst:
        yield []
        return
    first = lst[0]
    for i in range(1, len(lst)):
        second = lst[i]
        rest = lst[1:i] + lst[i+1:]
        for sub in pairings(rest):
            yield [(first, second)] + sub

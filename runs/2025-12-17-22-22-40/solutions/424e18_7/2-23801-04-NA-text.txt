def pairings(lst):
    # return all ways to partition list into unordered pairs (perfect matchings)
    # recursion: pick first element, pair it with each of the others
    if not lst:
        yield []
    else:
        first = lst[0]
        for i in range(1,len(lst)):
            second = lst[i]
            rest = lst[1:i] + lst[i+1:]
            for sub in pairings(rest):
                yield [(first,second)] + sub

def tournament_orderings(num_bits):
    n = 2**num_bits
    # speed ranking: label 0 fastest, 1 second, ..., n-1 slowest
    runners = list(range(n))
    # each runner has unique speed based on label (lower label faster)
    orderings = set()
    # define function to simulate tournament recursively
    def rec(scores, round_idx):
        # scores: dict runner->score
        if round_idx == num_bits:
            # ranking by scores descending; tiebreaker not needed since distinct
            ranking = tuple(sorted(runners, key=lambda r: -scores[r]))
            orderings.add(ranking)
            return
        # group by same score
        groups = {}
        for r in runners:
            s = scores[r]
            groups.setdefault(s, []).append(r)
        # For each group, iterate over all pairings
        # we need to iterate over product of pairings across groups
        # do recursively across groups
        # first list groups keys
        group_keys = list(groups.keys())
        # recursive function across groups
        def rec_groups(idx, new_scores):
            if idx == len(group_keys):
                rec(new_scores, round_idx+1)
                return
            key = group_keys[idx]
            members = groups[key]
            # generate all matchings for this group's members
            for pairing in pairings(members):
                # compute new scores after this group
                scores2 = new_scores.copy()
                # for each pair, assign winner
                for a,b in pairing:
                    # lower label is faster (winner)
                    if a < b:
                        winner, loser = a, b
                    else:
                        winner, loser = b, a
                    scores2[winner] = scores2.get(winner, scores[winner]) + 2**(num_bits-1 - round_idx)

                rec_groups(idx+1, scores2)
        # start recursion over groups
        rec_groups(0, scores.copy())
    # initial scores all zero
    rec({r:0 for r in runners}, 0)
    return orderings

    # return list of list of pairs (each pair a tuple of two players)

    if len(group) == 0:
        return [[]]

    first = group[0]
    result = []
    for i in range(1, len(group)):

        remaining = group[1:i] + group[i+1:]

def all_tournaments(m):
    n = 2**m
    players = list(range(1, n+1))
    # each tournament = list of pairings per round (list of pairs)
    # generate recursively
    def rec(scores, round_i):
        # scores dict mapping player->score
        if round_i > m:
            yield []
            return
        # group players by same score
        groups = collections.defaultdict(list)
        for p in players:
            groups[scores[p]].append(p)
        # for each group compute all possible pairings
        group_pairings = []
        for group in groups.values():
            # need to ensure even size
            if len(group) % 2 != 0:
                raise ValueError('odd group')
            # generate all matchings for this group
            # to avoid huge explosion, maybe we can generate iteratively
            group_pairings.append(all_pairings(group))
        # now combine pairings across groups (Cartesian product)
        for combo in itertools.product(*group_pairings):
            # combo: tuple of lists of pairs from each group
            all_pairs = [pair for sublist in combo for pair in sublist]
            # apply winners and update scores
            new_scores = scores.copy()

                winner = a if a > b else b

                new_scores[winner] += 2**(m - round_i)

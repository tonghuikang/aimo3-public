        random.seed(seed)
    n = 2**k
    speeds = list(range(n))  # 0 (slow) to n-1 (fast)

def all_pairings(group):
    # returns all ways to partition the group (list) into unordered pairs
    # group size is even
    if not group:
        return [[]]
    # take first element
    first = group[0]
    rest = group[1:]
    pairings = []
    for i, partner in enumerate(rest):
        # pair first with partner
        remaining = rest[:i] + rest[i+1:]
        for sub in all_pairings(remaining):
            pairings.append([(first, partner)] + sub)
    return pairings

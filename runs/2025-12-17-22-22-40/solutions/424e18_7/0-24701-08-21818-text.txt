def ballot_winner_subsets(players):
    # players: list sorted descending speed (fastest first)
    n = len(players)
    half = n//2
    # we need subsets of size half such that for any prefix of players (fastest to slowest),
    # number of winners <= number of losers (i.e., prefix condition)
    # Represent as binary string of length n with 1 for winner.
    # Use recursion/backtracking.
    # Could generate all subsets of size half and filter; n up to 8 small.
    subsets=[]
    for comb in itertools.combinations(range(n), half):
        # comb is indices (0..n-1) of winners in players
        # ensure ballot condition
        ok=True
        win_set=set(comb)
        winners_sofar=0
        losers_sofar=0
        for i in range(n):
            if i in win_set:
                winners_sofar+=1
            else:
                losers_sofar+=1
            if winners_sofar>losers_sofar:
                ok=False
                break
        if ok:
            subsets.append([players[i] for i in comb])
    return subsets

def generate_orderings(players):
    # recursively generate all final orderings from tournament.
    # Returns set of permutations (tuple) representing order from highest score to lowest.
    n=len(players)
    if n==1:
        return {tuple(players)}  # only ordering of one element
    # round i: split into winners (size half) and losers (size half)
    orderings=set()
    winner_subsets = ballot_winner_subsets(players)
    # For each possible winners set
    for winners in winner_subsets:
        losers = [p for p in players if p not in winners]
        # Recursively get orderings for winners and losers
        winner_orders = generate_orderings(winners)
        loser_orders = generate_orderings(losers)
        # Final ordering is winners ordering then losers ordering (since winners have higher scores)
        for wo in winner_orders:
            for lo in loser_orders:
                orderings.add(wo+lo)
    return orderings

def v_p_factorial(n, p):
    # Legendre's formula: sum_{i=1}^{∞} floor(n/p^i)
    cnt = 0
    d = p
    while d <= n:
        cnt += n // d
        d *= p
    return cnt

def v_p_catalan(n, p):
    # n is integer
    # C_n = (2n)!/(n!*(n+1)!)
    return v_p_factorial(2*n, p) - v_p_factorial(n, p) - v_p_factorial(n+1, p)

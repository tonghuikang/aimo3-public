def circumcenter(p1, p2, p3):
    # p1,p2,p3 are numpy arrays of shape (2,)
    # Compute circumcenter using perpendicular bisector intersection
    # Using linear algebra: solve for center O = (x,y) such that |O-p1|^2 = |O-p2|^2 = |O-p3|^2
    # Subtract equations to get linear equations
    A = np.array([
        [2*(p2[0]-p1[0]), 2*(p2[1]-p1[1])],
        [2*(p3[0]-p1[0]), 2*(p3[1]-p1[1])]
    ])
    b = np.array([
        p2[0]**2 + p2[1]**2 - p1[0]**2 - p1[1]**2,
        p3[0]**2 + p3[1]**2 - p1[0]**2 - p1[1]**2
    ])
    # Solve A O = b
    # A might be singular if points collinear, but not here
    O = np.linalg.solve(A, b)
    R = np.linalg.norm(O - p1)
    return O, R

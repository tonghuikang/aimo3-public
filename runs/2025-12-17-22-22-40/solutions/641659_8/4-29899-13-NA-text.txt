def circumcenter(P, Q, R):
    # P, Q, R are tuples (x,y)
    (x1,y1) = P
    (x2,y2) = Q
    (x3,y3) = R
    # compute using perpendicular bisectors intersection
    # Solve linear equations:
    # (x - x1)*(x2 - x1) + (y - y1)*(y2 - y1) = ((x2^2 - x1^2)+(y2^2 - y1^2))/2
    # Actually derived from equation of perpendicular bisector: (x - midpoint)*(dx) + (y - midpoint)*(dy) = 0? Let's derive.
    # Better use determinant formula:
    d = 2*(x1*(y2 - y3) + x2*(y3 - y1) + x3*(y1 - y2))
    if abs(d) < 1e-12:
        return None
    ux = ((x1**2 + y1**2)*(y2 - y3) + (x2**2 + y2**2)*(y3 - y1) + (x3**2 + y3**2)*(y1 - y2)) / d
    uy = ((x1**2 + y1**2)*(x3 - x2) + (x2**2 + y2**2)*(x1 - x3) + (x3**2 + y3**2)*(x2 - x1)) / d
    return (ux, uy)

def intersect_circles(O1,R1,O2,R2):
    # returns intersection points (0,1 or 2)
    x1,y1 = O1
    x2,y2 = O2
    dx = x2 - x1
    dy = y2 - y1
    d = math.hypot(dx,dy)
    # no intersection or one
    if d > R1 + R2 + 1e-12 or d < abs(R1 - R2) - 1e-12:
        return []  # no intersections
    # compute a = (R1^2 - R2^2 + d^2) / (2d)
    a = (R1**2 - R2**2 + d**2) / (2*d)
    # height from line to intersection points
    h2 = R1**2 - a**2
    if h2 < -1e-12:
        return []
    h = math.sqrt(max(0.0, h2))
    # point P2 = O1 + a*(O2-O1)/d
    xm = x1 + a*dx/d
    ym = y1 + a*dy/d
    # intersection points
    rx = -dy * (h/d)
    ry = dx * (h/d)
    p1 = (xm + rx, ym + ry)
    p2 = (xm - rx, ym - ry)
    # if h approx 0, return one point
    if h < 1e-12:
        return [p1]
    else:
        return [p1, p2]

    if d > R1 + R2 or d < abs(R1 - R2):
        return []  # no intersection
    a = (R1**2 - R2**2 + d**2) / (2*d)
    h2 = R1**2 - a**2
    if h2 < 0:
        h = mp.mpf('0')
    else:
        h = mp.sqrt(h2)
    xm = x1 + a*dx/d
    ym = y1 + a*dy/d
    rx = -dy * (h/d)
    ry = dx * (h/d)
    p1 = (xm + rx, ym + ry)
    p2 = (xm - rx, ym - ry)
    if h == 0:
        return [p1]
    else:
        return [p1, p2]

def compute_geom_mp(n, s):
    Fn = fib_big(n)
    Fn1 = fib_big(n+1)
    a_len = fib_big(n+2)
    if s <= a_len:
        raise ValueError("s must be > a")
    b_len = s - Fn
    c_len = s - Fn1
    # incircle radius
    r = mp.sqrt((s - a_len) * Fn * Fn1 / s)
    # points
    B = (mp.mpf('0'), mp.mpf('0'))
    C = (a_len, mp.mpf('0'))
    D_pt = (Fn, mp.mpf('0'))
    I_pt = (Fn, r)
    # A coordinates
    u = c_len * (Fn**2 - r**2) / (Fn**2 + r**2)
    v = 2 * c_len * Fn * r / (Fn**2 + r**2)
    A_pt = (u, v)
    # F point
    F_pt = (Fn * (Fn**2 - r**2) / (Fn**2 + r**2), 2 * Fn**2 * r / (Fn**2 + r**2))

    AI_vec = (I_pt[0] - A_pt[0], I_pt[1] - A_pt[1])

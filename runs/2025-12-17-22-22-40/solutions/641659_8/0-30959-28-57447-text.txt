def circumcenter(P1, P2, P3):
    # P are tuples (x,y)
    (x1,y1),(x2,y2),(x3,y3) = P1, P2, P3
    # Compute using determinant method
    d = 2*(x1*(y2 - y3) + x2*(y3 - y1) + x3*(y1 - y2))
    if abs(d) < 1e-12:
        # points collinear
        return None
    ux = ((x1**2 + y1**2)*(y2 - y3) + (x2**2 + y2**2)*(y3 - y1) + (x3**2 + y3**2)*(y1 - y2)) / d
    uy = ((x1**2 + y1**2)*(x3 - x2) + (x2**2 + y2**2)*(x1 - x3) + (x3**2 + y3**2)*(x2 - x1)) / d
    return (ux, uy)

def circle_from_three(P1,P2,P3):
    O = circumcenter(P1,P2,P3)
    if O is None:
        return None, None
    R = math.hypot(P1[0] - O[0], P1[1] - O[1])
    return O, R

def intersect_circles(O1,R1,O2,R2):
    # returns up to two intersection points
    x1,y1 = O1; x2,y2 = O2
    dx = x2 - x1; dy = y2 - y1
    d = math.hypot(dx, dy)
    if d > R1 + R2 + 1e-12 or d < abs(R1 - R2) - 1e-12:
        return []  # no intersection
    # handle d=0 (coincident) not relevant.
    a = (R1**2 - R2**2 + d**2) / (2*d) if d != 0 else 0
    # height from point P2 to intersection points
    h2 = R1**2 - a**2
    if h2 < -1e-12:
        return []
    h = math.sqrt(max(0.0, h2))
    # point P2
    xm = x1 + a * dx / d if d != 0 else x1
    ym = y1 + a * dy / d if d != 0 else y1
    # perpendicular direction
    rx = -dy * (h / d) if d != 0 else 0
    ry = dx * (h / d) if d != 0 else 0
    p1 = (xm + rx, ym + ry)
    p2 = (xm - rx, ym - ry)
    if h < 1e-12:
        return [p1]
    else:
        return [p1,p2]

def foot_of_perp_point(D, P, Q):
    # foot of perpendicular from D onto line PQ
    # D,P,Q are tuples
    x0,y0 = D
    x1,y1 = P
    x2,y2 = Q
    # line vector
    vx = x2 - x1
    vy = y2 - y1
    # projection parameter t
    t = ((x0 - x1)*vx + (y0 - y1)*vy) / (vx*vx + vy*vy)
    xN = x1 + t * vx
    yN = y1 + t * vy
    return (xN, yN)

def reflect_across_line(K, P, Q):
    # reflect K across line PQ
    # line PQ: param eq P + t*v
    # find projection M of K onto line
    xk, yk = K
    x1, y1 = P
    x2, y2 = Q
    vx = x2 - x1
    vy = y2 - y1
    # projection parameter t
    t = ((xk - x1)*vx + (yk - y1)*vy) / (vx*vx + vy*vy)
    mx = x1 + t*vx
    my = y1 + t*vy
    # reflection
    xkp = 2*mx - xk
    ykp = 2*my - yk
    return (xkp, ykp)

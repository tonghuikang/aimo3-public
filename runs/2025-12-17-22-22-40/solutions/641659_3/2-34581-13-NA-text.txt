    # compute circumcenter of triangle defined by points P1,P2,P3
    x1,y1 = P1[0], P1[1]
    x2,y2 = P2[0], P2[1]
    x3,y3 = P3[0], P3[1]
    # Set up linear equations:
    # 2(x2 - x1) x + 2(y2 - y1) y = x2^2 + y2^2 - x1^2 - y1^2
    # 2(x3 - x1) x + 2(y3 - y1) y = x3^2 + y3^2 - x1^2 - y1^2
    A1 = 2*(x2 - x1)
    B1 = 2*(y2 - y1)
    C1 = x2**2 + y2**2 - x1**2 - y1**2

    B2 = 2*(y3 - y1)
    C2 = x3**2 + y3**2 - x1**2 - y1**2

def circle_intersections(O1, r1, O2, r2):
    # return list of intersection points (as mp.matrix) of two circles
    x1,y1 = O1[0], O1[1]
    x2,y2 = O2[0], O2[1]
    dx = x2 - x1
    dy = y2 - y1
    d = mp.sqrt(dx*dx + dy*dy)
    if d == 0:
        return []  # coincident centers
    # Check for no intersection or one intersection
    # Compute a
    a = (r1**2 - r2**2 + d**2) / (2*d)
    # Compute h^2 = r1^2 - a^2
    h2 = r1**2 - a**2
    if h2 < -1e-12:
        return []  # no intersection
    h = mp.sqrt(max(h2, 0))
    # point P2 = O1 + a*(O2-O1)/d
    xm = x1 + a * dx / d
    ym = y1 + a * dy / d
    # offset vector for intersection points
    rx = -dy * (h/d)
    ry = dx * (h/d)
    p1 = mp.matrix([xm + rx, ym + ry])
    p2 = mp.matrix([xm - rx, ym - ry])
    if h < 1e-12:
        return [p1]
    else:
        return [p1, p2]

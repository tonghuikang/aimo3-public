    b = np.linalg.norm(np.subtract(C, A)) # CA opposite B
    c = np.linalg.norm(np.subtract(A, B)) # AB opposite C
    per = a + b + c
    # incenter weighted coordinates
    I = (a*np.array(A) + b*np.array(B) + c*np.array(C))/per
    # radius = area / s
    s = per/2
    # area via cross product
    area = 0.5*abs(np.cross(np.subtract(B, A), np.subtract(C, A)))
    r = area / s
    return I, r, a, b, c

def foot_point(P, Q, R):
    # foot of perpendicular from P to line QR
    # line QR: Q + t*(R-Q)
    Q = np.array(Q); R = np.array(R); P = np.array(P)
    v = R - Q
    t = np.dot(P - Q, v) / np.dot(v, v)
    foot = Q + t*v
    return foot

def circumcenter(P, Q, R):
    # return center and radius of circumcircle
    # using perpendicular bisectors
    # compute using formula from coordinates
    P = np.array(P); Q = np.array(Q); R = np.array(R)
    # matrix approach
    # Solve for center O = (x, y) such that |O-P|^2 = |O-Q|^2 = |O-R|^2
    # Using linear equations subtract eq for P and Q, P and R
    A = np.array([
        2*(Q-P),
        2*(R-P)
    ])
    b = np.array([
        np.dot(Q, Q) - np.dot(P, P),
        np.dot(R, R) - np.dot(P, P)
    ])
    # Solve A O = b
    O = np.linalg.solve(A, b)
    r = np.linalg.norm(O - P)
    return O, r

def intersect_circles(O1, r1, O2, r2):
    # return intersection points of circles (if any)
    # return list of points (np arrays)
    O1 = np.array(O1); O2 = np.array(O2)
    d = np.linalg.norm(O2 - O1)
    if d > r1 + r2 + 1e-9 or d < abs(r1 - r2) - 1e-9:
        return []  # no intersection
    # else find intersection(s)
    # distance from O1 to line joining intersection points
    a = (r1**2 - r2**2 + d**2) / (2*d)
    h_sq = max(r1**2 - a**2, 0)
    h = math.sqrt(h_sq)
    # point P2 is foot from O1 towards O2 at distance a
    P2 = O1 + a * (O2 - O1) / d
    # intersection points:
    # offset perpendicular to O1O2
    # unit perp vector
    perp = np.array([-(O2 - O1)[1], (O2 - O1)[0]]) / d
    inter1 = P2 + h * perp
    inter2 = P2 - h * perp
    return [inter1, inter2]

def angle(a,b,c):
    # angle at b formed by points a,b,c
    v1 = np.array(a) - np.array(b)
    v2 = np.array(c) - np.array(b)
    # compute angle using dot product
    dot = np.dot(v1, v2)
    norm1 = np.linalg.norm(v1)
    norm2 = np.linalg.norm(v2)
    # protect division by zero
    if norm1*norm2 == 0: return 0
    cosang = max(min(dot/(norm1*norm2), 1), -1)
    return math.acos(cosang)

def reflect_point_across_line(P, A, B):
    # reflect point P across line AB
    # formula: projection of P onto line AB, then symmetric
    A = np.array(A); B = np.array(B); P = np.array(P)
    # foot of perpendicular from P onto AB
    foot = foot_point(P, A, B)
    # reflect: P' = 2*foot - P
    return 2*foot - P

def compute_K(A,B,C,F,E):

    O2,r2 = circumcenter(A,F,E)

    for p in pts:
        if np.linalg.norm(p - np.array(A)) > 1e-6:
            return p
    # if both are A (coincident?), return None
    return None

    # angle between line P1-P2 and a direction vector L_dir (unit or not)
    v = np.array(P2) - np.array(P1)

def fib(n):
    # return fibonacci number with F0=0, F1=1
    a,b=0,1
    for _ in range(n):
        a,b=b,a+b
    return a

def foot_point(P, A, B):
    # foot from point P onto line AB (A,B distinct)
    # returns foot point Q
    # Compute projection of vector AP onto AB
    # using formula Q = A + ((P-A).dot(B-A))/|B-A|^2 * (B-A)
    ax,ay = A
    bx,by = B
    px,py = P
    vx,vy = bx-ax, by-ay
    wx,wy = px-ax, py-ay
    dot = vx*wx + vy*wy
    norm2 = vx*vx + vy*vy
    t = dot / norm2
    qx = ax + t*vx
    qy = ay + t*vy
    return (qx,qy)

def reflect_point(P, A, B):
    # reflect point P across line AB
    # compute foot Q of P onto AB, then reflect: P' = 2Q - P
    Q = foot_point(P, A, B)
    return (2*Q[0]-P[0], 2*Q[1]-P[1])

def circumcenter(A,B,C):
    # compute circumcenter of triangle ABC
    (x1,y1),(x2,y2),(x3,y3)=A,B,C
    # Using perpendicular bisector method
    d = 2*(x1*(y2-y3)+x2*(y3-y1)+x3*(y1-y2))
    if abs(d) < 1e-12:
        raise ValueError("Collinear")
    ux = ((x1*x1+y1*y1)*(y2-y3)+(x2*x2+y2*y2)*(y3-y1)+(x3*x3+y3*y3)*(y1-y2))/d
    uy = ((x1*x1+y1*y1)*(x3-x2)+(x2*x2+y2*y2)*(x1-x3)+(x3*x3+y3*y3)*(x2-x1))/d
    return (ux,uy)

def circle_intersections(O1,R1,O2,R2):

    # no intersection if d > R1+R2 or d < |R1-R2|

    if d > R1+R2+1e-12 or d < abs(R1-R2)-1e-12:

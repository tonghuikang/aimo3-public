def tournament_outcome(speed_order, pairing):
    # speed_order: list of participants in order fastest->slowest (i.e., index lower means faster)
    # we will map participant ID to rank (lower rank = faster)
    # participants are IDs 0..n-1
    n = len(speed_order)
    # create rank dict: participant->rank (0 fastest)
    rank = {p:i for i,p in enumerate(speed_order)} # rank lower = faster
    
    # scores initially zero
    scores = {p:0 for p in speed_order}
    # round1 matching: pairing is list of pairs (a,b)
    # In each match, faster wins
    # award points 2^{m-1} where m=2, points=2
    points = [2**(2-1), 2**(2-2)] # round 1 points 2, round2 points 1
    # round 1
    for a,b in pairing:
        # determine winner
        if rank[a] < rank[b]:
            winner, loser = a,b
        else:
            winner, loser = b,a
        scores[winner] += points[0]
        # loser gets nothing
    # after round1, groups by scores: values {0,2}
    # round2: for each score group of size 2, pair them arbitrarily (only one pairing)
    # Determine the two participants with score0, pair them; similarly for score2.
    # In each group, we need to decide pairing ordering: only one way for 2 participants.
    # So just do the matches.
    # find participants by score
    groups = {}
    for p,s in scores.items():
        groups.setdefault(s, []).append(p)
    # round2 matches
    for s, participants in groups.items():
        assert len(participants)==2
        a,b = participants
        # faster wins
        if rank[a] < rank[b]:
            winner, loser = a,b
        else:
            winner, loser = b,a
        scores[winner] += points[1]
    # final ranking by scores descending, ties broken? No ties.
    ranking = sorted(scores.items(), key=lambda x: x[1], reverse=True)
    return tuple(p for p,_ in ranking)

Then for each round i (from 1 to m), for each group of size 2^{m-i+1} (i.e., groups with same prefix bits for previous rounds), pair the participants in the upper half of the group (according to ranking) with those in lower half of the group, matching them arbitrarily (e.g., in order). This yields winners exactly the upper half participants (since they are faster). The process yields exactly the desired final ranking with binary patterns corresponding to the ranking order.

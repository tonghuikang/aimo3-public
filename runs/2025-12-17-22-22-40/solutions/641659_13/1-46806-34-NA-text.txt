def incircle_geom(b, c):
    # points as complex numbers
    D = 1+0j
    B = 1 + 1j*b
    C = 1 - 1j*c
    # Compute F and E coordinates
    xf = (1 - b*b) / (b*b + 1)
    yf = 2*b / (b*b + 1)
    F = xf + 1j*yf
    xe = (1 - c*c) / (c*c + 1)
    ye = -2*c / (c*c + 1)
    E = xe + 1j*ye
    # Compute A as intersection of lines x xf + y yf = 1, x xe + y ye = 1
    # Solve linear equations: real variables x,y
    # Write as matrix [[xf, yf], [xe, ye]] * [x, y]^T = [1,1]
    # Compute via linear solve
    import numpy as np
    M = np.array([[xf, yf], [xe, ye]], dtype=float)
    rhs = np.array([1.0,1.0])
    sol = np.linalg.solve(M, rhs)
    A = sol[0] + 1j*sol[1]
    # Return points
    return A, B, C, D, E, F

def intersect_circles(c1,r1,c2,r2):
    # returns intersection points as complex numbers (list of 0,1,2 points)
    # Using geometry: distance d between centers
    d = abs(c2 - c1)
    # No intersection if d > r1 + r2 or d < |r1 - r2|
    if d > r1 + r2 + 1e-12 or d < abs(r1 - r2) - 1e-12:
        return []
    # Solve via circle-circle intersection formula
    # ex = (c2 - c1) / d
    ex = (c2 - c1) / d
    # a = (r1^2 - r2^2 + d^2) / (2d)
    a = (r1*r1 - r2*r2 + d*d) / (2*d)
    # h = sqrt(r1^2 - a^2)
    h_sq = r1*r1 - a*a
    if h_sq < 0: h_sq = 0
    h = math.sqrt(h_sq)
    # point p2 = c1 + a * ex
    p2 = c1 + a * ex
    # intersection points:
    # offset = h * ( -ex.imag + i*ex.real )
    offset = h * (-ex.imag + 1j*ex.real)
    inter1 = p2 + offset
    inter2 = p2 - offset
    if h < 1e-12:
        return [inter1]  # tangent
    else:
        return [inter1, inter2]

def compute_K(b,c):
    A,B,C,D,E,F = incircle_geom(b,c)

    O_abc, R_abc = circle_through_three(A,B,C)
    # circle of A,E,F

    pts = intersect_circles(O_abc,R_abc,O_aef,R_aef)

def foot_of_perp(P, line_pt1, line_pt2):
    # foot of perpendicular from point P to line defined by two points (line_pt1, line_pt2)
    # Using projection formula
    v = line_pt2 - line_pt1
    t = ((P - line_pt1).conjugate() * v).real / (abs(v)**2)  # projection scalar onto v (real)
    proj = line_pt1 + t * v
    # foot of perpendicular: projection onto line.
    return proj

def foot_of_perp_to_line(P, L1, L2):
    # return foot of perpendicular from P to line L1-L2 (complex)
    # Use formula: foot = L1 + ((P-L1).dot(v) / |v|^2) * v, where v = L2 - L1, dot as real inner product.
    v = L2 - L1
    # inner product as real: Re((P - L1) * conj(v))
    t = ( (P - L1) * v.conjugate() ).real / (abs(v)**2)
    foot = L1 + t * v
    return foot

def reflect_across_line(P, L1, L2):

def is_cyclic(P, Q, R, S):
    # test if four points are concyclic: compute cross ratio of directed angles? Use determinant.
    # Condition: oriented area of quadrilateral? Use complex condition: (p - q)(r - s) / ((p - s)(r - q)) is real.
    # For points p,q,r,s, they are concyclic iff cross ratio (p,q;r,s) is real.
    p,q,r,s = P,Q,R,S
    val = (p - q)*(r - s) / ((p - s)*(r - q))
    return abs(val.imag) < 1e-6

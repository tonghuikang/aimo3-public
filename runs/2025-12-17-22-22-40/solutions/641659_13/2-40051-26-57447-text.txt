def solve2x2(A, b):
    # A is list of lists [[a11,a12],[a21,a22]], b is [b1,b2]
    a11, a12 = A[0]
    a21, a22 = A[1]
    det = a11*a22 - a12*a21
    if abs(det) < 1e-15:
        raise ValueError("singular")
    x = (b[0]*a22 - a12*b[1]) / det
    y = (a11*b[1] - b[0]*a21) / det
    return x, y

    c_max = max(10*a, d*10 + 10)  # start with moderate

    prev_c = cs[0]

    interval = None
    for ci in cs[1:]:
        val = f_cross(ci)
        if val is None or prev_val is None:

        if val == 0:

        if prev_val * val < 0:
            interval = (prev_c, ci)

                prev_c, prev_val = ci, val
            if interval is not None:
                break
    if interval is None:
        # No root found
        return None
    # now root refine using bisection
    a_int, b_int = interval
    # ensure f(a_int) and f(b_int) have opposite signs
    fa = f_cross(a_int)
    fb = f_cross(b_int)
    if fa is None or fb is None:
        return None
    # Bisection loop
    for _ in range(60):
        mid = (a_int + b_int) / 2
        fm = f_cross(mid)
        if fm is None:

            a_int = mid
            continue

            fb = fm
        else:
            a_int = mid
            fa = fm
    c_sol = (a_int + b_int) / 2
    # compute final geometry at c_sol
    result = geometry_vals(c_sol)
    return result, c_sol

from math import sqrt

def circumcenter(A,B,C):
    # compute circumcenter of triangle ABC
    # returns (ox, oy)
    # using perpendicular bisector intersection formula
    (x1,y1)=A
    (x2,y2)=B
    (x3,y3)=C
    # compute determinant
    d = 2*(x1*(y2-y3) + x2*(y3-y1) + x3*(y1-y2))
    if abs(d) < 1e-12:
        return None
    ux = ((x1**2+y1**2)*(y2-y3) + (x2**2+y2**2)*(y3-y1) + (x3**2+y3**2)*(y1-y2))/d
    uy = ((x1**2+y1**2)*(x3-x2) + (x2**2+y2**2)*(x1-x3) + (x3**2+y3**2)*(x2-x1))/d
    return (ux, uy)

def intersect_line_circle(P0, dir_vec, center, radius):
    # line: P = P0 + t*dir_vec, circle: |P - center| = radius
    # solve quadratic for t: |P0 + t*v - O|^2 = R^2
    (x0,y0) = P0
    (dx,dy) = dir_vec
    (ox,oy) = center
    # coefficients a t^2 + b t + c = 0
    a = dx*dx + dy*dy
    b = 2*(dx*(x0-ox) + dy*(y0-oy))
    c = (x0-ox)**2 + (y0-oy)**2 - radius*radius
    disc = b*b - 4*a*c
    if disc < -1e-12:
        return []  # no intersection
    disc = max(disc,0.0)
    sqrt_disc = sqrt(disc)
    t1 = (-b + sqrt_disc)/(2*a)
    t2 = (-b - sqrt_disc)/(2*a)
    return [(x0+dx*t1, y0+dy*t1), (x0+dx*t2, y0+dy*t2)]

def foot_of_perpendicular(P, A, B):
    # foot from point P to line AB
    # line AB param: A + t*(B-A)
    (px,py) = P
    (ax,ay) = A
    (bx,by) = B
    vx = bx - ax
    vy = by - ay
    # projection t = ((P-A)·v) / (v·v)
    t = ((px-ax)*vx + (py-ay)*vy) / (vx*vx + vy*vy)
    # foot
    return (ax + t*vx, ay + t*vy)

def reflect_point_across_line(P, A, B):
    # reflect point P across line through A, B
    # find foot H of perpendicular from P onto line AB
    H = foot_of_perpendicular(P, A, B)
    # reflection: P' = 2*H - P
    return (2*H[0]-P[0], 2*H[1]-P[1])

def angle_between(v1, v2):
    # returns angle between vectors v1 and v2 in radians between 0 and pi
    (x1,y1) = v1
    (x2,y2) = v2
    dot = x1*x2 + y1*y2
    norm1 = sqrt(x1*x1 + y1*y1)
    norm2 = sqrt(x2*x2 + y2*y2)
    if norm1*norm2 == 0:
        return 0.0
    val = max(min(dot/(norm1*norm2),1.0),-1.0)
    return math.acos(val)

    v1 = (P[0]-Q[0], P[1]-Q[1])
    v2 = (R[0]-Q[0], R[1]-Q[1])

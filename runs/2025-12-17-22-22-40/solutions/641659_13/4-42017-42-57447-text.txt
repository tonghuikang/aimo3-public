def circumcenter(p1, p2, p3):
    # Return center of circumcircle of three points p1,p2,p3 and radius
    # points are tuples (x,y)
    (x1,y1),(x2,y2),(x3,y3) = p1,p2,p3
    # Compute using perpendicular bisectors method (determinants)
    d = 2*(x1*(y2-y3) + x2*(y3-y1) + x3*(y1-y2))
    if abs(d) < 1e-12:
        return None, None
    ux = ((x1**2 + y1**2)*(y2-y3) + (x2**2 + y2**2)*(y3-y1) + (x3**2 + y3**2)*(y1-y2)) / d
    uy = ((x1**2 + y1**2)*(x3-x2) + (x2**2 + y2**2)*(x1-x3) + (x3**2 + y3**2)*(x2-x1)) / d
    R = math.hypot(ux-x1, uy-y1)
    return (ux, uy), R

def intersect_circles(O1,R1,O2,R2):
    # Return intersection points of two circles (list of tuples)
    x1,y1 = O1
    x2,y2 = O2
    d = math.hypot(x2-x1, y2-y1)
    if d > R1+R2+1e-12 or d < abs(R1-R2)-1e-12 or d == 0:
        # no intersection or infinite
        return []
    # a = (R1^2 - R2^2 + d^2) / (2d)
    a = (R1**2 - R2**2 + d**2) / (2*d)
    # height
    h_sq = R1**2 - a**2
    if h_sq < -1e-12:
        return []
    h = math.sqrt(max(0,h_sq))
    # point P2
    x3 = x1 + a*(x2-x1)/d
    y3 = y1 + a*(y2-y1)/d
    # offset vector perpendicular to line O1O2
    rx = -(y2-y1)*(h/d)
    ry = (x2-x1)*(h/d)
    p_int1 = (x3+rx, y3+ry)
    p_int2 = (x3-rx, y3-ry)
    return [p_int1, p_int2]

def reflect_point_across_line(P, L1, L2):
    # reflect point P across line through L1 and L2
    # compute projection onto line
    x0,y0 = P
    x1,y1 = L1
    x2,y2 = L2
    # direction unit vector of line
    dx = x2 - x1
    dy = y2 - y1
    norm = dx*dx + dy*dy
    if norm == 0:
        raise ValueError("Line points coincide")
    # projection scalar
    t = ((x0-x1)*dx + (y0-y1)*dy) / norm
    projx = x1 + t*dx
    projy = y1 + t*dy
    # reflection
    rx = 2*projx - x0
    ry = 2*projy - y0
    return (rx, ry)

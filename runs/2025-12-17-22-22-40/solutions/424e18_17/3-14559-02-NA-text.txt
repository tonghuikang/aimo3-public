    # group: list of participants (e.g., ints)
    # returns list of list of pairs (list of tuples) that partition group into unordered pairs.
    # For group size m (even), number of perfect matchings = (m-1)!! (double factorial)
    # We can generate recursively.
    if not group:
        return [[]]
    # Choose first element, pair with any other
    first = group[0]
    rest = group[1:]
    matchings = []
    for i, other in enumerate(rest):
        # pair first with other

        # remaining elements: remove other
        remaining = rest[:i] + rest[i+1:]
        for sub in generate_matchings(remaining):
            matchings.append([pair] + sub)
    return matchings

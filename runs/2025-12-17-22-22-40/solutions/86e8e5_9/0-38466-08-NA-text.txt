def smallest_divisor_not_div_by3(n):
    # return smallest divisor d>=3, d not divisible by 3, dividing n.
    # If none, return None.
    # we can iterate from 3 up to sqrt(n)
    # skip multiples of 3

    # If n itself is >=3 and n%3 !=0, then n is divisor

    return None

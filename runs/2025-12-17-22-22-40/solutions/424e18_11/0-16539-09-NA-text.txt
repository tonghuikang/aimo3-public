def generate_pairings(group):
    # group list
    if not group:
        return [[]]
    first = group[0]
    rest = group[1:]
    result = []
    for i, partner in enumerate(rest):
        new_rest = rest[:i] + rest[i+1:]
        for sub in generate_pairings(new_rest):
            result.append([(first, partner)] + sub)
    return result

def tournament_outcomes(num_participants):

            outcomes.add(ranking)

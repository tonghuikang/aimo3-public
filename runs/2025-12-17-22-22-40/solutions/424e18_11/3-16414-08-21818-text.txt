    n = len(group)

    # Generate all subsets of size half that satisfy prefix condition (no prefix more winners than losers)

    # generate combinations of half indices, then test condition.

        # quick test prefix condition
        ok = True
        subset_set = set(subset_idx)
        win_count = 0
        for i in range(n):

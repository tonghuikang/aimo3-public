def f_brutal_opt_limit(n):
    best=None
    best_trip=None
    for a in range(1,n//3+1):
        for b in range(a+1,(n-a)//2+1):
            c=n-a-b
            if c<=b: continue
            l=math.lcm(a,b,c)
            if best is None or l<best:
                best=l
                best_trip=(a,b,c)
    return best,best_trip

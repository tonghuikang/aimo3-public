    f_val = None
    f_trip = None
    from math import gcd
    def lcm(x, y):
        return x // gcd(x, y) * y
    for a in range(1, S//3+1):
        for b in range(a+1, (S - a)//2 + 1):
            c = S - a - b
            if c <= b:
                continue
            L = lcm(lcm(a,b),c)
            if f_val is None or L < f_val:
                f_val = L
                f_trip = (a,b,c)
    return f_val, f_trip

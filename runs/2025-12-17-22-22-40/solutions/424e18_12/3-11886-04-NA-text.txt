def random_tournament(n):
    num = 2**n
    # participants: speeds 0..num-1 (higher faster)
    scores = [0]*num
    for i in range(n): # round i from 0..n-1
        pts = 2**(n-1-i)
        # group indices by current score
        groups = {}
        for idx, sc in enumerate(scores):
            groups.setdefault(sc, []).append(idx)
        # For each group, random perfect matching
        new_scores = scores[:]
        for sc, idxs in groups.items():
            random.shuffle(idxs)
            # pair them
            for j in range(0, len(idxs), 2):
                a, b = idxs[j], idxs[j+1]
                winner = a if a > b else b # higher index faster
                new_scores[winner] += pts
        scores = new_scores
    return sorted(scores) # sorted final scores

    return outcomes

def all_pairings_for_group(group):
    # returns list of matchings (list of pairs)
    if not group:
        return [[]]
    first = group[0]
    rest = group[1:]
    result = []
    for i, second in enumerate(rest):
        pair = (first, second)
        remaining = rest[:i] + rest[i+1:]
        for sub in all_pairings_for_group(remaining):
            result.append([pair] + sub)
    return result

def find_root_c(n, c_min_factor=1.001, c_max_factor=500, steps=2000):
    d = fib(n)
    a = fib(n+2)
    c_min = d * c_min_factor
    c_max = d * c_max_factor
    # sample many points to find sign change
    cs = []
    rs = []
    for i in range(steps+1):
        c = c_min + (c_max - c_min) * i / steps
        out = compute_triangle(c, d, a)
        if out is None:
            continue
        cs.append(c)
        rs.append(out['cyclic_residual'])
    # find intervals
    intervals = []
    for i in range(1, len(cs)):
        if rs[i-1] * rs[i] < 0:
            intervals.append((cs[i-1], cs[i]))
    # refine each interval
    solutions = []
    for lo,hi in intervals:
        for _ in range(50):
            mid = (lo+hi)/2
            out_mid = compute_triangle(mid, d, a)
            if out_mid is None:
                break
            fmid = out_mid['cyclic_residual']
            out_lo = compute_triangle(lo, d, a)
            flo = out_lo['cyclic_residual']
            if flo * fmid <= 0:
                hi = mid
            else:
                lo = mid
        sol = compute_triangle((lo+hi)/2, d, a)
        if sol:
            solutions.append(sol)
    return solutions

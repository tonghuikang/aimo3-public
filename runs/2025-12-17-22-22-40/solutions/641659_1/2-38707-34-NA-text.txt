def circumcenter(P1, P2, P3):
    # returns (ox, oy, r)
    x1, y1 = P1
    x2, y2 = P2
    x3, y3 = P3
    # Use determinant method
    d = 2*(x1*(y2 - y3) + x2*(y3 - y1) + x3*(y1 - y2))
    if abs(d) < 1e-12:
        raise ValueError("Collinear points")
    ux = ((x1**2 + y1**2)*(y2 - y3) + (x2**2 + y2**2)*(y3 - y1) + (x3**2 + y3**2)*(y1 - y2)) / d
    uy = ((x1**2 + y1**2)*(x3 - x2) + (x2**2 + y2**2)*(x1 - x3) + (x3**2 + y3**2)*(x2 - x1)) / d
    r = mp.sqrt((ux - x1)**2 + (uy - y1)**2)
    return (ux, uy, r)

def circle_intersections(O1, r1, O2, r2):
    x1, y1 = O1
    x2, y2 = O2
    dx = x2 - x1
    dy = y2 - y1
    d = mp.sqrt(dx*dx + dy*dy)
    if d == 0:
        raise ValueError("Concentric circles")
    # check for intersection
    if d > r1 + r2 + 1e-12 or d < abs(r1 - r2) - 1e-12:
        return []  # no intersection
    a = (r1*r1 - r2*r2 + d*d) / (2*d)
    h_sq = r1*r1 - a*a
    if h_sq < 0:
        h = 0
    else:
        h = mp.sqrt(h_sq)
    # point P2 = O1 + a*(O2 - O1)/d
    xm = x1 + a*dx/d
    ym = y1 + a*dy/d
    # intersection points:
    rx = -dy * (h/d)
    ry = dx * (h/d)
    p1 = (xm + rx, ym + ry)
    p2 = (xm - rx, ym - ry)
    return [p1, p2]

def reflect_across_line(P, A, B):
    # reflect point P across line through A,B
    # line equation via vector AB
    ax, ay = A
    bx, by = B
    # compute projection of P onto line AB
    # using formula for reflection: P' = 2*proj - P
    # proj = A + ((P-A).dot(AB) / |AB|^2) * AB
    ABx = bx - ax
    ABy = by - ay
    APx = P[0] - ax
    APy = P[1] - ay
    dot = APx*ABx + APy*ABy
    len2 = ABx*ABx + ABy*ABy
    projx = ax + dot*ABx/len2
    projy = ay + dot*ABy/len2
    # reflection:
    Rx = 2*projx - P[0]
    Ry = 2*projy - P[1]
    return (Rx, Ry)

def foot_of_perpendicular(P, A, B):
    # foot from P to line AB
    # same as projection of P onto AB
    ax, ay = A
    bx, by = B
    ABx = bx - ax
    ABy = by - ay
    APx = P[0] - ax
    APy = P[1] - ay
    dot = APx*ABx + APy*ABy
    len2 = ABx*ABx + ABy*ABy
    t = dot/len2
    footx = ax + t*ABx
    footy = ay + t*ABy
    return (footx, footy)

s2 = (a*dy - dx*c)/det = (-BNy*dy - dx*BNx) / det.

Thus s1 formula.

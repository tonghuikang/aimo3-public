def compute_triangle(n, s):
    # Fibonacci numbers
    # Using standard definition F0=0, F1=1
    # compute upto n+2
    fib = [0,1]
    for i in range(2,n+3):
        fib.append(fib[-1]+fib[-2])
    Fn = fib[n]
    Fn1 = fib[n+1]
    Fn2 = fib[n+2]
    # side lengths
    a = Fn2
    b = s - Fn
    c = s - Fn1
    # check triangle inequalities
    if not (b + c > a and a + b > c and a + c > b):
        raise ValueError("Triangle inequality fails")
    # coordinates
    B = sp.Point(0,0)
    C = sp.Point(a,0)
    # compute x_A:
    x_A = (a**2 - b**2 + c**2) / (2*a)
    # y_A
    # using squared length, ensure non-negative
    y_sq = c**2 - x_A**2
    if y_sq <= 0:
        raise ValueError("non-positive y^2")
    y_A = math.sqrt(float(y_sq))
    A = sp.Point(x_A, y_A)
    # area via Heron
    ssem = (a+b+c)/2
    area = math.sqrt(ssem*(ssem-a)*(ssem-b)*(ssem-c))
    r = area / ssem
    # incircle center should be (Fn, r)
    I = sp.Point(Fn, r)
    # contact points
    # F on AB with BF = Fn
    # AB length = c
    F = sp.Point(Fn/c * A.x, Fn/c * A.y)  # B + (Fn/c)*(A - B) = (Fn/c) * A
    # E on AC with CE = Fn1
    t = Fn1 / b
    E = sp.Point(C.x + t*(A.x - C.x), C.y + t*(A.y - C.y))
    # foot N from D to line EF
    D = sp.Point(Fn, 0)
    # line EF: define as line
    lineEF = sp.Line(E, F)
    N = lineEF.projection(D)  # foot of perpendicular
    # circles
    # Circle through A, F, E
    circ1 = sp.Circle(A, F, E)
    # Circumcircle of ABC
    circ2 = sp.Circle(A, B, C)
    # Intersection of circles
    inters = circ1.intersection(circ2)
    # inters includes A and K
    # filter out A
    K_candidates = [pt for pt in inters if not pt.equals(A)]
    if len(K_candidates)==0:
        raise ValueError("No second intersection")
    K = K_candidates[0]
    # reflect K across line EF
    # reflection: compute foot of perpendicular onto EF then reflect
    # or use sympy reflect method?
    K_ref = K.reflect(lineEF)
    Kp = K_ref
    # Check cyclic condition for B, K, N, Kp
    Bpt = B
    # Find circumcircle of B, K, N
    circ_BKN = sp.Circle(Bpt, K, N)
    # Check if Kp lies on it (distance from center equals radius)
    on_circle = abs(circ_BKN.center.distance(Kp) - circ_BKN.radius) < 1e-8
    return {
        'Fn':Fn, 'Fn1':Fn1, 'Fn2':Fn2,
        's':s,
        'a':a, 'b':b, 'c':c,
        'A':A, 'B':B, 'C':C, 'I':I, 'F':F, 'E':E,
        'D':D, 'N':N, 'K':K, 'Kp':Kp,
        'cyclic':on_circle,
        'circ_BKN_center':circ_BKN.center,
        'circ_BKN_radius':circ_BKN.radius
    }

    A = np.array(p1, dtype=float)

    M = np.column_stack((n1, -n2))  # 2x2 matrix

def circle_intersections(O1, r1, O2, r2):

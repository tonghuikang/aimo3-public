def geometry_compute(t, lam):
    # incircle radius = 1
    # Points:
    D = mp.matrix([1, 0])
    B = mp.matrix([1, t])
    C = mp.matrix([1, -lam*t])
    # angles
    beta = 2*mp.atan(t)
    gamma = 2*mp.atan(lam*t)
    # contact points on incircle
    F = mp.matrix([mp.cos(beta), mp.sin(beta)])
    E = mp.matrix([mp.cos(gamma), -mp.sin(gamma)])  # angle -gamma
    # lines AB: x cos(beta) + y sin(beta) = 1
    # line AC: x cos(gamma) - y sin(gamma) = 1
    # solve for A
    # Solve linear system:
    # [cos(beta), sin(beta)] [x; y] = 1
    # [cos(gamma), -sin(gamma)] [x; y] = 1
    M = mp.matrix([[mp.cos(beta), mp.sin(beta)],
                   [mp.cos(gamma), -mp.sin(gamma)]])
    rhs = mp.matrix([1,1])
    A = mp.linalg.solve(M, rhs)
    # Compute circumcenter of ABC (Omega)
    # Use general formula: find circle through three points A,B,C
    # Solve for center (ox, oy) and radius squared R2.
    # (x - ox)^2 + (y - oy)^2 = R2 for each point
    # Subtract equations to eliminate R2.
    # Using linear algebra.
    # Solve:
    # 2*(B - A)·O = |B|^2 - |A|^2
    # 2*(C - A)·O = |C|^2 - |A|^2
    # O = (ox, oy)
    # Setup matrix eq: 
    BA = B - A
    CA = C - A
    M2 = mp.matrix([[2*BA[0], 2*BA[1]],
                    [2*CA[0], 2*CA[1]]])
    rhs2 = mp.matrix([mp.norm(B)**2 - mp.norm(A)**2,
                      mp.norm(C)**2 - mp.norm(A)**2])
    O = mp.linalg.solve(M2, rhs2)
    # Compute radius squared if needed
    R2 = (A[0] - O[0])**2 + (A[1] - O[1])**2
    # Circle gamma: through A, E, F
    # Solve similarly for its center G and radius
    # Points A, E, F
    EA = E - A
    FA = F - A
    M3 = mp.matrix([[2*EA[0], 2*EA[1]],

    NE = mp.sqrt((N[0] - E[0])**2 + (N[1] - E[1])**2)
    # Ratio:
    ratio = (CT * NB) / (BT * NE)
    return det_val, ratio, {"A":A, "B":B, "C":C, "D":D, "E":E, "F":F, "K":K, "Kprime":Kprime, "N":N, "T":T}

def solve2x2(a11,a12,a21,a22,b1,b2):
    det = a11*a22 - a12*a21
    if abs(det) < mp.mpf('1e-30'):
        raise ValueError("Determinant too small")
    x = (b1*a22 - a12*b2) / det
    y = (a11*b2 - b1*a21) / det
    return x, y

    # incircle radius = 1
    D = mp.matrix([1,0])
    B = mp.matrix([1,t])
    C = mp.matrix([1,-lam*t])
    beta = 2*mp.atan(t)
    gamma = 2*mp.atan(lam*t)
    # contact points
    F = mp.matrix([mp.cos(beta), mp.sin(beta)])
    E = mp.matrix([mp.cos(gamma), -mp.sin(gamma)])
    # compute A solving linear system
    c1 = mp.cos(beta); s1 = mp.sin(beta)
    c2 = mp.cos(gamma); s2 = mp.sin(gamma)
    # eq1: c1*x + s1*y = 1
    # eq2: c2*x - s2*y = 1
    # Solve:
    # Using formula above
    # y = (c2 - c1) / (c2*s1 + s2*c1)
    denom_y = c2*s1 + s2*c1
    if abs(denom_y) < mp.mpf('1e-30'):
        raise ValueError("Denom y zero")
    yA = (c2 - c1) / denom_y
    xA = (1 - s1*yA) / c1
    A = mp.matrix([xA, yA])
    # circumcenter of ABC
    # Solve O = (ox, oy)
    # Equations: (B-O)^2 = (A-O)^2 and (C-O)^2 = (A-O)^2
    # Equivalent: 2*(B-A)·O = |B|^2 - |A|^2
    #            2*(C-A)·O = |C|^2 - |A|^2
    BA = B - A
    CA = C - A
    a11 = 2*BA[0]; a12 = 2*BA[1]; b1 = (mp.norm(B)**2 - mp.norm(A)**2)
    a21 = 2*CA[0]; a22 = 2*CA[1]; b2 = (mp.norm(C)**2 - mp.norm(A)**2)
    ox, oy = solve2x2(a11,a12,a21,a22,b1,b2)
    O = mp.matrix([ox, oy])
    R2 = (A[0]-ox)**2 + (A[1]-oy)**2
    # circle through A, E, F
    EA = E - A
    FA = F - A
    a11 = 2*EA[0]; a12 = 2*EA[1]; b1 = (mp.norm(E)**2 - mp.norm(A)**2)
    a21 = 2*FA[0]; a22 = 2*FA[1]; b2 = (mp.norm(F)**2 - mp.norm(A)**2)
    gx, gy = solve2x2(a11,a12,a21,a22,b1,b2)
    G = mp.matrix([gx, gy])
    Rg2 = (A[0]-gx)**2 + (A[1]-gy)**2
    # intersection of circles O,R2 and G,Rg2
    # distance between centers:
    d = mp.sqrt((ox-gx)**2 + (oy-gy)**2)
    # a = (R2 - Rg2 + d^2) / (2d)
    a_val = (R2 - Rg2 + d**2) / (2*d)
    # h^2 = R2 - a^2
    h2 = R2 - a_val**2
    if h2 < 0:
        # small negative due to numeric
        h2 = max(h2, 0)
    h = mp.sqrt(h2)
    # point P2
    P2x = ox + a_val*(gx-ox)/d
    P2y = oy + a_val*(gy-oy)/d
    # offset
    offset_x = -(gy - oy) * (h/d)
    offset_y = (gx - ox) * (h/d)
    X1 = mp.matrix([P2x + offset_x, P2y + offset_y])
    X2 = mp.matrix([P2x - offset_x, P2y - offset_y])
    # choose K as point not equal to A
    if mp.norm(X1 - A) < mp.mpf('1e-12'):
        K = X2
    elif mp.norm(X2 - A) < mp.mpf('1e-12'):
        K = X1
    else:
        # pick one that is not A (choose far)
        if mp.norm(X1 - A) > mp.norm(X2 - A):
            K = X1
        else:
            K = X2
    # reflection K' across line EF
    u = F - E
    u_norm_sq = u[0]**2 + u[1]**2
    dotK = (K[0] - E[0])*u[0] + (K[1] - E[1])*u[1]
    proj = mp.matrix([E[0] + dotK/u_norm_sq * u[0],
                      E[1] + dotK/u_norm_sq * u[1]])
    Kprime = mp.matrix([2*proj[0] - K[0], 2*proj[1] - K[1]])
    # foot N from D onto EF
    dotD = (D[0] - E[0])*u[0] + (D[1] - E[1])*u[1]
    N = mp.matrix([E[0] + dotD/u_norm_sq * u[0],
                    E[1] + dotD/u_norm_sq * u[1]])
    # determinant for concyclicity
    def det4(P, Q, R, S):
        M = mp.matrix([[P[0], P[1], P[0]**2+P[1]**2, 1],
                       [Q[0], Q[1], Q[0]**2+Q[1]**2, 1],
                       [R[0], R[1], R[0]**2+R[1]**2, 1],
                       [S[0], S[1], S[0]**2+S[1]**2, 1]])
        return mp.det(M)
    det_val = det4(B, K, Kprime, N)
    # circle tangent to BN at B through K
    BN = N - B
    # perpendicular direction n = (-BN_y, BN_x)
    n_vec = mp.matrix([-BN[1], BN[0]])
    # solve t0 = -|BK|^2 / (2 * n·(B - K))
    BK = B - K
    numerator = - (BK[0]**2 + BK[1]**2)
    denominator = 2 * (n_vec[0]*BK[0] + n_vec[1]*BK[1])
    t0 = numerator / denominator
    Oprime = mp.matrix([B[0] + t0*n_vec[0], B[1] + t0*n_vec[1]])
    Rprime = mp.sqrt((Oprime[0] - B[0])**2 + (Oprime[1] - B[1])**2)
    # intersect with x=1
    d2 = Rprime**2 - (1 - Oprime[0])**2
    if d2 < 0:
        d2 = max(d2, mp.mpf('0'))
    dval = mp.sqrt(d2)
    y_T1 = Oprime[1] + dval
    y_T2 = Oprime[1] - dval
    # choose the one not equal to B.y
    if mp.almosteq(y_T1, B[1], rel_eps=mp.mpf('1e-12')):
        y_T = y_T2
    else:
        y_T = y_T1
    T = mp.matrix([1, y_T])
    BT = abs(y_T - B[1])  # vertical distance
    CT = abs(y_T - C[1])
    NB = mp.sqrt((N[0] - B[0])**2 + (N[1] - B[1])**2)
    NE = mp.sqrt((N[0] - E[0])**2 + (N[1] - E[1])**2)
    ratio = (CT * NB) / (BT * NE)
    return det_val, ratio, {"A":A,"B":B,"C":C,"D":D,"E":E,"F":F,"K":K,"Kprime":Kprime,"N":N,"T":T}

def circumcenter(P1, P2, P3):
    # solve for O: |O-P1|^2 = |O-P2|^2 and |O-P1|^2 = |O-P3|^2
    # This yields linear equations:
    # 2*(P2-P1)·O = |P2|^2 - |P1|^2
    # 2*(P3-P1)·O = |P3|^2 - |P1|^2
    P1 = np.asarray(P1, dtype=float)
    P2 = np.asarray(P2, dtype=float)
    P3 = np.asarray(P3, dtype=float)
    A = np.vstack([2*(P2-P1), 2*(P3-P1)])
    b = np.array([np.dot(P2,P2) - np.dot(P1,P1), np.dot(P3,P3) - np.dot(P1,P1)], dtype=float)
    # solve A O = b
    # Use np.linalg.lstsq
    O, residuals, rank, s = np.linalg.lstsq(A, b, rcond=None)
    return O

def circle_intersections(O1, r1, O2, r2):
    O1 = np.asarray(O1, dtype=float)
    O2 = np.asarray(O2, dtype=float)
    d = np.linalg.norm(O2 - O1)
    if d < 1e-12:
        return []  # same center
    # check for no intersection
    if d > r1 + r2 + 1e-12 or d < abs(r1 - r2) - 1e-12:
        return []  # no intersection
    # point P2
    a = (r1**2 - r2**2 + d**2) / (2*d)
    P2 = O1 + a * (O2 - O1) / d
    # height h
    # handle floating errors
    h_sq = max(r1**2 - a**2, 0.0)
    h = np.sqrt(h_sq)
    # unit vector perpendicular to O2-O1
    # rotate (dx,dy) by 90 deg: (-dy, dx)
    dx, dy = (O2 - O1) / d
    perp = np.array([-dy, dx])
    inter1 = P2 + h * perp
    inter2 = P2 - h * perp
    # If h ~0, both same
    if h < 1e-12:
        return [inter1]
    else:
        return [inter1, inter2]

def circle_intersections(c1, r1, c2, r2):
    # Return intersection points of two circles, if any.
    (x0, y0), (x1, y1) = c1, c2
    dx = x1 - x0
    dy = y1 - y0
    d = math.hypot(dx, dy)
    # No solutions if d > r0 + r1 or d < |r0 - r1|
    if d > r1 + r2 + 1e-12 or d < abs(r1 - r2) - 1e-12:
        return []  # no intersection
    # distance from c0 to point p2 where line through intersection points crosses line of centers
    a = (r1**2 - r2**2 + d**2) / (2*d)
    h_sq = max(r1**2 - a**2, 0.0)
    h = math.sqrt(h_sq)
    # point p2
    x2 = x0 + a * dx / d
    y2 = y0 + a * dy / d
    # intersection points
    rx = -dy * (h / d)
    ry = dx * (h / d)
    p_int1 = (x2 + rx, y2 + ry)
    p_int2 = (x2 - rx, y2 - ry)
    return [p_int1, p_int2]

def cyclic_det_points(P1, P2, P3, P4):
    # compute determinant of matrix [x^2+y^2, x, y, 1] for points
    M = []
    for (x,y) in [P1,P2,P3,P4]:

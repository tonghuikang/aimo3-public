def all_matchings(lst):
    # returns list of matchings (list of tuple pairs) for given list of items (even length)
    if not lst:
        return [[]]
    first = lst[0]
    rest = lst[1:]
    matchings = []
    for i, other in enumerate(rest):
        # pair first with other
        remaining = rest[:i] + rest[i+1:]
        for sub in all_matchings(remaining):
            matchings.append([(first, other)] + sub)
    return matchings

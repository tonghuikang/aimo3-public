def possible_pairings(group):
    # group: tuple of player indices
    # returns list of list of pairs (each pair tuple)
    n = len(group)
    if n == 0:
        return [[]]
    # take first element, pair with any other
    first = group[0]
    rest = group[1:]
    pairings = []
    for i, other in enumerate(rest):
        # pair first with other
        remaining = rest[:i] + rest[i+1:]
        for sub in possible_pairings(remaining):
            pairings.append([(first, other)] + sub)
    return pairings

def all_valid_winner_subsets(participants):
    # participants is a list of participant IDs sorted by speed (fastest first)
    n = len(participants)
    assert n%2==0
    half = n//2
    # generate all subsets of size half that satisfy matching condition (#w >= #l in any prefix)
    # Represent by binary string of length n, 1 for winner, 0 for loser. Must have exactly half ones.
    # Use recursion / generate Catalan sequences.
    # Use known algorithm: generate all Dyck words of length n (where n=2*half).
    # For each Dyck word (prefix condition #1 >= #0), winners set is positions where bit=1.
    # We'll produce as set of indices.
    # For small n, we can generate all combinations and test.
    # Use brute force for now.
    res=[]
    for combo in itertools.combinations(range(n), half):
        # Build sequence: 1 if index in combo else 0.
        ok=True
        w=0
        l=0
        combo_set=set(combo)
        for i in range(n):
            if i in combo_set:
                w+=1
            else:
                l+=1
            if w<l:
                ok=False
                break
        if ok:
            res.append(set([participants[i] for i in combo]))
    return res

def count_orderings(participants):
    # participants sorted by speed (fastest first) list of IDs.
    n=len(participants)
    if n==1:
        # leaf, single participant.
        return {tuple([participants[0]]):1}
    # else internal node
    result_counts={}
    # generate all valid winner subsets for this group
    for winners_set in all_valid_winner_subsets(participants):
        losers_set = set(participants) - winners_set
        # recursively compute possible orderings for each child
        win_orderings = count_orderings(sorted(winners_set, key=lambda x:x)) # speed order preserved
        lose_orderings = count_orderings(sorted(losers_set, key=lambda x:x))
        # For each combination of suborderings, combine them to overall ordering.
        # The final ranking will be: first the rankings of winners (they have higher scores) in order
        # then the rankings of losers (lower scores) in order.
        # Since scores differ at this round: any winner's score > any loser, because they get extra bit.
        # So overall ordering is concatenation: winners ranking (in order determined by deeper bits) then losers ranking.
        for win_seq,_ in win_orderings.items():
            for lose_seq,_ in lose_orderings.items():
                overall = win_seq + lose_seq
                result_counts[overall] = result_counts.get(overall,0) + 1
    return result_counts

def compute_N_bruteforce(n):
    participants = list(range(1,2**n+1)) # speed rank: 1 fastest
    orderings = count_orderings(participants)
    return len(orderings)

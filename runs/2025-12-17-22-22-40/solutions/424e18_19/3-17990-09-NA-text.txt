def all_matchings(players):
    # return list of matchings as list of pairs (list of tuples)
    # players is a tuple/list of distinct items
    if not players:
        return [[]]
    # pick first player
    first = players[0]
    matchings = []
    for i in range(1,len(players)):
        second = players[i]
        # pair first with second, recursively match rest
        rest = players[1:i] + players[i+1:]
        for sub in all_matchings(rest):
            matchings.append([(first,second)] + sub)
    return matchings

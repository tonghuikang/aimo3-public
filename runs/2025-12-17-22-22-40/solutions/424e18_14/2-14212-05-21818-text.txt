def dyck_partitions(elements):
    # elements: list of participants in decreasing speed order (fastest first)
    # returns list of (winners_set, losers_set) partitions satisfying condition
    # size must be even, len = 2n
    # each partition corresponds to a Dyck path: choose n winners such that in any prefix #w>=#l
    # We'll generate by recursion: standard way to generate Dyck words of length 2n and assign participants accordingly.
    n = len(elements)//2
    # generate all Dyck words of length 2n using recursion/backtracking
    results=[]
    def backtrack(pos, wcount, lcount, prefix, used_winners):
        # pos is index in elements (starting from 0)
        if pos == len(elements):
            # all assigned
            # winners are those indices where prefix char 'W' (or 1)
            winners = [elements[i] for i, ch in enumerate(prefix) if ch == 'W']
            losers  = [elements[i] for i, ch in enumerate(prefix) if ch == 'L']
            results.append((tuple(winners), tuple(losers)))
            return
        # we can assign next element as winner (W) if we haven't reached n winners
        if wcount < n:
            backtrack(pos+1, wcount+1, lcount, prefix+('W',), used_winners + [elements[pos]])
        # we can assign as loser (L) if #L < #W so far
        if lcount < wcount:
            backtrack(pos+1, wcount, lcount+1, prefix+('L',), used_winners)
    backtrack(0,0,0,tuple(),[])
    return results

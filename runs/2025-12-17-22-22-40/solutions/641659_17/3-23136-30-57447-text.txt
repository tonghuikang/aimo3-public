def circumcenter(p1, p2, p3):
    (x1,y1),(x2,y2),(x3,y3) = p1,p2,p3
    d = 2*(x1*(y2-y3)+x2*(y3-y1)+x3*(y1-y2))
    if abs(d) < 1e-12:
        # points collinear or nearly
        return None
    ux = ((x1*x1+y1*y1)*(y2-y3)+(x2*x2+y2*y2)*(y3-y1)+(x3*x3+y3*y3)*(y1-y2))/d
    uy = ((x1*x1+y1*y1)*(x3-x2)+(x2*x2+y2*y2)*(x1-x3)+(x3*x3+y3*y3)*(x2-x1))/d
    return (ux, uy)

def circle_intersections(o1,r1,o2,r2):
    # return list of intersection points (0, 1, or 2)
    x1,y1 = o1
    x2,y2 = o2
    dx = x2-x1
    dy = y2-y1
    d = math.hypot(dx,dy)
    # no intersection if d > r1+r2 or d < |r1-r2| or d ==0
    if d < 1e-12:
        return []
    if d > r1+r2+1e-9:
        return []
    if d < abs(r1-r2)-1e-9:
        return []
    # distance from o1 to point p2 along line o1->o2
    a = (r1*r1 - r2*r2 + d*d)/(2*d)
    # height from p to intersection points
    h2 = r1*r1 - a*a
    if h2 < -1e-12:
        return []
    h = math.sqrt(max(h2,0.0))
    # point p = o1 + a*(o2-o1)/d
    xm = x1 + a*dx/d
    ym = y1 + a*dy/d
    # offset vector perpendicular to line
    rx = -dy * (h/d)
    ry = dx * (h/d)
    p1 = (xm+rx, ym+ry)
    p2 = (xm-rx, ym-ry)
    if h < 1e-12:
        return [p1]
    else:
        return [p1, p2]

def foot_of_point_to_line(P, A, B):
    # foot of perpendicular from P to line AB
    # param t of projection onto AB: t = ((P-A)·(B-A))/|B-A|^2
    (px,py) = P
    (ax,ay) = A
    (bx,by) = B
    vx = bx-ax
    vy = by-ay
    # if AB is degenerate:
    denom = vx*vx + vy*vy
    if denom == 0:
        return A
    t = ((px-ax)*vx + (py-ay)*vy)/denom
    projx = ax + t*vx
    projy = ay + t*vy
    return (projx, projy)

def reflect_point_across_line(P, A, B):
    # reflect P across line AB
    proj = foot_of_point_to_line(P, A, B)
    # reflection: P' = 2*proj - P
    return (2*proj[0] - P[0], 2*proj[1] - P[1])

def dist(p,q):
    return math.hypot(p[0]-q[0], p[1]-q[1])

def triangle_geometry(r, s):
    # BD=1, CD=r
    BD = 1.0
    CD = r
    a = BD + CD # BC length
    # s must be > a

    # ensure positivity

    D = (BD, 0.0)

    E = foot_of_point_to_line(I, C, A)

    # circumcenter of AFE

    if O1 is None:

    # intersect circles (O1,R1) and (O2,R2), get points
    pts = circle_intersections(O1, R1, O2, R2)

    # Find point not equal to A (within tolerance)
    K = None

            K = pt
            break

        # maybe only one intersection (tangent)? Then K=A, degenerate
        raise ValueError("No K found, circles intersect only at A")

    Kp = reflect_point_across_line(K, E, F)

    N = foot_of_point_to_line(D, E, F)

        raise ValueError("B,K,N collinear")

    # Solve for intersection of two lines parametric: B + t*dir1 = M + u*dir2.
    # Solve 2x2 linear system: t*dir1 - u*dir2 = M - B
    # Represent as matrix [dir1, -dir2] * [t; u] = M - B.
    # Solve.
    A_mat = [[dir1[0], -dir2[0]], [dir1[1], -dir2[1]]]
    rhs = [M[0]-B[0], M[1]-B[1]]
    # Solve via determinant.
    det = A_mat[0][0]*A_mat[1][1] - A_mat[0][1]*A_mat[1][0]

        'O4':O4,'T':T,'CT':CT,'BT':BT,'NB':NB,'NE':NE

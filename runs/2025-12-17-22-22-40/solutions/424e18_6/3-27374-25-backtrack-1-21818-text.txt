def perfect_matchings(elements):
    # elements is a tuple/list of items (must be even length)
    if not elements:
        yield []
        return
    # take first element, pair with each other
    first = elements[0]
    rest = elements[1:]
    for i, partner in enumerate(rest):
        remaining = list(rest[:i] + rest[i+1:])
        for submatch in perfect_matchings(tuple(remaining)):
            yield [(first, partner)] + submatch

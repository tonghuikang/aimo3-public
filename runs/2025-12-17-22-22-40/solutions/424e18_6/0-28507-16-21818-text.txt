def generate_matchings(lst):
    # return list of list of pairs (as tuples), covering all perfect matchings of lst
    # lst is a tuple/list of distinct items.
    # Use recursion: pick first element, pair with any other, then recurse on remaining.
    if len(lst) == 0:
        return [[]]
    first = lst[0]
    rest = lst[1:]
    matchings = []
    for i, other in enumerate(rest):
        pair = (first, other)
        remaining = rest[:i] + rest[i+1:]
        for sub in generate_matchings(remaining):
            matchings.append([pair] + sub)
    return matchings

def simulate_tournament(participants):
    # participants is a list of participant IDs (speed rank ints, 1 fastest)
    n = len(participants).bit_length() - 1  # log2
    # recursively simulate all possible pairings.
    # Returns set of rankings (list of participants in order)
    # We'll use recursion to generate all possible final orderings.
    # We'll need to handle large number but n=3 small.
    def rec(current_set):
        # current_set is a tuple of participants (unsorted)
        size = len(current_set)
        if size == 1:
            # only one participant, its ranking list is just that participant
            return { (current_set[0],) }
        # generate all perfect matchings of current_set
        matchings = generate_matchings(list(current_set))
        rankings = set()
        for matching in matchings:
            # determine winners and losers
            winners = []
            losers = []
            for a,b in matching:
                # smaller rank = faster
                if a < b:
                    winners.append(a)
                    losers.append(b)
                else:
                    winners.append(b)
                    losers.append(a)
            # recursively order winners and losers
            for win_order in rec(tuple(winners)):
                for lose_order in rec(tuple(losers)):
                    # final ranking: winners first in their order, then losers in their order
                    rankings.add(win_order + lose_order)
        return rankings
    return rec(tuple(participants))

def count_rankings(n):
    participants = tuple(range(1, 2**n + 1))
    rankings = simulate_tournament(participants)
    return len(rankings)

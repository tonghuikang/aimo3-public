    for i in range(rounds):
        points = 2**(rounds - i - 1)  # 2^{r-i-1}

def simulate_tournament(n, schedule, speeds):
    # speeds: list of speed values, higher speed = faster
    scores = [0]*n
    r = n.bit_length() - 1
    for i, pairs in enumerate(schedule):
        pts = 2**(r - i - 1)
        for a,b in pairs:
            if speeds[a] > speeds[b]:
                winner = a
            else:
                winner = b
            scores[winner] += pts
    # final ordering
    ordering = sorted(range(n), key=lambda i: scores[i], reverse=True)
    return ordering, scores

    N = 2**n
    players = list(range(N)) # 0 fastest, N-1 slowest? Let's treat lower index = faster speed.
    # speed ranking: index 0 fastest, 1 second fastest, ...
    # Pairing each round: players are partitioned by current score (which is integer).
    # At each round i (0-indexed for i=0..n-1), the point value for winners is 2^{n-1-i}
    # We need to consider all possible ways to pair within each group of equal scores.
    # Number of ways per group: if group size = 2*m, we have (2*m-1)!! = (2*m)!/(2^m * m!) possible perfect matchings.
    # For n small, we can brute force enumerating all pairings.
    from math import factorial
    from functools import lru_cache

    # Helper: generate all perfect matchings of a list of players.
    def matchings(lst):
        if not lst:
            yield []
        else:
            # take first element, pair with any other
            a = lst[0]
            for i in range(1, len(lst)):
                b = lst[i]
                rest = lst[1:i] + lst[i+1:]
                for rest_match in matchings(rest):
                    yield [(a,b)] + rest_match

    # Recursively generate all possible tournaments and final scores
    final_permutations = set()
    # scores dict: player->score

    def dfs(round_idx, scores):
        if round_idx == n:
            # final ordering: sort players by descending score (higher score first)
            # Since scores are unique for n rounds, we can order.
            ordering = tuple(sorted(players, key=lambda p: scores[p], reverse=True))
            final_permutations.add(ordering)
            return
        # group players by current score
        groups = {}
        for p in players:
            s = scores[p]
            groups.setdefault(s, []).append(p)
        # For each group, generate all possible pairings
        # We'll need to consider independent matchings per group and combine them.
        # Use recursion over groups.
        # Create list of groups (score, list)
        group_items = list(groups.items())
        # For each group, get all possible matchings
        matchings_per_group = []
        for s, group in group_items:
            # group size must be even
            assert len(group) % 2 == 0
            # all matchings
            matchings_per_group.append(list(matchings(group)))
        # combine matchings across groups via cartesian product
        import itertools
        for combo in itertools.product(*matchings_per_group):
            # combo is a tuple of list of pairs per group
            new_scores = scores.copy()
            # point value for this round:
            points = 2**(n-1-round_idx)
            # process each group matches
            for matches in combo:
                for a, b in matches:
                    # winner is faster (lower index)
                    winner = a if a < b else b
                    # loser = other
                    # assign points
                    new_scores[winner] += points
                    # loser unchanged
            # recurse to next round
            dfs(round_idx+1, new_scores)

    # initial scores
    init_scores = {p:0 for p in players}
    dfs(0, init_scores)
    return final_permutations

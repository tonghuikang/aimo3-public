def has_perfect_matching(winners, losers):
    # winners and losers are distinct integers
    # edges: winner > loser
    # we can treat bipartite graph with winners on left, losers on right.
    # Use Hungarian/DFS for maximum bipartite matching; but small size.
    n = len(winners)
    # build adjacency list
    adj = {i: [] for i in range(n)}
    # map winners list to indices
    for i,w in enumerate(winners):
        for j,l in enumerate(losers):
            if w>l:
                adj[i].append(j)
    # attempt to find matching using DFS augmenting path
    matchR = [-1]*n  # loser matched to which winner index
    def bpm(u, seen):
        for v in adj[u]:
            if not seen[v]:
                seen[v]=True
                if matchR[v]==-1 or bpm(matchR[v], seen):
                    matchR[v]=u
                    return True
        return False
    result=0
    for u in range(n):
        seen=[False]*n
        if bpm(u, seen):
            result+=1
    return result==n

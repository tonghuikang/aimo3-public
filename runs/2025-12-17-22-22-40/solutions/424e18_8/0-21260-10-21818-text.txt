def all_matchings(lst):
    # return all perfect matchings of list lst (list of elements)
    if not lst:
        return [[]]
    # pick first element
    first = lst[0]
    matchings = []
    for i in range(1, len(lst)):
        second = lst[i]
        rest = lst[1:i] + lst[i+1:]
        for sub in all_matchings(rest):
            matchings.append([(first, second)] + sub)
    return matchings

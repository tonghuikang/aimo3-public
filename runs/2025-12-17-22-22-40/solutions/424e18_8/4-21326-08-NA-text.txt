def all_matchings(participants):
    if not participants:
        return [[]]
    first = participants[0]
    rest = participants[1:]
    matchings = []
    for i, other in enumerate(rest):
        new_rest = rest[:i] + rest[i+1:]
        for sub in all_matchings(new_rest):
            matchings.append([(first, other)] + sub)
    return matchings

def winners_of_matching(matching):
    return {min(a,b) for a,b in matching}  # winner = smaller index (faster)

def is_valid_winner_set(winners, total):
    # total participants list sorted 1..total
    # check prefix condition #w >= #l
    wins = set(winners)
    w = 0
    l = 0
    for i in range(1,total+1):
        if i in wins:
            w+=1
        else:
            l+=1
        if w<l:
            return False
    return True

def test_n(n):
    total = 2*n
    participants = list(range(1,total+1))
    matchings = all_matchings(participants)
    ok=True
    for m in matchings:
        wins = winners_of_matching(m)
        if not is_valid_winner_set(wins,total):
            ok=False
            break
    return ok, len(matchings)

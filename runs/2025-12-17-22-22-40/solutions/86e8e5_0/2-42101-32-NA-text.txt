def divisors(n):
    divs=set()
    for i in range(1,int(math.isqrt(n))+1):
        if n%i==0:

            divs.add(n//i)

def f_by_Ksearch(N, maxK=None):
    # find minimal K > N/2 satisfying condition
    start = N//2 + 1
    if maxK is None:
        maxK = N-1
    for K in range(start, maxK+1):
        d = N - K
        # need distinct proper divisors a,b of K summing to d
        divs = divisors(K)
        # remove K itself
        divs.discard(K)
        # find pairs a<b in divs such that a+b=d
        # quick check using set
        for a in divs:
            b = d - a
            if b > a and b in divs:
                # found
                return K
    return None

def compute_ratio(x, y, h):
    # returns ratio R for given x=BD, y=CD, h>0 if cyclic condition holds? 
    # Actually compute ratio without verifying condition; we'll compute points anyway.
    a = x + y
    # side lengths
    c = x + h # AB
    b = y + h # AC
    
    # coordinates
    B = mp.matrix([0,0])
    C = mp.matrix([a,0])
    # Compute X coordinate of A
    X = (c**2 - b**2 + a**2) / (2*a)
    # Y coordinate positive
    # Ensure X is between 0 and a
    # Y^2 = c^2 - X^2
    Y_sq = c**2 - X**2
    if Y_sq <= 0:
        # degenerate
        return None
    Y = mp.sqrt(Y_sq)
    A = mp.matrix([X, Y])
    # incircle radius
    r = mp.sqrt(h * x * y / (x + y + h))
    I = mp.matrix([x, r])
    # Points F and E
    F = (x/(x+h)) * A  # (x/c) * A
    # E = (h*C + y*A)/(y+h) = (h*C + y*A)/b
    E = (h*C + y*A) / b
    # line EF direction vector
    v = F - E
    # D point
    D = mp.matrix([x, 0])
    # foot N of D onto line EF
    t_proj = ((D - E).dot(v)) / (v.dot(v))
    N = E + t_proj * v
    
    # Compute symmedian intersection L
    # L divides BC: BL:LC = c^2 : b^2
    Lx = (c**2 * a) / (b**2 + c**2)
    L = mp.matrix([Lx, 0])
    
    # circumcenter O
    # O_x = a/2
    O_x = a/2
    # compute O_y as derived: O_y = Y/2 - X*(a - X)/(2*Y)
    O_y = Y/2 - (X*(a - X))/(2*Y)
    O = mp.matrix([O_x, O_y])
    # radius
    R = mp.sqrt((O_x - B[0])**2 + (O_y - B[1])**2)
    
    # find K: second intersection of line A-L with circumcircle
    u = L - A
    w = A - O
    # t_K = -2 * dot(w,u) / dot(u,u)
    t_K = -2 * (w.dot(u)) / (u.dot(u))
    K = A + t_K * u
    # reflect K across line EF to get K'
    # projection of K onto line EF:
    t_proj_K = ((K - E).dot(v)) / (v.dot(v))
    Q = E + t_proj_K * v
    Kp = 2*Q - K
    
    # compute ratio R_val = (CT * NB) / (BT * NE)
    # find O_T, center of circle through B and K tangent to BN at B.
    # line BN direction:
    BN = N - B
    # line perpendicular to BN through B:
    # direction vector of this line is perp = (-BN_y, BN_x)
    perp = mp.matrix([-BN[1], BN[0]])
    # perpendicular bisector of BK: 
    mid_BK = (B + K)/2
    # direction of BK:
    BK = K - B
    # slope of perpendicular bisector is perpendicular to BK: direction = (-BK_y, BK_x)
    perp_BK = mp.matrix([-BK[1], BK[0]])
    # solve for intersection of line1: B + s*perp with line2: mid_BK + t*perp_BK
    # Solve B + s*perp = mid_BK + t*perp_BK
    # Two equations, unknown s,t.
    # Write as s*perp - t*perp_BK = mid_BK - B
    # Solve linear system 2x2.
    M = mp.matrix([[perp[0], -perp_BK[0]],
                   [perp[1], -perp_BK[1]]])
    rhs = mid_BK - B
    # Solve M*[s t]^T = rhs
    try:
        sol = mp.linalg.solve(M, rhs)
    except Exception as e:
        # singular maybe? return None
        return None
    s = sol[0]
    # O_T is B + s*perp
    O_T = B + s*perp
    # O_T.x should be positive
    # Compute T point: x-coordinate 2*O_T.x, y=0
    T_x = 2*O_T[0]
    # CT = |C.x - T_x|
    CT = abs(a - T_x)
    BT = abs(T_x)  # positive
    # NB and NE distances
    NB = mp.sqrt((N[0] - B[0])**2 + (N[1] - B[1])**2)
    NE = mp.sqrt((N[0] - E[0])**2 + (N[1] - E[1])**2)
    # ratio
    ratio = (CT * NB) / (BT * NE)
    # also compute cyclic condition value f (determinant)
    # compute determinant for points N, K, K'
    mat = mp.matrix([[N[0], N[1], N[0]**2 + N[1]**2],
                     [K[0], K[1], K[0]**2 + K[1]**2],
                     [Kp[0], Kp[1], Kp[0]**2 + Kp[1]**2]])
    # compute determinant
    f_val = mp.det(mat)
    return {
        'ratio': ratio,
        'f': f_val,
        'h': h,
        'N': N,
        'K': K,
        'Kp': Kp,
        'O_T': O_T,
        'A': A,
        'B': B,
        'C': C,
        'E': E,
        'F': F,
        'I': I,
        'r': r,
        'x': x,
        'y': y,
        'a': a,
        'b': b,
        'c': c
    }

    fs = []
    for h in hs:
        try:
            f_val = compute_ratio(x,y,h)['f']
        except Exception as e:
            f_val = None
        fs.append(f_val)
    # Find sign changes where f not None and sign changes between consecutive points.
    for i in range(len(hs)-1):
        f1 = fs[i]
        f2 = fs[i+1]
        if f1 is None or f2 is None:
            continue
        # check if signs are opposite or one close to zero
        if f1 == 0:
            return hs[i]
        if f2 == 0:
            return hs[i+1]
        if f1*f2 < 0:
            # bracket found
            # Use findroot via bisection
            a = hs[i]
            b = hs[i+1]
            # Use bisection
            for _ in range(50):
                m = (a+b)/2
                fmid = compute_ratio(x,y,m)['f']
                if fmid == 0:
                    return m
                if fmid*f1 < 0:
                    b = m
                    f2 = fmid
                else:
                    a = m
                    f1 = fmid
            return (a+b)/2
    # if no sign change found, maybe f does not cross zero? Return None
    return None

Instead of using determinant, we can write equation: The distance from O to B squared equals distance from O to N squared, which yields O lying on perpendicular bisector of BN. Since O also lies on EF, O is intersection of line EF with perpendicular bisector of BN. Then distance OB must also equal distance OK.

Thus condition reduces to: If O = intersection of EF and perpendicular bisector of BN, then OK^2 = OB^2.

Thus we can compute O analytically as function of x,y,h, then impose OK^2 - OB^2 = 0.

    # incircle center
    I = (p, r)
    # F point
    xF = p*(p**2 - r**2)/(p**2 + r**2)
    yF = 2*p**2 * r/(p**2 + r**2)
    F = (xF, yF)
    # E point
    q = 1 - p
    xE = 1 - q*(q**2 - r**2)/(q**2 + r**2)
    yE = 2*q**2 * r/(q**2 + r**2)
    E = (xE, yE)
    # line EF: compute direction vector
    # compute N foot of D onto EF
    # D = (p, 0)
    # projection formula:
    # param t: projection of D onto line EF: N = F + ((D-F)·(E-F))/|E-F|^2 * (E-F)
    dx = E[0] - F[0]
    dy = E[1] - F[1]
    # dot product (D-F)·(E-F)
    dfx = p - F[0]
    dfy = -F[1]
    dot = dfx*dx + dfy*dy
    norm2 = dx*dx + dy*dy
    t = dot / norm2

    Ny = F[1] + t*dy

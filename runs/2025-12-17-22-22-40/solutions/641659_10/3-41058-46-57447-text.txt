    # p,q,r are floats
    a = p + q
    # points
    B = sp.Point(0,0)
    C = sp.Point(a,0)
    # slopes
    mB = (2*p*r) / (p**2 - r**2)
    mC = - (2*q*r) / (q**2 - r**2)
    # intersection of lines AB and AC for A
    xA = -mC * a / (mB - mC)
    yA = mB * xA
    A = sp.Point(xA, yA)
    # incircle center
    I = sp.Point(p, r)
    # touchpoints F and E
    # F on AB:
    F_x = (p + mB * r) / (mB**2 + 1)
    F_y = (r * mB**2 + mB * p) / (mB**2 + 1)
    F = sp.Point(F_x, F_y)
    # E on AC:
    E_x = (a * mC**2 + p + mC * r) / (mC**2 + 1)
    E_y = (r * mC**2 - mC * q) / (mC**2 + 1)
    E = sp.Point(E_x, E_y)
    # circles
    circ_ABC = sp.Circle(A, B, C)
    circ_AFE = sp.Circle(A, F, E)

    O1 = circ_ABC.center
    R1 = circ_ABC.radius
    O2 = circ_AFE.center
    R2 = circ_AFE.radius
    # vector v = O2 - O1
    vx = O2.x - O1.x
    vy = O2.y - O1.y
    # constant C0 = ((R1**2 - R2**2) + (O2.x**2 + O2.y**2 - O1.x**2 - O1.y**2))/2
    C0 = ((R1**2 - R2**2) + (O2.x**2 + O2.y**2 - O1.x**2 - O1.y**2))/2
    # solve intersection of line vx*x + vy*y = C0 with circle centered O1 radius R1
    # param solve quadratic
    x, y = sp.symbols('x y')
    # solve linear for y if vy != 0 else solve for x
    if abs(vy) > 1e-12:
        # y = (C0 - vx*x)/vy
        y_expr = (C0 - vx*x)/vy
        eq = (x - O1.x)**2 + (y_expr - O1.y)**2 - R1**2
        # solve quadratic eq for x
        sol_x = sp.solve(eq, x)
        # Two solutions; compute points
        pts = []
        for sol in sol_x:
            xv = sp.N(sol)
            yv = sp.N(y_expr.subs(x, sol))
            pts.append(sp.Point(xv, yv))
    else:

        # x = (C0 - vy*y)/vx = C0/vx
        x_val = C0 / vx
        # Solve circle eq for y: (x_val - O1.x)^2 + (y - O1.y)^2 = R1**2

        pts = []

            pts.append(sp.Point(float(x_val), sp.N(sol)))

    K = None
    for pt in pts:

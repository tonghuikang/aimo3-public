def compute_f_float_mp(r, alpha):
    d = alpha
    u = 1 - alpha
    m1 = 2*d*r/(d**2 - r**2)
    m2 = -2*u*r/(u**2 - r**2)
    x = -m2/(m1 - m2)
    y = m1*x
    xf = (d + m1*r)/(1 + m1**2)
    yf = m1*xf
    xe = (d + m2**2 + r*m2)/(1 + m2**2)
    ye = (-u*m2 + r*m2**2)/(1 + m2**2)
    a_line = ye - yf
    b_line = xf - xe
    c_line = -a_line*xf - b_line*yf
    # circumcenter ABC
    y_O = (x**2 - x + y**2) / (2*y)
    O_Omega = [mp.mpf('0.5'), y_O]
    R_Omega_sq = (mp.mpf('0.5'))**2 + y_O**2
    # circumcenter AEF
    A_mat = mp.matrix([[2*(xe - x), 2*(ye - y)], [2*(xf - x), 2*(yf - y)]])
    b_vec = mp.matrix([xe**2 + ye**2 - x**2 - y**2, xf**2 + yf**2 - x**2 - y**2])
    sol = mp.lu_solve(A_mat, b_vec)
    U = sol[0]; V = sol[1]
    O1 = [U, V]
    dvec = [O1[0] - O_Omega[0], O1[1] - O_Omega[1]]
    vx = -dvec[1]; vy = dvec[0]
    # intersection line with circumcircle
    dx0 = x - O_Omega[0]; dy0 = y - O_Omega[1]
    a_quad = vx**2 + vy**2
    b_quad = 2*(dx0*vx + dy0*vy)
    c_quad = dx0**2 + dy0**2 - R_Omega_sq
    disc = b_quad**2 - 4*a_quad*c_quad
    sqrt_disc = mp.sqrt(disc)
    t1 = (-b_quad + sqrt_disc) / (2*a_quad)
    t2 = (-b_quad - sqrt_disc) / (2*a_quad)
    t_val = t1 if abs(t1) > mp.mpf('1e-30') else t2
    K = [x + t_val*vx, y + t_val*vy]
    # foot N
    Dp = [d, mp.mpf('0')]
    Fp = [xf, yf]
    Ep = [xe, ye]
    vEF = [Ep[0] - Fp[0], Ep[1] - Fp[1]]
    s_proj = dot([Dp[0] - Fp[0], Dp[1] - Fp[1]], vEF) / dot(vEF, vEF)
    Np = [Fp[0] + s_proj * vEF[0], Fp[1] + s_proj * vEF[1]]
    # circumcenter of B,K,N
    Bp = [mp.mpf('0'), mp.mpf('0')]
    # solve linear equations: 2*(K - B).O = |K|^2, 2*(N - B).O = |N|^2
    A_mat2 = mp.matrix([[2*(K[0] - Bp[0]), 2*(K[1] - Bp[1])],
                        [2*(Np[0] - Bp[0]), 2*(Np[1] - Bp[1])]])
    b_vec2 = mp.matrix([K[0]**2 + K[1]**2, Np[0]**2 + Np[1]**2])
    O_circ = mp.lu_solve(A_mat2, b_vec2)
    # distance from O_circ to line EF
    dist = (a_line*O_circ[0] + b_line*O_circ[1] + c_line) / mp.sqrt(a_line**2 + b_line**2)
    return dist

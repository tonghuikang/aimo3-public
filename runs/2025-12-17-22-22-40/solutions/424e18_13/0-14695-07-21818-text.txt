def all_matchings(participants):
    # participants is a tuple/list of even length
    # generate all perfect matchings as list of list of pairs (i,j)
    # For enumeration, we can use recursion: pick first participant, pair with any other, and recurse on remaining.
    participants = tuple(participants)
    if not participants:
        yield []
        return
    first = participants[0]
    for i in range(1, len(participants)):
        second = participants[i]
        rest = participants[1:i] + participants[i+1:]
        for rest_match in all_matchings(rest):
            yield [(first, second)] + rest_match

def perfect_matchings(elements):
    # elements must be tuple/list of even length
    if len(elements) == 0:
        yield []
        return
    # choose first element
    a = elements[0]
    for i in range(1, len(elements)):
        b = elements[i]
        rest = elements[1:i] + elements[i+1:]
        for sub in perfect_matchings(rest):
            yield [(a,b)] + sub

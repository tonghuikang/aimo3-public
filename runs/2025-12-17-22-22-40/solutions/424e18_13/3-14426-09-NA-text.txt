def all_pairings(lst):
    # lst: list of items (labels)
    # generate all possible ways to pair up the elements in lst (list length even)
    # return a list of list of pairs (tuple)
    # Use recursion: pick first element, pair with each other, then recurse on rest
    if not lst:
        yield []
        return
    first = lst[0]
    for i in range(1, len(lst)):
        second = lst[i]
        rest = lst[1:i] + lst[i+1:]
        for subpairings in all_pairings(rest):
            yield [(first, second)] + subpairings

def poly_mul(p, q):
    # p, q are dict exponent->coeff (int)
    r = {}
    for i, a in p.items():
        for j, b in q.items():
            r[i+j] = r.get(i+j, 0) + a*b
    # remove zero
    r = {i:c for i,c in r.items() if c != 0}
    return r

def poly_add(p, q):
    r = p.copy()
    for i,b in q.items():
        r[i] = r.get(i,0)+b
        if r[i]==0:
            del r[i]
    return r

def poly_sub(p,q):

        r[i] = r.get(i,0)-b

            del r[i]

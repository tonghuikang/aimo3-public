def all_matchings(participants):
    # participants is a tuple/list of items
    # produce list of list of pairs (unordered pairs)
    # we can generate recursively.
    if not participants:
        yield []
        return
    # take first participant, pair with any later participant
    first = participants[0]
    for i in range(1, len(participants)):
        second = participants[i]
        rest = participants[1:i] + participants[i+1:]
        for rest_pairs in all_matchings(rest):
            yield [(first, second)] + rest_pairs

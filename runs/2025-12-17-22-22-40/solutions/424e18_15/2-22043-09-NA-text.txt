def all_matchings(lst):
    # return list of matchings (list of pairs) for even-length list
    # recursively generate all perfect matchings
    if not lst:
        return [[]]
    # pick first element
    first = lst[0]
    rest = lst[1:]
    matchings = []
    for i, other in enumerate(rest):
        # pair first with other
        pair = (first, other)
        # remaining list without other
        new_rest = rest[:i] + rest[i+1:]
        for submatching in all_matchings(new_rest):
            matchings.append([pair] + submatching)
    return matchings

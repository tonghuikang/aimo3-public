    # speeds: label from 0 (fastest) to num_players-1 (slowest)
    speeds = list(range(num_players))
    # each player represented by dict: speed, score
    players = [{'speed': s, 'score': 0} for s in speeds]
    m = num_players.bit_length() - 1
    for i in range(1, m+1):
        point = 2**(m - i)
        # group by score
        groups = {}
        for p in players:
            groups.setdefault(p['score'], []).append(p)
        new_players = []
        for score, group in groups.items():
            # sort group by speed (ascending speed label = faster)
            group.sort(key=lambda x: x['speed'])
            half = len(group)//2
            # winners: top half (fastest)
            for idx, p in enumerate(group):
                if idx < half:
                    p['score'] += point
                # else loser stays same
        # after updating scores in-place, continue next round
    # after tournament, return list of (speed, score)
    return [(p['speed'], p['score']) for p in players]

def get_orderings(num_players):
    # produce final ordering by score descending (fastest highest score)
    scores = tournament_scores(num_players)
    # sort by score descending
    sorted_by_score = sorted(scores, key=lambda x: (-x[1], x[0]))
    # return ordering of speeds
    return [s for s,_ in sorted_by_score]

def pairings(participants):
    # returns list of possible perfect matchings (list of list of pairs)
    # participants is a list
    if not participants:
        return [[]]
    # pick first participant
    a = participants[0]
    rest = participants[1:]
    matchings = []
    for i,b in enumerate(rest):
        # pair a with b
        remaining = rest[:i] + rest[i+1:]
        for sub in pairings(remaining):
            matchings.append([(a,b)] + sub)
    return matchings

def ranking_for_pairings(initial_pairing):

    # compute L(m) using brute search over a<b<c sum=m
    best = None
    best_trip = None
    import math
    for a in range(1,m):
        for b in range(a+1,m-a):
            c=m-a-b
            if c<=b: continue
            l = math.lcm(a,b,c)
            if best is None or l < best:

                best_trip = (a,b,c)

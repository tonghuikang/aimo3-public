    # trial division up to sqrt
    factors = {}
    d=2
    while d*d <= n:
        while n % d == 0:
            factors[d] = factors.get(d, 0)+1
            n //= d
        d += 1 if d==2 else 2  # skip even after 2
    if n>1:
        factors[n] = 1
    return factors

def all_divisors_from_factors(factors):
    # factors is dict prime->exp
    # returns list of divisors
    primes = list(factors.keys())
    exps = [range(e+1) for e in factors.values()]
    divs = [1]
    for p, e_range in zip(primes, exps):
        new_divs = []
        for d in divs:
            for e in e_range:
                new_divs.append(d * (p**e))
        divs = new_divs
    return sorted(divs)

def compute_r_primitive(limit):
    r_prim = {}
    for s in range(6, limit+1):
        # find minimal ratio for primitive triples with sum s
        # we need to consider triples (a,b,c) with gcd=1 (i.e., primitive)
        min_r = float('inf')
        best = None
        for a in range(1, s-2):
            for b in range(a+1, s-a-1):
                c = s - a - b
                if c <= b:
                    continue
                if math.gcd(a, math.gcd(b, c)) != 1:
                    continue
                l = math.lcm(a,b,c)
                r = l / s
                if r < min_r:
                    min_r = r
                    best = (a,b,c,l)
        r_prim[s] = (min_r, best)
    return r_prim

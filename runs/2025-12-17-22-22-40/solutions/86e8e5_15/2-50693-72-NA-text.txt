def factorial_mod(n, mod):
    res = 1 % mod
    for i in range(2, n+1):
        res = (res * i) % mod
    return res

def M_mod_s(s):
    # compute M = 3^K mod s, where K = 2025!
    # s can be up to maybe 1000
    # factor s into 3^a * t, t coprime to 3
    a = 0
    t = s
    while t % 3 == 0 and t > 0:
        a += 1
        t //= 3
    # now s = 3^a * t, t coprime to 3
    # M is multiple of 3^K, with K >= a, so M mod 3^a = 0
    if t == 1:
        # s is power of 3, M divisible by s (since K >= a)
        return 0
    # compute phi(t)
    # phi(t) = t * product_{p|t} (1-1/p)
    # compute phi straightforward
    def phi(n):
        result = n
        nn = n
        p=2
        while p*p <= nn:
            if nn%p==0:
                while nn%p==0:
                    nn//=p
                result -= result//p
            p+=1
        if nn>1:
            result -= result//nn
        return result
    phit = phi(t)
    # compute K_mod_phi = 2025! mod phit
    K_mod_phi = factorial_mod(2025, phit)
    v = pow(3, K_mod_phi, t)
    # combine via CRT: find x ≡ 0 (mod 3^a) and x ≡ v (mod t)
    mod1 = 3**a
    mod2 = t
    # we need inv_mod = inverse of mod1 modulo mod2
    inv = pow(mod1, -1, mod2)  # modular inverse
    u = (v * inv) % mod2
    x = (mod1 * u) % (mod1 * mod2)
    return x

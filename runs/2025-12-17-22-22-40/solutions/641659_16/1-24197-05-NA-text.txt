def geometry_numeric(x, y, r):
    # x = BD, y = CD, r incircle radius
    a = x + y
    # incircle center
    I = (x, r)
    # slopes
    if abs(x**2 - r**2) < 1e-12 or abs(y**2 - r**2) < 1e-12:
        return None
    mB = 2 * x * r / (x**2 - r**2)
    mC = -2 * y * r / (y**2 - r**2)
    # Compute A
    # x_A = - mC * a / (mB - mC)
    denom = (mB - mC)
    if abs(denom) < 1e-12:
        return None
    xA = - mC * a / denom
    yA = mB * xA
    A = (xA, yA)
    # Compute F (projection onto AB line through origin)
    # Using formula
    # proj = ((x + mB * r) / (1 + mB**2)) * (1, mB)
    denomAB = 1 + mB**2
    proj_factor = (x + mB * r) / denomAB
    Fx = proj_factor
    Fy = mB * proj_factor
    F = (Fx, Fy)
    # Compute E (projection onto line AC passing through C)
    # Compute vector from C to I: (x - a, r)
    dxCI = x - a
    dyCI = r
    denomAC = 1 + mC**2
    proj_factor2 = (dxCI + mC * dyCI) / denomAC
    Ex = a + proj_factor2  # C.x + proj_factor2 * 1
    Ey = 0 + proj_factor2 * mC
    E = (Ex, Ey)
    # D
    D = (x, 0)
    # line EF: direction vector w = E - F
    w = (Ex - Fx, Ey - Fy)
    w_norm2 = w[0]**2 + w[1]**2
    if w_norm2 < 1e-18:
        return None
    # projection of D onto line EF
    # vector from F to D:
    vFD = (x - Fx, - Fy)
    t = (vFD[0] * w[0] + vFD[1] * w[1]) / w_norm2
    Nx = Fx + t * w[0]
    Ny = Fy + t * w[1]
    N = (Nx, Ny)
    # circumcenter of triangle ABC
    # Use formula for circumcenter from coordinates.
    # Compute using perpendicular bisectors intersection
    # Let's define function circumcenter(p1,p2,p3):
    def circumcenter(P, Q, R):
        (x1,y1),(x2,y2),(x3,y3) = (P,Q,R)
        d = 2*(x1*(y2-y3)+x2*(y3-y1)+x3*(y1-y2))
        if abs(d) < 1e-12:
            return None
        ux = ((x1**2+y1**2)*(y2-y3)+(x2**2+y2**2)*(y3-y1)+(x3**2+y3**2)*(y1-y2))/d
        uy = ((x1**2+y1**2)*(x3-x2)+(x2**2+y2**2)*(x1-x3)+(x3**2+y3**2)*(x2-x1))/d
        return (ux, uy)
    O_ABC = circumcenter(A, (0,0), (a,0))

    # Intersection of circles: solve for K
    # Circles: center O_ABC radius R_ABC, center O_AFE radius R_AFE.
    # Intersection points can be computed analytically.
    # Use formula for intersection of two circles.

    O1 = O_ABC

def circumcenter(p1, p2, p3):
    # returns (center_x, center_y) and radius
    x1,y1 = p1
    x2,y2 = p2
    x3,y3 = p3
    d = 2*(x1*(y2-y3) + x2*(y3-y1) + x3*(y1-y2))
    if abs(d) < 1e-12:
        return None, None
    ux = ((x1**2+y1**2)*(y2-y3) + (x2**2+y2**2)*(y3-y1) + (x3**2+y3**2)*(y1-y2))/d
    uy = ((x1**2+y1**2)*(x3-x2) + (x2**2+y2**2)*(x1-x3) + (x3**2+y3**2)*(x2-x1))/d
    R = mp.sqrt((ux-x1)**2 + (uy-y1)**2)
    return (ux, uy), R

def intersect_circles(O1,R1,O2,R2):
    # returns two intersection points as tuples
    x1,y1 = O1
    x2,y2 = O2
    d = mp.sqrt((x2-x1)**2 + (y2-y1)**2)
    # if d > R1+R2 or d < |R1-R2|, no intersect; assume intersect
    if d == 0:
        return []
    a = (R1**2 - R2**2 + d**2) / (2*d)
    h_sq = R1**2 - a**2
    if h_sq < -1e-12:
        # no intersection
        return []
    h = mp.sqrt(max(0, h_sq))
    # point P2 = O1 + a * (O2-O1) / d
    ex = (x2-x1)/d
    ey = (y2-y1)/d
    x3 = x1 + a*ex
    y3 = y1 + a*ey
    # intersection points:
    # (x3 +/- h*(-ey), y3 +/- h*ex)
    p1 = (x3 + h*(-ey), y3 + h*ex)
    p2 = (x3 - h*(-ey), y3 - h*ex)
    return [p1, p2]

def reflect_point_across_line(P, A, B):
    # reflect point P across line AB (through points A and B)
    # compute projection onto AB:
    # vector AB
    ax,ay = A
    bx,by = B
    px,py = P
    # direction vector u = B-A
    ux = bx-ax
    uy = by-ay
    # compute projection scalar t = ((P-A)·u) / (u·u)
    t = ((px-ax)*ux + (py-ay)*uy) / (ux**2 + uy**2)
    projx = ax + t*ux
    projy = ay + t*uy
    # reflection:
    rx = 2*projx - px
    ry = 2*projy - py
    return (rx, ry)

def foot_point_on_line(P, A, B):
    # projection of point P onto line AB
    ax,ay = A
    bx,by = B
    px,py = P
    ux = bx-ax
    uy = by-ay
    t = ((px-ax)*ux + (py-ay)*uy) / (ux**2 + uy**2)
    return (ax + t*ux, ay + t*uy)

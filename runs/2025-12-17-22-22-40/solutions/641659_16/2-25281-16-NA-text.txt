    ux = ((x1**2 + y1**2)*(y2 - y3) + (x2**2 + y2**2)*(y3 - y1) + (x3**2 + y3**2)*(y1 - y2))/d
    uy = ((x1**2 + y1**2)*(x3 - x2) + (x2**2 + y2**2)*(x1 - x3) + (x3**2 + y3**2)*(x2 - x1))/d
    center = mp.matrix([ux, uy])
    radius = mp.sqrt((ux - x1)**2 + (uy - y1)**2)
    return (center, radius)

def intersect_circles(c1,c2):
    # c1 = (center1,r1), c2 = (center2,r2)
    O1,R1 = c1
    O2,R2 = c2
    x1,y1 = O1[0],O1[1]
    x2,y2 = O2[0],O2[1]
    d = mp.sqrt((x2-x1)**2 + (y2-y1)**2)
    # no intersection or one intersection, assume two intersections
    if d > R1 + R2 or d < abs(R1 - R2):
        return []  # no intersection
    # a = distance from O1 to line joining intersection points
    a = (R1**2 - R2**2 + d**2) / (2*d)
    h_sq = R1**2 - a**2
    if h_sq < -1e-12:
        return []
    h = mp.sqrt(max(0, h_sq))
    # point P2 is midpoint along line O1O2 at distance a from O1
    P2 = O1 + a*(O2 - O1)/d
    # intersection points
    # offset vector perpendicular to O1O2
    rx = -(y2 - y1)*(h/d)
    ry = (x2 - x1)*(h/d)
    inter1 = P2 + mp.matrix([rx, ry])
    inter2 = P2 - mp.matrix([rx, ry])
    return [inter1, inter2]

def foot_of_perpendicular(P, line_pt1, line_pt2):
    # returns foot of perpendicular from P to line through line_pt1 and line_pt2
    # vector line direction
    v = line_pt2 - line_pt1
    w = P - line_pt1
    t = (v[0]*w[0] + v[1]*w[1])/(v[0]**2 + v[1]**2)
    proj = line_pt1 + t*v
    # foot is proj
    return proj

def reflect_point_across_line(P, line_pt1, line_pt2):
    # reflect P across line
    foot = foot_of_perpendicular(P, line_pt1, line_pt2)
    # reflected point: 2*foot - P
    return 2*foot - P

def is_cyclic(B,N,K,Kp):
    # Check if points B,N,K,Kp are concyclic: compute determinant
    # matrix of [x y x^2+y^2 1]
    def row(P):
        return [P[0], P[1], P[0]**2+P[1]**2, 1]
    M = mp.matrix([row(B), row(N), row(K), row(Kp)])
    # compute determinant
    det = mp.det(M)
    return det

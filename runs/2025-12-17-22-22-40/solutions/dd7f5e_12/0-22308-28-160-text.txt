    # Euler's totient
    result = n
    p = 2
    nn = n
    while p*p <= nn:
        if nn % p == 0:
            while nn % p == 0:
                nn //= p
            result -= result // p
        p += 1
    if nn > 1:
        result -= result // nn
    return result

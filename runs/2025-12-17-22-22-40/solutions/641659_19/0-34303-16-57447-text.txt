def cyclic_det(B, K, N, Kprime):
    # compute determinant for concyclicity
    # points as arrays [x,y]
    # matrix rows: [x, y, x^2+y^2, 1]
    M = np.array([
        [B[0], B[1], B[0]**2 + B[1]**2, 1],
        [K[0], K[1], K[0]**2 + K[1]**2, 1],
        [N[0], N[1], N[0]**2 + N[1]**2, 1],
        [Kprime[0], Kprime[1], Kprime[0]**2 + Kprime[1]**2, 1]
    ])
    return np.linalg.det(M)

def ratio(d,c,r):

    # BN direction
    w = N - B

    n = np.array([-w[1], w[0]])  # rotate by +90 degrees
    norm_n = np.linalg.norm(n)
    if norm_n == 0:
        return None
    n_u = n / norm_n
    # compute R = |K|^2 / (2 * (K·n_u))
    Kvec = K - B
    dot_val = np.dot(Kvec, n_u)
    if dot_val <= 0:
        # maybe choose opposite normal?
        n_u = -n_u
        dot_val = np.dot(Kvec, n_u)
    R = np.dot(Kvec, Kvec) / (2 * dot_val)

    # ratio
    if BT == 0 or NE == 0:
        return None
    val = (CT * NB) / (BT * NE)
    return val

def circle_center_from_three_points(P1, P2, P3):
    # Solve for circumcenter using perpendicular bisector intersection
    # Using linear algebra.
    # Coordinates:
    x1, y1 = P1
    x2, y2 = P2
    x3, y3 = P3
    # Compute determinant:
    D = 2*(x1*(y2 - y3) + x2*(y3 - y1) + x3*(y1 - y2))
    if abs(D) < 1e-12:
        return None  # collinear
    # Compute center coordinates:
    ux = ((x1**2 + y1**2)*(y2 - y3) + (x2**2 + y2**2)*(y3 - y1) + (x3**2 + y3**2)*(y1 - y2)) / D
    uy = ((x1**2 + y1**2)*(x3 - x2) + (x2**2 + y2**2)*(x1 - x3) + (x3**2 + y3**2)*(x2 - x1)) / D
    return np.array([ux, uy])

def circumcircle_radius(P1, P2, P3, center):
    # radius = distance from center to any point
    return np.linalg.norm(center - P1)

def intersect_line_line(P1, d1, P2, d2):
    # intersect lines: P1 + t*d1 = P2 + s*d2
    # Solve 2x2 for t,s
    # d1, d2 vectors
    A = np.column_stack((d1, -d2))
    b = P2 - P1
    # Solve for t,s:
    # Might be singular if lines parallel.
    try:
        sol = np.linalg.lstsq(A, b, rcond=None)[0]
    except Exception as e:
        return None
    t = sol[0]
    return P1 + t*d1

def intersect_line_circle(line_point, line_dir, circle_center, radius):
    # Solve (P + t*d - O)^2 = r^2
    # t variable.
    # Let w = line_point - O
    w = line_point - circle_center
    d = line_dir
    # quadratic: (d·d) t^2 + 2(w·d) t + (w·w - r^2) = 0
    a = np.dot(d, d)
    b = 2*np.dot(w, d)
    c = np.dot(w, w) - radius**2
    disc = b**2 - 4*a*c
    if disc < -1e-12:
        return []  # no real
    disc = max(disc, 0.0)
    t1 = (-b + math.sqrt(disc)) / (2*a)
    t2 = (-b - math.sqrt(disc)) / (2*a)
    p1 = line_point + t1 * d
    p2 = line_point + t2 * d
    return [p1, p2]

def compute_geom(n, s):
    # Returns diff (distance error for cyclic condition) and ratio for given n, s.
    a = fib(n+2)
    BD = fib(n)
    CD = fib(n+1)
    b = s - BD
    c = s - CD
    if b <= 0 or c <= 0:
        return None
    # coordinates:
    B = np.array([0.0, 0.0])
    C = np.array([float(a), 0.0])
    # compute A:
    xA = (c**2 - b**2 + a**2) / (2.0 * a)
    val = c**2 - xA**2
    if val <= 0:
        return None
    h = math.sqrt(val)
    A = np.array([xA, h])
    # incircle radius r = area / s
    area = 0.5 * a * h
    r = area / s
    I = np.array([float(BD), r])
    D = np.array([float(BD), 0.0])
    # contact points:
    # foot from I onto line AC:
    # compute projection
    E = projection_point_to_line(I, A, C)
    # foot onto AB:
    F = projection_point_to_line(I, A, B)
    # circumcenter of ABC: but we need circle Omega. Use three points:
    O = circle_center_from_three_points(A, B, C)
    if O is None:
        return None
    R = np.linalg.norm(O - A)
    # circle through A, E, F: compute center:
    O2 = circle_center_from_three_points(A, E, F)
    if O2 is None:
        return None
    R2 = np.linalg.norm(O2 - A)
    # Find intersection of circles Omega and circle AEF: one is A, the other is K.
    # Solve for intersection of two circles: given two centers O, O2 and radii R, R2.
    # Intersection points can be computed analytically.
    # Compute distance d between centers:
    d = np.linalg.norm(O - O2)
    if d > R + R2 + 1e-9 or d < abs(R - R2) - 1e-9:
        # no intersection (should intersect at two points)
        return None
    # Compute a = (R^2 - R2^2 + d^2) / (2d)
    a_dist = (R**2 - R2**2 + d**2) / (2.0 * d)
    # Height from line connecting centers to intersection points:
    h_inter = math.sqrt(max(R**2 - a_dist**2, 0.0))
    # point P2 = O + a_dist * (O2-O)/d (point along line between centers)
    P2 = O + a_dist * (O2 - O) / d

    # unit perpendicular vector:
    perp = np.array([-(O2 - O)[1], (O2 - O)[0]]) / d
    inter1 = P2 + h_inter * perp
    inter2 = P2 - h_inter * perp
    # Determine which is A (or close). Choose K as the one not equal to A.
    # Check distances:
    if np.linalg.norm(inter1 - A) < 1e-6:
        K = inter2
    elif np.linalg.norm(inter2 - A) < 1e-6:

        if np.linalg.norm(inter1 - A) < np.linalg.norm(inter2 - A):

        else:

    N = projection_point_to_line(D, E, F)

    O_BNK = circle_center_from_three_points(B, N, K)
    if O_BNK is None:

    # distance from O_BNK to Kprime:
    diff = np.linalg.norm(O_BNK - Kprime) - R_BNK
    # Compute ratio:
    # Circle tangent to BN at B passing through K: center is intersection of perp bisector of BK and line through B perpendicular to BN.
    # Compute mid of BK:
    mid_BK = (B + K) / 2.0
    # direction of BK:
    dir_BK = K - B
    # perpendicular bisector line: passes through mid_BK, direction perp to dir_BK
    perp_dir_BK = np.array([-dir_BK[1], dir_BK[0]])  # direction vector
    # line L1: point = mid_BK, direction = perp_dir_BK
    # line through B perpendicular to BN:
    dir_BN = N - B
    perp_dir_BN = np.array([-dir_BN[1], dir_BN[0]])

    val = R_circle**2 - O_center[1]**2

    sqrt_val = math.sqrt(val)
    x1 = O_center[0] + sqrt_val
    x2 = O_center[0] - sqrt_val
    # one should be near 0 (B). Choose the other:
    # Determine which is close to 0:
    if abs(x1) < 1e-6:
        x_T = x2
    elif abs(x2) < 1e-6:
        x_T = x1
    else:
        # whichever is not 0:
        # compare which is farther from 0?
        # The correct T is the other intersection distinct from B (0).
        # Choose the one further away from 0.
        if abs(x1) > abs(x2):
            x_T = x1

    T = np.array([x_T, 0.0])

    NB = np.linalg.norm(N - B)

    if BT == 0 or NE == 0:

    ratio = (CT * NB) / (BT * NE)

def circumcenter(p1, p2, p3):
    # Compute circumcenter using linear algebra
    # Solve for center O = (ox, oy) such that |O-p1|^2 = |O-p2|^2 = |O-p3|^2
    # Use formula for intersection of perpendicular bisectors.
    # Use sympy? But numeric.
    A = np.array(p1); B = np.array(p2); C = np.array(p3)
    # Solve linear equations: (O - A).dot(O - A) = (O - B).dot(O - B) and = (O - C)
    # Subtract pairwise gives linear equations.
    # (O·O - 2 O·A + A·A) = (O·O - 2 O·B + B·B) => -2 O·A + A·A = -2 O·B + B·B => 2 O·(B - A) = B·B - A·A
    # Similarly for C-A.
    M = np.vstack([2*(B - A), 2*(C - A)])
    rhs = np.array([np.dot(B,B) - np.dot(A,A), np.dot(C,C) - np.dot(A,A)])
    # Solve linear system M * O = rhs
    # If M is singular, return None.
    try:
        O = np.linalg.solve(M, rhs)
    except np.linalg.LinAlgError:
        O = None
    return O

def reflect_point_across_line(P, L1, L2):
    # reflect point P across line defined by points L1, L2.
    # Using vector projection.
    # Let v = L2 - L1 (direction)
    # Normalize v? Not needed.
    v = L2 - L1
    # Project vector from L1 to P onto v: proj = L1 + ((P-L1)·v)/|v|^2 * v
    # The foot of perpendicular (closest point) is Q = L1 + ((P-L1)·v)/|v|^2 * v
    # Then reflected point P' = 2Q - P
    v_norm_sq = np.dot(v, v)
    if v_norm_sq == 0:
        return None
    t = np.dot(P - L1, v) / v_norm_sq
    Q = L1 + t * v
    P_ref = 2 * Q - P
    return P_ref

def foot_of_perpendicular(P, L1, L2):
    # foot from point P to line L1L2
    v = L2 - L1
    v_norm_sq = np.dot(v, v)
    if v_norm_sq == 0:
        return None
    t = np.dot(P - L1, v) / v_norm_sq
    Q = L1 + t * v
    return Q

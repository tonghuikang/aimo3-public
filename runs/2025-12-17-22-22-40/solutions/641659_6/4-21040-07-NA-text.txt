def triangle_coords(a,b,c):
    # side lengths: BC = a, CA = b, AB = c
    # place B at (0,0), C at (a,0)
    # find coordinates of A
    # using law of cos
    # x_A = (c^2 - b^2 + a^2) / (2*a)
    # y_A = sqrt(c^2 - x_A^2)
    xA = (c**2 - b**2 + a**2) / (2*a)
    y_sq = c**2 - xA**2
    if y_sq < 0:
        # degenerate triangle
        yA = 0
    else:
        yA = np.sqrt(y_sq)
    B = np.array([0.0,0.0])
    C = np.array([a,0.0])
    A = np.array([xA, yA])
    return A,B,C

def incircle_center_radius(a,b,c):
    # returns incircle center I (x,y) and radius r
    # using formula: incircle coordinates: I = (ax_A + bx_B + cx_C)/(a+b+c) etc? Actually formula for coordinates of incenter:
    # I = (a*A + b*B + c*C) / (a+b+c), where a=BC length opposite A, etc.
    # side lengths a,b,c correspond to BC, CA, AB respectively.
    # So assign vertices A,B,C as above.
    A,B,C = triangle_coords(a,b,c)
    # side lengths: a opposite A, so length a=BC, b opposite B (i.e., CA), c opposite C (i.e., AB)
    # Actually typical notation: side a = BC opposite A; side b = CA opposite B; side c = AB opposite C.
    # Use that.
    # Incenter coordinates: weighted sum by side lengths: (a*A + b*B + c*C) / (a+b+c)
    I = (a*A + b*B + c*C) / (a+b+c)

    # area via Heron
    area = math.sqrt(s*(s-a)*(s-b)*(s-c))
    r = area / s
    return I, r

def foot_point(P, Q, R):
    # foot of perpendicular from point P onto line QR
    # Q,R points on line
    Q = np.array(Q); R = np.array(R); P = np.array(P)
    # vector QR
    v = R - Q
    # projection of PQ onto v
    t = np.dot(P - Q, v) / np.dot(v, v)
    foot = Q + t*v
    return foot

def line_through(P,Q):
    # return coefficients (A,B,C) of line passing through P and Q: Ax + By + C = 0
    P = np.array(P); Q = np.array(Q)

    dx = Q[0]-P[0]; dy = Q[1]-P[1]
    A = dy; B = -dx; C = -(A*P[0] + B*P[1])

def intersect_lines(L1, L2):

def circle_from_three_points(P1,P2,P3):
    # return center, radius
    # using circumcenter
    P1 = np.array(P1); P2 = np.array(P2); P3 = np.array(P3)

    # Solve via linear algebra
    # Center O solves |O-P1|^2 = |O-P2|^2 = |O-P3|^2
    # Use equations
    A = 2*(P2-P1)
    B = 2*(P3-P1)
    C = np.array([np.dot(P2,P2) - np.dot(P1,P1), np.dot(P3,P3) - np.dot(P1,P1)])

def intersect_circles(O1,r1,O2,r2):
    # returns intersection points of two circles (two points or one)
    # using geometry.
    d = np.linalg.norm(O2 - O1)
    if d > r1 + r2 + 1e-12 or d < abs(r1 - r2) - 1e-12:
        return []  # no intersections
    # distance from O1 to line of intersection points
    a = (r1**2 - r2**2 + d**2) / (2*d)
    # height from that line to intersection points
    h_sq = r1**2 - a**2
    if h_sq < -1e-12:
        return []
    h = np.sqrt(max(h_sq,0.0))
    # point P2 = O1 + a*(O2-O1)/d
    P2 = O1 + a * (O2 - O1) / d
    # intersection points:
    # offset vector orthogonal to O2-O1
    # unit perpendicular
    # choose basis: (dx,dy) = O2-O1, perp = (-dy,dx)
    dx = (O2 - O1)[0] / d
    dy = (O2 - O1)[1] / d

    perp = np.array([-dy, dx])
    inter1 = P2 + h * perp
    inter2 = P2 - h * perp

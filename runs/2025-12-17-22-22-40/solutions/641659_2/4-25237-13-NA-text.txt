def compute_triangle_params(Fn, Fn1, Fn2):

def triangle_coords(a, d, c):
    # compute b = a + c - 2d
    b = a + c - 2*d
    # compute A coordinates
    # x_A = (a^2 + c^2 - b^2) / (2a)
    xA = (a*a + c*c - b*b) / (2*a)
    # yA = sqrt(c^2 - xA^2)
    # ensure positive
    temp = c*c - xA*xA
    if temp < 0:
        # numerical issues
        yA = 0.0
    else:
        yA = sqrt(temp)
    # points
    B = (0.0, 0.0)
    C = (float(a), 0.0)
    A = (xA, yA)
    return A, B, C, b

def point_on_AB(A, B, ratio):
    # return point on AB at distance BF = ratio from B
    # ratio is BF, B at (0,0)
    # AB vector = A - B = A
    c = sqrt(A[0]**2 + A[1]**2)
    t = ratio / c
    return (t*A[0], t*A[1])

def point_on_AC(A, C, ratio):
    # point on AC at distance CE = ratio from C towards A
    # vector from C to A = A - C
    # b = length of AC
    dx = A[0] - C[0]
    dy = A[1] - C[1]
    b = sqrt(dx*dx + dy*dy)
    t = ratio / b
    return (C[0] + t*dx, C[1] + t*dy)

def circumcircle_center(p1, p2, p3):
    # returns center (ox,oy) and radius R
    # using perpendicular bisectors
    (x1,y1), (x2,y2), (x3,y3) = p1, p2, p3
    # compute via solving linear equations for center (ox,oy):
    # (ox - x1)^2 + (oy - y1)^2 = (ox - x2)^2 + (oy - y2)^2
    # (ox - x1)^2 + (oy - y1)^2 = (ox - x3)^2 + (oy - y3)^2
    # which reduces to linear equations.
    d = 2*(x2 - x1)
    e = 2*(y2 - y1)
    f = x2*x2 + y2*y2 - x1*x1 - y1*y1
    g = 2*(x3 - x1)
    h = 2*(y3 - y1)
    i = x3*x3 + y3*y3 - x1*x1 - y1*y1
    # solve:
    # d*ox + e*oy = f
    # g*ox + h*oy = i
    det = d*h - e*g
    if abs(det) < 1e-12:
        # points collinear?
        return None, None
    ox = (f*h - e*i) / det
    oy = (d*i - f*g) / det
    R = sqrt((ox - x1)**2 + (oy - y1)**2)
    return (ox, oy), R

def intersect_circles(c1_center, R1, c2_center, R2):
    # returns intersection points of two circles (list of 0,1,2 points)
    (x0,y0) = c1_center
    (x1,y1) = c2_center
    d = sqrt((x1-x0)**2 + (y1-y0)**2)
    if d > R1 + R2 + 1e-12 or d < abs(R1 - R2) - 1e-12:
        # no intersection
        return []
    # a = (R1^2 - R2^2 + d^2) / (2 d)
    a = (R1*R1 - R2*R2 + d*d) / (2*d)
    # h = sqrt(R1^2 - a^2)
    h_sq = max(R1*R1 - a*a, 0.0)
    h = sqrt(h_sq)
    # point P2 = (x0 + a*(x1-x0)/d, y0 + a*(y1-y0)/d)
    x2 = x0 + a*(x1 - x0)/d
    y2 = y0 + a*(y1 - y0)/d
    # intersection points:
    # (x2 +- h*(y1 - y0)/d, y2 -+ h*(x1 - x0)/d)
    rx = -(y1 - y0) * (h / d)
    ry = (x1 - x0) * (h / d)
    p_int1 = (x2 + rx, y2 + ry)
    p_int2 = (x2 - rx, y2 - ry)
    # Might be same if h~0
    if h < 1e-12:
        return [p_int1]
    else:
        return [p_int1, p_int2]

def foot_of_perpendicular(P, Q, R):
    # foot of perpendicular from point P onto line QR
    # line Q->R direction vector
    (xP,yP) = P
    (xQ,yQ) = Q
    (xR,yR) = R
    vx = xR - xQ
    vy = yR - yQ
    # projection t = ((P - Q)·v) / (v·v)
    t = ((xP - xQ)*vx + (yP - yQ)*vy) / (vx*vx + vy*vy)
    # foot = Q + t*v
    return (xQ + t*vx, yQ + t*vy)

def reflect_point_across_line(P, Q, R):
    # reflect P across line QR
    # foot of perpendicular from P onto QR
    foot = foot_of_perpendicular(P, Q, R)
    # reflection: P' = foot*2 - P
    return (2*foot[0] - P[0], 2*foot[1] - P[1])

def concyclicity_det(B, K, N, Kp):
    # determinant for concyclicity: compute 4x4 matrix determinant
    import numpy as np
    mat = []
    for (x,y) in [B, K, N, Kp]:
        mat.append([x, y, x*x + y*y, 1.0])
    M = np.array(mat)
    det = np.linalg.det(M)
    return det

    def f_det(c):
        # compute determinant for cyclic condition
        # compute A,B,C,b
        try:
            A,B,C,b = triangle_coords(a,d,c)
        except Exception as e:
            return None
        # incircle contact points
        F = point_on_AB(A,B,d)

        D = (float(d), 0.0)
        N = foot_of_perpendicular(D, E, F)
        # circles
        O_ABC,R_ABC = circumcircle_center(A,B,C)
        O_AEF,R_AEF = circumcircle_center(A,E,F)
        if O_AEF is None:

        inters = intersect_circles(O_ABC,R_ABC,O_AEF,R_AEF)
        # find K distinct from A
        K = None
        for P in inters:
            if sqrt((P[0]-A[0])**2 + (P[1]-A[1])**2) > 1e-7:
                K = P
                break
        if K is None:
            return None
        Kp = reflect_point_across_line(K,E,F)
        det = concyclicity_det(B,K,N,Kp)
        return det
    return f_det

def circumcenter(p1, p2, p3):
    # p1,p2,p3 are tuples (x,y)
    (x1, y1), (x2, y2), (x3, y3) = p1, p2, p3
    D = 2 * (x1*(y2 - y3) + x2*(y3 - y1) + x3*(y1 - y2))
    if abs(D) < 1e-12:
        # points collinear or degenerate
        return None
    x1sq = x1*x1 + y1*y1
    x2sq = x2*x2 + y2*y2
    x3sq = x3*x3 + y3*y3
    ux = (x1sq*(y2 - y3) + x2sq*(y3 - y1) + x3sq*(y1 - y2)) / D
    uy = (x1sq*(x3 - x2) + x2sq*(x1 - x3) + x3sq*(x2 - x1)) / D
    return (ux, uy)

def compute_K(s, d1, d2):
    a = d1 + d2
    s_val = s
    b = s_val - d1
    c = s_val - d2
    if b <= 0 or c <= 0:
        return None
    # coordinates
    xA = (a*a - b*b + c*c) / (2*a)
    yA_sq = c*c - xA*xA
    if yA_sq <= 0:
        return None
    yA = math.sqrt(yA_sq)
    A = (xA, yA)
    B = (0.0, 0.0)
    C = (a, 0.0)
    # compute F
    F = (d1/c * xA, d1/c * yA)

def foot_point(P, Q, R):
    # foot of perpendicular from point P onto line QR
    # returns point as tuple
    # Q,R are points on line
    # vector QR = R - Q
    (xP, yP) = P
    (xQ, yQ) = Q
    (xR, yR) = R
    ux = xR - xQ
    uy = yR - yQ
    # projection scalar t = ((P-Q)·(U)) / (U·U)
    t = ((xP - xQ)*ux + (yP - yQ)*uy) / (ux*ux + uy*uy)
    # foot = Q + t*U
    return (xQ + t*ux, yQ + t*uy)

def reflect_point_across_line(P, Q, R):
    # reflect point P across line through Q,R
    # find foot H of P onto line QR, then reflected point = 2*H - P
    H = foot_point(P, Q, R)
    return (2*H[0] - P[0], 2*H[1] - P[1])

    A = (xA, yA)

    val = (math.hypot(Kprime[0] - O3[0], Kprime[1] - O3[1])**2 - R3**2)
    return val

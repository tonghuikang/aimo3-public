def circle_center_and_radius(p1, p2, p3):
    # compute circle from 3 points
    # Using perpendicular bisectors intersection
    (x1,y1),(x2,y2),(x3,y3)=p1,p2,p3
    # Solve linear equations for circumcenter (cx,cy)
    # (cx - x1)^2 + (cy - y1)^2 = (cx - x2)^2 + (cy - y2)^2
    # (cx - x1)^2 + (cy - y1)^2 = (cx - x3)^2 + (cy - y3)^2
    # Expand, simplify:
    A = np.array([[2*(x2-x1), 2*(y2-y1)],
                  [2*(x3-x1), 2*(y3-y1)]], dtype=float)
    B = np.array([x2**2 - x1**2 + y2**2 - y1**2,
                  x3**2 - x1**2 + y3**2 - y1**2], dtype=float)
    cx, cy = np.linalg.solve(A, B)
    R = math.hypot(cx-x1, cy-y1)
    return (cx,cy), R

def intersect_circles(O1,R1, O2,R2):
    # returns intersection points (x,y) list
    (x0,y0) = O1
    (x1,y1) = O2
    # distance between centers
    d = math.hypot(x1-x0, y1-y0)
    if d > R1+R2 or d < abs(R1-R2):
        return [] # no intersection
    # find a = (R1^2 - R2^2 + d^2) / (2d)
    a = (R1**2 - R2**2 + d**2) / (2*d)
    # height h = sqrt(R1^2 - a^2)
    h_sq = R1**2 - a**2
    if h_sq < 0:
        h = 0.0
    else:
        h = math.sqrt(h_sq)
    # point P2 = O1 + a*(O2-O1)/d
    x2 = x0 + a*(x1 - x0)/d
    y2 = y0 + a*(y1 - y0)/d
    # offset vectors for intersection points
    rx = -(y1 - y0) * (h/d)
    ry = (x1 - x0) * (h/d)
    # intersection points
    p_int1 = (x2 + rx, y2 + ry)
    p_int2 = (x2 - rx, y2 - ry)
    return [p_int1, p_int2]

def cyclic_determinant(B, K, N, Kp):
    # returns determinant of 4x4 matrix
    pts = [B, K, N, Kp]
    M = []
    for (x,y) in pts:
        M.append([x*x+y*y, x, y, 1])
    M = np.array(M, dtype=float)
    det = la.det(M)
    return det

    # M is 2x2 list of lists or matrix, RHS is list [rhs0, rhs1]

def circumcenter(P1,P2,P3):
    # Solve linear equations to find circumcenter
    # 2*(P2-P1).dot(O) = |P2|^2 - |P1|^2
    # 2*(P3-P1).dot(O) = |P3|^2 - |P1|^2
    a1 = 2*(P2[0]-P1[0])
    b1 = 2*(P2[1]-P1[1])
    c1 = P2[0]**2 + P2[1]**2 - (P1[0]**2 + P1[1]**2)
    a2 = 2*(P3[0]-P1[0])
    b2 = 2*(P3[1]-P1[1])
    c2 = P3[0]**2 + P3[1]**2 - (P1[0]**2 + P1[1]**2)
    # Solve [a1 b1; a2 b2] * [x; y] = [c1; c2]
    sol = solve_linear_2x2([[a1,b1],[a2,b2]],[c1,c2])
    if sol is None:
        return None
    return (sol[0], sol[1])

def intersect_circles(O1,R1,O2,R2):
    # O1, O2 as tuples/lists
    dx = O2[0] - O1[0]
    dy = O2[1] - O1[1]
    d = mp.sqrt(dx**2 + dy**2)
    if d == 0:
        return []
    # check for intersection
    if d > R1 + R2 + 1e-12 or d < abs(R1 - R2) - 1e-12:
        return []
    a = (R1**2 - R2**2 + d**2) / (2*d)
    h_sq = R1**2 - a**2
    if h_sq < 0:
        h = mp.mpf('0')
    else:
        h = mp.sqrt(h_sq)
    # point P2
    ex = dx / d
    ey = dy / d
    P2x = O1[0] + a * ex
    P2y = O1[1] + a * ey
    # perpendicular vector
    rx = -ey
    ry = ex
    inter1 = (P2x + h * rx, P2y + h * ry)
    inter2 = (P2x - h * rx, P2y - h * ry)
    return [inter1, inter2]

def reflect_point_across_line(P, L1, L2):
    # reflect P across line L1-L2
    v = (L2[0] - L1[0], L2[1] - L1[1])
    # projection t
    t = ((P[0] - L1[0])*v[0] + (P[1] - L1[1])*v[1]) / (v[0]**2 + v[1]**2)
    Q = (L1[0] + t*v[0], L1[1] + t*v[1])
    # reflection
    return (2*Q[0] - P[0], 2*Q[1] - P[1])

def compute_ratio_and_condition(n, r):
    Fn = fibonacci(n)
    Fn1 = fibonacci(n+1)
    B = (-Fn, 0.0)
    C = (Fn1, 0.0)
    # incircle center I = (0,r)
    # F coordinates:
    denomF = r**2 + Fn**2
    xF = -2*Fn * r**2 / denomF
    yF = 2*Fn**2 * r / denomF
    F = (xF, yF)
    denomE = r**2 + Fn1**2
    xE = 2*Fn1 * r**2 / denomE
    yE = 2*Fn1**2 * r / denomE
    E = (xE, yE)
    # line EF direction
    vEF = (E[0] - F[0], E[1] - F[1])
    # N projection of origin onto line EF
    # projection of O onto line through F with direction vEF
    t_param = ((0 - F[0])*vEF[0] + (0 - F[1])*vEF[1]) / (vEF[0]**2 + vEF[1]**2)
    N = (F[0] + t_param*vEF[0], F[1] + t_param*vEF[1])
    # Intersection A of lines BF and CE
    # line BF: B + s*(F-B)
    # line CE: C + t*(E-C)
    u = (F[0] - B[0], F[1] - B[1])
    w = (E[0] - C[0], E[1] - C[1])
    # Solve B + s*u = C + t*w -> s*u - t*w = C - B
    M = [[u[0], -w[0]], [u[1], -w[1]]]
    RHS = (C[0] - B[0], C[1] - B[1])
    sol = solve_linear_2x2(M, RHS)
    if sol is None:
        return None
    s_param, t_param2 = sol
    A = (B[0] + s_param*u[0], B[1] + s_param*u[1])
    # circumcenters
    O_abc = circumcenter(A, B, C)
    if O_abc is None:
        return None
    O_afe = circumcenter(A, F, E)
    if O_afe is None:
        return None
    R_abc = norm2((A[0] - O_abc[0], A[1] - O_abc[1]))
    R_afe = norm2((A[0] - O_afe[0], A[1] - O_afe[1]))
    # intersection circles
    inters = intersect_circles(O_abc, R_abc, O_afe, R_afe)
    if len(inters) == 0:
        return None
    # Choose K not equal to A
    K = None
    for pt in inters:
        if norm2((pt[0] - A[0], pt[1] - A[1])) > 1e-8:
            K = pt
            break
    if K is None:
        return None
    # K'
    Kprime = reflect_point_across_line(K, F, E)
    # Circle through K, B, Kprime
    O_circ = circumcenter(K, B, Kprime)
    if O_circ is None:
        return None
    R_circ = norm2((K[0] - O_circ[0], K[1] - O_circ[1]))
    circ_err = norm2((N[0] - O_circ[0], N[1] - O_circ[1])) - R_circ
    # Compute T: circle through B and K tangent to BN at B.
    # BN direction vector
    BN = (N[0] - B[0], N[1] - B[1])
    # perpendicular direction v_perp (unit)
    v_perp = (-BN[1], BN[0])
    v_perp_norm = norm2(v_perp)
    if v_perp_norm == 0:
        return None
    v_perp = (v_perp[0]/v_perp_norm, v_perp[1]/v_perp_norm)
    # unknown center O_t = B + s * v_perp
    # condition |O_t - B| = |O_t - K|
    # => s^2 = |B + s*v_perp - K|^2 => |B-K + s*v_perp|^2 = s^2
    # Expand: |B-K|^2 + 2s (B-K)·v_perp + s^2 = s^2 => |B-K|^2 + 2s (B-K)·v_perp = 0
    BK = (B[0] - K[0], B[1] - K[1])
    denom = 2 * (BK[0]*v_perp[0] + BK[1]*v_perp[1])
    if abs(denom) < 1e-12:
        return None
    s_val = - (BK[0]**2 + BK[1]**2) / denom
    O_t = (B[0] + s_val*v_perp[0], B[1] + s_val*v_perp[1])
    R_t = norm2((O_t[0] - B[0], O_t[1] - B[1]))
    # Intersect with BC (y=0)
    # Solve (x - O_t.x)^2 + O_t.y^2 = R_t^2 -> (x - O_t.x)^2 = R_t^2 - O_t.y^2
    val = R_t**2 - O_t[1]**2
    if val < -1e-12:
        return None
    if val < 0:
        val = 0
    sqrt_val = mp.sqrt(val)
    x1 = O_t[0] + sqrt_val
    x2 = O_t[0] - sqrt_val
    # Choose x_T not equal to B.x = -Fn
    if abs(x1 + Fn) < 1e-7:
        x_T = x2
    elif abs(x2 + Fn) < 1e-7:
        x_T = x1
    else:
        # choose the one farther from B maybe
        if abs(x1 + Fn) < abs(x2 + Fn):
            x_T = x2
        else:
            x_T = x1
    CT = abs(Fn1 - x_T)
    BT = abs(x_T + Fn)
    NB = norm2((N[0] - B[0], N[1] - B[1]))
    NE = norm2((N[0] - E[0], N[1] - E[1]))
    ratio = (CT * NB) / (BT * NE)
    return float(ratio), float(circ_err), {
        'B':B, 'C':C, 'F':F, 'E':E, 'N':N, 'A':A, 'K':K, 'Kprime':Kprime,
        'O_circ':O_circ, 'R_circ':R_circ, 'circ_err':circ_err,
        'ratio':ratio, 'x_T':x_T, 'CT':CT, 'BT':BT, 'NB':NB, 'NE':NE,
        'O_abc':O_abc, 'O_afe':O_afe
    }

def find_r_for_n(n, r_low=1e-6, r_high=10):

    lo = r_low
    hi = r_high
    f_lo = f(lo)
    f_hi = f(hi)

    while (f_lo is None or f_hi is None or f_lo*f_hi > 0) and it<50:
        hi *= 2
        f_hi = f(hi)
        it+=1
    if f_lo is None or f_hi is None:
        return None
    if f_lo*f_hi > 0:
        # try decreasing lo
        lo = 1e-12
        f_lo = f(lo)
        if f_lo is None or f_lo*f_hi > 0:
            # no sign change
            return None
    # Bisection
    for _ in range(80):

            lo = mid

    r_sol = (lo+hi)/2
    ratio, err, _ = compute_ratio_and_condition(n, r_sol)
    return r_sol, ratio, err

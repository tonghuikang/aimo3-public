def circumcenter(p1, p2, p3):
    # compute circumcenter of triangle given as (x,y)
    (x1,y1),(x2,y2),(x3,y3) = p1,p2,p3
    # using perpendicular bisector intersection formula
    d = 2*(x1*(y2-y3) + x2*(y3-y1) + x3*(y1-y2))
    if abs(d) < 1e-12:
        return None
    ux = ((x1**2+y1**2)*(y2-y3) + (x2**2+y2**2)*(y3-y1) + (x3**2+y3**2)*(y1-y2))/d
    uy = ((x1**2+y1**2)*(x3-x2) + (x2**2+y2**2)*(x1-x3) + (x3**2+y3**2)*(x2-x1))/d
    return (ux, uy)

def intersect_circles(o1, r1, o2, r2):
    # return two intersection points (or one if tangent)
    x1,y1 = o1
    x2,y2 = o2
    dx = x2 - x1
    dy = y2 - y1
    d = math.hypot(dx, dy)
    if d > r1 + r2 + 1e-12 or d < abs(r1 - r2) - 1e-12:
        return []  # no intersection
    # a = (r1^2 - r2^2 + d^2) / (2d)
    a = (r1*r1 - r2*r2 + d*d) / (2*d)
    # height from line to intersection points
    h_sq = r1*r1 - a*a
    if h_sq < 0:
        h = 0.0
    else:
        h = math.sqrt(h_sq)
    # point P0 = o1 + a*(o2-o1)/d
    xm = x1 + a*dx/d
    ym = y1 + a*dy/d

    rx = -dy * (h/d)

    p_int1 = (xm + rx, ym + ry)

    return [p_int1, p_int2]

def foot_of_perp(P, A, B):

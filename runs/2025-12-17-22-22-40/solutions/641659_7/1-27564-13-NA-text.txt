from math import sqrt

def circumcenter(P, Q, R):
    # P,Q,R are tuples (x,y)
    (x1,y1) = P
    (x2,y2) = Q
    (x3,y3) = R
    D = 2*(x1*(y2 - y3) + x2*(y3 - y1) + x3*(y1 - y2))
    if abs(D) < 1e-12:
        return None
    Ux = ((x1**2 + y1**2)*(y2 - y3) + (x2**2 + y2**2)*(y3 - y1) + (x3**2 + y3**2)*(y1 - y2)) / D
    Uy = ((x1**2 + y1**2)*(x3 - x2) + (x2**2 + y2**2)*(x1 - x3) + (x3**2 + y3**2)*(x2 - x1)) / D
    return (Ux, Uy)

def dist(P, Q):
    return sqrt((P[0]-Q[0])**2 + (P[1]-Q[1])**2)

def circle_intersections(O1, R1, O2, R2):
    # return intersection points of circles O1,R1 and O2,R2

        return []

        # no intersection
        return []

def foot_of_perpendicular(P, Q1, Q2):
    # foot of perpendicular from P to line through Q1 and Q2
    # returns point
    # vector Q2-Q1
    (x0,y0) = P
    (x1,y1) = Q1
    (x2,y2) = Q2
    dx = x2 - x1
    dy = y2 - y1
    if dx == 0 and dy == 0:
        return Q1
    # param t for projection onto line
    t = ((x0 - x1)*dx + (y0 - y1)*dy) / (dx*dx + dy*dy)
    proj_x = x1 + t*dx
    proj_y = y1 + t*dy
    return (proj_x, proj_y)

    foot = foot_of_perpendicular(P, Q1, Q2)

    return (2*foot[0] - P[0], 2*foot[1] - P[1])

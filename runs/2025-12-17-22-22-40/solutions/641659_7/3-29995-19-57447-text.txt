def circle_center(p1, p2, p3):
    # return center (x,y) and radius of circumcircle of three points p1,p2,p3
    (x1,y1),(x2,y2),(x3,y3) = p1,p2,p3

def foot_point_to_line(P, L1, L2):
    # foot from point P onto line L1-L2 (line infinite)
    (x0,y0) = P
    (x1,y1) = L1
    (x2,y2) = L2
    # projection formula
    dx = x2 - x1
    dy = y2 - y1
    if dx==0 and dy==0:
        return None
    t = ((x0 - x1)*dx + (y0 - y1)*dy) / (dx*dx + dy*dy)
    # foot point
    xf = x1 + t*dx
    yf = y1 + t*dy
    return (xf, yf)

def intersect_line_circle(line_point, line_dir, circle_center, radius):

    # circle2: circumcircle of A, B, C
    center2, r2 = circle_center(A, B, C)
    # Compute intersection of two circles; we need the point other than A
    # Use geometry: solve intersection of circles
    (x0,y0), R0 = center1, r1
    (x1,y1), R1 = center2, r2
    # Compute distance between centers
    d = sqrt((x1-x0)**2 + (y1-y0)**2)
    if d == 0:
        return None
    # Find intersection points (two) using standard formula
    # a = (R0^2 - R1^2 + d^2) / (2d)
    a = (R0**2 - R1**2 + d**2) / (2*d)
    # height h = sqrt(R0^2 - a^2)
    h_sq = R0**2 - a**2
    if h_sq < -1e-9:
        # no intersection
        return None
    if h_sq < 0:
        h_sq = 0
    h = sqrt(h_sq)
    # point P2 = P0 + a*(P1-P0)/d
    x2 = x0 + a*(x1-x0)/d
    y2 = y0 + a*(y1-y0)/d
    # intersection points:
    rx = -(y1 - y0) * (h/d)
    ry = (x1 - x0) * (h/d)
    intersect1 = (x2 + rx, y2 + ry)
    intersect2 = (x2 - rx, y2 - ry)
    # Choose the one not equal (close) to A
    def dist(p,q):
        return sqrt((p[0]-q[0])**2 + (p[1]-q[1])**2)
    if dist(intersect1, A) > 1e-6:
        return intersect1
    else:
        return intersect2

def reflect_across_line(P, L1, L2):
    # reflect point P across line defined by L1-L2
    # using formula: projection onto line then symmetrical
    proj = foot_point_to_line(P, L1, L2)
    # reflect: P' = (2*proj - P)
    return (2*proj[0] - P[0], 2*proj[1] - P[1])

def check_cyclic_condition(geom):
    K = compute_K(geom)
    if K is None:
        return False, None
    N = compute_N(geom)
    B = geom['B']
    O = compute_O_circumcenter_BKN(B,K,N)
    # Check if O lies on line EF
    E = geom['E']; F = geom['F']
    # Compute distance of O from line EF
    # using cross product
    # line EF: vector v = (F - E)
    v = (F[0] - E[0], F[1] - E[1])
    # vector from E to O
    w = (O[0] - E[0], O[1] - E[1])
    # cross product magnitude
    area2 = abs(v[0]*w[1] - v[1]*w[0])
    # distance = area2 / |v|
    line_len = sqrt(v[0]**2 + v[1]**2)
    dist = area2 / line_len if line_len != 0 else None
    # Check near zero
    ok = (dist is not None) and (dist < 1e-6)
    return ok, (K,N,O,dist)

def compute_T_and_ratio(geom):
    # compute T and ratio (CT*NB)/(BT*NE)
    B = geom['B']; C = geom['C']; 
    K = compute_K(geom)
    if K is None:
        return None
    N = compute_N(geom)
    # line BN: direction from B to N
    BN_dir = (N[0] - B[0], N[1] - B[1])
    # Perpendicular direction to BN (for radius direction at B)
    # Circle center O' lies on line through B perpendicular to BN
    # So direction vector of that line is perpendicular to BN: ( -BN_dir[1], BN_dir[0] )
    perp_dir = (-BN_dir[1], BN_dir[0])
    # Also O' must be on perpendicular bisector of BK
    # So O' is intersection of line L1: B + t1 * perp_dir and line L2: midpoint of BK + u * direction of perpendicular to BK?
    # Actually perpendicular bisector of BK passes through midpoint M of BK and has direction perpendicular to BK.
    # So we solve for intersection.
    M_mid = ((B[0]+K[0])/2.0, (B[1]+K[1])/2.0)
    # direction of BK
    BK_dir = (K[0]-B[0], K[1]-B[1])
    # direction of perpendicular bisector = perp to BK_dir = (-BK_dir[1], BK_dir[0])
    perp_BK_dir = (-BK_dir[1], BK_dir[0])
    # Solve for intersection of lines: B + t*perp_dir = M_mid + u*perp_BK_dir
    # Write as B + t*perp_dir - M_mid = u*perp_BK_dir
    # Two equations linear in t and u.
    # Solve using matrix.
    import numpy as np
    A_mat = np.array([[perp_dir[0], -perp_BK_dir[0]],
                      [perp_dir[1], -perp_BK_dir[1]]], dtype=float)
    rhs = np.array([M_mid[0] - B[0], M_mid[1] - B[1]], dtype=float)
    try:
        sol = np.linalg.solve(A_mat, rhs)
    except np.linalg.LinAlgError:
        # lines parallel (maybe degenerate)
        return None
    t = sol[0]
    # center O_prime:
    O_prime = (B[0] + t*perp_dir[0], B[1] + t*perp_dir[1])
    # radius = distance O_prime to B
    R = sqrt((O_prime[0]-B[0])**2 + (O_prime[1]-B[1])**2)
    # Intersect circle with BC line (y=0). BC from B (0,0) to C (a,0). So intersection with x-axis.
    # Solve (x - O'_x)^2 + (0 - O'_y)^2 = R^2 => (x - O'_x)^2 = R^2 - O'_y^2.
    # Solutions: x = O'_x +/- sqrt(R^2 - O'_y^2).
    # One solution is x = 0 (point B). The other is T (x_T).
    # Let's compute both, pick the one not B.
    disc = R**2 - O_prime[1]**2
    if disc < -1e-9:
        return None
    if disc < 0:
        disc = 0.0
    sqrt_disc = sqrt(disc)
    x1 = O_prime[0] + sqrt_disc
    x2 = O_prime[0] - sqrt_disc
    # Choose the one that is not 0 (within tolerance)
    T_x = x1 if abs(x1) > 1e-6 else x2
    T = (T_x, 0.0)
    # Now compute distances needed:
    # CT = distance from C to T = |C_x - T_x|
    CT = abs(C[0] - T_x)
    BT = abs(T_x - B[0]) # should be just T_x
    NB = sqrt((N[0]-B[0])**2 + (N[1]-B[1])**2)
    NE = sqrt((N[0]-geom['E'][0])**2 + (N[1]-geom['E'][1])**2)
    ratio = (CT * NB) / (BT * NE)
    return ratio, {'K':K, 'N':N, 'T':T, 'CT':CT, 'BT':BT,'NB':NB,'NE':NE}

    # compute r^2 using formula
    r_sq = a * c * F_n / (c + F_np1) - F_n**2
    if r_sq <= 0:

    r = sp.sqrt(r_sq)
    # compute A coordinates using formulas
    x_A = c * (F_n**2 - r_sq) / (F_n**2 + r_sq)
    y_A = (2 * c * r * F_n) / (F_n**2 + r_sq)
    # Ensure that AB length is c (should be)
    # Compute B (0,0), C (a,0)

def compute_diff_numeric(n, c_val):
    # Compute Fibonacci numbers for n
    def fib(k):
        a, b = 0, 1
        for _ in range(k):
            a, b = b, a + b
        return a
    F_n = fib(n)
    F_np1 = fib(n+1)
    a_len = fib(n+2)
    # Compute r^2
    r_sq = a_len * c_val * F_n / (c_val + F_np1) - F_n**2
    if r_sq <= 0:
        return None
    r = mp.sqrt(r_sq)
    # Coordinates
    x_A = c_val * (F_n**2 - r_sq) / (F_n**2 + r_sq)
    y_A = (2 * c_val * r * F_n) / (F_n**2 + r_sq)
    # Points
    B = mp.matrix([0,0])
    C = mp.matrix([a_len,0])
    I = mp.matrix([F_n, r])
    A = mp.matrix([x_A, y_A])
    # F point
    F_pt = mp.matrix([F_n * x_A / c_val, F_n * y_A / c_val])
    # E point
    # compute b length
    b_len = mp.sqrt((x_A - a_len)**2 + y_A**2)
    t = F_np1 / b_len
    E_pt = mp.matrix([a_len + t * (x_A - a_len), t * y_A])
    # D point
    D_pt = mp.matrix([F_n, 0])
    # N foot from D to line EF
    # line EF param: point F_pt + s*(E_pt - F_pt)
    # projection of vector D-F onto direction of line
    dir_EF = E_pt - F_pt
    # projection scalar s = dot((D-F), dir) / |dir|^2
    s = mp.dot(D_pt - F_pt, dir_EF) / mp.dot(dir_EF, dir_EF)
    N_pt = F_pt + s * dir_EF
    # Circle ABC center O_ABC (circumcenter)
    # We can compute intersection of circles (ABC) and (AEF) to find K.
    # Compute circumcircle of ABC:
    # Using formula for circumcenter from coordinates
    # function to compute circumcenter given three points
    def circumcenter(P, Q, R):
        # returns center (x,y) and radius
        # Using perpendicular bisectors intersection
        D = 2*(P[0]*(Q[1]-R[1]) + Q[0]*(R[1]-P[1]) + R[0]*(P[1]-Q[1]))
        if abs(D) < 1e-12:
            return None
        ux = ((P[0]**2+P[1]**2)*(Q[1]-R[1]) + (Q[0]**2+Q[1]**2)*(R[1]-P[1]) + (R[0]**2+R[1]**2)*(P[1]-Q[1]))/D
        uy = ((P[0]**2+P[1]**2)*(R[0]-Q[0]) + (Q[0]**2+Q[1]**2)*(P[0]-R[0]) + (R[0]**2+R[1]**2)*(Q[0]-P[0]))/D
        O = mp.matrix([ux, uy])
        Rrad = mp.sqrt((P[0]-ux)**2 + (P[1]-uy)**2)
        return O, Rrad
    O_ABC, R_ABC = circumcenter(A, B, C)
    O_AEF, R_AEF = circumcenter(A, E_pt, F_pt)
    # Intersection of two circles: find K (other than A)
    # Compute intersection using formula: intersection points of circles centered O1,R1 and O2,R2.
    # Vector between centers:
    O1 = O_ABC
    O2 = O_AEF
    d = mp.sqrt((O2[0]-O1[0])**2 + (O2[1]-O1[1])**2)
    # There are two intersection points. Compute using standard formula:
    # a = (R1^2 - R2^2 + d^2) / (2d)
    a_dist = (R_ABC**2 - R_AEF**2 + d**2) / (2*d)
    # Height from line connecting centers to intersection points:
    h_sq = R_ABC**2 - a_dist**2
    if h_sq < 0:
        # No intersection or tangent
        h = 0
    else:
        h = mp.sqrt(h_sq)

    P2y = O1[1] + a_dist * (O2[1]-O1[1]) / d
    # Intersection points:
    # x3 = P2x +- h*(O2[1]-O1[1])/d
    # y3 = P2y -+ h*(O2[0]-O1[0])/d
    # There are two solutions; we need one that is not A
    # Compute both:
    offset_x = h * (O2[1]-O1[1]) / d
    offset_y = -h * (O2[0]-O1[0]) / d
    K1 = mp.matrix([P2x + offset_x, P2y + offset_y])
    K2 = mp.matrix([P2x - offset_x, P2y - offset_y])

    diff = mp.norm(Kp_pt - O_BNK) - R_BNK

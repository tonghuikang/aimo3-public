def reflect_point_across_line(P, P1, P2):
    """
    Reflect point P across line defined by points P1,P2.
    Return reflected point.
    """
    # line vector
    x1,y1 = P1
    x2,y2 = P2
    vx = x2 - x1
    vy = y2 - y1
    # compute projection of (P - P1) onto line
    # param t = ((P-P1)·v) / (v·v)
    px,py = P
    wx = px - x1
    wy = py - y1
    vnorm2 = vx*vx + vy*vy
    if vnorm2 == 0:
        raise ValueError("Line degenerate")
    t = (wx*vx + wy*vy) / vnorm2
    # foot point
    fx = x1 + t*vx
    fy = y1 + t*vy
    # reflected point = 2*foot - P
    rx = 2*fx - px
    ry = 2*fy - py
    return (rx, ry)

def cyclic_condition_det(B, K, N, Kp):
    """
    Determine if points B, K, N, K' are concyclic using determinant condition.
    Compute 4x4 determinant:
    | x y x^2+y^2 1 |
    """
    # construct matrix
    def row(P):
        x,y = P
        return [x, y, x*x + y*y, 1.0]
    M = np.array([row(B), row(K), row(N), row(Kp)], dtype=float)
    # Compute determinant
    det = np.linalg.det(M)
    return det

def compute_ratio(a, B, C, T, N, E):

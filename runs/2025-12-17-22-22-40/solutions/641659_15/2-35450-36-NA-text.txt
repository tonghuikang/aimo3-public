    # foot of perpendicular from P onto line AB
    (px, py) = P
    (ax, ay) = A
    (bx, by) = B
    ABx = bx - ax
    ABy = by - ay
    APx = px - ax
    APy = py - ay
    # t projection
    denom = ABx**2 + ABy**2
    if denom == 0:
        return A  # degenerate
    t = (APx*ABx + APy*ABy) / denom
    return (ax + t*ABx, ay + t*ABy)

def circle_from_three(p1,p2,p3):
    (x1,y1), (x2,y2), (x3,y3) = p1,p2,p3
    d = 2*(x1*(y2-y3) + x2*(y3-y1) + x3*(y1-y2))
    if abs(d) < 1e-12:
        raise ValueError("collinear")
    ux = ((x1**2+y1**2)*(y2-y3) + (x2**2+y2**2)*(y3-y1) + (x3**2+y3**2)*(y1-y2)) / d
    uy = ((x1**2+y1**2)*(x3-x2) + (x2**2+y2**2)*(x1-x3) + (x3**2+y3**2)*(x2-x1)) / d
    r = sqrt((ux-x1)**2 + (uy-y1)**2)
    return (ux, uy), r

def intersect_circles(c1, r1, c2, r2):
    (x0,y0) = c1
    (x1,y1) = c2
    d = sqrt((x1-x0)**2 + (y1-y0)**2)
    if d < 1e-12:
        return []
    if d > r1 + r2 + 1e-12 or d < abs(r1 - r2) - 1e-12:
        return []
    a = (r1**2 - r2**2 + d**2) / (2*d)
    h2 = r1**2 - a**2
    if h2 < -1e-12:
        return []
    h = sqrt(max(h2,0.0))
    x2 = x0 + a*(x1-x0)/d
    y2 = y0 + a*(y1-y0)/d
    rx = -(y1-y0) * (h / d)
    ry = (x1-x0) * (h / d)
    p3 = (x2 + rx, y2 + ry)
    p4 = (x2 - rx, y2 - ry)
    return [p3, p4]

    fa = func(a)
    fb = func(b)
    if fa is None or fb is None:
        raise ValueError("Function undefined at endpoints")
    if fa * fb > 0:
        raise ValueError("Root not bracketed")

            a = c

            fb = fc
        else:
            a = c
            fa = fc
    return (a+b)/2

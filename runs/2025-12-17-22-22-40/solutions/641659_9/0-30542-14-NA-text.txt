from math import sqrt, isclose

def fib(n):
    # compute fib numbers F_n with F_0 = 0, F_1 = 1
    if n < 0:
        raise ValueError
    a, b = 0, 1
    for i in range(n):
        a, b = b, a + b
    return a

def incircle_data(Fn, Fnp1, r):
    # Fn = BD, Fnp1 = CD
    # Coordinates
    B = (0.0, 0.0)
    a = Fn + Fnp1  # BC length = F_{n+2}
    C = (float(a), 0.0)
    D = (float(Fn), 0.0)
    I = (float(Fn), r)
    # contact point F:
    denomF = Fn*Fn + r*r
    Fx = Fn * (Fn*Fn - r*r) / denomF
    Fy = 2 * r * Fn*Fn / denomF
    F = (Fx, Fy)
    # contact point E:
    # using C as origin shift
    denomE = Fnp1*Fnp1 + r*r
    Ex = a - Fnp1 * (Fnp1*Fnp1 - r*r) / denomE
    Ey = 2 * r * Fnp1*Fnp1 / denomE
    E = (Ex, Ey)
    return B, C, D, I, F, E

def intersect_lines(p1, d1, p2, d2):
    # line1: p1 + t * d1, line2: p2 + s * d2
    # returns intersection point (assuming not parallel)
    # solve p1 + t d1 = p2 + s d2 => two equations
    # solve for t,s via 2x2
    # d1 = (dx1, dy1), d2 = (dx2, dy2)
    dx1, dy1 = d1
    dx2, dy2 = d2
    # Solve:
    # p1.x + t*dx1 = p2.x + s*dx2
    # p1.y + t*dy1 = p2.y + s*dy2
    # => t*dx1 - s*dx2 = p2.x - p1.x
    #    t*dy1 - s*dy2 = p2.y - p1.y
    # Solve for t using linear equations
    # Represent as matrix [[dx1, -dx2],[dy1, -dy2]] * [t,s]^T = [p2.x - p1.x, p2.y - p1.y]^T
    A = [[dx1, -dx2],[dy1, -dy2]]
    Bvec = [p2[0]-p1[0], p2[1]-p1[1]]
    # Compute determinant
    det = A[0][0]*A[1][1] - A[0][1]*A[1][0]
    if abs(det) < 1e-12:
        return None
    inv_det = 1/det
    t = ( Bvec[0]*A[1][1] - Bvec[1]*A[0][1] ) * inv_det
    # s = (A[0][0]*Bvec[1] - A[1][0]*Bvec[0]) / det # not needed
    # Intersection point
    return (p1[0] + t*dx1, p1[1] + t*dy1)

def circumcenter(p1,p2,p3):
    # Return circumcenter and radius
    # Using formula based on perpendicular bisectors
    # Solve linear equations for center O = (ox, oy) such that |O-p1|=|O-p2|=|O-p3|
    # Use pairwise equations
    x1,y1 = p1
    x2,y2 = p2
    x3,y3 = p3
    # Compute using determinant method
    d = 2*(x1*(y2-y3) + x2*(y3-y1) + x3*(y1-y2))
    if abs(d) < 1e-12:
        return None
    ux = ((x1*x1 + y1*y1)*(y2 - y3) + (x2*x2 + y2*y2)*(y3 - y1) + (x3*x3 + y3*y3)*(y1 - y2)) / d
    uy = ((x1*x1 + y1*y1)*(x3 - x2) + (x2*x2 + y2*y2)*(x1 - x3) + (x3*x3 + y3*y3)*(x2 - x1)) / d
    O = (ux, uy)
    R = sqrt( (ux-x1)**2 + (uy-y1)**2 )
    return O, R

def circle_intersections(O1,R1,O2,R2):
    # Return list of intersection points (0, 1, or 2)
    x1,y1 = O1
    x2,y2 = O2
    d = sqrt((x2-x1)**2 + (y2-y1)**2)
    if d > R1+R2+1e-12 or d < abs(R1-R2)-1e-12:
        return []  # no intersection
    if d < 1e-12 and abs(R1-R2) < 1e-12:
        # coincident circles: infinite intersections
        return []
    # compute a = (R1^2 - R2^2 + d^2) / (2d)
    a = (R1*R1 - R2*R2 + d*d) / (2*d)
    # h = sqrt(R1^2 - a^2)
    # note numeric issues: sometimes due to precision, negative small
    tmp = R1*R1 - a*a
    if tmp < -1e-12:
        return []
    h = sqrt(max(0.0, tmp))
    # point P2 = O1 + a*(O2-O1)/d
    xm = x1 + a*(x2-x1)/d
    ym = y1 + a*(y2-y1)/d
    # intersection points:
    rx = -(y2 - y1) * (h/d)
    ry = (x2 - x1) * (h/d)
    p_int1 = (xm + rx, ym + ry)
    p_int2 = (xm - rx, ym - ry)
    if h < 1e-12:
        return [p_int1]
    else:
        return [p_int1, p_int2]

def reflect_point_across_line(P, A, B):
    # reflect point P across line AB
    # line AB defined by points A(x1,y1), B(x2,y2)
    # projection formula: foot Q = A + ((AP·AB)/|AB|^2) * AB
    # then reflect P: P' = 2Q - P
    x1,y1 = A
    x2,y2 = B
    px,py = P
    # vector AB:
    ABx = x2 - x1
    ABy = y2 - y1
    # vector AP:
    APx = px - x1
    APy = py - y1
    AB_sq = ABx*ABx + ABy*ABy
    if AB_sq == 0:
        raise ValueError("Zero length line for reflection")
    t = (APx*ABx + APy*ABy) / AB_sq
    Qx = x1 + t*ABx
    Qy = y1 + t*ABy
    # reflection:
    P_rx = 2*Qx - px
    P_ry = 2*Qy - py
    return (P_rx, P_ry)

def foot_of_perpendicular(P, A, B):
    # foot from point P to line AB
    # similar to projection: foot H = A + ((AP·AB)/|AB|^2) * AB
    x1,y1 = A
    x2,y2 = B
    px,py = P
    ABx = x2 - x1
    ABy = y2 - y1
    APx = px - x1
    APy = py - y1
    AB_sq = ABx*ABx + ABy*ABy
    if AB_sq == 0:
        raise ValueError("Zero length line")
    t = (APx*ABx + APy*ABy) / AB_sq
    Hx = x1 + t*ABx
    Hy = y1 + t*ABy
    return (Hx, Hy)

def cyclic_condition(B, K, N, Kp):
    # check if B,K,N,K' concyclic via determinant condition
    # B at (0,0) might simplify
    # Compute determinant:
    # | x y x^2+y^2 1 |
    import numpy as np
    M = np.array([
        [B[0], B[1], B[0]**2+B[1]**2, 1.0],
        [K[0], K[1], K[0]**2+K[1]**2, 1.0],
        [N[0], N[1], N[0]**2+N[1]**2, 1.0],
        [Kp[0], Kp[1], Kp[0]**2+Kp[1]**2, 1.0]
    ])
    det = np.linalg.det(M)
    return det

def ratio_R(a, u, NB, NE):
    # a = BC length (a), u = coordinate of center O_T's x-coordinate
    # BT = 2u, CT = a - 2u
    # Ratio = (CT*NB)/(BT*NE)
    BT = 2*u
    CT = a - 2*u
    if BT == 0 or NE == 0:
        return None
    return (CT * NB) / (BT * NE)

                mid = (lo+hi)/2
                data_mid = compute_ratio_for_r(Fn, Fnp1, mid)
                if data_mid is None:
                    # adjust maybe?
                    # if mid is too small causing degenerate, move lo up
                    lo = mid
                    continue
                fm = data_mid["det"]
                if fm == 0:
                    lo = hi = mid
                    break
                if v1*fm < 0:
                    hi = mid
                else:
                    lo = mid
            root = (lo+hi)/2
            solutions.append(root)
    # Also check if any solutions near 0? we ignore.
    # Remove duplicates within tolerance
    uniq = []
    for r in solutions:
        if all(abs(r - r2) > 1e-6 for r2 in uniq):
            uniq.append(r)
    return uniq

def compute_a_n(n):
    Fn = fib(n)
    Fnp1 = fib(n+1)
    sols = find_solutions(Fn, Fnp1)
    # For each solution compute ratio
    ratios = []
    for r in sols:
        data = compute_ratio_for_r(Fn, Fnp1, r)
        if data and data["ratio"] is not None:
            ratios.append(data["ratio"])
    if not ratios:
        return None
    return max(ratios), sols, ratios

    return {
        "A":A,"K":K,"Kp":Kp,"N":N,"det":det,"u":u,"v":v,"NB":NB,"NE":NE,"R":R
    }

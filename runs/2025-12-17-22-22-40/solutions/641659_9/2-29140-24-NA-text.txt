def circumcenter(p1, p2, p3):
    # coordinates as tuples (x,y)
    (x1, y1), (x2, y2), (x3, y3) = p1, p2, p3
    # Using determinant method
    d = 2*(x1*(y2 - y3) + x2*(y3 - y1) + x3*(y1 - y2))
    if abs(d) < 1e-12:
        return None  # collinear
    ux = ((x1**2 + y1**2)*(y2 - y3) + (x2**2+y2**2)*(y3 - y1) + (x3**2+y3**2)*(y1 - y2)) / d
    uy = ((x1**2 + y1**2)*(x3 - x2) + (x2**2+y2**2)*(x1 - x3) + (x3**2+y3**2)*(x2 - x1)) / d
    return (ux, uy)

def circle_center_radius(p1, p2, p3):
    cen = circumcenter(p1,p2,p3)
    if cen is None:
        return None, None
    r = math.hypot(p1[0]-cen[0], p1[1]-cen[1])
    return cen, r

def intersect_circles(o1, r1, o2, r2):
    x1, y1 = o1
    x2, y2 = o2
    dx = x2 - x1
    dy = y2 - y1
    d = math.hypot(dx, dy)
    if d < 1e-12:
        return []  # concentric or identical
    # check if no intersection
    if d > r1 + r2 + 1e-12 or d < abs(r1 - r2) - 1e-12:
        return []
    # a = (r1^2 - r2^2 + d^2) / (2d)
    a = (r1**2 - r2**2 + d**2) / (2*d)
    # height h
    h_sq = r1**2 - a**2
    if h_sq < -1e-12:
        return []
    h = math.sqrt(max(0.0, h_sq))
    # point P2
    xm = x1 + a * dx / d
    ym = y1 + a * dy / d
    # intersection points
    rx = -dy * (h / d)
    ry = dx * (h / d)
    p_int1 = (xm + rx, ym + ry)
    p_int2 = (xm - rx, ym - ry)
    if h < 1e-12:  # one point (tangent)
        return [p_int1]
    else:
        return [p_int1, p_int2]

def foot_point(P, line_pt1, line_pt2):
    # projection of P onto line defined by line_pt1 and line_pt2
    x0, y0 = P
    x1, y1 = line_pt1
    x2, y2 = line_pt2
    # direction
    dx = x2 - x1
    dy = y2 - y1
    # if line degenerate
    if abs(dx) + abs(dy) < 1e-12:
        return (x1, y1)
    # projection parameter t
    t = ((x0 - x1)*dx + (y0 - y1)*dy) / (dx*dx + dy*dy)
    return (x1 + t*dx, y1 + t*dy)

def reflect_point(P, line_pt1, line_pt2):
    # reflect point P across line defined by line_pt1,line_pt2
    # Using projection: foot H, then P' = 2H - P
    H = foot_point(P, line_pt1, line_pt2)
    return (2*H[0] - P[0], 2*H[1] - P[1])

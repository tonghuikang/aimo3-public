    # return list of possible outcomes: each outcome is a tuple (winners_set, losers_set)
    # Actually we need to return all possible pairings, each pairing yields winners and losers sets
    # but we need also the assignment of each participant to winner/loser
    # generate all perfect matchings of group, then assign winner based on speed (lower index = faster)
    # Since speed ordering is total, winner is min index in each pair
    group = list(group)
    n = len(group)
    if n==0:
        return [({},{})] # empty
    # generate matchings recursively: pick first element, pair with any other
    first = group[0]
    outcomes=[]

        # pair (first,second)

            lose=second
        else:

def tournament_outcomes(participants, rounds, points):
    # participants is list of participant ids (speed descending)
    # points is list of point values per round, length=rounds
    # recursively process rounds
    # returns dict mapping participant -> final score for each possible outcome
    # We'll accumulate list of final score dictionaries
    outcomes = []
    # state: for each participant, current score
    # we do recursion on round index
    def rec(current_group, round_idx, scores):
        # current_group: list of participants (all participants) but they are not needed; scores dict holds scores.
        # Actually we need groups based on scores.
        # For each unique score value in current scores, we collect group of participants with that score.
        # Then for each group, we generate all matchings and for each combination across groups produce outcomes.
        if round_idx == rounds:
            # done, record scores
            outcomes.append(scores.copy())
            return
        # group participants by current score
        groups = {}
        for p in participants:
            s = scores[p]
            groups.setdefault(s, []).append(p)
        # For each group, generate all possible splits (matchings)
        # We'll need to take cartesian product across groups.
        # For each group, get list of possible winner sets (subsets) produced by matchings; each includes winner set and loser set.
        # We'll combine across groups.
        group_matchings = []
        group_keys = []
        for s, group in groups.items():
            # group size should be power of two
            # generate all possible winner/loser splits
            matchings = generate_matchings(tuple(group))
            group_matchings.append(matchings)
            group_keys.append(s)
        # Now iterate over product
        for combo in itertools.product(*group_matchings):
            new_scores = scores.copy()
            # Apply each group's winners/losers to update scores for this round
            for (wset,lset), score_val in zip(combo, group_keys):
                # winners get points[round_idx] added
                for p in wset:
                    new_scores[p] += points[round_idx]
                # losers unchanged
            # proceed to next round
            rec(current_group, round_idx+1, new_scores)
    # init scores zero
    init_scores = {p:0 for p in participants}
    rec(participants,0,init_scores)
    return outcomes

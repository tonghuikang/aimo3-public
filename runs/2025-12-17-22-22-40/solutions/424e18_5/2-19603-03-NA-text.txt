def possible_orderings(n):
    # n is power of 2, number of participants. We'll assume participants labeled by speed ranking 0..n-1 (0 fastest)
    participants = list(range(n))
    # recursive simulation of all possible pairings consistent with scores.
    # Represent state as a dict: participant->score (int)
    # But we only need to track grouping by scores (prefix bits).
    # We'll simulate round by round, choosing matchings for each group.
    from functools import lru_cache
    
    # Precompute all possible perfect matchings for a given set size (list of participants)
    # But number of matchings grows huge; for n=4 maybe okay; for n=8 may be large but we can brute for small n.
    
    def matchings(lst):
        # return list of list of pairs (tuple of two)
        if not lst:
            return [[]]
        first = lst[0]
        res = []
        for i in range(1, len(lst)):
            second = lst[i]
            rest = lst[1:i] + lst[i+1:]
            for sub in matchings(rest):
                res.append([(first, second)] + sub)
        return res
    
    @lru_cache(None)
    def all_matchings_tuple(tup):
        # tuple of sorted participants
        lst = list(tup)
        return [tuple(sorted(pair) for pair in m) for m in matchings(lst)]
    
    # Let's brute for n=4
    def simulate(state, round_index):
        # state: dict participant->score (int)
        # round_index from 0 to log2(n)-1 (i.e., total rounds)
        if round_index == int(math.log2(n)):
            # finished, compute ranking by scores descending, tie-breaking not needed (scores distinct)
            # ranking as tuple of participants sorted by descending score
            sorted_part = sorted(state.keys(), key=lambda x: state[x], reverse=True)
            return {tuple(sorted_part)}
        # Group participants by current score
        groups = collections.defaultdict(list)
        for p,s in state.items():
            groups[s].append(p)
        # For each group, we need to consider all possible perfect matchings
        # We'll generate all combinations across groups via product
        group_matchings = []
        group_keys = []
        for score, participants_list in groups.items():
            if len(participants_list) % 2 != 0:
                raise ValueError("odd group size")
            # generate all perfect matchings for this group
            # For performance, for n=4 groups are small.

            winlose_options = []
            for m in matchlist:

                    if a < b: # faster

                    else:
                        wins.append((b,a))
                winlose_options.append(wins)

        def recurse(idx, cur_state):
            if idx == len(group_matchings):

                all_rankings.update(simulate(cur_state, round_index+1))
                return

def possible_winner_sets(n):
    # n = number of pairs, total size = 2n
    m=2*n
    result=[]
    for S in itertools.combinations(range(1,m+1), n):
        # S sorted list of winner positions (1-indexed)
        # complement L = remaining positions
        # need matching exists: assign each winner to a larger loser distinct.
        # This is a bipartite matching problem. We'll just try to find matching via greedy/backtracking.
        winners = list(S)
        losers = [i for i in range(1,m+1) if i not in S]
        # construct bipartite edges: winner i connects to losers j > i
        # try to find perfect matching using simple backtracking or max flow.
        # use a greedy algorithm: sort winners ascending, assign each to smallest available loser > winner.
        avail = sorted(losers)
        feasible=True
        for w in sorted(winners):
            # find smallest loser > w
            found = None
            for idx, l in enumerate(avail):
                if l > w:
                    found = idx
                    break
            if found is None:
                feasible=False
                break
            # assign and remove
            avail.pop(found)
        if feasible:
            result.append(S)
    return result

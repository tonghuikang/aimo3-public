def all_pairings(lst):
    # generate all ways to partition lst (list of items) into unordered pairs
    # assume len(lst) even.
    if not lst:
        yield []
        return
    # take first element and pair it with any other
    a = lst[0]
    for i in range(1, len(lst)):
        b = lst[i]
        rest = lst[1:i] + lst[i+1:]
        for pairs in all_pairings(rest):
            yield [(a,b)] + pairs

def random_tournament(num_players):
    # players labeled by speed ranking 0..num_players-1 (slowest to fastest)
    # each round i from 1 to log2(num_players)
    # maintain list of participants as list of (player_id, score)
    participants = [(i, 0) for i in range(num_players)]
    log = num_players.bit_length() - 1
    for i in range(1, log+1):
        # Group participants by score
        groups = {}
        for pid, sc in participants:
            groups.setdefault(sc, []).append(pid)
        new_participants = []
        points = 2**(log - i)  # winner gets this many points (i-th round)
        for sc, pids in groups.items():
            # pair arbitrarily: random shuffle then pair successive
            random.shuffle(pids)
            assert len(pids)%2==0
            for a,b in zip(pids[::2], pids[1::2]):
                # determine winner: higher speed (higher pid)
                winner = max(a,b)
                loser = min(a,b)
                new_participants.append((winner, sc+points))
                new_participants.append((loser, sc))
        participants = new_participants
    # return list of final scores
    scores = {pid: sc for pid, sc in participants}
    return scores

def catalan_subsets(size, participants):
    # participants: list of ids sorted by speed (ascending)
    # size must be even: 2n
    n = size // 2
    # generate all subsets of size n satisfying prefix condition (Catalan condition)
    # Use recursion/backtracking
    # Represent subset as binary string of length size: 1 if winner.
    # prefix condition: #1 <= #0 at all prefixes.
    # Need exactly n ones.
    result = []
    def backtrack(pos, ones, zeros, cur):
        if pos == size:
            if ones == n and zeros == n:
                result.append(tuple(cur))
            return
        # We can put a loser (0) if zeros < n
        if zeros < n:
            cur.append(0)
            backtrack(pos+1, ones, zeros+1, cur)
            cur.pop()
        # We can put a winner (1) if ones < n and zeros > ones (prefix condition)
        if ones < n and zeros > ones:
            cur.append(1)
            backtrack(pos+1, ones+1, zeros, cur)
            cur.pop()
    backtrack(0,0,0,[])
    # Convert binary pattern to actual winner set (list of ids)
    subsets = []
    for pattern in result:
        winner_set = [participants[i] for i,bit in enumerate(pattern) if bit==1]
        subsets.append(winner_set)
    return subsets

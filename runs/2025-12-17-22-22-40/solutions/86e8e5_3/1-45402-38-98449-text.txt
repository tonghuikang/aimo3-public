    min_lcm = None
    best_trip = None
    # iterate a from 1 to S-3
    for a in range(1, S-1):

def carmichael(n):
    # compute Carmichael function λ(n) for odd n (or any n)
    # using factorization via trial division
    orig = n
    result = 1
    temp = n
    p = 2
    while p * p <= temp:
        if temp % p == 0:
            k = 0
            while temp % p == 0:
                temp //= p
                k += 1
            if p == 2:
                if k == 1:
                    lam = 1
                elif k == 2:
                    lam = 2
                else:
                    lam = 2**(k-2)
            else:
                lam = (p-1) * (p**(k-1))
            result = math.lcm(result, lam)
        p += 1 if p==2 else 2
    if temp > 1:

        k = 1
        if p == 2:

        else:

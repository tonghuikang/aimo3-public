def factor_pf(n):
    # returns set of prime factors
    pf=set()
    d=2
    while d*d<=n:
        while n%d==0:
            pf.add(d)
            n//=d
        d+=1 if d==2 else 2
    if n>1:

def is_3_5_power(n):

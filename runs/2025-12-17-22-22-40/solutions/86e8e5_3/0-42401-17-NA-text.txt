    import math
    best = None
    best_triple = None
    # iterate a<b<c distinct, sum=n
    for a in range(1, n-2):
        # ensure a <= b-1
        # b starts from a+1
        for b in range(a+1, n-a-1):
            c = n - a - b
            if c <= b:

            # distinct automatically
            # compute lcm of a,b,c
            l = a*b // math.gcd(a,b)

            if best is None or l < best:

                best_triple = (a,b,c)

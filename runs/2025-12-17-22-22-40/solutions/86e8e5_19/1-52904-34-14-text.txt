# define function to compute M mod d
def M_mod_d(d):
    # M = 3**(2025!)
    # If d divisible by 3, compute power mod d using Chinese remainder
    # factor d = 3^k * t (t coprime to 3)
    # M mod 3^k = 0 if exponent >= k (which holds for any k <= 2025!)
    # M mod t = 1 (since exponent is multiple of phi(t))
    # Combine using CRT
    if d == 1:
        return 0
    # compute exponent for t part: exponent large multiple of phi(t) => M mod t = 1
    # Determine power of 3 dividing d
    k = 0
    while d % 3 == 0:
        d //= 3
        k += 1
    # now d is t (coprime to 3)
    t = d
    # compute remainder modulo 3^k: if k>0, M ≡ 0 mod 3^k
    # compute remainder modulo t: M ≡ 1 mod t
    # combine
    # if k ==0: return 1 mod t
    if k == 0:
        return 1 % t
    else:
        # compute result modulo 3**k * t using CRT
        mod1 = 3**k
        r1 = 0
        mod2 = t
        r2 = 1 % mod2 if mod2 != 1 else 0
        # solve x ≡ r1 (mod mod1), x ≡ r2 (mod mod2)
        # Use Chinese remainder (mod1 and mod2 are coprime)
        # Compute inverse of mod1 mod mod2
        # Solve: x = r1 + mod1 * k such that x ≡ r2 (mod mod2) => mod1*k ≡ r2 - r1 (mod mod2)
        if mod2 == 1:
            return r1 % mod1
        inv = pow(mod1, -1, mod2)  # modular inverse
        k_val = ((r2 - r1) * inv) % mod2
        x = r1 + mod1 * k_val
        return x % (mod1 * mod2)

def f0(S):
    best = None
    best_trip=None
    # a,b,c distinct positive integers sum S
    for a in range(1, S-2):
        for b in range(a+1, S-1):
            c = S - a - b
            if c <= b: # need distinct, so c > b
                continue
            # distinct positive
            # c > b > a >=1
            l = math.lcm(a,b,c)
            if best is None or l < best:
                best = l
                best_trip = (a,b,c)
    return best, best_trip

def compute_ratios(limit=200):
    ratios={}
    for S in range(6, limit+1):
        best, trip = f0(S)
        ratios[S] = (best, best/ S, trip)
    return ratios

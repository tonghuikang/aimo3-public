    # brute with max_x limit; returns f(n) or None
    for x in range(1, max_x+1):
        # quick divisors
        divs = set()
        for d in range(1, int(x**0.5)+1):
            if x % d == 0:
                divs.add(d)
                divs.add(x//d)
        divs = sorted(divs)
        if len(divs) >= 3:
            import itertools
            for combo in itertools.combinations(divs, 3):
                if sum(combo) == n:
                    return x
    return None

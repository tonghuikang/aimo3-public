    # compute geometry and error for cyclic condition; returns signed distance diff
    # compute y, x
    denominator = d**2 - d + r**2
    if denominator == 0:
        return 1e9
    y = -2 * r * d * (1 - d) / denominator
    x = - y * (r**2 - d**2) / (2 * r * d)
    # Points
    B = mp.matrix([0,0])
    C = mp.matrix([1,0])
    A = mp.matrix([x,y])
    I = mp.matrix([d, r])
    # compute F,E
    v = A - B
    t = (I[0]*v[0] + I[1]*v[1])/(v[0]**2 + v[1]**2)
    F = B + t*v
    w = A - C
    s = ((I[0]-1)*w[0] + (I[1])*w[1])/(w[0]**2 + w[1]**2)
    E = C + s*w
    # N
    D = mp.matrix([d,0])
    u = E - F
    tn = ((D[0]-F[0])*u[0] + (D[1]-F[1])*u[1])/(u[0]**2 + u[1]**2)
    N = F + tn*u
    # K
    def circumcenter(P,Q,R):
        x1,y1 = P; x2,y2 = Q; x3,y3 = R
        Dd = 2*(x1*(y2 - y3) + x2*(y3 - y1) + x3*(y1 - y2))
        if abs(Dd) < 1e-12:
            return None
        Ux = ((x1**2 + y1**2)*(y2 - y3) + (x2**2 + y2**2)*(y3 - y1) + (x3**2 + y3**2)*(y1 - y2))/Dd
        Uy = ((x1**2 + y1**2)*(x3 - x2) + (x2**2 + y2**2)*(x1 - x3) + (x3**2 + y3**2)*(x2 - x1))/Dd
        return mp.matrix([Ux, Uy])
    O = circumcenter(A,B,C)
    R = mp.norm(O - A)
    O2 = circumcenter(A,F,E)
    R2 = mp.norm(O2 - A)
    dC = O2 - O
    d_len = mp.norm(dC)
    a = (R**2 - R2**2 + d_len**2)/(2*d_len)
    h_sq = R**2 - a**2
    h = mp.sqrt(max(h_sq,0))
    P2 = O + a*(dC/d_len)
    perp = mp.matrix([-dC[1], dC[0]])/d_len
    K1 = P2 + h*perp
    K2 = P2 - h*perp
    K = K1 if mp.norm(K1 - A) > 1e-8 else K2
    # Compute K' reflection across EF
    # direction of EF
    u_dir = u / mp.norm(u)
    n_vec = mp.matrix([-u_dir[1], u_dir[0]]) # unit normal
    # reflect K across line EF (through point E with direction u)
    # Using formula K' = K - 2 * ((K - E)·n) * n
    dot = (K[0]-E[0])*n_vec[0] + (K[1]-E[1])*n_vec[1]
    Kprime = K - 2*dot*n_vec
    # compute circle through B,K,N and check Kprime lies on it
    O3 = circumcenter(B, K, N)
    if O3 is None:
        return 1e9
    err = mp.norm(O3 - Kprime) - mp.norm(O3 - B)
    return err

def find_r(d):
    # root find r in (0, sqrt(d*(1-d))) such that cyclic_error = 0.
    # Determine sign at ends
    low = 1e-6

                root = mp.findroot(lambda rr: cyclic_error(rr, d), guess)

            except: pass
    return None

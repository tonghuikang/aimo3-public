def circumcenter(P, Q, R):
    # returns center and radius of circumcircle of triangle PQR
    # using perpendicular bisector intersection
    # convert to vectors
    A = np.array(P); B = np.array(Q); C = np.array(R)
    # compute using linear algebra
    # Solve for center O: |O-A|^2 = |O-B|^2 = |O-C|^2
    # Equivalent to linear equations: 2*(B-A).O = |B|^2 - |A|^2, similarly with C-A
    M = np.vstack([2*(B-A), 2*(C-A)])
    rhs = np.array([np.dot(B,B) - np.dot(A,A), np.dot(C,C) - np.dot(A,A)])
    O = np.linalg.solve(M, rhs)
    R = np.linalg.norm(O - A)
    return O, R

def point_reflection(P, Q, R):

    # line defined by points Q,R

    B = np.array([0.0,0.0])
    C = np.array([float(BD+CD),0.0])
    D = np.array([float(BD),0.0])
    I = np.array([float(BD), r])
    # compute F, E as earlier

    denom = 1 + t**2
    cosθ = (1 - t**2)/denom
    sinθ = 2*t/denom
    F = I + r*np.array([cosθ, sinθ])
    u = (CD - r)/(CD + r)
    denom2 = 1 + u**2
    cosψ = (1 - u**2)/denom2
    sinψ = 2*u/denom2
    E = I + r*np.array([cosψ, sinψ])
    # Intersection A
    M = np.array([[F[0], -(E[0]-C[0])],
                  [F[1], -(E[1]-C[1])]])
    RHS = np.array([C[0], C[1]])
    sol = np.linalg.solve(M, RHS)
    s = sol[0]
    A = s*F
    # circumcenters
    def circumcenter(P,Q,R):
        Ax,Ay = P[0],P[1]
        Bx,By = Q[0],Q[1]
        Cx,Cy = R[0],R[1]
        Ddet = 2*(Ax*(By-Cy) + Bx*(Cy-Ay) + Cx*(Ay-By))
        ux = ((Ax**2+Ay**2)*(By-Cy) + (Bx**2+By**2)*(Cy-Ay) + (Cx**2+Cy**2)*(Ay-By))/Ddet
        uy = ((Ax**2+Ay**2)*(Cx-Bx) + (Bx**2+By**2)*(Ax-Cx) + (Cx**2+Cy**2)*(Bx-Ax))/Ddet
        return np.array([ux, uy])

    O2 = circumcenter(A,F,E)
    R = np.linalg.norm(O - A)
    R2 = np.linalg.norm(O2 - A)
    d = np.linalg.norm(O - O2)
    ex = (O2 - O)/d
    a_len = (R**2 - R2**2 + d**2)/(2*d)
    h = math.sqrt(max(R**2 - a_len**2, 0))
    P2 = O + a_len*ex
    perp = np.array([-ex[1], ex[0]])
    K1 = P2 + h*perp
    K2 = P2 - h*perp
    K = K1 if np.linalg.norm(K1 - A) > 1e-6 else K2
    # N foot
    vEF = E - F
    t_par = np.dot(D - F, vEF) / np.dot(vEF, vEF)
    N = F + t_par*vEF
    # Compute T using the tangent circle at B to BN
    BN = N - B

def is_convex_quad(pts):
    # pts list of four points in order (assume closed shape). Compute cross product signs.
    n = len(pts)
    signs = []
    for i in range(n):
        a = pts[i]
        b = pts[(i+1)%n]
        c = pts[(i+2)%n]
        cross = np.cross(b-a, c-b)
        signs.append(cross)
    # Quadrilateral is convex if all cross products have same sign (non-zero)
    return all(s > 0 for s in signs) or all(s < 0 for s in signs)

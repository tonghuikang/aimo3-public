def is_concyclic(P1, P2, P3, P4):
    # check if points are concyclic using determinant method: matrix of coordinates and squares.
    x1,y1 = P1.x, P1.y
    x2,y2 = P2.x, P2.y
    x3,y3 = P3.x, P3.y
    x4,y4 = P4.x, P4.y
    M = sp.Matrix([
        [x1, y1, x1**2 + y1**2, 1],
        [x2, y2, x2**2 + y2**2, 1],
        [x3, y3, x3**2 + y3**2, 1],
        [x4, y4, x4**2 + y4**2, 1]
    ])
    return sp.simplify(M.det()) == 0

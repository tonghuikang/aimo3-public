    min_lcm = None
    best = None
    # iterate over a<b<c distinct positive ints summing to N
    # naive O(N^2) for N up to maybe 100 or 200.
    for a in range(1, N//3 + 1):
        for b in range(a+1, (N-a)//2 + 1):
            c = N - a - b
            if c <= b:
                continue
            # compute lcm
            import math
            l = a*b // math.gcd(a,b)
            l = l*c // math.gcd(l,c)
            if min_lcm is None or l < min_lcm:
                min_lcm = l
                best = (a,b,c)
    return min_lcm, best

def compute_ratios(limit):
    ratios = {}
    for N in range(6, limit+1):
        min_lcm,best = min_lcm_for_sum(N)
        ratios[N] = (min_lcm, best, min_lcm/N)
    return ratios

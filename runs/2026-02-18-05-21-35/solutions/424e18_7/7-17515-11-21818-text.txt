def possible_winner_subsets(n):
    # n is number of pairs = size/2
    size = 2*n
    result = []
    for winners in itertools.combinations(range(size), n):
        # check Hall condition: for any k prefix, #winners in first k <= floor(k/2)
        ok = True
        winners_set = set(winners)
        for k in range(1, size+1):
            w = sum(1 for i in range(k) if i in winners_set)
            if w > k//2:
                ok = False
                break
        if ok:
            result.append(winners)
    return result

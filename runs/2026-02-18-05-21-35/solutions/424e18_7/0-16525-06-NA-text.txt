def all_matchings(group):
    # group is list of participants (identifiers)
    # returns list of list of pairs (as tuples)
    n = len(group)
    if n == 0:
        return [[]]
    # generate matchings by picking first element and pairing with any other
    first = group[0]
    rest = group[1:]
    matchings = []
    for i, partner in enumerate(rest):
        remaining = rest[:i] + rest[i+1:]
        for sub in all_matchings(remaining):
            matchings.append([(first, partner)] + sub)
    return matchings

    participants = list(range(n))

    # We'll use recursion similar as before but we will track final rankings directly,
    # but we only need count distinct rankings.
    def rec(round_idx, scores):
        if round_idx == rounds:
            ranking = tuple(sorted(participants, key=lambda p: (scores[p], p), reverse=True))
            final_rankings.add(ranking)
            return
        groups = {}
        for p in participants:
            groups.setdefault(scores[p], []).append(p)
        # generate matchings per group
        group_options = []
        for score, group in groups.items():
            matchings = all_matchings(group)
            opts = []
            points = 2**(rounds-1 - round_idx)
            for matching in matchings:
                delta = {}
                for a,b in matching:
                    winner = a if a > b else b
                    loser = b if winner == a else a
                    delta[winner] = points
                    delta[loser] = 0
                opts.append(delta)
            group_options.append(opts)
        for combo in itertools.product(*group_options):
            new_scores = scores.copy()
            for upd in combo:
                for p,add in upd.items():
                    new_scores[p] += add
            rec(round_idx+1, new_scores)
    rec(0, scores0)
    return len(final_rankings), final_rankings

def circumcenter(P, Q, R):
    # return circumcenter of three points
    # using perpendicular bisector intersection
    A = np.array(P); B = np.array(Q); C = np.array(R)
    # compute using formula

    ux = ((np.dot(A, A))*(B[1]-C[1]) + (np.dot(B,B))*(C[1]-A[1]) + (np.dot(C,C))*(A[1]-B[1]))/D

def circle_intersections(center1, radius1, center2, radius2):
    # returns intersection points of two circles
    # assume they intersect
    O1 = np.array(center1); O2 = np.array(center2)

    a = (radius1**2 - radius2**2 + d**2) / (2*d)

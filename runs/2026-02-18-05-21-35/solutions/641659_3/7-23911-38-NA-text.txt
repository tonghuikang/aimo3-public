def circle_intersections(c1, r1, c2, r2):
    # returns list of intersection points (0,1,2)
    # compute distance between centers
    c1 = np.array(c1); c2 = np.array(c2)
    d = np.linalg.norm(c2 - c1)
    if d > r1 + r2 + 1e-9 or d < abs(r1 - r2) - 1e-9:
        return []  # no intersect
    # handle degenerate case
    # compute a = (r1^2 - r2^2 + d^2) / (2d)
    a = (r1**2 - r2**2 + d**2) / (2*d) if d != 0 else 0
    h_sq = r1**2 - a**2
    if h_sq < -1e-9:
        return []

    if d == 0:

        return []

def reflect_point_across_line(P, L1, L2):
    # reflect P across line through L1 L2
    # line vector
    v = L2 - L1
    v = v / np.linalg.norm(v)
    # projection of (P - L1) onto v
    proj = np.dot(P - L1, v) * v
    # perpendicular component
    perp = P - L1 - proj
    # reflection: L1 + proj - perp
    return L1 + proj - perp

def foot_of_perpendicular(P, L1, L2):
    # foot from point P to line through L1 L2
    v = L2 - L1
    v = v / np.linalg.norm(v)
    return L1 + np.dot(P - L1, v) * v

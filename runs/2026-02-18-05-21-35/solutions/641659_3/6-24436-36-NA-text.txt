    solutions = []

    prev_theta = None

        tan_half_theta = math.tan(th/2)
        tan_half_phi = r * tan_half_theta
        phi = 2*math.atan(tan_half_phi)
        # ensure phi within (0, pi)
        if phi <= 0 or phi >= math.pi:
            val = None
        else:

        if val is None:

            prev_theta = None
            continue
        if prev_val is not None and prev_val*val < 0:
            # root between prev_theta and th
            lo, hi = prev_theta, th
            # bisection refine
            for _ in range(40):

def is_shifty_factor(poly):
    # poly is sympy Poly
    if poly.is_zero:
        return False
    # shift out monomial factor
    d = poly.as_dict()
    min_exp = min(exp[0] for exp in d.keys())
    # if min_exp > 0 (monomial factor), fine. shift out.
    new_terms = {}
    for exp, coeff in d.items():
        new_exp = exp[0] - min_exp
        new_terms[(new_exp,)] = coeff
    dprime = sp.Poly.from_dict(new_terms, x, domain='ZZ')
    # constant term must be ±1
    const_term = dprime.TC()
    if const_term not in (1, -1):
        return False
    # factor out cyclotomics
    factors = []
    # We need to check each even d whether its cyclotomic divides dprime.
    # We'll test in arbitrary order; but we also need to ensure all factors accounted.
    remaining = dprime
    for d in even_d_list:
        Phi = cyclo_dict[d]
        # Check if Phi divides remaining
        while True:
            q, r = sp.div(remaining, Phi)
            if r.is_zero:
                factors.append(d)
                remaining = q
            else:
                break
    # after removing all possible cyclotomics, if remaining is not 1, fail.
    if not remaining.is_one:
        return False
    # now check v2 uniform across factors (if any)
    if len(factors) == 0:
        # no cyclotomic factors, only monomial; that's fine.
        return True
    v2_set = { (d & -d).bit_length() - 1 for d in factors }  # compute exponent of 2
    # Actually compute v2(d)
    def v2(d):
        cnt = 0
        while d % 2 == 0:
            d //= 2
            cnt += 1
        return cnt
    v2_vals = {v2(d) for d in factors}
    return len(v2_vals) == 1

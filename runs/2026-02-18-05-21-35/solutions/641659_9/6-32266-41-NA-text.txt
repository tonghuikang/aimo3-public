<|end|><|start|>assistant<|channel|>analysis to=python code<|message|>def is_convex_quad(B, K, Kp, N):
    # Determine if quadrilateral B-K-Kp-N is convex (non-self-intersecting) with vertices in order.
    # Compute cross product of successive edges; they should have same sign for convex polygon.
    pts = [B, K, Kp, N]
    signs = []
    n = len(pts)
    for i in range(n):
        p0 = pts[i]
        p1 = pts[(i+1)%n]
        p2 = pts[(i+2)%n]
        # vector p0->p1 and p1->p2
        v1 = (p1[0]-p0[0], p1[1]-p0[1])
        v2 = (p2[0]-p1[0], p2[1]-p1[1])
        cross = v1[0]*v2[1] - v1[1]*v2[0]
        signs.append(cross)
    # all cross products should have same sign (all positive or all negative)
    pos = [c > 1e-9 for c in signs]
    neg = [c < -1e-9 for c in signs]
    return all(pos) or all(neg)

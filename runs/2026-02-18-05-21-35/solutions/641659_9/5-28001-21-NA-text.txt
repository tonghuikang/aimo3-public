def circumcenter(p1, p2, p3):
    # return center (x,y) and radius
    # Using perpendicular bisectors intersection
    # Solve for center (cx,cy) such that |c-p1|=|c-p2|=|c-p3|
    # We'll use linear algebra.
    x1, y1 = p1
    x2, y2 = p2
    x3, y3 = p3
    # compute matrix
    A = np.array([[2*(x2 - x1), 2*(y2 - y1)],
                  [2*(x3 - x1), 2*(y3 - y1)]], dtype=float)
    b = np.array([x2**2 + y2**2 - x1**2 - y1**2,
                  x3**2 + y3**2 - x1**2 - y1**2], dtype=float)

    try:

    except np.linalg.LinAlgError:
        return None, None
    r = math.hypot(cx - x1, cy - y1)
    return (cx, cy), r

def intersect_circles(c1, r1, c2, r2):

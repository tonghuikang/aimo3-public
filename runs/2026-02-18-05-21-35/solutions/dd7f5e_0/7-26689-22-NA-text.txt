def is_shifty_bruteforce(alpha_coeffs):
    # alpha_coeffs dict of exponent->coeff (including zeros)
    # Determine min exponent t
    exp_nonzero = [e for e,c in alpha_coeffs.items() if c!=0]
    if not exp_nonzero:
        return False
    t = min(exp_nonzero)
    # compute C coefficients after shift
    max_exp = max(exp_nonzero)
    degC = max_exp - t
    coeffsC = [0]*(degC+1)
    for e,c in alpha_coeffs.items():
        if c!=0:
            coeffsC[e - t] = c
    # constant term must be ±1
    if abs(coeffsC[0]) != 1:
        return False
    # check if C divides 1+x^m for some m up to 2520
    C = Poly(sum(coeffsC[i]*x**i for i in range(len(coeffsC))), x, domain='ZZ')
    for m in range(1,2521):
        target = Poly(1 + x**m, x, domain='ZZ')
        if target.rem(C) == Poly(0, x, domain='ZZ'):
            return True
    return False

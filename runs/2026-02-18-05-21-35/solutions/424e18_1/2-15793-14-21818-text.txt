def pairings(players):
    # return list of matchings (list of tuples (a,b))
    # players is a tuple/list of items
    n=len(players)
    if n==0:
        return [[]]
    first=players[0]
    results=[]
    for i in range(1,n):
        second=players[i]
        rest=players[1:i]+players[i+1:]
        for sub in pairings(rest):
            results.append([(first,second)]+sub)
    return results

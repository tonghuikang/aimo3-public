def simulate(pairings):

    # speeds: ranking 0 fastest < 1 < ... slower.
    speeds = list(range(4))  # 0 fastest, 1 second, 2 third, 3 fourth
    scores = [0]*4
    points = [2, 1]  # round1: 2, round2:1
    for i in range(2):
        p = points[i]
        for a,b in pairings[i]:
            # winner is lower speed index
            if speeds[a] < speeds[b]:
                winner = a
                loser = b
            else:
                winner = b
                loser = a
            scores[winner] += p
    # final ranking descending scores (tie break by speed? Not needed)
    # produce order list of indices sorted by scores descending, ties by speed (faster first).
    order = sorted(range(4), key=lambda x:(-scores[x], speeds[x]))
    return tuple(order)

def all_pairings(players):
    # generate all possible pairings (unordered pairs) of list players
    if not players:
        yield []
        return
    a = players[0]
    for i in range(1,len(players)):
        b = players[i]
        rest = players[1:i]+players[i+1:]
        for sub in all_pairings(rest):
            yield [(a,b)] + sub

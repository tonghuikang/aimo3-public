def perfect_matchings(elements):
    # returns list of list of pairs (as tuples) covering all elements
    elements = list(elements)
    if not elements:
        yield []
    else:
        first = elements[0]
        for i in range(1, len(elements)):
            second = elements[i]
            rest = elements[1:i] + elements[i+1:]
            for matching in perfect_matchings(rest):
                yield [(first, second)] + matching

def generate_matchings(elements):
    # returns list of matchings (list of pairs)
    # For small sets only
    if not elements:
        yield []
        return
    elements = list(elements)

        rest = elements[1:i] + elements[i+1:]

        # group participants by current score

                rec(cur_scores, round_idx+1)
                return
            s, group = group_items[idx]
            # group size must be even
            for matching in generate_matchings(group):

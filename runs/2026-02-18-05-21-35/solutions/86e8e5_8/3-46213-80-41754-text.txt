    # compute minimal lcm over distinct triples sum n
    best = None
    best_lcm = None
    from math import gcd
    for a in range(1, n-1):
        for b in range(a+1, n):
            c = n - a - b
            if c <= b:
                continue
            # distinct a<b<c
            # compute lcm
            def lcm(x,y):
                return x*y//gcd(x,y)
            l = lcm(lcm(a,b), c)
            if best_lcm is None or l < best_lcm:
                best_lcm = l
                best = (a,b,c)
    return best_lcm, best

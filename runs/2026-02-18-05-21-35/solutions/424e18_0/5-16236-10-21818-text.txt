def pairings(lst):
    """Yield all perfect matchings (list of pairs) for list lst (list of elements). Order of pairs and within pair doesn't matter."""
    # recursion
    if not lst:
        yield []
        return
    first = lst[0]
    for i in range(1, len(lst)):
        second = lst[i]
        rest = lst[1:i] + lst[i+1:]
        for rest_pair in pairings(rest):
            yield [(first, second)] + rest_pair

    # returns set of final score assignments as tuple of scores per player index (0 fastest)
    players = list(range(2**n))
    # We'll recursively simulate all possible pairing decisions (matchings) each round.
    # Use recursion: given current groups (list of lists of player indices), round index i from 1..n.
    from functools import lru_cache
    
    final_score_sets = set()
    
    def recurse(groups, round_i, scores):
        if round_i > n:
            # all rounds done
            final_score_sets.add(tuple(scores))
            return
        # For each group, generate all possible pairings and assign bits
        # We need to iterate over all possible matchings across all groups simultaneously.
        # Could use product of matchings for each group.
        # For each group, generate all possible ways to split into winners and losers (bit assignments) and produce next groups.
        # We'll treat groups independently.
        group_options = []
        for grp in groups:
            m = len(grp)
            # generate all matchings and outcomes for this group
            # We'll produce list of (winners, losers, next_winners_group, next_losers_group) where next groups are lists of players
            # winners get bit 1 at this round, losers bit 0.
            matchings = []
            for pairing in generate_matchings(grp):
                # Determine winner for each pair (min index faster)
                winners = []
                losers = []
                for a,b in pairing:
                    if a < b: # faster (lower index) wins
                        winners.append(a)
                        losers.append(b)
                    else:
                        winners.append(b)
                        losers.append(a)
                # Sort for consistency
                winners_sorted = tuple(sorted(winners))
                losers_sorted = tuple(sorted(losers))
                matchings.append((winners_sorted, losers_sorted))
            # Remove duplicates (since multiple matchings may produce same split)
            matchings = list(set(matchings))
            group_options.append(matchings)
        # Now iterate over product
        for combo in itertools.product(*group_options):
            # combo is a tuple for each group: (winners, losers)
            new_groups = []
            # update scores
            new_scores = list(scores)
            for (winners, losers) in combo:
                # set bit for each player
                bit = 1 << (n - round_i)  # weight of this round
                for p in winners:
                    new_scores[p] += bit
                # next groups for next round: winners group and losers group each go to next round as separate groups.
                new_groups.append(list(winners))
                new_groups.append(list(losers))
            # continue recursion
            recurse(new_groups, round_i+1, new_scores)
    # initial groups: one group of all players
    recurse([players], 1, [0]*len(players))
    return final_score_sets

def count_orderings(score_sets):
    total = 0
    for scores in score_sets:
        # compute multiplicities of each score
        from collections import Counter
        cnt = Counter(scores)
        # number of total orders consistent: product of factorials of counts
        ways = 1
        for c in cnt.values():
            ways *= math.factorial(c)
        total += ways
    return total

    N = count_orderings(score_sets)

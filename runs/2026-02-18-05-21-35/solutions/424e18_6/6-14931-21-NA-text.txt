        if not group:

        res = []

            remaining = group[1:i] + group[i+1:]

    # precompute matchings for groups of size up to n

    def recurse(state, round_idx):

            ordered = sorted(state, key=lambda x: x[0], reverse=True)
            order = tuple(s for _, s in ordered)
            final_orders.add(order)
            return
        # points for this round not needed.

        from collections import defaultdict
        groups = defaultdict(list)

            groups[sc].append((sc, sp))
        # for each group, we must produce pairings and decide outcomes (determined by speeds)
        # Since winners are faster, we need to consider all possible pairings; results are deterministic.
        # For each group, generate all possible matchings (pairings).

        group_items = list(groups.items())

        # We'll generate list of possible next states for each group, then combine.
        group_next_states = []

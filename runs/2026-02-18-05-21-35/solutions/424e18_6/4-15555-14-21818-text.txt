def matchings(lst):
    # returns list of list of pairs (list of tuples)
    if not lst:
        return [[]]
    first = lst[0]
    rest = lst[1:]
    res = []
    for i, other in enumerate(rest):
        for sub in matchings(rest[:i] + rest[i+1:]):
            res.append([(first, other)] + sub)
    return res

def possible_scores(players):

def count_rankings(n):

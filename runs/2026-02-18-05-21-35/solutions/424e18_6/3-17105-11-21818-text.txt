def pairings_of_list(lst):
    # returns list of possible pairings (list of pairs) for given list (even length)
    # treat as unordered sets, produce each unique pairing (order of pairs doesn't matter)
    # recursion: choose first element, pair with any other, then recursively pair remaining.
    if not lst:
        return [[]]
    first = lst[0]
    result = []
    for i in range(1, len(lst)):
        second = lst[i]
        rest = lst[1:i] + lst[i+1:]
        for sub in pairings_of_list(rest):
            result.append([(first, second)] + sub)
    return result

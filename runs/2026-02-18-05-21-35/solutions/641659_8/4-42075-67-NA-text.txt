def incircle_touch_points(A, B, C, s_b, s_c):

def circle_intersections(center1, r1, center2, r2):

def foot_of_perpendicular(P, A, B):
    # foot of P onto line AB
    (x0,y0) = P
    (x1,y1) = A
    (x2,y2) = B
    # line AB vector
    dx = x2 - x1
    dy = y2 - y1
    if dx == dy == 0:
        return (x1, y1)
    t = ((x0 - x1)*dx + (y0 - y1)*dy) / (dx*dx + dy*dy)
    # projection onto AB
    projx = x1 + t*dx
    projy = y1 + t*dy
    # foot of perpendicular is same as projection because we want foot onto line
    return (projx, projy)

    signs = []
    for i in range(4):
        a = p[i]
        b = p[(i+1)%4]
        c = p[(i+2)%4]
        # cross product (b-a) x (c-b)
        cross = (b[0]-a[0])*(c[1]-b[1]) - (b[1]-a[1])*(c[0]-b[0])
        signs.append(cross)
    # check all same sign (allow small tolerance)
    pos = sum(1 for v in signs if v > 1e-9)
    neg = sum(1 for v in signs if v < -1e-9)
    return (pos==4 or neg==4)

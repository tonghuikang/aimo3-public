def intersect_circles(c1, r1, c2, r2):
    # returns intersection points of circles c1,r1 and c2,r2 (list of 0,1,2 points)
    x0,y0 = c1
    x1,y1 = c2
    dx = x1-x0
    dy = y1-y0
    d = math.hypot(dx, dy)
    if d > r1+r2+1e-12 or d < abs(r1-r2)-1e-12 or d==0:
        return []
    # point 2 is distance a from c1 along line to c2
    a = (r1**2 - r2**2 + d**2) / (2*d)
    h_sq = max(r1**2 - a**2, 0.0)
    h = math.sqrt(h_sq)
    xm = x0 + a*dx/d
    ym = y0 + a*dy/d
    # intersection points
    xs1 = xm + h*(-dy)/d
    ys1 = ym + h*(dx)/d
    xs2 = xm - h*(-dy)/d
    ys2 = ym - h*(dx)/d
    if h < 1e-12:
        return [(xs1, ys1)]
    else:
        return [(xs1, ys1), (xs2, ys2)]

def reflect_point_across_line(P, A, B):
    # reflect point P across line AB
    # line defined by points A, B
    # compute projection
    ax, ay = A
    bx, by = B
    px, py = P
    # line vector
    vx, vy = bx-ax, by-ay
    # normalize?
    # projection of vector AP onto line direction
    # compute dot product
    apx, apy = px-ax, py-ay
    dot = apx*vx + apy*vy
    len2 = vx*vx + vy*vy
    projx = ax + dot*vx/len2
    projy = ay + dot*vy/len2
    # reflection
    rx = 2*projx - px
    ry = 2*projy - py
    return (rx, ry)

def line_through_points(P, Q):

    x1,y1 = P
    x2,y2 = Q
    A = y1 - y2
    B = x2 - x1
    C = x1*y2 - x2*y1
    return (A, B, C)

    x = (B1*C2 - B2*C1)/det
    y = (C1*A2 - C2*A1)/det

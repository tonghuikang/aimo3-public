def solve_linear(A, b):
    # A 2x2 matrix as list of lists or mp.matrix
    a,b1 = A[0][0], A[0][1]
    c,d = A[1][0], A[1][1]
    det = a*d - b1*c
    if abs(det) < 1e-12:
        raise ValueError("Singular")
    x = (b[0]*d - b1*b[1]) / det
    y = (a*b[1] - b[0]*c) / det

def circumcenter(P, Q, R):

def circle_intersections(O1, r1, O2, r2):

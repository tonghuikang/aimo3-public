We need O·d1 = mid1·d1, O·d2 = mid2·d2

Solve linear system: [d1_x d1_y; d2_x d2_y] * O = [mid1·d1; mid2·d2]

Thus O = matrix_inv * rhs. Use determinant to solve.

    return ax*by - ay*bx

def circumcenter(P, Q, R):
    # P, Q, R as mp.matrix size 2
    d1 = Q - P
    d2 = R - P
    mid1 = (P + Q) * 0.5
    mid2 = (P + R) * 0.5
    # Solve linear equations: d1·O = d1·mid1, d2·O = d2·mid2
    # Write as matrix [[d1x, d1y],[d2x,d2y]] * O = [dot(d1,mid1), dot(d2,mid2)]
    a1 = d1[0]; b1 = d1[1]
    a2 = d2[0]; b2 = d2[1]
    det = a1*b2 - a2*b1
    if abs(det) < 1e-12:
        return None
    rhs1 = a1*mid1[0] + b1*mid1[1]
    rhs2 = a2*mid2[0] + b2*mid2[1]
    ox = (rhs1*b2 - rhs2*b1) / det
    oy = (a1*rhs2 - a2*rhs1) / det
    return mp.matrix([ox, oy])

def circle_intersections(O1, r1, O2, r2):
    d = mp.norm(O2 - O1)
    if d > r1 + r2 + 1e-12 or d < abs(r1 - r2) - 1e-12:
        return []
    a = (r1**2 - r2**2 + d**2)/(2*d)
    h_sq = max(r1**2 - a**2, 0)
    h = mp.sqrt(h_sq)
    # base point
    P2 = O1 + a*(O2 - O1)/d
    # offset perpendicular
    offset = h * mp.matrix([-(O2[1]-O1[1])/d, (O2[0]-O1[0])/d])
    return [P2 + offset, P2 - offset]

def compute_ratio(r, d, c, a):
    B = mp.matrix([0.0,0.0])
    C = mp.matrix([float(a),0.0])
    D = mp.matrix([float(d),0.0])
    # points F, E
    denom_f = d**2 + r**2
    xF = d*(d**2 - r**2)/denom_f
    yF = 2*d**2 * r / denom_f
    F = mp.matrix([xF, yF])
    denom_e = c**2 + r**2
    xE = d + 2 * r**2 * c / denom_e
    yE = 2 * r * c**2 / denom_e
    E = mp.matrix([xE, yE])
    # line BF and CE intersection A
    u = F - B
    v = E - C
    w = C - B
    det_uv = cross(u[0], u[1], v[0], v[1])
    if abs(det_uv) < 1e-12:
        return None
    t = cross(w[0], w[1], v[0], v[1]) / det_uv
    A = B + u * t
    # circumcenters
    Oabc = circumcenter(A, B, C)
    if Oabc is None:
        return None
    Rabc = mp.norm(A - Oabc)
    Oafe = circumcenter(A, F, E)
    if Oafe is None:
        return None
    Rafe = mp.norm(A - Oafe)
    inters = circle_intersections(Oabc, Rabc, Oafe, Rafe)
    K = None
    for P in inters:
        if mp.norm(P - A) > 1e-6:
            K = P
            break
    if K is None:
        return None
    # reflection of K across line EF
    def reflect(P, L1, L2):
        v = L2 - L1
        t = ((P - L1).dot(v)) / (v.dot(v))
        foot = L1 + v*t
        return foot*2 - P
    Kp = reflect(K, E, F)
    # foot from D to EF (projection)
    def foot(P, L1, L2):
        v = L2 - L1
        t = ((P - L1).dot(v)) / (v.dot(v))
        return L1 + v*t
    N = foot(D, E, F)
    # cyclic condition determinant
    def cyclic_det(P1,P2,P3,P4):
        mat = mp.matrix([[P1[0]**2+P1[1]**2, P1[0], P1[1], 1],
                         [P2[0]**2+P2[1]**2, P2[0], P2[1], 1],
                         [P3[0]**2+P3[1]**2, P3[0], P3[1], 1],
                         [P4[0]**2+P4[1]**2, P4[0], P4[1], 1]])
        return mp.det(mat)
    cyclic_det = cyclic_det(B, K, Kp, N)
    # compute T as intersection of circle through B and K tangent to BN at B
    # compute perpendicular direction
    dx = N[0] - B[0]; dy = N[1] - B[1]
    norm = mp.sqrt(dx**2 + dy**2)
    if norm == 0:
        return None
    perp = mp.matrix([-dy, dx]) / norm
    dot = K[0]*perp[0] + K[1]*perp[1]  # K·perp
    if abs(dot) < 1e-12:
        return None
    t_val = (K[0]**2 + K[1]**2) / (2*dot)
    O2 = B + perp * t_val
    # intersection with y=0
    rad2_minus_y2 = t_val**2 - O2[1]**2
    if rad2_minus_y2 < -1e-12:
        return None
    dx_root = mp.sqrt(max(rad2_minus_y2, 0))
    x1 = O2[0] + dx_root
    x2 = O2[0] - dx_root
    # choose the root not near zero
    if abs(x1) < 1e-6:
        xT = x2
    elif abs(x2) < 1e-6:
        xT = x1
    else:
        # pick one not near zero
        if abs(x1) < abs(x2):
            xT = x2
        else:
            xT = x1
    T = mp.matrix([xT, 0.0])
    # lengths
    CT = mp.norm(C - T)
    BT = mp.norm(T - B)
    NB = mp.norm(N - B)
    NE = mp.norm(N - E)
    ratio = (CT * NB) / (BT * NE)

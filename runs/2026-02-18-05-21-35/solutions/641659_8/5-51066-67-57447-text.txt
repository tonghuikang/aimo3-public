        A = np.array(p1); Bp = np.array(p2); Cc = np.array(p3)
        d = 2*(A[0]*(Bp[1]-Cc[1]) + Bp[0]*(Cc[1]-A[1]) + Cc[0]*(A[1]-Bp[1]))
        if abs(d) < 1e-12:
            return None
        ux = ((np.dot(A,A))*(Bp[1]-Cc[1]) + (np.dot(Bp,Bp))*(Cc[1]-A[1]) + (np.dot(Cc,Cc))*(A[1]-Bp[1]))/d
        uy = ((np.dot(A,A))*(Cc[0]-Bp[0]) + (np.dot(Bp,Bp))*(A[0]-Cc[0]) + (np.dot(Cc,Cc))*(Bp[0]-A[0]))/d
        return np.array([ux, uy])
    O_ABC = circumcenter(A,B,C)
    O_AFE = circumcenter(A,F,E)
    if O_ABC is None or O_AFE is None:
        return None
    R1 = np.linalg.norm(A - O_ABC)
    R2 = np.linalg.norm(A - O_AFE)
    # intersect circles
    d_centers = np.linalg.norm(O_AFE - O_ABC)
    if d_centers < 1e-12:
        return None
    a_len = (R1**2 - R2**2 + d_centers**2)/(2*d_centers)
    h_sq = R1**2 - a_len**2
    if h_sq < 0:
        h = 0.0
    else:
        h = math.sqrt(h_sq)
    P2 = O_ABC + a_len*(O_AFE - O_ABC)/d_centers
    perp = np.array([-(O_AFE-O_ABC)[1], (O_AFE-O_ABC)[0]])/d_centers
    inter1 = P2 + h*perp
    inter2 = P2 - h*perp
    # Choose K not equal to A
    if np.linalg.norm(inter1 - A) < 1e-6:
        K = inter2
    else:
        K = inter1
    # reflection across EF
    vEF = F - E
    # reflect point K across line through E,F
    # projection onto EF
    proj_len = np.dot(K - E, vEF) / np.dot(vEF, vEF)
    proj = E + proj_len * vEF
    Kp = 2*proj - K
    # circle through B,K tangent to BN at B: center O_T intersection of:
    # perp bisector of BK and line through B perpendicular to BN.
    midBK = (B + K)/2
    vBK = K - B
    dir1 = np.array([-vBK[1], vBK[0]])  # direction of perp bisector
    vBN = N - B
    dir2 = np.array([-vBN[1], vBN[0]])  # direction of line through B perpendicular to BN
    # solve for O_T
    A_mat = np.column_stack((dir2, -dir1))
    b_vec = midBK - B
    # solve linear system
    try:
        sol = np.linalg.solve(A_mat, b_vec)
    except np.linalg.LinAlgError:
        return None
    s = sol[0]
    O_T = B + s*dir2
    # T point is second intersection with BC (x-axis)
    x_T = 2*O_T[0]
    T = np.array([x_T, 0.0])
    # compute ratio
    CT = np.linalg.norm(C - T)
    BT = np.linalg.norm(T - B)
    NB = np.linalg.norm(N - B)
    NE = np.linalg.norm(N - E)
    ratio = (CT * NB) / (BT * NE)
    return {
        "A": A, "E": E, "F": F, "N": N,
        "K": K, "Kp": Kp, "T": T,
        "ratio": ratio, "BK": np.linalg.norm(B-K),
        "BN": np.linalg.norm(B-N)
    }

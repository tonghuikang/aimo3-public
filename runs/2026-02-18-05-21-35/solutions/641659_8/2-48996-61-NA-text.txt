def circle_intersections(O1, r1, O2, r2):
    # return list of intersection points (0,1,2)
    O1 = np.array(O1); O2=np.array(O2)
    d = np.linalg.norm(O2-O1)
    if d > r1+r2+1e-9 or d < abs(r1-r2)-1e-9 or d < 1e-12:
        return []  # no intersection or coincident
    # distance from O1 to line of centers intersection
    a = (r1**2 - r2**2 + d**2) / (2*d)
    h_sq = r1**2 - a**2
    if h_sq < -1e-9:
        return []
    h = math.sqrt(max(h_sq,0.0))
    P2 = O1 + a*(O2-O1)/d
    # intersection points
    offset = h * np.array([-(O2-O1)[1], (O2-O1)[0]]) / d
    I1 = P2 + offset
    I2 = P2 - offset
    if h < 1e-9:
        return [I1]
    else:
        return [I1, I2]

def reflect_across_line(P, L1, L2):
    # reflect point P across line through L1,L2
    L1 = np.array(L1); L2=np.array(L2); P=np.array(P)
    # projection of P onto line
    v = L2 - L1
    if np.linalg.norm(v) < 1e-12:
        return None
    proj = L1 + np.dot(P - L1, v)/np.dot(v,v) * v
    refl = proj*2 - P
    return refl

def compute_T(B, K, N, line_BC):
    # compute the circle tangent to line BN at B and passing through K.
    # Circle passes through B and K, and line BN is tangent at B.
    # Center O is intersection of perpendicular to BN at B (call L1) and perpendicular bisector of BK (call L2)
    B = np.array(B); K = np.array(K); N = np.array(N)
    # line BN direction
    vBN = N - B
    # line perpendicular at B: direction = perp of vBN
    # we can param compute O as B + t * perp(vBN)
    # The perpendicular bisector of BK: midpoint M = (B+K)/2, direction perp(BK)
    M = (B+K)/2
    vBK = K - B
    # Solve for intersection: B + t * perp(vBN) = M + s * perp(vBK)
    # Represent as linear equations for t,s.
    # perp vector in 2D: perp(v) = [-v[1], v[0]]
    perpBN = np.array([-vBN[1], vBN[0]])
    perpBK = np.array([-vBK[1], vBK[0]])
    # Solve: B + t*perpBN = M + s*perpBK
    # => t*perpBN - s*perpBK = M - B
    A_mat = np.column_stack((perpBN, -perpBK))
    rhs = M - B
    # Solve linear system for t,s
    try:
        ts = np.linalg.solve(A_mat, rhs)
    except np.linalg.LinAlgError:
        return None
    t = ts[0]
    O = B + t*perpBN
    R = np.linalg.norm(O - B)
    # Intersect circle with line BC (y=0), find other intersection besides B
    # param eq for line BC: (x,0) for x real. Solve (x-Ox)^2 + (0-Oy)^2 = R^2
    # => (x-Ox)^2 = R^2 - Oy^2
    delta = R**2 - O[1]**2
    if delta < -1e-9:

    x2 = O[0] - sqrt_delta

    if abs(x1) > 1e-6:
        T = np.array([x1, 0.0])
    else:
        T = np.array([x2, 0.0])
    return T

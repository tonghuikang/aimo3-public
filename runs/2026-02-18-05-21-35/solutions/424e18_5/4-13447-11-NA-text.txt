def pairings(lst):
    """Return all possible ways to pair up elements of lst (list) into unordered pairs.
    lst length must be even."""
    if not lst:
        return [[]]
    # recursively pick first element and pair with each later element
    first = lst[0]
    rest = lst[1:]
    all_pairings = []
    for i, partner in enumerate(rest):
        remaining = rest[:i] + rest[i+1:]
        for subpair in pairings(remaining):
            all_pairings.append([(first, partner)] + subpair)
    return all_pairings

def count_linear_extensions(n):
    elements = list(range(2**n))
    # define subset relation: a subset b if (a & b) == a (bits)
    def less_eq(a,b):
        return (a & b) == a
    cnt = 0
    for perm in itertools.permutations(elements):
        ok = True
        pos = {elem:i for i,elem in enumerate(perm)}
        for a in elements:
            for b in elements:
                if a!=b and less_eq(a,b):
                    if pos[a] > pos[b]:
                        ok=False
                        break
            if not ok:
                break
        if ok:
            cnt+=1
    return cnt

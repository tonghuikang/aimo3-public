def simulate_pairings(order, n=4):
    # order: list of players sorted by speed descending
    # 0 fastest, etc.
    players = list(range(n))
    scores = [0]*n
    # rounds: i from 0 to log2 n -1
    rounds = int(math.log2(n))
    # At each round, players with same score must be paired arbitrarily.
    # We'll generate all possibilities recursively.
    results = set()
    def recurse(round_idx, scores):
        if round_idx == rounds:
            # final scores
            results.add(tuple(scores))
            return
        points = 2**(rounds-1-round_idx)  # round 0: 2^{rounds-1}
        # group players by current score
        score_to_players = {}
        for p,s in enumerate(scores):
            score_to_players.setdefault(s, []).append(p)
        # For each group, generate all matchings
        group_matchings_lists = []
        for group in score_to_players.values():
            # must be even size
            matchings = all_matchings(group)
            group_matchings_lists.append(matchings)
        # Cartesian product across groups
        for combo in itertools.product(*group_matchings_lists):
            # apply all matches
            new_scores = scores[:]
            for match in combo:
                for a,b in match:
                    # winner faster (lower index in order)
                    if order.index(a) < order.index(b):
                        winner = a
                    else:
                        winner = b
                    new_scores[winner] += points
            recurse(round_idx+1, new_scores)
    recurse(0, scores)
    return results

def final_rankings(order):

    # ties broken arbitrarily? We'll just record groupings)
    results=set()
    for scores in simulate_pairings(order):
        # produce tuple of groups sorted by score descending, each group as sorted tuple of players

        # Also sort groups by score descending.

            groups.setdefault(s, []).append(i)

def generate_scores(players, round_idx, max_rounds):
    # players: list of player ids (ordered by speed ascending)
    # round_idx: 0-index for round (starting 0)
    # returns list of dict {player:score}
    if len(players) == 1:
        return [{players[0]: 0}]
    half = len(players)//2
    results = []
    # generate valid partitions (L winners)
    for L in combinations(players, half):
        Lset = set(L)
        # ballot condition
        cntL = cntR = 0
        ok = True
        for p in players:
            if p in Lset:
                cntL += 1
            else:
                cntR += 1
            if cntL > cntR:
                ok = False
                break
        if not ok:

        R = [p for p in players if p not in Lset]
        left_maps = generate_scores(L, round_idx+1, max_rounds)
        right_maps = generate_scores(R, round_idx+1, max_rounds)
        point = 2**(max_rounds-1-round_idx)  # points for this round
        # combine
        for lm in left_maps:

                combined = {}
                for p, s in lm.items():

                for p, s in rm.items():
                    combined[p] = s
                results.append(combined)
    return results

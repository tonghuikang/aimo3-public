def all_orders(num):
    R = int(math.log2(num))
    participants = list(range(num))
    # initial groups: all participants in one group
    groups = [participants]  # each group list sorted by speed (fastest first)
    scores = {p:0 for p in participants}
    orders = set()
    def recurse(groups, round_idx):
        if round_idx == R:
            # final scores distinct; order participants by descending score
            ordered = sorted(participants, key=lambda p: scores[p], reverse=True)
            orders.add(tuple(ordered))
            return
        # For each group, generate possible winner subsets (Dyck)
        # Since groups are independent, we need to iterate over Cartesian product of choices for each group.
        # We'll use recursion over groups.
        def process_group(idx, new_groups):
            if idx == len(groups):
                # new_groups is list of groups after this round (winners groups + losers groups)
                recurse(new_groups, round_idx+1)
                return
            group = groups[idx]
            # group size may be 1? but if size 1, cannot split (should not happen before R)
            if len(group) == 1:
                # no competition, group stays as is
                process_group(idx+1, new_groups + [group])
                return
            subsets = dyck_subsets(group)
            for winners in subsets:
                winners_set = set(winners)
                losers = [p for p in group if p not in winners_set]
                # update scores
                inc = 2**(R-1-round_idx)  # points for this round
                for p in winners:
                    scores[p] += inc
                # proceed recursively
                process_group(idx+1, new_groups + [winners, losers])
                # revert scores
                for p in winners:
                    scores[p] -= inc
        process_group(0, [])
    recurse(groups, 0)
    return orders

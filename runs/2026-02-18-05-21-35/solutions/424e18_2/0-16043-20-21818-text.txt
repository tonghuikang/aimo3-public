def all_matchings(elements):
    # elements: tuple/list of distinct items
    # returns list of list of pairs (each pair as tuple)
    # recursively generate perfect matchings
    if not elements:
        yield []
        return
    first = elements[0]
    for i in range(1, len(elements)):
        second = elements[i]
        rest = elements[1:i] + elements[i+1:]
        for matching in all_matchings(rest):
            yield [(first, second)] + matching

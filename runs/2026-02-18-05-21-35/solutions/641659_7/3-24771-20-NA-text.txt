def incircle_contact_points(A,B,C):
    # B=(0,0), C=(c,0). General.
    # Compute side lengths
    a = math.dist(B, C)
    b = math.dist(C, A)
    c_len = math.dist(A, B)
    # semiperimeter
    s = (a+b+c_len)/2
    # distances from vertices to touchpoints: BD = s - b, CD = s - c
    BD = s - b
    CD = s - c_len
    D = (BD, 0)  # on BC from B
    # coordinates of incircle center I: formula using weighted average of vertices by side lengths
    # I = (ax_A + bx_B + cx_C)/perimeter where a opposite A = BC length etc.
    # Actually coordinates: (a*x_A + b*x_B + c*x_C)/perimeter ...
    per = a+b+c_len
    I = ((a*A[0] + b*B[0] + c_len*C[0])/per,
         (a*A[1] + b*B[1] + c_len*C[1])/per)
    # radius r = area / s
    # But we need E and F: foot from I to sides CA and AB.
    # Compute projection of I onto line CA etc.
    def proj_point(P, Q, R): # foot from R to line PQ
        # vector PQ, PR
        (x1,y1)=P; (x2,y2)=Q; (x0,y0)=R
        vx,vy=x2-x1,y2-y1
        # project vector (R-P) onto v
        t = ((x0-x1)*vx + (y0-y1)*vy)/(vx*vx+vy*vy)
        # foot point
        return (x1 + t*vx, y1 + t*vy)
    E = proj_point(C, A, I)
    F = proj_point(A, B, I)
    return D, E, F, I

def circumcenter(P, Q, R):
    # return center of circumscribed circle of triangle PQR
    (x1,y1)=P; (x2,y2)=Q; (x3,y3)=R
    d = 2*(x1*(y2-y3)+x2*(y3-y1)+x3*(y1-y2))
    if abs(d) < 1e-9:
        return None
    ux = ((x1**2+y1**2)*(y2-y3)+(x2**2+y2**2)*(y3-y1)+(x3**2+y3**2)*(y1-y2))/d
    uy = ((x1**2+y1**2)*(x3-x2)+(x2**2+y2**2)*(x1-x3)+(x3**2+y3**2)*(x2-x1))/d
    return (ux, uy)

def circle_intersections(c1, r1, c2, r2):
    (x0,y0)=c1; (x1,y1)=c2
    dx, dy = x1-x0, y1-y0
    d = math.hypot(dx, dy)
    if d > r1+r2+1e-9 or d < abs(r1-r2)-1e-9:
        return []  # no intersection
    # a = (r1^2 - r2^2 + d^2) / (2d)
    a = (r1**2 - r2**2 + d**2) / (2*d)
    # h = sqrt(r1^2 - a^2)
    h_sq = r1**2 - a**2
    if h_sq < -1e-9:
        return []
    h = math.sqrt(max(h_sq,0))
    xm = x0 + a*dx/d
    ym = y0 + a*dy/d
    xs1 = xm + h*dy/d
    ys1 = ym - h*dx/d
    xs2 = xm - h*dy/d
    ys2 = ym + h*dx/d
    if h < 1e-9:
        return [(xs1,ys1)]
    else:
        return [(xs1,ys1),(xs2,ys2)]

def is_cyclic(P,Q,R,S, eps=1e-6):
    # check if four points lie on a circle (non-collinear)
    # compute circumcenter of PQR, compare radius to distance to S
    O = circumcenter(P,Q,R)
    if O is None:
        return False
    r = math.dist(O, P)
    return abs(math.dist(O, S) - r) < eps

    A = None
    # Solve 2x2: s*dirB - t*dirC = C - B
    A_mat = np.array([[dirB[0], -dirC[0]],[dirB[1], -dirC[1]]], float)
    rhs = np.array([C[0]-B[0], C[1]-B[1]], float)

        A = (B[0] + s*dirB[0], B[1] + s*dirB[1])

        return None
    return A, B, C, D, I

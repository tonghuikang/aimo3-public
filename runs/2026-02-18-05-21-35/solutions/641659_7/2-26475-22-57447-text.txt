def intersect_circles(c1, c2):
    # returns intersection points of two circles (list of tuples)
    (x0,y0,r0) = c1
    (x1,y1,r1) = c2
    # distance between centers
    d = mp.sqrt((x1-x0)**2 + (y1-y0)**2)
    if d > r0+r1 or d < abs(r0-r1):
        return []  # no intersect
    # point 2 intersection
    a = (r0**2 - r1**2 + d**2) / (2*d)
    h_sq = r0**2 - a**2
    if h_sq < 0:
        h = 0
    else:
        h = mp.sqrt(h_sq)
    # point p2 = base point
    x2 = x0 + a*(x1-x0)/d
    y2 = y0 + a*(y1-y0)/d
    # intersection points
    rx = -(y1-y0) * (h/d)
    ry = (x1-x0) * (h/d)
    p_int1 = (x2 + rx, y2 + ry)
    p_int2 = (x2 - rx, y2 - ry)
    return [p_int1, p_int2]

def foot_of_perpendicular(P, Q, R):
    # foot from P to line QR
    (x0,y0) = P
    (x1,y1) = Q
    (x2,y2) = R
    # vector QR
    dx = x2 - x1
    dy = y2 - y1
    # projection t of vector QP onto QR
    t = ((x0 - x1)*dx + (y0 - y1)*dy) / (dx*dx + dy*dy)
    # foot
    return (x1 + t*dx, y1 + t*dy)

def triangle_from_params(d, a, r):
    u = a - d
    m_ab = 2*d*r/(d**2 - r**2)
    m_ac = -2*u*r/(u**2 - r**2)
    # intersection of lines
    xA = - m_ac * a / (m_ab - m_ac)
    yA = m_ab * xA
    B = (0.0, 0.0)
    C = (a, 0.0)
    D = (d, 0.0)
    # foot F
    tF = (d + m_ab * r) / (1 + m_ab**2)
    F = (tF, m_ab*tF)
    # foot E
    u_param = ((d - a) + m_ac * r) / (1 + m_ac**2)
    E = (a + u_param, m_ac * u_param)
    return {'A': (xA, yA), 'B': B, 'C': C, 'D': D, 'E': E, 'F': F, 'I': (d, r), 'm_ab': m_ab, 'm_ac': m_ac}

def circumcenter(p1,p2,p3):
    # returns center and radius of circle through three points
    A = np.array(p1)
    B = np.array(p2)
    C = np.array(p3)
    # compute using perpendicular bisectors
    # Solve linear equations (B-A)·X = (|B|^2 - |A|^2)/2 etc.
    # Use matrix
    M = np.array([B - A, C - A])
    if np.linalg.matrix_rank(M) < 2:
        return None,None
    # Right side
    rhs = np.array([(np.dot(B,B)-np.dot(A,A))/2.0,
                    (np.dot(C,C)-np.dot(A,A))/2.0])
    # Solve for X
    center = np.linalg.solve(M.T, rhs)
    radius = np.linalg.norm(center - A)
    return center, radius

def line_intersection(p1, d1, p2, d2):
    # returns intersection of lines p1 + t*d1 and p2 + s*d2
    # solve p1 + t d1 = p2 + s d2
    A = np.column_stack((d1, -d2))
    if np.linalg.matrix_rank(A) < 2:
        return None
    t_s = np.linalg.solve(A, p2 - p1)

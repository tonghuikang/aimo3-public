def line_through(P,Q):
    # returns (A,B,C) for line Ax+By+C=0
    (x1,y1),(x2,y2)=P,Q
    A = y1 - y2
    B = x2 - x1
    C = x1*y2 - x2*y1
    return (A,B,C)

def foot_of_perpendicular(P, line):
    # line as (A,B,C). foot from point P onto line
    x0,y0 = P
    A,B,C = line
    d = A*A + B*B
    x = (B*(B*x0 - A*y0) - A*C)/d
    y = (A*(-B*x0 + A*y0) - B*C)/d
    return (x,y)

def reflect_point_across_line(P, line):
    # reflect P across line (A,B,C)
    x0,y0=P
    A,B,C=line
    d = A*A + B*B
    # formula for reflection
    x = x0 - 2*A*(A*x0 + B*y0 + C)/d
    y = y0 - 2*B*(A*x0 + B*y0 + C)/d
    return (x,y)

def are_cyclic(P,Q,R,S):
    # check if four points are concyclic (determinant ~0)
    # compute determinant
    def det4(m):
        return np.linalg.det(np.array(m))
    mat = []
    for (x,y) in [P,Q,R,S]:
        mat.append([x, y, x*x+y*y, 1])
    return abs(det4(mat)) < 1e-6

def triangle_from_sides(a,b,c):
    # place B at (0,0), C at (a,0). Place A at coordinates using law of cos
    # using side lengths: AB=c, AC=b
    # we need coordinates of A = (x_A, y_A) such that distance to B is c, to C is b.
    # Solve x^2 + y^2 = c^2, (x-a)^2 + y^2 = b^2
    x, y = sp.symbols('x y', real=True)
    sol = sp.solve([x**2 + y**2 - c**2, (x-a)**2 + y**2 - b**2], (x,y))
    # Choose solution with y>0
    for s in sol:
        if s[1].evalf() > 0:
            return (float(s[0]), float(s[1]))
    # fallback
    return (float(sol[0][0]), float(sol[0][1]))

def incircle_touch_points(A,B,C):
    # compute incircle center I and radius r, then foot of perpendiculars onto sides
    Ax,Ay = A; Bx,By = B; Cx,Cy = C
    # side lengths
    a = sp.sqrt((Bx-Cx)**2 + (By-Cy)**2) # BC
    b = sp.sqrt((Cx-Ax)**2 + (Cy-Ay)**2) # CA
    c = sp.sqrt((Ax-Bx)**2 + (Ay-By)**2) # AB
    s = (a+b+c)/2
    # incenter coordinates:
    I = ((a*Ax + b*Bx + c*Cx)/(a+b+c), (a*Ay + b*By + c*Cy)/(a+b+c))
    # foot to BC:
    # line BC: param t from B to C, foot D is projection of I onto BC
    # using vector projection
    import numpy as np
    Bnp = np.array([Bx,By]); Cnp = np.array([Cx,Cy]); Inp = np.array([float(I[0]), float(I[1])])
    BC = Cnp - Bnp
    t = np.dot(Inp - Bnp, BC) / np.dot(BC, BC)
    D = Bnp + t*BC
    # foot to CA:
    Cnp = np.array([Cx,Cy]); Anp = np.array([Ax,Ay])
    CA = Anp - Cnp
    t2 = np.dot(Inp - Cnp, CA) / np.dot(CA, CA)
    E = Cnp + t2*CA
    # foot to AB:
    Anp = np.array([Ax,Ay]); Bnp = np.array([Bx,By])
    AB = Bnp - Anp
    t3 = np.dot(Inp - Anp, AB) / np.dot(AB, AB)
    F = Anp + t3*AB
    return tuple(map(tuple, (D,E,F, I)))

def circumcircle_center_and_radius(A,B,C):
    Ax,Ay = A; Bx,By = B; Cx,Cy = C
    # using sympy geometry
    circ = sp.Circle(sp.Point(Ax,Ay), sp.Point(Bx,By), sp.Point(Cx,Cy))
    return (float(circ.center.x), float(circ.center.y), float(circ.radius))

def circle_intersections(c1_center, r1, c2_center, r2):
    # returns intersection points as list of tuples
    x1,y1 = c1_center; x2,y2 = c2_center
    d = ((x2-x1)**2 + (y2-y1)**2)**0.5
    if d > r1 + r2 + 1e-9 or d < abs(r1 - r2) - 1e-9:
        return []
    # compute intersection points
    a = (r1**2 - r2**2 + d**2) / (2*d)
    h_sq = r1**2 - a**2
    if h_sq < 0:
        h = 0.0
    else:
        h = h_sq**0.5
    xm = x1 + a*(x2-x1)/d
    ym = y1 + a*(y2-y1)/d
    xs1 = xm + h*(y2-y1)/d
    ys1 = ym - h*(x2-x1)/d
    xs2 = xm - h*(y2-y1)/d
    ys2 = ym + h*(x2-x1)/d
    return [(xs1,ys1),(xs2,ys2)]

def reflect_point_across_line(P, A, B):
    # reflect point P across line AB
    Px,Py = P; Ax,Ay = A; Bx,By = B
    # line vector
    dx, dy = Bx-Ax, By-Ay
    # normalize?
    # formula from projection
    # compute projection of P onto line AB
    t = ((Px-Ax)*dx + (Py-Ay)*dy) / (dx*dx + dy*dy)
    proj = (Ax + t*dx, Ay + t*dy)
    # reflection: P' = 2*proj - P
    return (2*proj[0]-Px, 2*proj[1]-Py)

def foot_of_perpendicular_from_point_to_line(P, A, B):
    # returns foot from P to line AB
    Px,Py = P; Ax,Ay = A; Bx,By = B
    dx, dy = Bx-Ax, By-Ay
    t = ((Px-Ax)*dx + (Py-Ay)*dy) / (dx*dx + dy*dy)
    foot = (Ax + t*dx, Ay + t*dy)
    return foot

    d = math.hypot(dx, dy)
    if d > r1 + r2 + 1e-9 or d < abs(r1 - r2) - 1e-9:
        return []
    if d == 0 and abs(r1 - r2) < 1e-9:
        # coincident circles
        return []
    # distance from center1 to line of intersection points
    a = (r1**2 - r2**2 + d**2) / (2*d)
    h_sq = r1**2 - a**2
    if h_sq < 0:
        h = 0.0
    else:
        h = math.sqrt(h_sq)
    xm = x1 + a*dx/d
    ym = y1 + a*dy/d
    xs1 = xm + h*dy/d
    ys1 = ym - h*dx/d
    xs2 = xm - h*dy/d
    ys2 = ym + h*dx/d
    if h < 1e-12:
        return [(xs1, ys1)]
    else:
        return [(xs1, ys1), (xs2, ys2)]

def reflect_point_across_line(P, A, B):
    # reflect point P across line AB
    # using vector projection
    # formula from geometry
    # line direction vector
    dx = B[0] - A[0]
    dy = B[1] - A[1]
    # compute projection parameter t of P onto line AB
    t = ((P[0] - A[0])*dx + (P[1] - A[1])*dy) / (dx*dx + dy*dy)
    proj = (A[0] + t*dx, A[1] + t*dy)
    # reflection
    return (2*proj[0] - P[0], 2*proj[1] - P[1])

def foot_of_perpendicular(P, A, B):
    # foot from P to line AB
    dx = B[0] - A[0]
    dy = B[1] - A[1]
    t = ((P[0] - A[0])*dx + (P[1] - A[1])*dy) / (dx*dx + dy*dy)
    return (A[0] + t*dx, A[1] + t*dy)

def incircle_center_and_radius(A, B, C):
    # compute incenter coordinates and radius
    # using side lengths
    a = abs(B-C)
    b = abs(C-A)
    c = abs(A-B)
    # coordinates as complex numbers
    # incenter formula: weighted average of vertices with side lengths
    I = (a*A + b*B + c*C) / (a + b + c)
    # radius = area / s
    # area via cross product
    area = abs((B-A).conjugate() * (C-A)).imag / 2.0
    s = (a + b + c)/2.0
    r = area / s
    return I, r

def line_intersection(p1, p2, p3, p4):
    # return intersection of lines (p1,p2) and (p3,p4)
    # using complex numbers
    # solve p1 + t*(p2-p1) = p3 + u*(p4-p3)
    # cross product
    x1, x2, x3, x4 = p1, p2, p3, p4
    # convert to real vectors
    def to_vec(z):
        return np.array([z.real, z.imag])
    A = np.column_stack((to_vec(x2-x1), -to_vec(x4-x3)))
    b = to_vec(x3 - x1)
    try:
        t_u = np.linalg.solve(A, b)
    except np.linalg.LinAlgError:
        return None
    t = t_u[0]
    return x1 + t*(x2-x1)

def circle_intersections(O1, r1, O2, r2):
    # returns intersection points of two circles
    d = abs(O2-O1)
    if d > r1 + r2 + 1e-9 or d < abs(r1 - r2) - 1e-9:
        return [] # no intersection
    # handle near tangency
    a = (r1**2 - r2**2 + d**2) / (2*d)
    h_sq = max(r1**2 - a**2, 0)
    h = math.sqrt(h_sq)
    # point P2
    P2 = O1 + a*(O2-O1)/d
    # intersection points
    rx = -(O2-O1).imag * (h/d)
    ry = (O2-O1).real * (h/d)
    # Using complex arithmetic:
    inter1 = P2 + complex(rx, ry)
    inter2 = P2 - complex(rx, ry)
    return [inter1, inter2]

def circumcircle(A,B,C):
    # returns (center, radius)
    # using perpendicular bisectors
    # Solve using linear algebra
    D = 2*(A.real*(B.imag - C.imag) + B.real*(C.imag - A.imag) + C.real*(A.imag - B.imag))
    if abs(D) < 1e-9:
        return None, None
    Ux = ((abs(A)**2)*(B.imag - C.imag) + (abs(B)**2)*(C.imag - A.imag) + (abs(C)**2)*(A.imag - B.imag)) / D
    Uy = ((abs(A)**2)*(C.real - B.real) + (abs(B)**2)*(A.real - C.real) + (abs(C)**2)*(B.real - A.real)) / D
    O = complex(Ux, Uy)
    r = abs(A-O)
    return O, r

def foot_point(P, A, B):
    # foot of perpendicular from P to line AB
    # using projection
    AB = B - A
    t = ((P - A).real * AB.real + (P - A).imag * AB.imag) / (AB.real**2 + AB.imag**2)
    proj = A + t*AB
    # compute foot perpendicular: projection onto AB, then foot is projection (since we want perpendicular to AB)
    # Actually foot is projection onto line AB: same as proj
    return proj

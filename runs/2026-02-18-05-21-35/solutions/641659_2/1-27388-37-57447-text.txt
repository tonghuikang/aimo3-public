def circumcenter(p1,p2,p3):
    # points as (x,y)
    A = np.array(p1)
    B = np.array(p2)
    C = np.array(p3)
    # compute circumcenter using perpendicular bisectors
    # Solve for O such that |O-A|^2=|O-B|^2 and |O-A|^2=|O-C|^2
    # Linear equations: 2*(B-A)·O = |B|^2 - |A|^2
    # similarly for C-A
    BA = B - A
    CA = C - A
    rhs1 = (np.dot(B,B) - np.dot(A,A)) / 2.0
    rhs2 = (np.dot(C,C) - np.dot(A,A)) / 2.0
    M = np.vstack([BA, CA])
    rhs = np.array([rhs1, rhs2])
    O = np.linalg.lstsq(M, rhs, rcond=None)[0]
    # radius
    r = np.linalg.norm(O - A)
    return O, r

def circle_intersection(O1,r1,O2,r2):
    # returns list of intersection points (0,1,2) as list of (x,y)
    d = np.linalg.norm(O2 - O1)
    if d > r1 + r2 + 1e-12 or d < abs(r1 - r2) - 1e-12:
        return []  # no intersections
    # distance from O1 to line joining intersections
    a = (r1**2 - r2**2 + d**2) / (2*d)
    h_sq = r1**2 - a**2
    if h_sq < -1e-12:
        return []
    h = math.sqrt(max(h_sq,0.0))
    P = O1 + a * (O2 - O1) / d
    # offsets
    offset = h * np.array([-(O2[1]-O1[1]), O2[0]-O1[0]]) / d
    inter1 = P + offset
    inter2 = P - offset
    if np.allclose(inter1, inter2):
        return [tuple(inter1)]
    else:
        return [tuple(inter1), tuple(inter2)]

    # foot from point P to line AB
    # project onto line
    A = np.array(A); B = np.array(B); P = np.array(P)
    AB = B - A
    t = np.dot(P - A, AB) / np.dot(AB, AB)
    foot = A + t * AB
    return foot

def reflect_point_across_line(P, A, B):
    # reflect P across line AB
    foot = foot_point(P, A, B)
    # vector from P to foot
    v = foot - np.array(P)
    return tuple(np.array(P) + 2*v)

def distance(P, Q):
    return np.linalg.norm(np.array(P)-np.array(Q))

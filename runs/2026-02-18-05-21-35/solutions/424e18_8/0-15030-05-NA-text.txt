            current.pop()
        # place zero if zeros_left >0 and balance >0 (otherwise prefix condition violated)
        if zeros_left > 0 and balance > 0:
            current.append(0)
            rec(pos+1, ones_left, zeros_left-1, balance-1, current)
            current.pop()
    rec(0, n, n, 0, [])
    return result

def count_permutations(size):
    players = list(range(size))
    memo = {}
    def rec(seq):
        # seq is tuple of players ordered by speed
        if len(seq) == 1:
            return 1
        if seq in memo:
            return memo[seq]
        n = len(seq)//2
        total = 0
        for mask in generate_balanced_masks(n):
            # mask list length 2n: 1 for left group (winners), 0 for right group
            left = tuple(seq[i] for i, b in enumerate(mask) if b==1)
            right = tuple(seq[i] for i, b in enumerate(mask) if b==0)
            total += rec(left) * rec(right)
        memo[seq] = total
        return total
    return rec(tuple(players))

def pairings(lst):
    # generate all perfect matchings for list lst (even length)
    if not lst:
        yield []
        return
    first = lst[0]
    for i in range(1, len(lst)):
        second = lst[i]
        rest = lst[1:i] + lst[i+1:]
        for rest_match in pairings(rest):
            yield [(first, second)] + rest_match

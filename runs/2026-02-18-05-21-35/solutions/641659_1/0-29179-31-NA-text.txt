def intersect_circles(c1, r1, c2, r2):

    # point p = c1 + x*ex
    p = c1 + x*ex
    # offset vector perpendicular to ex
    offset = h*ex*1j
    return [p + offset, p - offset]

def triangle_points(B, C):
    A = math.pi - B - C
    if A <= 0: 
        return None
    # incircle radius r=1
    d = 1+0j  # angle 0
    f = cmath.rect(1, math.pi - B)  # angle pi - B
    e = cmath.rect(1, math.pi + A)  # angle pi + A
    # compute vertices
    r = 1.0
    A_pt = 2*r*e*f/(e+f)
    B_pt = 2*r*f*d/(f+d)
    C_pt = 2*r*d*e/(d+e)
    return (A_pt, B_pt, C_pt, d, e, f)

def circumcenter(p1, p2, p3):
    x1, y1 = p1.real, p1.imag
    x2, y2 = p2.real, p2.imag
    x3, y3 = p3.real, p3.imag
    d = 2*(x1*(y2-y3) + x2*(y3-y1) + x3*(y1-y2))
    if abs(d) < 1e-12:
        return None
    ux = ((x1**2+y1**2)*(y2-y3) + (x2**2+y2**2)*(y3-y1) + (x3**2+y3**2)*(y1-y2))/d
    uy = ((x1**2+y1**2)*(x3-x2) + (x2**2+y2**2)*(x1-x3) + (x3**2+y3**2)*(x2-x1))/d
    center = ux + uy*1j
    radius = abs(p1 - center)
    return center, radius

def intersect_circles(c1, r1, c2, r2):
    d = abs(c2 - c1)
    if d < 1e-12:
        return []
    if d > r1 + r2 + 1e-9 or d < abs(r1 - r2) - 1e-9:
        return []
    # ex unit vector

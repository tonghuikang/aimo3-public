def tangent_points(P, O, R):
    # Returns two tangent points from external point P to circle (O,R)
    v = P - O
    d2 = (v.real**2 + v.imag**2)

def foot_point(P, A, B):

def circumcenter(P, Q, R):
    # returns center as complex
    # Use perpendicular bisectors method.
    # Solve for center O such that |O-P|=|O-Q| and |O-P|=|O-R|
    # Equivalent linear equations from squared distances difference
    # We'll solve using linear algebra.
    Ax = 2*(Q.real - P.real)
    Ay = 2*(Q.imag - P.imag)
    Bx = 2*(R.real - P.real)
    By = 2*(R.imag - P.imag)
    C1 = Q.real**2 + Q.imag**2 - (P.real**2 + P.imag**2)
    C2 = R.real**2 + R.imag**2 - (P.real**2 + P.imag**2)
    A = np.array([[Ax, Ay], [Bx, By]])
    C = np.array([C1, C2])
    try:
        ox, oy = np.linalg.solve(A, C)
    except np.linalg.LinAlgError:
        return None
    return complex(ox, oy)

def circumcircle_intersection(P, Q, R, S, T, U):
    # intersect two circumcircles (P,Q,R) and (S,T,U). Return two points (list)
    # Compute circle centers and radii, then intersect circles.
    O1 = circumcenter(P, Q, R)
    O2 = circumcenter(S, T, U)
    if O1 is None or O2 is None:
        return []
    r1 = abs(P - O1)
    r2 = abs(S - O2)
    d = abs(O2 - O1)
    if d > r1 + r2 + 1e-9 or d < abs(r1 - r2) - 1e-9:
        return []  # no intersections
    # compute intersection points
    # from circle-circle intersection formulas
    # a = (r1^2 - r2^2 + d^2) / (2d)
    a = (r1**2 - r2**2 + d**2) / (2*d)
    h_sq = max(r1**2 - a**2, 0.0)
    h = math.sqrt(h_sq)
    # point P2 = O1 + a*(O2-O1)/d
    P2 = O1 + a * (O2 - O1) / d
    # offset vector
    offset = h * (O2 - O1) / d * 1j  # rotate by 90 degrees (multiply by i)
    inter1 = P2 + offset
    inter2 = P2 - offset
    return [inter1, inter2]

    return -eps <= t <= 1+eps

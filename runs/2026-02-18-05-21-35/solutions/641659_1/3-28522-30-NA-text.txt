def triangle_geometry(p,q,s):
    a = p+q
    b = q + s - a
    c = p + s - a
    # ensure b>0, c>0

def circumcenter(p1, p2, p3):
    # returns center (x,y) of circumcircle of triangle p1,p2,p3
    # using perpendicular bisector intersection.
    (x1,y1),(x2,y2),(x3,y3) = p1,p2,p3
    d = 2*(x1*(y2 - y3) + x2*(y3 - y1) + x3*(y1 - y2))
    if abs(d) < 1e-12:
        return None
    ux = ((x1**2 + y1**2)*(y2 - y3) + (x2**2 + y2**2)*(y3 - y1) + (x3**2 + y3**2)*(y1 - y2))/d
    uy = ((x1**2 + y1**2)*(x3 - x2) + (x2**2 + y2**2)*(x1 - x3) + (x3**2 + y3**2)*(x2 - x1))/d
    return (ux, uy)

def circle_center_radius(p1,p2,p3):

def foot_point(P, A, B):
    # foot of perpendicular from point P to line AB
    # using projection
    (x0,y0) = P
    (x1,y1) = A
    (x2,y2) = B
    dx = x2 - x1
    dy = y2 - y1
    if dx==0 and dy==0:
        return A
    t = ((x0 - x1)*dx + (y0 - y1)*dy)/(dx*dx + dy*dy)
    # projection onto line
    projx = x1 + t*dx
    projy = y1 + t*dy
    # foot is projection? Wait foot is projection onto line, yes
    # Since we want perpendicular foot, it's same as projection onto line.
    return (projx, projy)

def intersect_circles(c1, r1, c2, r2):
    # returns list of intersection points (0,1,2)
    x1,y1 = c1
    x2,y2 = c2
    d = math.hypot(x2-x1, y2-y1)
    if d > r1 + r2 + 1e-12 or d < abs(r1 - r2) - 1e-12 or d == 0:
        return []
    # distance from c1 to line connecting intersection points
    a = (r1**2 - r2**2 + d**2) / (2*d)
    h_sq = max(r1**2 - a**2, 0.0)
    h = math.sqrt(h_sq)
    # point p2 = c1 + a*(c2-c1)/d
    x3 = x1 + a*(x2 - x1)/d
    y3 = y1 + a*(y2 - y1)/d
    # intersection points
    rx = -(y2 - y1) * (h/d)
    ry = (x2 - x1) * (h/d)
    p_int1 = (x3 + rx, y3 + ry)
    p_int2 = (x3 - rx, y3 - ry)
    if h < 1e-12:
        return [p_int1]
    else:
        return [p_int1, p_int2]

def reflect_point_across_line(P, A, B):
    # reflect point P across line AB
    # formula: projection onto line, then symmetrical
    proj = foot_point(P, A, B)
    # vector from P to proj
    vx = proj[0] - P[0]
    vy = proj[1] - P[1]
    # reflected point = P + 2*(proj - P)
    return (P[0] + 2*vx, P[1] + 2*vy)

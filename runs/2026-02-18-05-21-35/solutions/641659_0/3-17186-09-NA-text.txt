def intersect_circles(c1, r1, c2, r2):
    # returns two intersection points as numpy arrays
    O1 = np.array(c1, dtype=float)
    O2 = np.array(c2, dtype=float)
    d = np.linalg.norm(O2 - O1)
    if d == 0:
        return []
    # check for no intersection
    if d > r1 + r2 + 1e-12 or d < abs(r1 - r2) - 1e-12:
        return []
    a = (r1**2 - r2**2 + d**2) / (2*d)
    h_sq = r1**2 - a**2
    if h_sq < -1e-12:
        return []
    h = math.sqrt(max(h_sq, 0.0))
    P2 = O1 + a * (O2 - O1) / d
    # perpendicular vector
    perp = np.array([-(O2 - O1)[1], (O2 - O1)[0]]) / d
    inter1 = P2 + h * perp
    inter2 = P2 - h * perp
    return [inter1, inter2]

def circle_center_radius(p1, p2, p3):
    # compute circumcenter and radius via linear algebra
    A = np.array(p1, dtype=float)
    B = np.array(p2, dtype=float)
    C = np.array(p3, dtype=float)
    # using perpendicular bisectors intersection
    # Solve using determinants (circumcenter formula)
    # Compute using matrix method
    d = 2 * (A[0]*(B[1]-C[1]) + B[0]*(C[1]-A[1]) + C[0]*(A[1]-B[1]))
    if abs(d) < 1e-12:
        raise ValueError("Collinear")
    ux = ((np.linalg.norm(A)**2)*(B[1]-C[1]) + (np.linalg.norm(B)**2)*(C[1]-A[1]) + (np.linalg.norm(C)**2)*(A[1]-B[1])) / d
    uy = ((np.linalg.norm(A)**2)*(C[0]-B[0]) + (np.linalg.norm(B)**2)*(A[0]-C[0]) + (np.linalg.norm(C)**2)*(B[0]-A[0])) / d
    center = np.array([ux, uy])
    radius = np.linalg.norm(center - A)
    return center, radius

def proj_point_on_line(P, L1, L2):
    # projection of P onto line L1 L2
    P = np.array(P, dtype=float)
    L1 = np.array(L1, dtype=float)
    L2 = np.array(L2, dtype=float)
    v = L2 - L1
    t = np.dot(P - L1, v) / np.dot(v, v)
    proj = L1 + t * v
    return proj

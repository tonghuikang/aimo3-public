def intersect_circles(c1, r1, c2, r2):
    # return intersection points (list of complex)
    # based on circle intersection formulas
    d = abs(c2 - c1)
    if d > r1 + r2 + 1e-9 or d < abs(r1 - r2) - 1e-9:
        return []  # no intersection
    # compute a = (r1^2 - r2^2 + d^2)/(2d)
    a = (r1**2 - r2**2 + d**2) / (2*d)
    h_sq = r1**2 - a**2
    if h_sq < -1e-9:
        return []
    h = math.sqrt(max(h_sq, 0.0))
    # point p = c1 + a*(c2 - c1)/d
    p = c1 + a * (c2 - c1) / d
    # offset vector
    offset = h * (c2 - c1) * 1j / d
    return [p + offset, p - offset]

def foot_point(P, A, B):
    # foot from P to line AB
    # compute projection
    # Using complex numbers: foot = A + ((P-A).dot(B-A))/|B-A|^2 * (B-A)
    v = B - A
    t = ((P - A).real * v.real + (P - A).imag * v.imag) / (abs(v)**2)
    return A + t * v

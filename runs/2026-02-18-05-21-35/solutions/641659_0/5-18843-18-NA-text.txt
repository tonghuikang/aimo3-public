def intersection_circles(c1, r1, c2, r2):
    # returns intersection points of two circles (list)
    c1 = np.array(c1); c2 = np.array(c2)
    d = np.linalg.norm(c2-c1)

        return []  # no intersection
    # point along line connecting centers at distance a from c1
    a = (r1**2 - r2**2 + d**2) / (2*d)
    h_sq = r1**2 - a**2
    if h_sq < -1e-9:
        return []
    h = math.sqrt(max(0.0, h_sq))
    # base point
    P2 = c1 + a*(c2-c1)/d
    # offset vector perpendicular
    offset = h * np.array([-(c2[1]-c1[1])/d, (c2[0]-c1[0])/d])

def reflect_point_across_line(P, L1, L2):

    proj = L1 + d * np.dot(P - L1, d) / d_norm2
    # reflected point

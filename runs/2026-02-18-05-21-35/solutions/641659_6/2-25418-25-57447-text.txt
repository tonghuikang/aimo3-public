def tangent_points(P, I, r):
    # returns two tangent points from external point P to circle (I,r)
    # P and I are tuples (x,y)
    Px, Py = P
    Ix, Iy = I
    vx = Px - Ix
    vy = Py - Iy
    d2 = vx*vx + vy*vy
    d = math.sqrt(d2)
    if d <= r:
        return []  # inside or on circle
    a = r*r / d2
    b = r * math.sqrt(d2 - r*r) / d2
    # rotate v 90 degrees: ( -vy, vx)
    # two solutions: I + a*v +/- b*rot90(v)
    t1 = (Ix + a*vx - b*vy, Iy + a*vy + b*vx)
    t2 = (Ix + a*vx + b*vy, Iy + a*vy - b*vx)
    return [t1, t2]

def line_intersection(p1,p2,p3,p4):
    # intersection of lines p1p2 and p3p4 (not segments)
    x1,y1 = p1
    x2,y2 = p2
    x3,y3 = p3
    x4,y4 = p4
    # Solve using cross product
    denom = (x1-x2)*(y3-y4)-(y1-y2)*(x3-x4)
    if abs(denom) < 1e-12:
        return None
    x = ((x1*y2 - y1*x2)*(x3-x4) - (x1-x2)*(x3*y4 - y3*x4))/denom
    y = ((x1*y2 - y1*x2)*(y3-y4) - (y1-y2)*(x3*y4 - y3*x4))/denom
    return (x,y)

def circumcenter(p1,p2,p3):
    # returns center and radius of circle through three points
    x1,y1 = p1
    x2,y2 = p2
    x3,y3 = p3
    # Use perpendicular bisector method
    d = 2*(x1*(y2 - y3) + x2*(y3 - y1) + x3*(y1 - y2))
    if abs(d) < 1e-12:
        return None
    ux = ((x1*x1 + y1*y1)*(y2 - y3) + (x2*x2 + y2*y2)*(y3 - y1) + (x3*x3 + y3*y3)*(y1 - y2))/d
    uy = ((x1*x1 + y1*y1)*(x3 - x2) + (x2*x2 + y2*y2)*(x1 - x3) + (x3*x3 + y3*y3)*(x2 - x1))/d
    r = math.hypot(ux - x1, uy - y1)
    return (ux, uy, r)

def reflect_point_across_line(P, A, B):
    # reflect P across line through points A,B
    # formula: projection onto line then mirror
    x0,y0 = P
    x1,y1 = A
    x2,y2 = B
    # line direction
    dx = x2 - x1
    dy = y2 - y1
    # if line degenerate, return P
    if abs(dx) + abs(dy) < 1e-12:
        return P
    # projection of P onto line
    t = ((x0 - x1)*dx + (y0 - y1)*dy) / (dx*dx + dy*dy)
    projx = x1 + t*dx
    projy = y1 + t*dy
    # reflect: P' = 2*proj - P
    return (2*projx - x0, 2*projy - y0)

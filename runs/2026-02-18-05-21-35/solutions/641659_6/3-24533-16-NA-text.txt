def reflect_across_line(P, line_pt1, line_pt2):
    # reflect point P across line defined by two points
    L = sp.Line(line_pt1, line_pt2)
    # projection of P onto line
    proj = L.projection(P)
    # reflect: Q = proj*2 - P
    Q = sp.Point(2*proj.x - P.x, 2*proj.y - P.y)
    return Q

def incircle_center(A,B,C):
    # compute side lengths a=BC,b=CA,c=AB
    a = math.hypot(B[0]-C[0], B[1]-C[1])
    b = math.hypot(C[0]-A[0], C[1]-A[1])
    c = math.hypot(A[0]-B[0], A[1]-B[1])
    # coordinates weighted by side lengths
    px = a*A[0] + b*B[0] + c*C[0]
    py = a*A[1] + b*B[1] + c*C[1]
    s = a+b+c
    return (px/s, py/s)

def foot_point(P, X, Y):
    # projection of P onto line XY
    # XY vector
    vx = Y[0] - X[0]
    vy = Y[1] - X[1]
    # vector from X to P
    wx = P[0] - X[0]
    wy = P[1] - X[1]
    # projection scalar t = (w·v)/(v·v)
    vdotv = vx*vx + vy*vy
    t = (wx*vx + wy*vy) / vdotv
    return (X[0] + t*vx, X[1] + t*vy)

def circumcenter(P1,P2,P3):
    # returns center (cx,cy) and radius r of circumcircle
    # using perpendicular bisector intersection
    (x1,y1),(x2,y2),(x3,y3) = P1,P2,P3
    d = 2*(x1*(y2 - y3) + x2*(y3 - y1) + x3*(y1 - y2))
    if abs(d) < 1e-12:
        return None
    ux = ((x1**2 + y1**2)*(y2 - y3) + (x2**2 + y2**2)*(y3 - y1) + (x3**2 + y3**2)*(y1 - y2))/d
    uy = ((x1**2 + y1**2)*(x3 - x2) + (x2**2 + y2**2)*(x1 - x3) + (x3**2 + y3**2)*(x2 - x1))/d
    r = math.hypot(ux - x1, uy - y1)
    return (ux, uy, r)

def circle_intersections(c1,c2):
    # c1,c2 each (cx,cy,r)
    x0,y0,r0 = c1
    x1,y1,r1 = c2
    # distance between centers
    d = math.hypot(x1-x0, y1-y0)
    if d > r0 + r1 + 1e-12 or d < abs(r0 - r1) - 1e-12:
        return []  # no intersection
    # find point p2 which is the point where line through intersection points crosses line between centers
    a = (r0**2 - r1**2 + d**2) / (2*d)
    h_sq = r0**2 - a**2
    if h_sq < -1e-12:
        return []
    h = math.sqrt(max(h_sq,0))
    # point p2
    x2 = x0 + a*(x1 - x0)/d
    y2 = y0 + a*(y1 - y0)/d
    # intersection points
    rx = -(y1 - y0) * (h/d)

    p_int1 = (x2 + rx, y2 + ry)
    p_int2 = (x2 - rx, y2 - ry)
    return [p_int1, p_int2]

def reflect_point_across_line(P, Q1, Q2):
    # reflection of P across line through Q1,Q2
    # compute projection of P onto line
    # vector along line
    vx = Q2[0] - Q1[0]

    # We need to find T on BC satisfying that BN is tangent to circle (B,K,T). Equivalent condition:
    # power of N: NB^2 = NK * NT? Wait T is on BC line, not on line BN.
    # Actually the condition of tangency: NB^2 = NK * NL where L is other intersection of line through N with the circle. But T is intersection of circle with BC, not with line through N.
    # We'll solve by finding circle center such that B,K,N are known: find circle through B and K whose tangent at B has direction BN. Then intersect with BC.
    # Let's compute circle via solving for center O = (ox, oy) such that:
    # |O-B| = |O-K| (radius)
    # vector (O - B) is perpendicular to BN (since radius to point of tangency is perpendicular to tangent line)
    # Actually radius to B is along line O B. For BN to be tangent at B, BN must be perpendicular to O B.
    # So O B dot BN = 0.
    # Solve for O.
    B_pt = B
    K_pt = K
    BN_vec = (N[0] - B_pt[0], N[1] - B_pt[1])
    # Unknown O = (ox, oy). Conditions:
    # (ox - Bx)^2 + (oy - By)^2 = (ox - Kx)^2 + (oy - Ky)^2
    # dot((ox - Bx, oy - By), BN_vec) = 0
    Bx, By = B_pt
    Kx, Ky = K_pt
    # first condition simplifies to:
    # (ox^2 - 2 ox Bx + Bx^2 + oy^2 - 2 oy By + By^2) = (ox^2 - 2 ox Kx + Kx^2 + oy^2 - 2 oy Ky + Ky^2)
    # cancel ox^2, oy^2:
    # -2 ox Bx + Bx^2 - 2 oy By + By^2 = -2 ox Kx + Kx^2 - 2 oy Ky + Ky^2
    # rearrange: -2 ox Bx + 2 ox Kx -2 oy By + 2 oy Ky = Kx^2 + Ky^2 - Bx^2 - By^2
    # 2 ox (Kx - Bx) + 2 oy (Ky - By) = Kx^2 + Ky^2 - Bx^2 - By^2
    # divide by 2
    # ox (Kx - Bx) + oy (Ky - By) = (Kx^2 + Ky^2 - Bx^2 - By^2)/2
    rhs = (Kx**2 + Ky**2 - Bx**2 - By**2)/2.0

    # solve linear equations for ox, oy
    A1 = Kx - Bx
    B1 = Ky - By
    C1 = rhs
    A2 = BN_vec[0]
    B2 = BN_vec[1]
    C2 = Bx*BN_vec[0] + By*BN_vec[1]

    if abs(det) < 1e-9:

    oy = (A1*C2 - A2*C1)/det
    # radius r
    r = math.hypot(ox - Bx, oy - By)
    # intersect circle with line BC (y=0) to find T (other than B)

def foot_point(P, L1, L2):
    # foot of perpendicular from point P to line through L1, L2
    x0,y0 = P
    x1,y1 = L1
    x2,y2 = L2
    # line direction
    dx = x2 - x1
    dy = y2 - y1
    # projection formula
    t = ((x0 - x1)*dx + (y0 - y1)*dy) / (dx*dx + dy*dy)
    proj = (x1 + t*dx, y1 + t*dy)
    return proj

def circle_center_radius(p1,p2,p3):
    # return center, radius of circumcircle of three points
    # using perpendicular bisectors
    (x1,y1),(x2,y2),(x3,y3) = p1,p2,p3
    # compute using determinant formula
    temp = x1*(y2 - y3) + x2*(y3 - y1) + x3*(y1 - y2)
    if abs(temp) < 1e-12:
        raise ValueError("collinear")
    A = x1**2 + y1**2
    B = x2**2 + y2**2
    Cc = x3**2 + y3**2
    D = 2*temp
    Ux = (A*(y2 - y3) + B*(y3 - y1) + Cc*(y1 - y2)) / D
    Uy = (A*(x3 - x2) + B*(x1 - x3) + Cc*(x2 - x1)) / D
    center = (Ux, Uy)
    radius = math.dist(center, p1)
    return center, radius

def reflect_point_across_line(P, L1, L2):
    # reflect point P across line L1-L2
    proj = foot_point(P, L1, L2)
    # reflected point = 2*proj - P
    return (2*proj[0] - P[0], 2*proj[1] - P[1])

def circle_through_BK_tangent_to_BN(B, K, N):
    # circle passes through B and K and tangent to line BN at B.
    # This circle is defined by its center lying on line perpendicular to BN through B
    # Let direction BN = N-B
    # The center O_c is at B + t * perp(BN) where perp is unit vector rotated 90 degrees.
    # The radius = distance O_c to B = distance O_c to K.
    # Solve for t.
    bx,by = B
    kx,ky = K
    nx,ny = N
    # vector BN
    vx = nx - bx
    vy = ny - by
    # perpendicular vector (rotate by 90 deg). Use ( -vy, vx )
    pvx = -vy
    pvy = vx
    # unit? we can keep scaling variable t
    # center = B + t * (pvx, pvy)
    # radius squared = (t*pvx)^2 + (t*pvy)^2 = t^2 * (pvx^2 + pvy^2)
    # also equal to distance from center to K squared:
    # (bx + t*pvx - kx)^2 + (by + t*pvy - ky)^2 = t^2*(pvx^2 + pvy^2)
    # Solve for t
    A = pvx**2 + pvy**2
    # Expand equation: (bx + t*pvx - kx)^2 + (by + t*pvy - ky)^2 - t^2*A = 0
    # Compute coefficients for t
    # Let dx = bx - kx, dy = by - ky
    dx = bx - kx
    dy = by - ky
    # Expand: (dx + t*pvx)^2 + (dy + t*pvy)^2 - t^2*A = 0
    # = dx^2 + 2dx*pvx*t + pvx^2*t^2 + dy^2 + 2dy*pvy*t + pvy^2*t^2 - A*t^2
    # = (dx^2 + dy^2) + 2t(dx*pvx + dy*pvy) + t^2*(pvx^2 + pvy^2 - A)
    # but A = pvx^2 + pvy^2, so coefficient of t^2 cancels! => we get linear equation
    # Indeed simplifies to (dx^2+dy^2) + 2t(dx*pvx + dy*pvy) = 0
    # Solve for t:
    num = -(dx**2 + dy**2)
    den = 2*(dx*pvx + dy*pvy)
    if abs(den) < 1e-12:
        return None
    t = num/den
    # center coordinates
    Ocx = bx + t*pvx
    Ocy = by + t*pvy
    # radius = distance to B
    r = math.hypot(Ocx - bx, Ocy - by)
    # Return center and radius
    return (Ocx, Ocy), r

def intersect_circle_line(circle_center, radius, point1, point2):
    # intersection of circle with line defined by point1-point2 (returns two points)
    # Return points on line.
    (cx,cy) = circle_center
    (x1,y1) = point1
    (x2,y2) = point2
    # param t such that P = point1 + t*(point2-point1)
    dx = x2 - x1
    dy = y2 - y1
    # solve (x1+dx*t - cx)^2 + (y1+dy*t - cy)^2 = r^2
    a = dx*dx + dy*dy
    b = 2*(dx*(x1-cx) + dy*(y1-cy))
    c = (x1-cx)**2 + (y1-cy)**2 - radius**2
    disc = b*b - 4*a*c
    if disc < 0:
        return []
    sqrt_disc = math.sqrt(disc)
    t1 = (-b + sqrt_disc) / (2*a)
    t2 = (-b - sqrt_disc) / (2*a)
    p1 = (x1 + dx*t1, y1 + dy*t1)
    p2 = (x1 + dx*t2, y1 + dy*t2)
    return [p1, p2]

def M_mod_k(k):
    # compute 3^K mod k, where K is huge factorial
    # if k has factor 3, then M divisible by 3^K >= k? Actually M = 3^K.
    # If k's prime factorization includes 3^t, then M mod 3^t =0.
    # So M mod k = 0 mod 3^t part, and 1 mod rest co-prime to 3.
    # Let's compute using Chinese remainder.
    import math
    # decompose k = 3^t * m, with gcd(m,3)=1
    t=0
    while k%3==0:
        t+=1
        k//=3
    m = k
    # compute residue modulo 3^t: 0
    # compute modulo m: 1 (since 3^K ≡1 mod m)
    # combine via CRT
    # we need to find x such that x≡0 mod 3^t and x≡1 mod m.
    # if t=0, residue is 1.
    if t==0:
        return 1 % m
    else:
        # modulus = 3^t * m
        mod = (3**t)*m
        # solution: x = 0 + 3^t * inv(3^t, m) * (1 - 0) mod mod
        # need inverse of 3^t modulo m
        inv = pow(3**t, -1, m)  # Python 3.8 supports
        x = (3**t * inv) % mod
        return x

def f_bruteforce(n):
    best = None
    for a in range(1, n//3+1):
        for b in range(a+1, (n-a)//2+1):
            c = n - a - b
            if c <= b: continue
            l = math.lcm(a,b,c)
            if best is None or l < best:
                best = l
    return best

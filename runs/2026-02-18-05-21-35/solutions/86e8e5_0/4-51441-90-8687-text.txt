def compute_primitives(limit_s):
    # returns list of (s, ratio) for primitive sums <= limit_s sorted by ratio
    best_ratio = {}
    for s in range(6, limit_s+1):
        # compute minimal lcm for sum s
        min_lcm = None
        for a in range(1, s-2):
            for b in range(a+1, s-a-1):
                c = s - a - b
                if c <= b:
                    continue
                l = a // math.gcd(a,b) * b
                l = l // math.gcd(l,c) * c
                if min_lcm is None or l < min_lcm:
                    min_lcm = l
        if min_lcm is not None:
            best_ratio[s] = Fraction(min_lcm, s)
    # now find primitives: those not multiples of any earlier s
    sorted_items = sorted(best_ratio.items(), key=lambda x: x[1])
    primitives = []
    covered = set()
    for s, r in sorted_items:
        # check if any divisor already in covered
        has_factor = False
        for s0 in covered:
            if s % s0 == 0:
                has_factor = True
                break
        if not has_factor:
            primitives.append((s, r))
            covered.add(s)
    # sort primitives by ratio again (they already are)
    return primitives

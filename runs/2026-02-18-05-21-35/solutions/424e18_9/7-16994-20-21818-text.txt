def generate_pairings(lst):
    """Return list of list of pairs for even-sized list lst (list of elements). Pairings order not important."""
    if not lst:
        return [[]]
    # pick first element
    first = lst[0]
    pairings = []
    for i in range(1, len(lst)):
        second = lst[i]
        rest = lst[1:i] + lst[i+1:]
        for sub in generate_pairings(rest):
            pairings.append([(first, second)] + sub)
    return pairings

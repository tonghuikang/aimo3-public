def possible_winner_sets(n):
    participants = list(range(1,n+1))  # 1 fastest
    # generate all perfect matchings (pairings)
    # Use algorithm: generate all ways to partition list into pairs, order not important.
    # We'll just brute force for n small.
    matchings = []
    def gen(p):
        if not p:
            yield []
            return
        a = p[0]
        for i in range(1,len(p)):
            b = p[i]
            rest = p[1:i] + p[i+1:]
            for tail in gen(rest):
                yield [(a,b)] + tail
    for pairs in gen(participants):
        # order of pairs irrelevant
        # compute winners
        winners = tuple(sorted([min(a,b) for a,b in pairs]))

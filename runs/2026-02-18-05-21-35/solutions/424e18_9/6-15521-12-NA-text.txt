                    sc = cur_scores[p]
                    score_to_players.setdefault(sc, []).append(p)
                new_groups = tuple(tuple(sorted(v)) for v in score_to_players.values())
                return rec(round_idx+1, new_groups, cur_scores)
            group = groups[g_idx]
            # generate all pairings (perfect matchings) of this group.
            pairings = all_pairings(list(group))
            # For each pairing, determine winners and update scores.
            res_set = set()
            for pairing in pairings:
                # compute new scores copy
                new_scores = list(cur_scores)
                for a,b in pairing:
                    # a,b are player indices; faster is min (since 0 fastest)
                    winner = a if a < b else b
                    loser = b if a < b else a
                    new_scores[winner] += point
                    # loser unchanged
                # Recurse to process next group
                sub_set = process_group(g_idx+1, tuple(new_scores))
                res_set.update(sub_set)
            return res_set
        
        # start with current scores tuple
        return process_group(0, scores)
    
    # initial scores all zero
    init_scores = tuple([0]*num_players)
    # initial groups: one group of all players
    init_groups = (players,)
    return rec(0, init_groups, init_scores)

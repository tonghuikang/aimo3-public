def pairings(lst):
    # generate all perfect matchings (pairings) of list lst (even length)
    if not lst:
        yield []
        return
    first = lst[0]
    for i in range(1, len(lst)):
        second = lst[i]
        rest = lst[1:i] + lst[i+1:]
        for rest_pairs in pairings(rest):
            yield [(first, second)] + rest_pairs

def simulate(pair_seq, speeds):
    # pair_seq: list of pairings for each round (list of list of tuples)
    # speeds: dict runner->rank (higher speed bigger rank)
    scores = {r: 0 for r in speeds}
    n_rounds = len(pair_seq)
    for i, pairs in enumerate(pair_seq):
        point = 2**(n_rounds - i - 1)  # round i (0-index) points
        # check that each pair have same score at start
        for a,b in pairs:
            assert scores[a]==scores[b]
        # each pair, winner gets points
        for a,b in pairs:
            winner = a if speeds[a] > speeds[b] else b
            scores[winner] += point
    return scores

def all_final_scores(num_players, speed_order):

def incircle_center(A,B,C):
    # A,B,C as np arrays
    a = np.linalg.norm(B-C)
    b = np.linalg.norm(C-A)
    c = np.linalg.norm(A-B)
    I = (a*A + b*B + c*C)/(a+b+c)
    return I

def foot_point(P, Q, R):
    # foot from P onto line QR
    # Q,R are points
    v = R-Q
    t = np.dot(P-Q, v) / np.dot(v,v)
    foot = Q + t*v
    return foot

def circumcenter(A,B,C):
    # compute circumcenter of triangle, using linear algebra
    # Using perpendicular bisectors intersection
    # Solve for center O: |O-A|^2 = |O-B|^2 = |O-C|^2
    # Equivalent to solving linear equations.
    # We'll set up using matrix
    # 2*(B-A)·O = |B|^2 - |A|^2 etc.
    A2 = np.dot(A,A)
    B2 = np.dot(B,B)
    C2 = np.dot(C,C)
    M = np.array([2*(B-A), 2*(C-A)])
    rhs = np.array([B2 - A2, C2 - A2])
    # solve M * O = rhs
    O = np.linalg.solve(M, rhs)
    return O

def circumradius(O, A):
    return np.linalg.norm(O - A)

def circle_intersections(O1, r1, O2, r2):
    # return list of intersection points (0,1,2)
    d = np.linalg.norm(O2 - O1)
    if d > r1 + r2 + 1e-9 or d < abs(r1 - r2) - 1e-9 or d == 0:
        return [] # no intersection or infinite
    # Compute a = (r1^2 - r2^2 + d^2) / (2d)
    a = (r1**2 - r2**2 + d**2) / (2*d)
    h_sq = r1**2 - a**2
    if h_sq < -1e-9:
        return []
    h = math.sqrt(max(h_sq, 0.0))
    # point P2 = O1 + a*(O2 - O1)/d
    P2 = O1 + a * (O2 - O1) / d
    # intersection points
    offset = h * np.array([-(O2[1]-O1[1]), (O2[0]-O1[0])]) / d
    inter1 = P2 + offset
    inter2 = P2 - offset
    if np.allclose(inter1, inter2, atol=1e-9):
        return [inter1]
    else:
        return [inter1, inter2]

def reflect_point(P, A, B):
    # reflect point P across line AB
    # using projection and symmetry
    # line defined by points A, B
    AB = B - A
    # param t for projection of (P-A) onto AB
    t = np.dot(P - A, AB) / np.dot(AB, AB)
    proj = A + t * AB
    # reflection: P' = 2*proj - P
    return 2*proj - P

    # Use linear algebra method

def circle_intersections(O1, r1, O2, r2):
    # return intersection points as list of numpy arrays (0,1,2)
    d = np.linalg.norm(O2 - O1)
    if d > r1 + r2 + 1e-9 or d < abs(r1 - r2) - 1e-9:
        return []
    if d < 1e-12:  # coincident
        return []
    a = (r1**2 - r2**2 + d**2) / (2*d)

    if h_sq < -1e-9:
        return []

    inter2 = P2 - offset

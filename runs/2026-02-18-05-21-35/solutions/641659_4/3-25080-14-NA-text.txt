        uy = ((np.linalg.norm(P1)**2)*(P3[0]-P2[0]) + (np.linalg.norm(P2)**2)*(P1[0]-P3[0]) + (np.linalg.norm(P3)**2)*(P2[0]-P1[0])) / d
        return np.array([ux, uy])

    if O is None:

    # solve for intersection points of two circles: O,R and O2,R2
    def circle_intersections(O1, r1, O2, r2):
        # returns list of intersection points (0,1,2)
        d = np.linalg.norm(O2 - O1)

        inter1 = P2 + offset
        inter2 = P2 - offset
        return [inter1, inter2]

        if np.linalg.norm(pt - A) > 1e-6:

            break

        return None
    # Compute K' reflection across line EF
    def reflect_point_across_line(P, Q, R):

        # compute projection onto line
        v = R - Q
        v_norm2 = np.dot(v, v)
        if v_norm2 == 0:
            return P
        # projection
        proj = Q + np.dot(P - Q, v) / v_norm2 * v

        return 2*proj - P

    NB = np.linalg.norm(N - B)
    NE = np.linalg.norm(N - E)
    ratio = (CT * NB) / (BT * NE)

def find_max_ratio_for_n(n):
    # s must be > a. Let's search for s in some range, perhaps [a+1, a+1000] maybe.
    a = fibonacci(n+2)
    # We'll sample s values and use root solving for cyclic condition.
    # Actually we need s such that cyclic condition holds.
    # We'll treat s as variable and find root of function f(s) = signed distance difference.
    # define function that returns residual of cyclic condition (distance from K' to circumcircle of B,N,K)
    def residual(s):
        # compute geometry and return difference (distance - radius)
        # compute as before but return signed difference
        a = fibonacci(n+2)
        b = s - fibonacci(n)
        c = s - fibonacci(n+1)
        if b <= 0 or c <= 0:
            return None
        # Compute positions
        B = np.array([0.0, 0.0])
        C = np.array([a, 0.0])
        xA = (c**2 - b**2 + a**2) / (2*a)
        y_sq = c**2 - xA**2
        if y_sq <= 0:
            return None
        yA = math.sqrt(y_sq)
        A = np.array([xA, yA])
        BF = fibonacci(n)
        CE = fibonacci(n+1)
        t = BF / c
        F = B + t*(A-B)
        u = CE / b
        E = C + u*(A-C)
        D = np.array([fibonacci(n), 0.0])
        N = foot_of_perpendicular(D, E, F)
        # compute circles
        O = circumcenter(A,B,C)
        O2 = circumcenter(A,F,E)
        if O is None or O2 is None:
            return None
        r1 = np.linalg.norm(A-O)
        r2 = np.linalg.norm(A-O2)
        inters = circle_intersections(O, r1, O2, r2)
        # find K not A
        K = None
        for pt in inters:
            if np.linalg.norm(pt - A) > 1e-6:
                K = pt
                break
        if K is None:
            return None
        K_prime = reflect_point_across_line(K, E, F)
        # circumcenter of B,N,K
        O3 = circumcenter(B, N, K)
        if O3 is None:
            return None
        r3 = np.linalg.norm(B - O3)
        diff = np.linalg.norm(K_prime - O3) - r3
        return diff
    # Use root-finding scanning range for s. We'll use sign changes.
    # Let's sample s from a+0.1 to a+1000
    s_vals = np.linspace(fibonacci(n+2)+0.1, fibonacci(n+2)+2000, 2000)
    vals = []
    for s in s_vals:
        res = residual(s)
        if res is None:
            vals.append(np.nan)
        else:
            vals.append(res)
    # Find intervals where sign changes
    intervals = []
    for i in range(len(s_vals)-1):
        if np.isnan(vals[i]) or np.isnan(vals[i+1]): continue
        if vals[i]==0: intervals.append((s_vals[i], s_vals[i]))
        if vals[i]*vals[i+1] < 0:
            intervals.append((s_vals[i], s_vals[i+1]))
    # For each interval, find root via bisect
    roots = []
    for a_int, b_int in intervals:
        lo, hi = a_int, b_int
        for _ in range(60):
            mid = (lo+hi)/2
            fmid = residual(mid)
            flo = residual(lo)
            if fmid is None or flo is None:
                break
            if flo*fmid <= 0:
                hi = mid
            else:
                lo = mid
        roots.append((lo+hi)/2)
    # For each root, compute ratio
    ratios = []
    for s_root in roots:
        r = compute_ratio(n, s_root)
        if r is not None:
            ratios.append(r)
    if not ratios:
        return None
    return max(ratios), ratios, roots

def incircle_touch_points(A,B,C):
    # Return D (BC), E (CA), F (AB)
    a = abs(B-C)
    b = abs(C-A)
    c = abs(A-B)
    s = (a+b+c)/2
    # D on BC
    D = ((s-c)*B + (s-b)*C)/a
    # E on CA
    E = ((s-a)*C + (s-c)*A)/b
    # F on AB
    F = ((s-b)*A + (s-a)*B)/c
    return D,E,F

def circumcenter(P,Q,R):
    # compute circumcenter using perpendicular bisector method
    # returns complex point
    # formula using complex numbers: see known formula
    # We'll compute using linear algebra
    A = np.array([[2*(Q.real-P.real), 2*(Q.imag-P.imag)],
                  [2*(R.real-P.real), 2*(R.imag-P.imag)]])
    B = np.array([abs(Q)**2 - abs(P)**2, abs(R)**2 - abs(P)**2])
    sol = np.linalg.solve(A,B)
    return complex(sol[0], sol[1])

def circle_center_radius(P,Q,R):
    O = circumcenter(P,Q,R)
    r = abs(P - O)
    return O,r

def intersect_circles(O1,r1,O2,r2):
    # returns intersection points of two circles (real)
    # may return 0,1,2 solutions
    d = abs(O2 - O1)
    if d > r1+r2+1e-9 or d < abs(r1-r2)-1e-9:
        return [] # no intersection
    # compute
    # ex = (O2-O1)/d
    ex = (O2 - O1) / d
    # x = (r1^2 - r2^2 + d^2) / (2d)
    x = (r1**2 - r2**2 + d**2) / (2*d)
    # y = sqrt(r1^2 - x^2)
    y_sq = r1**2 - x**2
    if y_sq < -1e-9:
        return []
    y = math.sqrt(max(y_sq,0.0))
    # point p = O1 + x*ex +/- y*ey where ey is perpendicular
    ey = complex(-ex.imag, ex.real)  # rotate 90 degrees
    p1 = O1 + x*ex + y*ey
    p2 = O1 + x*ex - y*ey
    return [p1,p2]

def second_intersection_of_circles(A, F, E, B, C):
    # returns K: second intersection of circumcircle of A,F,E and circumcircle of ABC (Ω)
    O1,r1 = circle_center_radius(A,F,E)
    O2,r2 = circle_center_radius(A,B,C)  # circumcircle
    pts = intersect_circles(O1,r1,O2,r2)
    # choose point not equal to A (within tolerance)
    # both circles intersect at A and another point
    K = None
    for p in pts:
        if abs(p - A) > 1e-6:
            K = p
    return K

def reflect_across_line(P, L1, L2):
    # reflect point P across line through L1,L2
    # formula using projection
    # line vector
    v = L2 - L1
    # projection of (P - L1) onto v
    proj = L1 + ( (P - L1).real * v.real + (P - L1).imag * v.imag ) / (abs(v)**2) * v
    # reflection: Q = 2*proj - P
    Q = 2*proj - P
    return Q

def foot_of_perpendicular(P, L1, L2):
    v = L2 - L1
    # projection of (P - L1) onto v
    proj = L1 + ((P - L1).real * v.real + (P - L1).imag * v.imag) / (abs(v)**2) * v
    return proj

def triangle_from_sides(a,b,c):
    # Place B at (0,0), C at (a,0). Determine A using law of cosines.
    # side lengths: a = BC, b = CA, c = AB.
    # coordinates: B (0,0), C (a,0). Find coordinates of A = (x,y) with distance to B = c, to C = b.
    # Solve:
    # x^2 + y^2 = c^2
    # (x - a)^2 + y^2 = b^2
    # subtract: (x - a)^2 - x^2 = b^2 - c^2 => x^2 - 2ax + a^2 - x^2 = b^2 - c^2 => -2a x + a^2 = b^2 - c^2

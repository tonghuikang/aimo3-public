    geom = compute_geometry(d1,d2,r)
    # Compute T: intersection of circle tangent to BN at B and through K with line BC (x-axis)
    B = geom['B']
    K = geom['K']
    # line BN direction
    N = geom['N']
    # line BN direction vector
    # Center O' lies at intersection of perpendicular bisector of BK and line through B perpendicular to BN.
    # Compute midpoint of BK
    M = sp.Point((B.x + K.x)/2, (B.y + K.y)/2)
    # Perpendicular bisector: line through M with direction perpendicular to BK vector.
    v_BK = sp.Matrix([K.x - B.x, K.y - B.y])
    # direction perpendicular to BK: rotate v_BK by 90 degrees: (-v_BK[1], v_BK[0])
    perp_dir = sp.Matrix([-v_BK[1], v_BK[0]])
    # Line eq for perpendicular bisector:
    line_perp_bis = sp.Line(M, M + sp.Point(perp_dir[0], perp_dir[1]))
    # line through B perpendicular to BN: BN direction:
    v_BN = sp.Matrix([N.x - B.x, N.y - B.y])
    perp_BN_dir = sp.Matrix([-v_BN[1], v_BN[0]])
    line_center = sp.Line(B, B + sp.Point(perp_BN_dir[0], perp_BN_dir[1]))
    O = line_perp_bis.intersection(line_center)[0]
    # radius = distance O to B
    R = O.distance(B)
    # Intersection with BC (x-axis y=0) besides B.
    # Circle eq: (x - O.x)^2 + (y - O.y)^2 = R^2.
    # Plug y=0, solve for x.
    x = sp.symbols('x')
    eq = (x - O.x)**2 + (0 - O.y)**2 - R**2
    sol_x = sp.solve(eq, x)
    # Choose solution not equal to B.x (0)
    T_x = None
    for s in sol_x:
        if not sp.simplify(s - B.x) == 0:
            T_x = s
            break
    if T_x is None:
        raise ValueError("No T found")
    T = sp.Point(T_x, 0)
    # Now compute lengths:
    CT = sp.sqrt((geom['C'].x - T.x)**2 + (geom['C'].y - T.y)**2)
    BT = sp.sqrt((B.x - T.x)**2 + (B.y - T.y)**2)
    NB = B.distance(geom['N'])
    NE = geom['E'].distance(geom['N'])
    ratio = (CT * NB) / (BT * NE)
    ratio_simplified = sp.nsimplify(ratio, [sp.sqrt(5)])
    return float(ratio.evalf()), ratio_simplified, ratio

def circumcenter(P, Q, R):
    # return center of circle through points P,Q,R
    # using sympy geometry
    circ = sp.Circle(P, Q, R)
    return circ.center, circ.radius

def second_intersection(P1, P2, P3, Q1, Q2, Q3):
    # compute second intersection of circumcircles of (P1,P2,P3) and (Q1,Q2,Q3) aside from common point if any (assumed to be P1=Q1 maybe)
    # assume P1 is common point.
    O1, R1 = circumcenter(P1, P2, P3)
    O2, R2 = circumcenter(Q1, Q2, Q3)
    # line for radical axis: passes through P1 and is perpendicular to O1O2
    dir_vec = sp.Matrix([O2.x - O1.x, O2.y - O1.y])
    # direction perpendicular to dir_vec
    perp = sp.Matrix([-dir_vec[1], dir_vec[0]])
    # param line: P1 + t * perp
    t = sp.symbols('t', real=True)
    X = sp.Point(P1.x + t*perp[0], P1.y + t*perp[1])
    # Solve for intersection with circle O1 radius R1
    eq = (X.x - O1.x)**2 + (X.y - O1.y)**2 - R1**2
    sol = sp.solve(eq, t)
    # sol will have two solutions: t=0 for P1, and t_other for second point
    sol_t = [s for s in sol if not sp.simplify(s) == 0][0]
    K = sp.Point(P1.x + sol_t*perp[0], P1.y + sol_t*perp[1])
    return sp.simplify(K)

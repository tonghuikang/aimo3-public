def triangle_from_side_lengths(b,c,a): # input side lengths AB=c, AC=b, BC=a. returns coordinates A,B,C as sympy points

    # Compute x-coordinate of A using law of cosines
    # x_A = (c**2 - b**2 + a**2) / (2*a)

def incircle_center_and_radius(A, B, C):

    # Coordinates: (ax_A + bx_B + cx_C)/perimeter?

def foot_of_point_to_line(P, Q, R):
    # foot of P onto line QR
    # line QR: param eq Q + t*(R-Q)
    # foot = projection of vector (P-Q) onto direction (R-Q)
    v = sp.Point(R.x - Q.x, R.y - Q.y)
    w = sp.Point(P.x - Q.x, P.y - Q.y)
    t = (w.dot(v)) / (v.dot(v))
    foot = sp.Point(Q.x + t*v.x, Q.y + t*v.y)
    return foot

def is_cyclic(P, Q, R, S):
    # check if four points are concyclic (co-circle)
    # condition: oriented area? Use determinant condition.
    # For points (x_i, y_i), they are concyclic iff determinant of matrix:
    # | x_i^2 + y_i^2, x_i, y_i, 1 | = 0 (for all 4 points)
    # Actually if determinant for 4 points =0, they lie on a circle.

                   [R.x**2 + R.y**2, R.x, R.y, 1],
                   [S.x**2 + S.y**2, S.x, S.y, 1]])

def circumcenter(P1, P2, P3):
    # Compute circumcenter of triangle P1,P2,P3 using linear algebra
    # P1, P2, P3 are numpy arrays shape (2,)
    # Solve for center O such that |O-P1|^2 = |O-P2|^2 = |O-P3|^2.
    # Using perpendicular bisector intersection.
    # Let midpoints of P1-P2 and P1-P3: M12, M13
    # Direction vectors: d12 = P2 - P1, d13 = P3 - P1.
    # Solve O = M12 + t * perp(d12) = M13 + s * perp(d13)
    # Use linear equations.
    # Another method: solve linear system A O = b.
    # Derive from equations: 2*(P2-P1)·O = |P2|^2 - |P1|^2, similarly for P3-P1.
    A = np.vstack([2*(P2 - P1), 2*(P3 - P1)])
    b = np.array([np.dot(P2, P2) - np.dot(P1, P1), np.dot(P3, P3) - np.dot(P1, P1)])
    # Solve linear system
    # Might be singular if points collinear (shouldn't happen)
    try:
        O = np.linalg.solve(A, b)
    except np.linalg.LinAlgError:
        return None
    return O

    # find a = (R1^2 - R2^2 + d^2) / (2d)
    a = (R1**2 - R2**2 + d**2) / (2*d) if d != 0 else 0
    # height h = sqrt(R1^2 - a^2)
    # handle potential floating errors
    h_sq = R1**2 - a**2
    if h_sq < -1e-12:
        return []
    h_sq = max(h_sq, 0.0)
    h = math.sqrt(h_sq)
    # point P2 = O1 + a*(O2-O1)/d
    P2 = O1 + a * (O2 - O1) / d if d != 0 else O1
    # intersection points:
    # offset vector perpendicular to O2-O1
    # perp = (-(O2-O1)[1], (O2-O1)[0])
    vec = O2 - O1
    perp = np.array([-vec[1], vec[0]]) / d if d != 0 else np.array([0,0])
    inter1 = P2 + h * perp
    inter2 = P2 - h * perp
    if h < 1e-12:
        return [inter1]  # one point
    else:
        return [inter1, inter2]

def project_point_onto_line(P, L1, L2):
    # Projection of P onto line L1-L2
    v = L2 - L1
    if np.allclose(v, 0):
        return L1.copy()
    t = np.dot(P - L1, v) / np.dot(v, v)
    proj = L1 + t * v
    return proj

def reflect_point_across_line(P, L1, L2):
    proj = project_point_onto_line(P, L1, L2)
    return 2*proj - P

def compute_cyclic_f(c_val, n):
    # c_val = AB length
    # n >= 2 for our definitions.
    # Returns difference radius-dist difference; zero if cyclic.
    # Compute triangle geometry numerically
    # Fibonacci numbers
    F = fibonacci_upto(n+2)
    x = F[n]   # BD
    y = F[n+1] # CD
    a = x + y   # BC length
    if c_val <= x:
        return None
    # side lengths:
    b_val = c_val + F[n-1]  # CA length
    # Validate triangle inequality:
    if not (c_val + b_val > a and a + c_val > b_val and a + b_val > c_val):
        return None
    # coordinates:
    B = np.array([0.0, 0.0])
    C = np.array([float(a), 0.0])
    # compute coordinates of A:
    xA = (c_val**2 - b_val**2 + a**2) / (2*a)
    tmp = c_val**2 - xA**2
    if tmp <= 0:
        return None
    yA = math.sqrt(tmp)
    A = np.array([float(xA), float(yA)])
    # compute side lengths again (maybe check)
    # incenter I: formula weighted by side lengths
    a_len = a
    b_len = b_val
    c_len = c_val
    p_len = a_len + b_len + c_len
    I = (a_len * A + b_len * B + c_len * C) / p_len
    # incircle radius
    # area via cross product: 0.5*| (B-A) x (C-A) |
    area = 0.5 * abs(np.cross(B - A, C - A))
    s = p_len/2
    r = area / s
    # contact points: foot from I to each side
    # D: foot onto BC (line B-C)
    D = project_point_onto_line(I, B, C)
    # E: foot onto CA (line C-A)
    E = project_point_onto_line(I, C, A)
    # F: foot onto AB (line A-B)
    Fp = project_point_onto_line(I, A, B)
    # circumcircle of ABC
    O_ABC = circumcenter(A, B, C)
    if O_ABC is None:
        return None
    R_ABC = np.linalg.norm(A - O_ABC)
    # circumcircle of AEF
    O_AEF = circumcenter(A, E, Fp)
    if O_AEF is None:
        return None
    R_AEF = np.linalg.norm(A - O_AEF)
    # find intersection of circles: two possible points: A and K
    inter = circle_intersections(O_ABC, R_ABC, O_AEF, R_AEF)
    # find K as intersection not equal to A (within tolerance)
    K = None
    for pt in inter:
        if np.linalg.norm(pt - A) > 1e-6:
            K = pt
            break
    if K is None:
        # maybe degenerate? return None
        return None
    # compute K' reflection across EF
    Kp = reflect_point_across_line(K, E, Fp)
    # N foot from D onto EF
    N = project_point_onto_line(D, E, Fp)
    # compute circumcircle of K,N,B
    O_KNB = circumcenter(K, N, B)
    if O_KNB is None:
        return None
    R_KNB = np.linalg.norm(K - O_KNB)
    # check distance of K' to that center
    diff = np.linalg.norm(Kp - O_KNB) - R_KNB
    return diff

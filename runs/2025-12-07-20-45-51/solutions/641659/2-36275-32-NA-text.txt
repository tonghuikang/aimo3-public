def geometry_data(n, r):

def circumcenter(P1,P2,P3):
    # return circumcenter of triangle P1,P2,P3 using linear equations
    # Solve:
    # 2*(x2 - x1)*x + 2*(y2 - y1)*y = x2^2+y2^2 - x1^2 - y1^2
    # similarly for (P3,P1)
    x1,y1 = P1
    x2,y2 = P2
    x3,y3 = P3
    A = 2*(x2 - x1)
    B = 2*(y2 - y1)
    C = x2*x2 + y2*y2 - x1*x1 - y1*y1
    D = 2*(x3 - x1)
    E = 2*(y3 - y1)
    F = x3*x3 + y3*y3 - x1*x1 - y1*y1
    # Solve linear system A*x + B*y = C, D*x + E*y = F
    det = A*E - B*D
    if abs(det) < 1e-12:
        return None
    x = (C*E - B*F) / det
    y = (A*F - C*D) / det
    return np.array([x,y])

def circle_intersections(center1,R1,center2,R2):
    # returns intersection points of two circles (if any)
    # compute distance between centers
    d = np.linalg.norm(center2 - center1)
    if d > R1+R2+1e-12 or d < abs(R1-R2)-1e-12:
        return [] # no intersection
    # find a = (R1^2 - R2^2 + d^2) / (2d)
    a = (R1*R1 - R2*R2 + d*d) / (2*d)
    # height h = sqrt(R1^2 - a^2)
    h_sq = R1*R1 - a*a
    if h_sq < -1e-12:
        return []
    h = sqrt(max(0.0, h_sq))
    # point P2 = center1 + a*(center2-center1)/d
    P2 = center1 + a*(center2 - center1)/d
    # intersection points
    # offset vector ( -dy, dx ) normalized * h/d
    dx, dy = (center2 - center1)/d
    # perpendicular
    perp = np.array([-dy, dx])
    inter1 = P2 + h*perp
    inter2 = P2 - h*perp
    return [inter1, inter2]

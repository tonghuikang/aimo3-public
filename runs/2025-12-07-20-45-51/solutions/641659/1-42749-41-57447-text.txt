def fibonacci(n):
    # returns Fibonacci number F_n where F_0=0, F_1=1
    # Use fast algorithm
    if n < 0:
        raise ValueError
    a,b = 0,1
    for i in range(n):
        a,b = b,a+b
    return a

def triangle_data(n, x):

    # B = (0,0), C = (a,0), D = (F_n,0)

def second_intersection_of_two_circles(c1_center, r1, c2_center, r2, P):
    # Compute other intersection point of two circles (c1, c2) besides point P (which lies on both)
    # Solve intersection.
    # We'll compute intersections and return the one not equal to P (within tolerance)
    # Use sympy solving.
    x, y = sp.symbols('x y')
    eqs = [
        (x - c1_center.x)**2 + (y - c1_center.y)**2 - r1**2,
        (x - c2_center.x)**2 + (y - c2_center.y)**2 - r2**2
    ]
    sols = sp.solve(eqs, (x,y), dict=True)
    # sol may contain two solutions
    pts = [sp.Point(sol[x], sol[y]) for sol in sols]
    # Filter out P (approx)
    for pt in pts:
        if pt.distance(P).evalf() > 1e-6:
            return pt
    # if not found return None
    return None

def compute_K(n, x):

    pts = triangle_data(n, x)

def circumcenter(P1, P2, P3):
    # P1,P2,P3 are tuples (x,y)
    x1,y1 = P1
    x2,y2 = P2
    x3,y3 = P3
    # Solve linear equations:
    # (x1 - x2)x + (y1 - y2)y = (x1^2 - x2^2 + y1^2 - y2^2)/2
    # (x1 - x3)x + (y1 - y3)y = (x1^2 - x3^2 + y1^2 - y3^2)/2
    A = [[x1 - x2, y1 - y2],
         [x1 - x3, y1 - y3]]
    Bv = [ (x1*x1 - x2*x2 + y1*y1 - y2*y2)/2.0,
           (x1*x1 - x3*x3 + y1*y1 - y3*y3)/2.0 ]
    # Solve linear system A * [x,y] = Bv
    detA = A[0][0]*A[1][1] - A[0][1]*A[1][0]
    if abs(detA) < 1e-12:
        # points collinear? return None
        return None
    x = (Bv[0]*A[1][1] - Bv[1]*A[0][1]) / detA
    y = (A[0][0]*Bv[1] - A[1][0]*Bv[0]) / detA
    return (x, y)

def circle_intersections(O1, R1, O2, R2):
    # Return intersection points of two circles, as list of tuples. May be empty or one or two.
    x1,y1 = O1
    x2,y2 = O2
    # distance between centers
    d = math.hypot(x2-x1, y2-y1)
    if d > R1+R2+1e-12 or d < abs(R1-R2)-1e-12:
        # no intersections
        return []
    # Compute a = (R1^2 - R2^2 + d^2)/(2d)
    a = (R1*R1 - R2*R2 + d*d) / (2*d)
    # height h = sqrt(R1^2 - a^2)
    h_sq = R1*R1 - a*a
    if h_sq < -1e-12:
        return []
    h = math.sqrt(max(0, h_sq))
    # point P2 = O1 + a*(O2 - O1)/d
    x3 = x1 + a*(x2-x1)/d
    y3 = y1 + a*(y2-y1)/d
    # offset vectors
    rx = -(y2-y1) * (h/d)
    ry = (x2-x1) * (h/d)
    # intersection points
    p_int1 = (x3 + rx, y3 + ry)
    p_int2 = (x3 - rx, y3 - ry)
    if h < 1e-12:
        return [p_int1]
    else:
        return [p_int1, p_int2]

def line_projection_point(P, A, B):
    # project point P onto line through A and B
    # Returns projected point (x,y)
    # vector AB
    ax, ay = A
    bx, by = B
    px, py = P
    # if A==B, return A
    if abs(ax-bx) < 1e-12 and abs(ay-by) < 1e-12:
        return A
    # compute t = ((P-A)·(B-A))/|B-A|^2
    dx = bx - ax
    dy = by - ay
    t = ((px-ax)*dx + (py-ay)*dy) / (dx*dx + dy*dy)
    proj = (ax + t*dx, ay + t*dy)
    return proj

def reflect_point_across_line(P, A, B):
    # reflect point P across line AB
    proj = line_projection_point(P, A, B)
    # reflected point K' = 2*proj - P
    return (2*proj[0] - P[0], 2*proj[1] - P[1])

def circumcenter(p1, p2, p3):
    (x1,y1) = p1
    (x2,y2) = p2
    (x3,y3) = p3
    D = 2*(x1*(y2 - y3) + x2*(y3 - y1) + x3*(y1 - y2))
    if abs(D) < 1e-12:
        # points collinear
        return None
    Ux = ((x1**2 + y1**2)*(y2 - y3) + (x2**2 + y2**2)*(y3 - y1) + (x3**2 + y3**2)*(y1 - y2))/D
    Uy = ((x1**2 + y1**2)*(x3 - x2) + (x2**2 + y2**2)*(x1 - x3) + (x3**2 + y3**2)*(x2 - x1))/D
    return (Ux, Uy)

def circle_intersections(O1, R1, O2, R2):
    # return list of intersection points (0,1,2)
    x1,y1 = O1
    x2,y2 = O2
    dx = x2 - x1
    dy = y2 - y1
    d = math.hypot(dx, dy)
    # no intersection
    if d > R1 + R2 + 1e-12 or d < abs(R1 - R2) - 1e-12:
        return []
    # compute a
    a = (R1**2 - R2**2 + d**2)/(2*d) if d != 0 else 0
    # compute h
    # protect negative due to rounding
    h_sq = max(R1**2 - a**2, 0)
    h = math.sqrt(h_sq)
    # point P2 = O1 + a*(O2-O1)/d
    xm = x1 + a*dx/d if d != 0 else x1
    ym = y1 + a*dy/d if d != 0 else y1
    # intersection points:
    # offset = h * (-dy/d, dx/d)
    rx = -dy * (h/d) if d != 0 else 0
    ry = dx * (h/d) if d != 0 else 0
    p1 = (xm + rx, ym + ry)
    p2 = (xm - rx, ym - ry)
    if h < 1e-12:
        return [p1]  # one point (tangent)
    else:
        return [p1, p2]

def project_point_onto_line(P, A, B):
    # project point P onto line AB
    # vector AP = P - A, AB = B - A
    (px, py) = P
    (ax, ay) = A
    (bx, by) = B
    ABx = bx - ax
    ABy = by - ay
    AB_len2 = ABx**2 + ABy**2
    if AB_len2 == 0:
        return A
    t = ((px - ax)*ABx + (py - ay)*ABy) / AB_len2
    proj = (ax + t*ABx, ay + t*ABy)
    return proj

def reflect_point_across_line(P, A, B):
    # reflect P across line AB
    # foot of perp = projection of P onto AB
    foot = project_point_onto_line(P, A, B)
    # reflection: 2*foot - P
    rx = 2*foot[0] - P[0]
    ry = 2*foot[1] - P[1]
    return (rx, ry)

            b=m

            return m

            b=m
            fb=fm
        else:
            a=m
            fa=fm
    return (a+b)/2

    C = (a, mp.mpf(0))

    yA = 2*s*r/a

        vx, vy = V
        dx = vx-ux
        dy = vy-uy
        tpar = ((P[0]-ux)*dx + (P[1]-uy)*dy) / (dx*dx + dy*dy)
        return (ux + tpar*dx, uy + tpar*dy)

    def circle_center(p1,p2,p3):
        (x1,y1),(x2,y2),(x3,y3) = p1,p2,p3
        # solve linear equations for circumcenter
        d = 2*(x1*(y2 - y3) + x2*(y3 - y1) + x3*(y1 - y2))
        if abs(d) < 1e-30:
            raise Exception("collinear")
        ux = ((x1**2 + y1**2)*(y2 - y3) + (x2**2 + y2**2)*(y3 - y1) + (x3**2 + y3**2)*(y1 - y2)) / d
        uy = ((x1**2 + y1**2)*(x3 - x2) + (x2**2 + y2**2)*(x1 - x3) + (x3**2 + y3**2)*(x2 - x1)) / d
        R = mp.sqrt((ux - x1)**2 + (uy - y1)**2)
        return (ux, uy), R
    circ1_center, circ1_r = circle_center(A, F, E)
    circ2_center, circ2_r = circle_center(A, B, C)
    # intersection of circles
    (cx1,cy1) = circ1_center
    (cx2,cy2) = circ2_center
    dcent = mp.sqrt((cx2-cx1)**2 + (cy2-cy1)**2)
    a_len = (circ1_r**2 - circ2_r**2 + dcent**2) / (2*dcent)
    rad = circ1_r**2 - a_len**2
    if rad < 0:
        rad = mp.mpf('0')
    h = mp.sqrt(rad)
    ex = (cx2 - cx1) / dcent
    ey = (cy2 - cy1) / dcent
    px = cx1 + a_len * ex
    py = cy1 + a_len * ey
    K1 = (px + h * (-ey), py + h * ex)
    K2 = (px - h * (-ey), py - h * ex)
    # choose K not equal A
    def dist(p,q):
        return mp.sqrt((p[0]-q[0])**2 + (p[1]-q[1])**2)
    d1 = dist(K1, A)
    d2 = dist(K2, A)
    if d1 < 1e-12:
        K = K2
    elif d2 < 1e-12:
        K = K1
    else:
        # choose whichever far from A (i.e., not equal)
        K = K2 if d2 > d1 else K1
    # reflection of K across line EF
    def reflect(P, U, V):
        ux, uy = U
        vx, vy = V
        dx = vx-ux
        dy = vy-uy
        # projection t
        tpar = ((P[0]-ux)*dx + (P[1]-uy)*dy) / (dx*dx + dy*dy)
        qx = ux + tpar*dx
        qy = uy + tpar*dy
        rx = 2*qx - P[0]
        ry = 2*qy - P[1]
        return (rx, ry)
    Kp = reflect(K, E, F)
    # circle through B,N,K
    circBNK_center, circBNK_r = circle_center(B, N, K)
    # distance from K' to this circle's center
    dKp = dist(Kp, circBNK_center)
    residual = dKp - circBNK_r
    return residual

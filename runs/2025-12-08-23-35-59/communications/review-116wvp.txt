def circumcenter(A,B,C):
    # returns (ox, oy)
    x1,y1 = A
    x2,y2 = B
    x3,y3 = C
    D = 2*(x1*(y2 - y3) + x2*(y3 - y1) + x3*(y1 - y2))
    if abs(D) < 1e-12:
        return None
    Ux = ((x1**2 + y1**2)*(y2 - y3) + (x2**2 + y2**2)*(y3 - y1) + (x3**2 + y3**2)*(y1 - y2)) / D
    Uy = ((x1**2 + y1**2)*(x3 - x2) + (x2**2 + y2**2)*(x1 - x3) + (x3**2 + y3**2)*(x2 - x1)) / D
    return (Ux, Uy)

def circle_intersections(O1,R1,O2,R2):
    # returns list of intersection points (x,y) of two circles
    (x1,y1) = O1
    (x2,y2) = O2
    dx = x2 - x1
    dy = y2 - y1
    d = math.hypot(dx, dy)
    if d > R1 + R2 + 1e-12 or d < abs(R1 - R2) - 1e-12:
        return []  # no intersection
    # handle near zero distance?
    if d == 0:
        return []
    a = (R1**2 - R2**2 + d**2) / (2*d)
    h_sq = max(R1**2 - a**2, 0.0)
    h = math.sqrt(h_sq)
    # point p2
    xm = x1 + a * dx / d
    ym = y1 + a * dy / d
    # intersection points
    rx = -dy * (h / d)
    ry = dx * (h / d)
    p1 = (xm + rx, ym + ry)
    p2 = (xm - rx, ym - ry)
    return [p1, p2]

def foot_of_perpendicular(P, A, B):
    # foot from point P to line AB
    # returns projection point
    # vector AB
    ax, ay = A
    bx, by = B
    px, py = P
    # compute projection parameter t = dot(P-A, B-A) / |B-A|^2
    vx = bx - ax
    vy = by - ay
    t = ((px - ax)*vx + (py - ay)*vy) / (vx*vx + vy*vy)
    proj = (ax + t*vx, ay + t*vy)
    return proj

def reflect_point_across_line(P, A, B):
    proj = foot_of_perpendicular(P, A, B)
    # reflection: P' = 2*proj - P
    return (2*proj[0] - P[0], 2*proj[1] - P[1])

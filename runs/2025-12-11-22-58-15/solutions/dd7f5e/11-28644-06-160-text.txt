def correlate(alpha_dict, beta_dict):
    # alpha and beta are dict exponent->coeff
    # compute C(n) = sum_t alpha(t+n) * beta(t)
    # Since supports are finite, compute for all n in range min_n..max_n
    # Determine bounds
    min_a = min(alpha_dict.keys())
    max_a = max(alpha_dict.keys())
    min_b = min(beta_dict.keys())
    max_b = max(beta_dict.keys())
    # n runs from (min_a - max_b) to (max_a - min_b)
    n_min = min_a - max_b
    n_max = max_a - min_b
    C = {}
    for n in range(n_min, n_max+1):
        total = 0
        for t, bcoeff in beta_dict.items():
            acoeff = alpha_dict.get(t+n, 0)
            total += acoeff * bcoeff
        C[n] = total
    return C

def compute_residual(d, s, r):
    # all values are floats
    a = d + s
    # slopes:
    mB = 2*d*r/(d**2 - r**2)
    mC = -2*s*r/(s**2 - r**2)
    # points:
    B = mp.matrix([0,0])
    C = mp.matrix([a,0])
    D = mp.matrix([d,0])
    # Intersection A:
    # Solve mB*x = mC*(x-a)
    # x = -mC*a/(mB - mC)
    xA = -mC*a/(mB - mC)
    yA = mB*xA
    A = mp.matrix([xA, yA])
    # Line AB: y=mB x
    # Project I onto AB:
    I = mp.matrix([d,r])
    # projection onto line passing through origin direction v = (1,mB)
    # Formula: proj = (I·v)/(v·v) * v
    vAB = mp.matrix([1,mB])
    proj_factor = (I[0]*vAB[0] + I[1]*vAB[1]) / (vAB[0]**2 + vAB[1]**2)
    F = proj_factor * vAB  # foot on AB
    # Projection onto line AC:
    # line passes through C, direction vC = (1, mC) (since slope mC)
    # Actually direction vector is (1, mC).
    vAC = mp.matrix([1,mC])
    # vector from C to I:
    CI = I - C
    proj_factor2 = (CI[0]*vAC[0] + CI[1]*vAC[1]) / (vAC[0]**2 + vAC[1]**2)
    E = C + proj_factor2 * vAC  # foot on AC
    
    # Compute circumcircle of A,F,E: center O1 and radius R1.
    # Compute circumcenter via intersection of perpendicular bisectors.
    def circumcenter(P, Q, R):
        # Using formula based on coordinates
        # Return center (x,y) as mp matrix
        # from sympy? We'll implement directly using linear algebra.
        # Solve for O: |O-P|^2 = |O-Q|^2 and |O-P|^2 = |O-R|^2
        # Write as linear equations in O = (x,y).

        # Simplify to: -2P.x + 2Q.x * x + -2P.y + 2Q.y * y = Q.x^2+Q.y^2 - P.x^2 - P.y^2
        # We'll set up linear system.
        # Actually we can solve using matrix.
        P = mp.matrix(P)
        Q = mp.matrix(Q)
        R = mp.matrix(R)
        # Build equations A*O = b
        # eq1: 2*(Q-P)·O = Q·Q - P·P
        # eq2: 2*(R-P)·O = R·R - P·P
        A = mp.matrix([[2*(Q[0]-P[0]), 2*(Q[1]-P[1])],
                       [2*(R[0]-P[0]), 2*(R[1]-P[1])]])

                       R[0]**2+R[1]**2 - (P[0]**2+P[1]**2)])
        # Solve
        O = mp.linalg.solve(A, b)
        return O

    R1 = mp.sqrt((O1[0]-A[0])**2 + (O1[1]-A[1])**2)

def circle_intersections(Oa, Ra, Ob, Rb):
    dx = Ob[0] - Oa[0]
    dy = Ob[1] - Oa[1]
    d = mp.sqrt(dx*dx + dy*dy)
    # handle degenerate
    if d == 0:
        return []
    a = (Ra**2 - Rb**2 + d**2) / (2*d)
    h_sq = Ra**2 - a**2
    h = mp.sqrt(max(h_sq, 0))
    xm = Oa[0] + a*dx/d
    ym = Oa[1] + a*dy/d
    rx = -dy * (h/d)
    ry = dx * (h/d)
    p1 = mp.matrix([xm + rx, ym + ry])
    p2 = mp.matrix([xm - rx, ym - ry])
    # If h is near zero, points coincide.
    return [p1, p2]

    a = d + s
    # compute points
    B = mp.matrix([0.0,0.0])
    C = mp.matrix([a,0.0])
    D = mp.matrix([d,0.0])
    # slopes
    mB = 2*d*r/(d**2 - r**2)
    mC = -2*s*r/(s**2 - r**2)
    # Intersection A:
    xA = -mC*a/(mB - mC)
    yA = mB*xA
    A = mp.matrix([xA, yA])
    # foot points
    I = mp.matrix([d,r])
    vAB = mp.matrix([1.0, mB])
    proj_factor = (I[0]*vAB[0] + I[1]*vAB[1]) / (vAB[0]**2 + vAB[1]**2)
    F = proj_factor * vAB
    vAC = mp.matrix([1.0, mC])
    CI = I - C
    proj_factor2 = (CI[0]*vAC[0] + CI[1]*vAC[1]) / (vAC[0]**2 + vAC[1]**2)
    E = C + proj_factor2 * vAC
    # compute K
    O1 = circumcenter(A, F, E)
    R1 = mp.sqrt((O1[0]-A[0])**2 + (O1[1]-A[1])**2)
    O2 = circumcenter(A, B, C)
    R2 = mp.sqrt((O2[0]-A[0])**2 + (O2[1]-A[1])**2)
    # Intersection points of circles O1,R1 and O2,R2
    inter = circle_intersections(O1,R1,O2,R2)
    # choose K not equal to A
    K = None
    for pt in inter:
        if mp.sqrt((pt[0]-A[0])**2 + (pt[1]-A[1])**2) > 1e-8:
            K = pt
            break
    if K is None:
        K = inter[0] if inter else None
    # foot N from D onto EF:
    EF = F - E
    t = ((D[0]-E[0])*EF[0] + (D[1]-E[1])*EF[1]) / (EF[0]**2 + EF[1]**2)
    N = E + t*EF
    
    # compute O_T (center of circle through B and K tangent to BN)
    BN = N - B
    # solve linear equations:
    # BNx*x + BNy*y = 0
    # Kx*x + Ky*y = (Kx**2 + Ky**2)/2
    BNx = BN[0]; BNy = BN[1]
    Kx = K[0]; Ky = K[1]
    # Solve 2x2 system:
    det = BNx*Ky - BNy*Kx
    if abs(det) < 1e-12:
        # degenerate
        return None
    # Using Cramer's rule:
    # [BNx BNy][x] = [0]
    # [Kx   Ky ][y]   [ (Kx^2+Ky^2)/2 ]
    # Solve for x,y:
    # x = (0*Ky - BNy*RHS) / det = (- BNy*RHS) / det
    # y = (BNx*RHS - 0*Kx) / det = (BNx*RHS) / det
    RHS = (Kx**2 + Ky**2)/2
    xO = -BNy * RHS / det
    yO = BNx * RHS / det
    O_T = mp.matrix([xO, yO])
    # radius:
    R_T = mp.sqrt(xO**2 + yO**2)
    # T is second intersection with x-axis, coordinate x_T = 2*O_T.x
    xT = 2*xO
    # compute distances:
    BT = abs(xT - B[0])  # = 2*|xO|
    CT = abs(C[0] - xT)
    NB = mp.sqrt((N[0]-B[0])**2 + (N[1]-B[1])**2)
    NE = mp.sqrt((N[0]-E[0])**2 + (N[1]-E[1])**2)
    # ratio:
    ratio = (CT * NB) / (BT * NE)
    return ratio

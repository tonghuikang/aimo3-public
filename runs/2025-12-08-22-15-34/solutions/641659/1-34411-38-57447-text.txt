def circumcenter(P, Q, R):
    # returns center (x,y) of circle through points P, Q, R
    # using perpendicular bisectors
    # Convert to mp.mpf
    (x1, y1), (x2, y2), (x3, y3) = P, Q, R
    # Compute using linear algebra solution
    # Solve for center (cx,cy) such that |C-P|^2 = |C-Q|^2 and |C-P|^2 = |C-R|^2
    # This gives linear equations:
    # 2*(x2-x1)*cx + 2*(y2-y1)*cy = x2^2 - x1^2 + y2^2 - y1^2
    # 2*(x3-x1)*cx + 2*(y3-y1)*cy = x3^2 - x1^2 + y3^2 - y1^2
    A = [[2*(x2-x1), 2*(y2-y1)],
         [2*(x3-x1), 2*(y3-y1)]]
    Bvec = [x2**2 - x1**2 + y2**2 - y1**2,
            x3**2 - x1**2 + y3**2 - y1**2]
    # Solve linear system
    det = A[0][0]*A[1][1] - A[0][1]*A[1][0]
    if abs(det) < 1e-12:
        raise ValueError("Points collinear")
    cx = (Bvec[0]*A[1][1] - Bvec[1]*A[0][1]) / det
    cy = (A[0][0]*Bvec[1] - A[1][0]*Bvec[0]) / det
    return (cx, cy)

def circle_center_radius(P, Q, R):
    O = circumcenter(P, Q, R)
    Rr = mp.sqrt((O[0]-P[0])**2 + (O[1]-P[1])**2)
    return O, Rr

def intersect_circles(O1, R1, O2, R2):
    # returns the two intersection points (as tuples)
    x1, y1 = O1
    x2, y2 = O2
    d = mp.sqrt((x2-x1)**2 + (y2-y1)**2)
    if d == 0:
        raise ValueError("Coincident centers")
    # compute a = (R1^2 - R2^2 + d^2) / (2d)
    a = (R1**2 - R2**2 + d**2) / (2*d)
    # height h
    h_sq = R1**2 - a**2
    if h_sq < -1e-12:
        # no intersection
        return []
    h = mp.sqrt(max(h_sq, 0))
    # point P2 = O1 + a*(O2-O1)/d
    x3 = x1 + a*(x2 - x1)/d
    y3 = y1 + a*(y2 - y1)/d
    # intersection points
    # offset (rx, ry) = (- (y2 - y1)*(h/d), (x2 - x1)*(h/d))
    rx = -(y2 - y1) * (h/d)
    ry = (x2 - x1) * (h/d)
    inter1 = (x3 + rx, y3 + ry)
    inter2 = (x3 - rx, y3 - ry)
    return [inter1, inter2]

def reflect_point_across_line(P, L1, L2):
    # reflect point P across line through L1 and L2
    # Use vector projection
    x0, y0 = P
    x1, y1 = L1
    x2, y2 = L2
    # line direction v = L2 - L1
    vx, vy = x2 - x1, y2 - y1
    # Normalize? Not required.
    # Compute projection of vector (P-L1) onto v
    # proj = L1 + ((P-L1)·v)/|v|^2 * v
    dot = (x0 - x1)*vx + (y0 - y1)*vy
    norm_sq = vx*vx + vy*vy
    proj_x = x1 + dot*vx / norm_sq
    proj_y = y1 + dot*vy / norm_sq
    # reflected point = 2*proj - P
    rx = 2*proj_x - x0
    ry = 2*proj_y - y0
    return (rx, ry)

def projection_point(P, L1, L2):
    x0, y0 = P
    x1, y1 = L1
    x2, y2 = L2
    vx, vy = x2 - x1, y2 - y1
    dot = (x0 - x1)*vx + (y0 - y1)*vy
    norm_sq = vx*vx + vy*vy
    proj_x = x1 + dot*vx / norm_sq
    proj_y = y1 + dot*vy / norm_sq
    return (proj_x, proj_y)

def foot_of_perpendicular(P, Q, R):
    # foot from point P onto line QR
    # Q, R are points defining line
    # compute projection of vector QP onto QR, then foot
    # Using vector formula: foot = Q + ((P-Q)·(R-Q) / |R-Q|^2)*(R-Q)
    Q = mp.matrix(Q)
    R = mp.matrix(R)
    P = mp.matrix(P)
    v = R - Q
    t = ((P - Q).dot(v)) / (v.dot(v))
    foot = Q + t * v
    return foot

    foot = foot_of_perpendicular(P, Q, R)
    # reflection: P' = 2*foot - P

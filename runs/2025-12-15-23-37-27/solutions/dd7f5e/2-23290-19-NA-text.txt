def compute_correlation(alpha,beta,n):
    # alpha list low->high
    total = 0
    for t, b in beta.items():
        idx = t + n
        if 0 <= idx < len(alpha):
            total += alpha[idx] * b
    return total

def is_shifty(alpha):
    # check existence of beta as per definition
    beta,k,l = construct_beta_from_alpha(alpha)
    if beta is None:
        return False
    # compute correlation for many n; we know only need up to maybe 20
    # we check that correlation yields 1 at k,l and 0 elsewhere
    for n in range(-20, 21):
        val = compute_correlation(alpha,beta,n)
        if n==k or n==l:
            if val != 1:
                return False
        else:
            if val != 0:
                return False
    return True

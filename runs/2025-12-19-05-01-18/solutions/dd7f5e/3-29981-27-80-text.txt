def phi(n):
    result = n
    p = 2
    nn = n
    while p * p <= nn:
        if nn % p == 0:
            while nn % p == 0:
                nn //= p
            result -= result // p
        p += 1 if p == 2 else 2
    if nn > 1:

    return result

    # residues_mods: list of tuples (r, m)
    # returns (x0, M) solution x ≡ x0 (mod M) if exists, else None
    from math import gcd
    x0, M = residues_mods[0]
    # Normalize x0 in [0, M-1)
    x0 %= M
    for (r, m) in residues_mods[1:]:
        r %= m
        g = math.gcd(M, m)
        if (r - x0) % g != 0:
            return None
        # Solve using CRT: find solution modulo lcm(M,m)
        # Compute combined solution
        # Use formula: x = x0 + M * t, need x0 + M*t ≡ r (mod m)
        # => M*t ≡ r - x0 (mod m)
        # Let M' = M/g, m' = m/g, diff = (r-x0)/g
        M1 = M // g
        m1 = m // g
        diff = (r - x0) // g
        # Find t s.t. M1*t ≡ diff (mod m1)
        # M1 and m1 are coprime
        # Compute inv = M1^{-1} mod m1
        inv = pow(M1, -1, m1)  # Python 3.8+ supports
        t = (diff * inv) % m1
        x0 = x0 + M * t
        M = M * m1  # lcm = M * m/g = M * (m/g) = M/g * m = M*m/g
        x0 %= M
    return (x0, M)

def has_solution(subset):
    if not subset:
        return True
    residues_mods = [(m//2, m) for m in subset]
    return crt(residues_mods) is not None

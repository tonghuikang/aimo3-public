def circumcenter_coords(P, Q, R):
    # returns (center, radius) as mp.mpf vectors
    # P, Q, R are 2-element mp.mpf vectors
    x1, y1 = P[0], P[1]
    x2, y2 = Q[0], Q[1]

    A = 2*(x2 - x1)
    B = 2*(y2 - y1)

    D = 2*(x3 - x1)

    # Zero indicates cyclic.
    F_n = fib_int(n)
    F_np1 = fib_int(n+1)
    L = fib_int(n+2)

        return None
    mB = 2*F_n*r/(F_n**2 - r**2)
    mC = -2*F_np1*r/(F_np1**2 - r**2)

    C = (float(L), 0.0)
    xA = -mC * L / (mB - mC)

    F_pt = foot_of_perpendicular(I, B, (1.0, mB))

    E_pt = foot_of_perpendicular(I, C, (L+1.0, mC))

    R1 = math.hypot(A[0]-O1[0], A[1]-O1[1])

            K = pt
    if K is None:
        return None

    K_prime = reflect_point_across_line(K, E_pt, F_pt)

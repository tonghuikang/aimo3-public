    # Using standard F_0=0, F_1=1
    # compute F_n,... up to n+2
    def fib(k):
        if k < 0:
            raise ValueError
        a,b = 0,1
        for _ in range(k):
            a,b = b, a+b
        return a

    F_n1 = fib(n+1)
    F_n2 = fib(n+2)

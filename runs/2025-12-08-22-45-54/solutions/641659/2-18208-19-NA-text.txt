    if n<0:
        raise ValueError
    a,b = 0,1
    for i in range(n):
        a,b = b,a+b
    return a

    CT = C.distance(T)
    NB = N.distance(B)
    BT = B.distance(T)

    # ratio
    R = CT * NB / (BT * NE)

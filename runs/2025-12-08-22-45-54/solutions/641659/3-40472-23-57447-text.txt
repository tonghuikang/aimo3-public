    # Solve linear system for s,t
    # Write matrix [[v1_x, -v2_x], [v1_y, -v2_y]] * [s, t]^T = [M_x, M_y]^T
    A_mat = mp.matrix([[v1[0], -v2[0]], [v1[1], -v2[1]]])
    b_vec = mp.matrix([M[0], M[1]])
    # Solve for [s,t]
    sol = mp.lu_solve(A_mat, b_vec)  # returns matrix [s, t]

    # O = B + s*v1
    O = (B[0] + s_val * v1[0], B[1] + s_val * v1[1])
    # radius
    R_tang = dist(O, B)
    # Intersection with BC: y=0 => (x - O_x)^2 + O_y^2 = R^2 => (x-Ox)^2 = R^2 - Oy^2
    disc = R_tang**2 - O[1]**2
    if disc < 0:
        # no intersection? shouldn't happen
        return diff, None
    sqrt_disc = mp.sqrt(disc)
    x1 = O[0] + sqrt_disc
    x2 = O[0] - sqrt_disc
    # choose x that is not B.x = 0
    # T must be on BC segment from B to C maybe between?
    # Choose the one not equal to 0 (within tolerance)
    if abs(x1) > 1e-8:
        xT = x1
    else:
        xT = x2
    T = (xT, 0.0)
    # compute ratio
    CT = mp.sqrt((C[0]-T[0])**2 + (C[1]-T[1])**2)  # just distance along x-axis
    BT = mp.sqrt((B[0]-T[0])**2 + (B[1]-T[1])**2)

    NE = mp.sqrt((N[0]-E[0])**2 + (N[1]-E[1])**2)
    ratio = (CT * NB) / (BT * NE)
    return diff, ratio

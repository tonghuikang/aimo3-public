<|end|><|start|>assistant<|channel|>analysis to=python code<|message|>def min_ratio(limit):
    import math
    best = 1e9
    best_trip = None
    for a in range(1, limit+1):
        for b in range(a+1, limit+1):
            for c in range(b+1, limit+1):
                s = a+b+c
                l = (a*b)//math.gcd(a,b)
                l = (l*c)//math.gcd(l,c)
                ratio = l/s
                if ratio < best:
                    best = ratio
                    best_trip = (a,b,c)
    return best, best_trip

def geometry_data(p,q,t):
    a = p+q
    # compute incircle radius
    s = a + t
    r = mp.sqrt(t * p * q / s)
    # points:
    B = mp.matrix([0.0, 0.0])
    C = mp.matrix([a, 0.0])
    D = mp.matrix([p, 0.0])
    I = mp.matrix([p, r])
    # slopes of AB and AC
    # mB = 2*p*r/(p**2 - r**2)
    # mC = -2*q*r/(q**2 - r**2)
    # compute line AB passing through B with slope mB
    mB = 2*p*r/(p**2 - r**2)
    # direction vector for AB = (1, mB)
    # line AC passing through C with slope mC
    mC = -2*q*r/(q**2 - r**2)
    # direction for AC = (1, mC)
    # Compute intersection point A
    # Solve B + u*(1,mB) = C + v*(1,mC)
    # => [u, u*mB] = [a+v, v*mC]
    # Actually B = (0,0)+u*(1,mB) = (u, u*mB)
    # C = (a,0) + v*(1,mC) = (a+v, v*mC)
    # Solve u = a+v and u*mB = v*mC
    # from second: v = u*mB / mC if mC not zero. Then plug into first: u = a + u*mB / mC -> u - u*mB/mC = a -> u*(1 - mB/mC) = a -> u = a / (1 - mB/mC)
    # Compute u:
    if abs(mC) < 1e-12:
        raise ValueError("mC zero")
    u = a / (1 - mB/mC)
    v = u * mB / mC
    A = mp.matrix([u, u*mB])
    # foot of perpendicular from I to AB: projection onto line AB
    # line AB direction vector dAB = (1,mB)
    dAB = mp.matrix([1.0, mB])
    # projection of vector I-B onto dAB: proj = dAB * dot(I, dAB) / dot(dAB, dAB)
    dot_I_dAB = I[0]*dAB[0] + I[1]*dAB[1]
    dot_dAB2 = dAB[0]**2 + dAB[1]**2
    proj = dAB * (dot_I_dAB / dot_dAB2)
    F = proj  # foot from I to AB (since B is origin)
    # foot from I to AC: need projection onto line through C direction dAC = (1, mC)
    # point vector from C to I: I - C

    CI = I - C
    dot_CI_dAC = CI[0]*dAC[0] + CI[1]*dAC[1]

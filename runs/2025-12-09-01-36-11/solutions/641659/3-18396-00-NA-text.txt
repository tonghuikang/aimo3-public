    # returns (x,y) of circumcenter of triangle P1,P2,P3 (non-collinear)
    x1, y1 = P1
    x2, y2 = P2
    x3, y3 = P3
    # Solve linear system:
    # (x - x1)^2 + (y - y1)^2 = (x - x2)^2 + (y - y2)^2
    # (x - x1)^2 + (y - y1)^2 = (x - x3)^2 + (y - y3)^2
    # Simplify each:
    # -2x1 x -2y1 y + x1^2 + y1^2 = -2x2 x -2y2 y + x2^2 + y2^2
    # => 2(x2 - x1) x + 2(y2 - y1) y = x2^2 + y2^2 - x1^2 - y1^2
    # Similarly for 3.
    A = [[2*(x2 - x1), 2*(y2 - y1)], [2*(x3 - x1), 2*(y3 - y1)]]
    B = [x2**2 + y2**2 - x1**2 - y1**2, x3**2 + y3**2 - x1**2 - y1**2]
    # Solve linear equations
    det = A[0][0]*A[1][1] - A[0][1]*A[1][0]
    if abs(det) < 1e-12:

    x = (B[0]*A[1][1] - B[1]*A[0][1]) / det
    y = (A[0][0]*B[1] - A[1][0]*B[0]) / det
    return (x, y)

def circle_intersections(O1, R1, O2, R2):
    # returns intersection points of two circles (maybe 0,1,2 points)
    x1,y1 = O1
    x2,y2 = O2
    dx = x2 - x1
    dy = y2 - y1
    d = math.hypot(dx, dy)
    if d > R1 + R2 + 1e-12 or d < abs(R1 - R2) - 1e-12:
        return [] # no intersection
    # distance from O1 to line through intersection points:
    # a = (R1^2 - R2^2 + d^2) / (2d)
    a = (R1**2 - R2**2 + d**2) / (2*d)
    # height from point to intersection
    h_sq = max(R1**2 - a**2, 0.0)
    h = math.sqrt(h_sq)
    # point P2 = O1 + a*(O2-O1)/d
    xm = x1 + a*dx/d
    ym = y1 + a*dy/d
    # intersection points:
    # (x3, y3) = (xm +- h*dy/d, ym -+ h*dx/d)
    # (rotate by +-90)
    xs1 = xm + h * dy / d
    ys1 = ym - h * dx / d
    xs2 = xm - h * dy / d
    ys2 = ym + h * dx / d
    return [(xs1, ys1), (xs2, ys2)]

def projection_point(P, line_pt1, line_pt2):
    # projection of point P onto line defined by points A,B
    x0,y0 = P
    x1,y1 = line_pt1
    x2,y2 = line_pt2
    # vector AB
    vx = x2 - x1
    vy = y2 - y1
    # vector AP
    wx = x0 - x1
    wy = y0 - y1
    # projection factor t = (AP·AB) / (AB·AB)
    t = (wx*vx + wy*vy) / (vx*vx + vy*vy)
    # projection point = A + t * AB
    return (x1 + t*vx, y1 + t*vy)

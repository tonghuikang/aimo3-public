            break
        if abs(r_mid) < 1e-12:
            return s_mid
        # decide which side
        r_lo = f(s_lo)
        if r_lo * r_mid <= 0:
            s_hi = s_mid
        else:
            s_lo = s_mid
    return (s_lo + s_hi) / 2.0

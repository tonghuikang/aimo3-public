    Gx = ((Ax**2 + Ay**2)*(Fy - Ey) + (Fx**2 + Fy**2)*(Ey - Ay) + (Ex**2 + Ey**2)*(Ay - Fy))/Ddet2
    Gy = ((Ax**2 + Ay**2)*(Ex - Fx) + (Fx**2 + Fy**2)*(Ax - Ex) + (Ex**2 + Ey**2)*(Fx - Ax))/Ddet2
    # radius of gamma
    r_gamma = math.sqrt((Ax - Gx)**2 + (Ay - Gy)**2)
    # Find intersection point K other than A between circles (center O radius R) and (center G radius r_gamma)
    # Solve intersection of circles:
    # vector O->G: d = distance between centers
    dx = Gx - Ux
    dy = Gy - Uy
    d = math.sqrt(dx**2 + dy**2)
    # If circles intersect at two points, find intersection points
    # Using formula for intersection of two circles:
    # a = (R^2 - r_gamma^2 + d^2) / (2d)
    # h = sqrt(R^2 - a^2)
    # point P2 = O + a*(G-O)/d
    # intersection points = P2 +/- (h/d) * (-(dy), dx)
    # choose the point not equal to A (close tolerance)
    if d == 0:
        return None
    a_len = (R**2 - r_gamma**2 + d**2) / (2*d)
    # Check for existence
    if R**2 - a_len**2 < -1e-12:
        # no intersection
        return None
    h_inter = max(0.0, math.sqrt(max(0.0, R**2 - a_len**2)))
    # point P2
    P2x = Ux + a_len * (dx) / d
    P2y = Uy + a_len * (dy) / d
    # intersection points
    inter1 = (P2x + h_inter * (-dy) / d, P2y + h_inter * (dx) / d)
    inter2 = (P2x - h_inter * (-dy) / d, P2y - h_inter * (dx) / d)

        return (p[0]-q[0])**2 + (p[1]-q[1])**2

    d2 = dist_sq(inter2, A)

    if d1 < 1e-6:
        K_point = inter2
    elif d2 < 1e-6:
        K_point = inter1
    else:
        # none close to A: choose whichever farther?
        # but one must be A due to construction, might be rounding errors.
        # Choose the one farther from D? Actually ignore.
        # Let's just return None
        # But we can compute K as the intersection other than A by solving intersection and checking closeness to A.
        # Instead compute both, then pick the one with larger distance from A.
        K_point = inter1 if d1 > d2 else inter2
    # Now compute K' reflection of K across line EF.
    # Reflection of point across line given by line defined by two points E and F.
    # formula: reflect K across line L.
    # Compute projection of K onto line EF, then reflect.
    def reflect_point_across_line(P, A1, A2):
        # reflect point P across line through A1, A2
        # Using vector projection.
        # line direction v = A2 - A1
        v = (A2[0] - A1[0], A2[1] - A1[1])
        # normalize?
        # Compute projection of vector AP onto v.
        # AP = P - A1
        AP = (P[0] - A1[0], P[1] - A1[1])
        # dot(AP, v) / dot(v, v)
        t = (AP[0] * v[0] + AP[1] * v[1]) / (v[0]**2 + v[1]**2)
        # projection point Q = A1 + t * v
        Q = (A1[0] + t * v[0], A1[1] + t * v[1])
        # reflection: P' = (2 Q - P)
        P_ref = (2*Q[0] - P[0], 2*Q[1] - P[1])
        return P_ref
    K_prime = reflect_point_across_line(K_point, E_point, F_point)

    # Check if B, K, N, K' are concyclic: i.e., cross ratio? Or check that distances to circumcircle of BKNK' equal.
    # We can compute circumcircle of B, K, N and verify that K' lies on it (within tolerance).
    B_point = B

    N_pt = N_point

    def circumcenter(P1, P2, P3):

        x2, y2 = P2
        x3, y3 = P3
        d = 2*(x1*(y2 - y3) + x2*(y3 - y1) + x3*(y1 - y2))
        if abs(d) < 1e-12:
            return None

        uy = ((x1**2 + y1**2)*(x3 - x2) + (x2**2 + y2**2)*(x1 - x3) + (x3**2 + y3**2)*(x2 - x1))/d
        return (ux, uy)

    if O_BKN is None:
        cyclic_condition = False

    # Let line BN direction vector: N - B.
    # Then center O2 must lie on line through B + t * perpendicular direction to BN.
    # Also O2 must be equidistant to B and K.
    # So O2 satisfies: |O2 - B| = |O2 - K| (circle radius) and O2 - B is perpendicular to BN (tangent condition).
    # Solve for O2.
    # Let BN vector v = (N_x - 0, N_y - 0) = (N_x, N_y).
    # Perpendicular direction w = (-v_y, v_x). So center O2 = B + s * w for some scalar s.
    # Then distance to B = |s w| = |s| * |w| = radius.
    # Distance to K must equal same: |O2 - K| = radius.
    # So |O2 - K| = |s| * |w|.
    # Solve for s: (O2 - K) dot (O2 - K) = s^2 * (w·w).

    # Solve quadratic for s.
    Nx, Ny = N_pt
    Kx, Ky = K_pt
    # w = (-Ny, Nx)
    w_norm_sq = Nx**2 + Ny**2
    # Solve ( -s*Ny - Kx )^2 + ( s*Nx - Ky )^2 = s^2 * w_norm_sq
    # Expand:
    # ( -s*Ny - Kx )^2 = s^2 * Ny^2 + 2 s Ny Kx + Kx^2
    # ( s*Nx - Ky )^2 = s^2 * Nx^2 - 2 s Nx Ky + Ky^2
    # Sum: s^2 (Ny^2 + Nx^2) + 2s (Ny Kx - Nx Ky) + (Kx^2 + Ky^2)
    # = s^2 * w_norm_sq + 2s (Ny Kx - Nx Ky) + (Kx^2 + Ky^2)
    # Set equal to s^2 * w_norm_sq, cancel s^2 terms.
    # So 2s (Ny Kx - Nx Ky) + (Kx^2 + Ky^2) = 0
    # => s = - (Kx^2 + Ky^2) / (2 (Ny Kx - Nx Ky) )
    # Provided denominator not zero.
    denom = 2 * (Ny * Kx - Nx * Ky)
    if abs(denom) < 1e-12:
        # degenerate case
        return None
    s_val = - (Kx**2 + Ky**2) / denom
    # Center O2 = s_val * (-Ny, Nx)
    O2 = ( -s_val * Ny, s_val * Nx )
    # radius = sqrt( O2.x^2 + O2.y^2 ) = |s_val| * sqrt(Nx^2 + Ny^2) = abs(s_val) * sqrt(w_norm_sq)
    radius2 = abs(s_val) * math.sqrt(w_norm_sq)
    # Circle eq: (x - O2.x)^2 + (y - O2.y)^2 = radius2^2.
    # Intersection with BC line y=0.
    # Solve (x - O2.x)^2 + (0 - O2.y)^2 = radius2^2 => (x - O2.x)^2 = radius2^2 - O2.y^2
    # Compute RHS:
    RHS = radius2**2 - O2[1]**2
    if RHS < -1e-9:
        # no real intersection
        return None
    # Solve x = O2.x +/- sqrt(RHS)
    sqrt_RHS = math.sqrt(max(0.0, RHS))
    T_candidates = [O2[0] + sqrt_RHS, O2[0] - sqrt_RHS]
    # Choose T that is not B (B at x=0). So if one candidate approx 0, the other is T.
    T = None
    for cand in T_candidates:
        if abs(cand) > 1e-6:
            T = cand
    if T is None:
        # both near zero? degenerate
        return None
    # So T point = (T,0)
    # Compute needed lengths:
    # CT = |C_x - T|
    CT = abs(C[0] - T)
    # BT = |B_x - T| = |0 - T| = abs(T)
    BT = abs(T)
    # NB = distance from N to B
    NB = math.sqrt(Nx**2 + Ny**2)
    # NE = distance from N to E
    NE = math.sqrt((Nx - E_point[0])**2 + (Ny - E_point[1])**2)
    # ratio
    ratio = (CT * NB) / (BT * NE)
    return ratio, cyclic_condition, {
        'A': A,'B': B,'C': C,'D': D,'I': I,'F': F_point,'E': E_point,
        'N': N_point,'K': K_point,'Kprime': K_prime,'T': (T,0)
    }

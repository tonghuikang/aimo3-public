def divisors_from_factors(factors):
    # factors: dict prime->exponent
    divs = [1]
    for p, e in factors.items():
        new_divs = []
        for d in divs:
            for k in range(1, e+1):
                new_divs.append(d * (p**k))
        divs.extend(new_divs)
    return sorted(set(divs))
